//    Наблюдение: линия ядра α на закрытии бара своей серии (роль KernelLine, кольцо «K») и подпись
//    состояния — прежде модуль ничего не рисовал. Эксперименты, выключены:
//    Э65 ExpThrSigma — порог наклона в долях σ приращений (AFML 3.3, с. 44): тиковый порог не
//        переносится между ТФ и режимами; σ — EWM-std приращений закрытий за 100 баров.
//    Э66 ExpTrendScan — ширина окна по наибольшему |t| наклона среди окон 8…nMax (MLAM 5.4,

//
//  v1.1 — ПРОХОД 2 ПОЛНОЙ ПРОВЕРКИ. Расчёт не изменён. По догме проекта сигналы порождает
//    только структура [MOD1]: в роли Signal этот модуль не голосует, его место — фильтр.
//    Эксперимент Э45 [K69], выключен нулём: СЖАТИЕ НАКЛОНА К НУЛЮ ПО ЕГО НЕОПРЕДЕЛЁННОСТИ.
//    Банковская книга, гл. 1 (с. pdf 10): апостериорное среднее — среднее априорного и оценки
//    с весами их точностей. Априорно наклон около нуля с разбросом τ = [K69] тиков на бар;
//    оценка — наклон локально-линейной регрессии с ошибкой se = |наклон / t|. Сжатый наклон =
//    наклон × τ²/(τ² + se²): шумный наклон тянется к нулю и не проходит порог, уверенный —
//    почти не меняется. Только для локально-линейной регрессии: у неё есть t.
//  Роль Filter по умолчанию, Signal в исследовательском режиме [K65].
//
//  ЧТО СЧИТАЕТ. Локальную линейную регрессию цены со взвешиванием ядром с
//  компактным носителем. На правом краю — там, где принимается решение, —
//  локальная прямая даёт оценку без смещения на линейном тренде, тогда как
//  среднее Надарая–Ватсона на тренде систематически отстаёт.
//
//  ВЕРДИКТ. Наклон прямой в тиках на бар. По модулю ниже порога [K59] —
//  нейтраль; выше — согласие в сторону наклона, и ядро коммутатора запрещает
//  вход против него.
//
//  ZERO-GC. Все массивы создаются один раз в Attach. На такте бара нет ни
//  одного new; строка диагностики пересобирается только при смене состояния.
//  Трассировка пишет строки лишь при событиях. Это гарантия модуля, а не всей
//  стратегии: панель коммутатора выделяет память на каждой отрисовке.
//
//  ЭКСПЕРИМЕНТАЛЬНЫЕ ФУНКЦИИ — выключены по умолчанию, требуют проверки:
//    [K61] значимость наклона: вход по наклону, только если t-статистика
//          наклона выше порога. Приближённая: остатки автокоррелированы, и
//          t завышена — порог нужно подбирать на истории.
//    [K63] пересечение быстрой и медленной прямых вместо знака наклона.
//    [K64] самопроверка предсказательной силы: прогноз прямой на бар вперёд
//          сравнивается с наивным «цена не изменится». Если прямая предсказывает
//          не лучше — модуль уходит в нейтраль. На случайном блуждании наивный
//          прогноз оптимален, поэтому эта проверка честно выключит модуль там,
//          где сглаживание не несёт информации.

namespace NinjaTrader.NinjaScript

    /// <summary>Ядро. Автовыбор по таймфрейму отклонён: см. описание модуля.</summary>
    public enum SwKernelType { Epanechnikov, Triangular, Tricube, Cosine, Gaussian }

    /// <summary>Тип регрессии.</summary>
    public enum SwRegressionType { LocalLinear, NadarayaWatson }

    /// <summary>Нормировка объёма для веса.</summary>
    public enum SwVolumeScale { Robust, Standard }

    /// <summary>Что делать, когда наклон разворачивается против открытой позиции.</summary>
    public enum SwKernelDivergenceAction { Neutral, Veto, RequestPartialExit }
}

namespace NinjaTrader.NinjaScript.Strategies

    public class SwAdaptiveKernelConfig
    {
        public SwKernelType Kernel = SwKernelType.Epanechnikov;
        public SwRegressionType Regression = SwRegressionType.LocalLinear;
        public int Slot = 1;
        public int Bandwidth = 20;
        public bool AdaptiveBandwidth = true;
        public double AdaptMin = 0.5, AdaptMax = 2.0;
        public bool VolumeWeighting = false;
        public double VolumeGamma = 1.0;
        public SwVolumeScale VolumeScale = SwVolumeScale.Robust;
        public int VolumeWindow = 50;
        public double SlopeThresholdTicks = 0.5;
        public SwKernelDivergenceAction Divergence = SwKernelDivergenceAction.Neutral;
        public bool ExpSlopeTGate = false;
        public double ExpShrinkTauTicks = 0;   // Э45: априорный разброс наклона, тиков на бар
        public double ExpTMin = 2.0;
        public bool ExpFastSlowCross = false;
        public bool ExpSkillCheck = false;
        public int SkillWindow = 100;
        public bool SignalMode = false;
        public double ExpThrSigma = 0;        // Э65: порог наклона в σ приращений; 0 — тиковый порог [K59]
        public bool ExpTrendScan = false;     // Э66: окно по наибольшему |t| наклона
    }

    /// <summary>
    /// Чистое вычислительное ядро без обращений к платформе. Подгонка взвешенной
    /// прямой в замкнутой форме: пять сумм за один проход, второй проход — для
    /// стандартной ошибки наклона. Памяти не выделяет.
    /// </summary>
    public sealed class SwAdaptiveKernelEngine
    {
        public double Alpha, Beta, TStat;
        public bool Ok;

        /// <summary>
        /// Значение ядра. Нормирующие множители опущены: в отношениях сумм они
        /// сокращаются. Гаусс усечён на ширине окна, вес на краю exp(−4.5) ≈ 0.011.
        /// </summary>
        public static double K(SwKernelType k, double u)
        {
            if (u < 0) u = -u;
            switch (k)
            {
                case SwKernelType.Triangular: return u < 1.0 ? 1.0 - u : 0.0;
                case SwKernelType.Tricube:
                {
                    if (u >= 1.0) return 0.0;
                    double a = 1.0 - u * u * u;
                    return a * a * a;
                }
                case SwKernelType.Cosine: return u < 1.0 ? Math.Cos(Math.PI * 0.5 * u) : 0.0;
                case SwKernelType.Gaussian: { double z = 3.0 * u; return Math.Exp(-0.5 * z * z); }
                default: return u < 1.0 ? 1.0 - u * u : 0.0;   // Епанечников
            }
        }

        /// <summary>
        /// y[i] — цена i баров назад, x_i = −i. mult[i] — дополнительный множитель
        /// веса, строго положительный. Решение:
        ///   α = (T0·S2 − T1·S1) / D,  β = (S0·T1 − S1·T0) / D,  D = S0·S2 − S1².
        /// При вырожденном D и в режиме Надарая–Ватсона α = T0/S0, β не определён.
        /// </summary>
        public void Fit(double[] y, double[] mult, int n, int h, SwKernelType k, SwRegressionType reg)
        {
            Ok = false; TStat = 0; Beta = double.NaN;
            if (n < 2 || h < 2) return;

            double s0 = 0, s1 = 0, s2 = 0, t0 = 0, t1 = 0;
            for (int i = 0; i < n; i++)
            {
                double w = K(k, (double)i / h) * mult[i];
                if (w <= 0) continue;
                double x = -i;
                s0 += w; s1 += w * x; s2 += w * x * x;
                t0 += w * y[i]; t1 += w * x * y[i];
            }
            if (s0 <= 0) return;

            double d = s0 * s2 - s1 * s1;
            if (reg == SwRegressionType.NadarayaWatson || d <= 1e-12 * s0 * s0)
            {
                Alpha = t0 / s0;
                Ok = true;
                return;
            }

            Alpha = (t0 * s2 - t1 * s1) / d;
            Beta = (s0 * t1 - s1 * t0) / d;

            // Стандартная ошибка наклона: β = Σ c_i·y_i, c_i = w_i·(S0·x_i − S1)/D,
            // Var(β) = σ²·Σc_i². σ² — взвешенная дисперсия остатков.
            double rss = 0, cc = 0;
            for (int i = 0; i < n; i++)
            {
                double w = K(k, (double)i / h) * mult[i];
                if (w <= 0) continue;
                double x = -i;
                double r = y[i] - (Alpha + Beta * x);
                rss += w * r * r;
                double c = w * (s0 * x - s1) / d;
                cc += c * c;
            }
            double se = Math.Sqrt(Math.Max(0.0, rss / s0 * cc));
            TStat = se > 0 ? Beta / se : 0.0;
            Ok = true;
        }
    }

    public class SwAdaptiveKernelModule : SwModule
    {
        private readonly SwAdaptiveKernelConfig cfg;
        private readonly SwAdaptiveKernelEngine eng = new SwAdaptiveKernelEngine();
        private readonly SwAdaptiveKernelEngine engFast = new SwAdaptiveKernelEngine();

        // --- буферы: создаются один раз в Attach ---
        private double[] yBuf, mult, volWin, scratch, trRing;
        private int nMax;

        private int trCount, trIdx;
        private double prevClose = double.NaN;

        // --- результат подгонки ---
        private double alpha = double.NaN, beta = double.NaN, tstat, alphaFast = double.NaN, prevAlpha = double.NaN;
        private int hUsed, fits;

        // --- самопроверка предсказательной силы ---
        private double fcPrev = double.NaN, lastClose = double.NaN, emaK, emaN;
        private int skillN;

        // --- расхождение с позицией ---
        private bool divActive;
        private int divEvents, partialRequests;

        // --- диагностика без выделения памяти на каждом баре ---
        private int lastDir = int.MinValue, lastCode = -1;
        private string textCache = "";

        // v1.2 — Э65: EWM дисперсия приращений; наблюдение: прежняя точка линии ядра
        private double retVar = double.NaN; private int retN;
        private DateTime visPrevT = DateTime.MinValue; private double visPrevA = double.NaN;
        private static readonly int[] ScanWindows = { 8, 12, 16, 24, 32, 48, 64, 96, 128, 160, 192, 256 };

        public SwAdaptiveKernelModule(string id, SwAdaptiveKernelConfig config)
        {
            Id = id;
            cfg = config ?? new SwAdaptiveKernelConfig();
            Role = cfg.SignalMode ? SwRole.Signal : SwRole.Filter;
        }

        public override void Attach(SwHostView host)
        {
            base.Attach(host);
            nMax = Math.Max(4, (int)Math.Ceiling(Math.Max(1, cfg.Bandwidth) * Math.Max(1.0, cfg.AdaptMax))) + 1;
            yBuf = new double[nMax];
            mult = new double[nMax];
            int vw = Math.Max(5, cfg.VolumeWindow);
            volWin = new double[vw];
            scratch = new double[vw];
            trRing = new double[20];
            for (int i = 0; i < nMax; i++) mult[i] = 1.0;

            Host.Trace(Id, "KER_ATTACHED", "ядро " + cfg.Kernel + ", " + cfg.Regression + ", слот " + cfg.Slot
                + ", окно " + cfg.Bandwidth + (cfg.AdaptiveBandwidth ? " адаптивно" : "")
                + (cfg.VolumeWeighting ? ", вес по объёму γ=" + cfg.VolumeGamma : "")
                + ", роль " + Role);
        }

        public override void Compute(int bip)
        {
            if (Host.IsBarCloseOf(cfg.Slot, bip)) UpdateFit();
            if (bip != Host.ExecBip) return;

            int bar = Host.CurrentBar;
            bool ready = fits > 0 && eng.Ok && (!cfg.AdaptiveBandwidth || trCount >= trRing.Length);
            if (!ready)
            {
                Health = SwHealth.Warmup;
                Emit(SwSignal.Empty(bar));
                return;
            }

            double tick = Host.TickSize > 0 ? Host.TickSize : 1.0;
            double slopeT = double.IsNaN(beta) ? 0.0 : beta / tick;
            double thr = cfg.SlopeThresholdTicks;                                            // Э65
            if (cfg.ExpThrSigma > 0 && retN >= 20 && !double.IsNaN(retVar)) thr = cfg.ExpThrSigma * Math.Sqrt(retVar) / tick;
            if (cfg.ExpShrinkTauTicks > 0 && cfg.Regression == SwRegressionType.LocalLinear
                && !double.IsNaN(tstat) && Math.Abs(tstat) > 1e-12)
            {
                double seT = Math.Abs(slopeT / tstat), tau = cfg.ExpShrinkTauTicks;   // Э45
                slopeT *= tau * tau / (tau * tau + seT * seT);
            }

            // --- направление ---
            int dir = 0, code = 0;                       // code: причина для кэша строки
            if (cfg.ExpFastSlowCross && !double.IsNaN(alphaFast))
            {
                double diff = (alphaFast - alpha) / tick;
                if (Math.Abs(diff) >= thr) { dir = Math.Sign(diff); code = 1; }
            }
            else if (Math.Abs(slopeT) >= thr) { dir = Math.Sign(slopeT); code = 1; }

            if (dir != 0 && cfg.ExpSlopeTGate && cfg.Regression == SwRegressionType.LocalLinear
                && Math.Abs(tstat) < cfg.ExpTMin) { dir = 0; code = 2; }

            double skill = SkillScore();
            if (dir != 0 && cfg.ExpSkillCheck && skillN >= cfg.SkillWindow && skill <= 0) { dir = 0; code = 3; }

            // --- расхождение с позицией: наклон разворачивается против неё ---
            int pos = Host.PositionSide;
            bool div = pos != 0 && Math.Abs(slopeT) >= thr && Math.Sign(slopeT) == -pos;
            if (div && !divActive)
            {
                divActive = true; divEvents++;
                Host.Trace(Id, "KER_DIVERGENCE", "наклон " + slopeT.ToString("F2") + " т/бар против позиции " + pos
                    + ", действие " + cfg.Divergence);
                if (cfg.Divergence == SwKernelDivergenceAction.RequestPartialExit)
                {
                    int q = Host.PositionQuantity * 2 / 3;
                    if (q >= 1)
                    {
                        SwAction a = SwAction.None(bar);
                        a.Kind = SwActionKind.PartialExit;
                        a.Quantity = q;
                        a.Tag = "KDIV";
                        a.Reason = "ядро: наклон против позиции";
                        Emit(a);
                        partialRequests++;
                    }
                }
            }
            if (!div) divActive = false;

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;

            if (divActive && cfg.Divergence == SwKernelDivergenceAction.Veto)
            {
                s.Verdict = SwVerdict.Veto; s.Direction = 0; s.Confidence = 1.0; code = 4;
            }
            else if (dir != 0)
            {
                s.Verdict = SwVerdict.Agree; s.Direction = dir;
                s.Confidence = Math.Min(1.0, Math.Abs(slopeT) / Math.Max(1e-9, 4.0 * thr));
            }
            else
            {
                s.Verdict = SwVerdict.Neutral; s.Direction = 0; s.Confidence = 0;
            }

            // Строка пересобирается только при смене состояния: на остальных
            // барах переиспользуется та же ссылка, и память не выделяется.
            if (dir != lastDir || code != lastCode)
            {
                lastDir = dir; lastCode = code;
                textCache = CodeText(code) + (dir > 0 ? " вверх" : dir < 0 ? " вниз" : "")
                          + ", наклон " + slopeT.ToString("F2") + " т/бар, окно " + hUsed;
            }
            s.Text = textCache;

            Health = SwHealth.Ok;
            Emit(s);
        }

        private static string CodeText(int code)
        {
            switch (code)
            {
                case 1: return "ядро: наклон";
                case 2: return "ядро: наклон незначим";
                case 3: return "ядро: нет предсказательной силы";
                case 4: return "ядро: вето, наклон против позиции";
                default: return "ядро: плоско";
            }
        }

        #region Подгонка на закрытии бара своей серии

        private void UpdateFit()
        {
            if (!Host.SlotReady(cfg.Slot, Math.Max(nMax, volWin.Length) + 1)) return;

            double h = Host.HighOf(cfg.Slot, 0), l = Host.LowOf(cfg.Slot, 0), c = Host.CloseOf(cfg.Slot, 0);
            if (double.IsNaN(c)) return;

            // Истинный диапазон для адаптации ширины окна.
            double tr = double.IsNaN(prevClose) ? h - l
                      : Math.Max(h - l, Math.Max(Math.Abs(h - prevClose), Math.Abs(l - prevClose)));
            if (!double.IsNaN(prevClose))                                                   // Э65: σ приращений
            {
                double chg = c - prevClose, a65 = 2.0 / 101.0;
                retVar = double.IsNaN(retVar) ? chg * chg : retVar + a65 * (chg * chg - retVar);
                retN++;
            }
            prevClose = c;
            trRing[trIdx] = tr; trIdx = (trIdx + 1) % trRing.Length;
            if (trCount < trRing.Length) trCount++;

            // Самопроверка: прогноз прошлого бара против фактического закрытия.
            if (!double.IsNaN(fcPrev) && !double.IsNaN(lastClose))
            {
                double a = 2.0 / (Math.Max(2, cfg.SkillWindow) + 1.0);
                double ek = Math.Abs(c - fcPrev), en = Math.Abs(c - lastClose);
                if (skillN == 0) { emaK = ek; emaN = en; }
                else { emaK += a * (ek - emaK); emaN += a * (en - emaN); }
                skillN++;
            }

            // Ширина окна. Высокая краткосрочная волатильность — окно короче,
            // быстрее реакция на импульс; затишье — длиннее, сильнее подавление шума.
            int hb = Math.Max(3, cfg.Bandwidth);
            if (cfg.AdaptiveBandwidth && trCount >= trRing.Length)
            {
                double atr5 = 0, atr20 = 0;
                for (int k = 1; k <= trRing.Length; k++)
                {
                    double x = trRing[(trIdx - k + trRing.Length) % trRing.Length];
                    if (k <= 5) atr5 += x;
                    atr20 += x;
                }
                atr5 /= 5.0; atr20 /= trRing.Length;
                double ratio = atr5 > 0 ? atr20 / atr5 : 1.0;
                ratio = Math.Max(cfg.AdaptMin, Math.Min(cfg.AdaptMax, ratio));
                hb = (int)Math.Round(cfg.Bandwidth * ratio);
            }
            hb = Math.Max(3, Math.Min(nMax - 1, hb));
            if (cfg.ExpTrendScan && cfg.Regression == SwRegressionType.LocalLinear)                   // Э66
            {
                int fill = nMax - 1;
                for (int i = 0; i < fill; i++) { yBuf[i] = Host.CloseOf(cfg.Slot, i); mult[i] = 1.0; }
                double bestT = -1; int bestW = hb;
                for (int k = 0; k < ScanWindows.Length; k++)
                {
                    int w = ScanWindows[k];
                    if (w > fill) break;
                    eng.Fit(yBuf, mult, w, w, cfg.Kernel, cfg.Regression);
                    if (eng.Ok && Math.Abs(eng.TStat) > bestT) { bestT = Math.Abs(eng.TStat); bestW = w; }
                }
                hb = Math.Max(3, Math.Min(nMax - 1, bestW));
            }
            hUsed = hb;

            for (int i = 0; i < hb; i++) { yBuf[i] = Host.CloseOf(cfg.Slot, i); mult[i] = 1.0; }
            if (cfg.VolumeWeighting) FillVolumeWeights(hb);

            eng.Fit(yBuf, mult, hb, hb, cfg.Kernel, cfg.Regression);
            if (!eng.Ok) return;

            alpha = eng.Alpha;
            if (cfg.Regression == SwRegressionType.NadarayaWatson)
                beta = double.IsNaN(prevAlpha) ? 0.0 : alpha - prevAlpha;   // наклон как приращение уровня
            else
                beta = eng.Beta;
            tstat = eng.TStat;

            if (cfg.ExpFastSlowCross)
            {
                int hf = Math.Max(3, hb / 2);
                engFast.Fit(yBuf, mult, hf, hf, cfg.Kernel, cfg.Regression);
                alphaFast = engFast.Ok ? engFast.Alpha : double.NaN;
            }

            fcPrev = double.IsNaN(beta) ? alpha : alpha + beta;
            lastClose = c;
            prevAlpha = alpha;
            fits++;

            // v1.2 — наблюдение: линия ядра и подпись; сбой отрисовки не влияет на расчёт
            try { Observe(); } catch { }
        }

        private void Observe()
        {
            if (!Host.VisualOn(SwVisualRole.KernelLine)) return;
            DateTime t = Host.TimeOf(cfg.Slot, 0);
            if (t == DateTime.MinValue || double.IsNaN(alpha)) return;
            if (visPrevT != DateTime.MinValue && !double.IsNaN(visPrevA))
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.KernelLine, visPrevT, visPrevA, t, alpha)
                    .WithVariant(double.IsNaN(beta) ? 0 : Math.Sign(beta)).InRing("K"));
            double tick = Host.TickSize > 0 ? Host.TickSize : 1.0;
            EmitVisual(SwVisualEvent.Label(SwVisualRole.KernelLine, t, alpha, "K " + hUsed + " β " + (double.IsNaN(beta) ? 0.0 : beta / tick).ToString("F2")).AsState("KLBL"));
            visPrevT = t; visPrevA = alpha;
        }

        /// <summary>
        /// Вес по объёму: множитель 1 + γ·z, ОГРАНИЧЕННЫЙ снизу 0.1 и сверху 4.
        /// В ТЗ множитель не ограничен и при z &lt; −1/γ становится отрицательным;
        /// отрицательные веса разрушают взвешенный МНК. Робастная нормировка —
        /// медиана и 1.4826·MAD: не искажается одиночными всплесками объёма.
        /// </summary>
        private void FillVolumeWeights(int hb)
        {
            int n = volWin.Length;
            for (int i = 0; i < n; i++) volWin[i] = Host.VolumeOf(cfg.Slot, i);

            double center, scale;
            if (cfg.VolumeScale == SwVolumeScale.Robust)
            {
                center = Median(volWin, n);
                for (int i = 0; i < n; i++) scratch[i] = Math.Abs(volWin[i] - center);
                scale = 1.4826 * MedianInPlace(scratch, n);
            }
            else
            {
                double s = 0; for (int i = 0; i < n; i++) s += volWin[i];
                center = s / n;
                double v = 0; for (int i = 0; i < n; i++) v += (volWin[i] - center) * (volWin[i] - center);
                scale = Math.Sqrt(v / Math.Max(1, n - 1));
            }
            if (!(scale > 0)) return;

            for (int i = 0; i < hb; i++)
            {
                double z = (Host.VolumeOf(cfg.Slot, i) - center) / scale;
                double m = 1.0 + cfg.VolumeGamma * z;
                mult[i] = Math.Max(0.1, Math.Min(4.0, m));
            }
        }

        /// <summary>Медиана без выделения памяти: копия в scratch и сортировка вставками.</summary>
        private double Median(double[] src, int n)
        {
            for (int i = 0; i < n; i++) scratch[i] = src[i];
            return MedianInPlace(scratch, n);
        }

        private static double MedianInPlace(double[] a, int n)
        {
            for (int i = 1; i < n; i++)
            {
                double key = a[i]; int j = i - 1;
                while (j >= 0 && a[j] > key) { a[j + 1] = a[j]; j--; }
                a[j + 1] = key;
            }
            return (n % 2 == 1) ? a[n / 2] : 0.5 * (a[n / 2 - 1] + a[n / 2]);
        }

        #endregion

        #region Диагностика

        /// <summary>Доля улучшения против наивного прогноза: 1 − ошибка_ядра / ошибка_наивная.</summary>
        public double SkillScore()
        {
            return emaN > 0 ? 1.0 - emaK / emaN : 0.0;
        }

        {
            get
            {
                double tick = Host != null && Host.TickSize > 0 ? Host.TickSize : 1.0;
                return string.Format("{0} {1} h={2} | наклон {3:F2} т/бар, t {4:F1} | предсказ. сила {5:F2} ({6}) | расхожд. {7}, частичных {8}",
                    cfg.Kernel, cfg.Regression == SwRegressionType.LocalLinear ? "LL" : "NW", hUsed,
                    double.IsNaN(beta) ? 0.0 : beta / tick, tstat, SkillScore(), skillN, divEvents, partialRequests);
            }
        }

        #endregion
    }

        // на коротком окне ширина следует шуму. Для устойчивости — окно 256 и больше.

    public class SwHurstKernelModule : SwModule
    {
        private readonly SwHurstKernelConfig cfg;
        private readonly SwAdaptiveKernelEngine eng = new SwAdaptiveKernelEngine();

        private double[] closeRing, retRing, yBuf, mult;
        private int cap, cCount, cHead, rCount, rHead;

        private int[] scaleN; private double[] logN, logE; private int nScales;

        private double hurst = 0.5, alpha = double.NaN, beta = double.NaN, prevAlpha = double.NaN;
        private int hUsed, fits;
        private string textCache = ""; private int lastDir = int.MinValue, lastCode = -1;
        private double retSd = double.NaN, vrZ = double.NaN;                               // v1.2: Э65, Э73
        private DateTime visPrevT = DateTime.MinValue; private double visPrevA = double.NaN;

        public SwHurstKernelModule(string id, SwHurstKernelConfig config)
        {
            Id = id;
            cfg = config ?? new SwHurstKernelConfig();
            Role = cfg.SignalMode ? SwRole.Signal : SwRole.Filter;
        }

        public override void Attach(SwHostView host)
        {
            base.Attach(host);
            int N = Math.Max(16, cfg.HurstPeriod);
            cap = Math.Max(cfg.MaxBandwidth, N) + 2;
            closeRing = new double[cap];
            retRing = new double[N];
            yBuf = new double[cfg.MaxBandwidth + 2];
            mult = new double[cfg.MaxBandwidth + 2];
            for (int i = 0; i < mult.Length; i++) mult[i] = 1.0;

            // Масштабы n = N, N/2, … ≥ 8 и ожидаемое E[R/S]ₙ для независимых приращений.
            int cnt = 0; for (int n = N; n >= 8; n /= 2) cnt++;
            scaleN = new int[cnt]; logN = new double[cnt]; logE = new double[cnt];
            nScales = 0;
            for (int n = N; n >= 8; n /= 2)
            {
                scaleN[nScales] = n; logN[nScales] = Math.Log(n); logE[nScales] = Math.Log(ExpectedRS(n)); nScales++;
            }
            Host.Trace(Id, "HK_ATTACHED", "окно Хёрста " + N + ", масштабов " + nScales + ", ширина " + cfg.MinBandwidth + "–"
                + cfg.MaxBandwidth + ", " + cfg.Regression + ", роль " + Role);
        }

        /// <summary>
        /// E[R/S]ₙ = ((n − ½)/n) · Γ((n−1)/2) / (√π·Γ(n/2)) · Σᵢ₌₁ⁿ⁻¹ √((n−i)/i) — Anis–Lloyd, Peters.
        /// Отношение гамма-функций — точной рекурсией r(n+2) = r(n)·(n−1)/n без логарифма гаммы:
        /// в .NET Framework его нет. Вызывается только в Attach.
        /// </summary>
        private static double ExpectedRS(int n)
        {
            double r = (n % 2 == 0) ? Math.Sqrt(Math.PI) : 2.0 / Math.Sqrt(Math.PI);
            for (int m = (n % 2 == 0) ? 2 : 3; m < n; m += 2) r *= (m - 1.0) / m;
            double s = 0; for (int i = 1; i < n; i++) s += Math.Sqrt((double)(n - i) / i);
            return ((n - 0.5) / n) * (r / Math.Sqrt(Math.PI)) * s;
        }

        public override void Compute(int bip)
        {
            if (Host.IsBarCloseOf(cfg.Slot, bip)) OnOwnBar();
            if (bip != Host.ExecBip) return;

            int bar = Host.CurrentBar;
            if (fits == 0 || rCount < retRing.Length)
            {
                Health = SwHealth.Warmup;
                Emit(SwSignal.Empty(bar));
                return;
            }

            double tick = Host.TickSize > 0 ? Host.TickSize : 1.0;
            double slopeT = double.IsNaN(beta) ? 0.0 : beta / tick;
            double thr = cfg.SlopeThresholdTicks;                                                            // Э65
            if (cfg.ExpThrSigma > 0 && !double.IsNaN(retSd)) thr = cfg.ExpThrSigma * retSd * Host.CloseOf(cfg.Slot, 0) / tick;
            int dir = Math.Abs(slopeT) >= thr ? Math.Sign(slopeT) : 0;
            int code = dir != 0 ? 1 : 0;
            if (dir != 0 && cfg.ExpRandomWalkGate && Math.Abs(hurst - cfg.RwLevel) < cfg.ExpGateBand) { dir = 0; code = 2; }
            if (dir != 0 && cfg.ExpVrGateZ > 0 && !(vrZ >= cfg.ExpVrGateZ)) { dir = 0; code = 3; }                 // Э73

            double conf = dir != 0 ? Math.Min(1.0, Math.Abs(slopeT) / Math.Max(1e-9, 4.0 * thr)) : 0.0;
            if (cfg.ExpHurstConfidence) conf *= Math.Max(0.0, Math.Min(1.0, (hurst - cfg.RwLevel) / 0.1));

            SwSignal s = new SwSignal();
            s.IsReady = true; s.ProducedAtBar = bar;
            s.EntryStopPrice = double.NaN; s.TrailTicksOverride = double.NaN;
            s.Direction = dir;
            s.Verdict = dir != 0 ? SwVerdict.Agree : SwVerdict.Neutral;
            s.Confidence = conf;
            if (dir != lastDir || code != lastCode)
            {
                lastDir = dir; lastCode = code;
                textCache = (code == 2 ? "Хёрст: блуждание" : code == 3 ? "Хёрст: VR незначим" : dir > 0 ? "Хёрст-ядро: вверх" : dir < 0 ? "Хёрст-ядро: вниз" : "Хёрст-ядро: плоско")
                          + ", H " + hurst.ToString("F2") + ", окно " + hUsed;
            }
            s.Text = textCache;
            Health = SwHealth.Ok;
            Emit(s);
        }

        private void OnOwnBar()
        {
            if (!Host.SlotReady(cfg.Slot, 2)) return;
            double c = Host.CloseOf(cfg.Slot, 0);
            if (double.IsNaN(c) || c <= 0) return;

            double prev = cCount > 0 ? closeRing[cHead] : double.NaN;
            cHead = (cHead + 1) % cap; closeRing[cHead] = c; if (cCount < cap) cCount++;
            if (!double.IsNaN(prev) && prev > 0)
            {
                rHead = (rHead + 1) % retRing.Length; retRing[rHead] = Math.Log(c / prev);
                if (rCount < retRing.Length) rCount++;
            }
            if (rCount >= retRing.Length)
            {
                hurst = HurstRS();
                VarianceRatio();                                                           // v1.2: Э65, Э73
                if (cfg.ExpPriorObs > 0 && !double.IsNaN(hurst))                     // Э46
                {
                    double n = retRing.Length;
                    hurst = cfg.RwLevel + (hurst - cfg.RwLevel) * n / (n + cfg.ExpPriorObs);
                }
            }

            // Отображение Хёрста в ширину окна.
            double t = (hurst - cfg.HLow) / Math.Max(1e-9, cfg.HHigh - cfg.HLow);
            t = Math.Max(0.0, Math.Min(1.0, t));
            if (cfg.ExpInverseMap) t = 1.0 - t;
            int hb = (int)Math.Round(cfg.MinBandwidth + t * (cfg.MaxBandwidth - cfg.MinBandwidth));
            hb = Math.Max(3, Math.Min(Math.Min(cCount, yBuf.Length - 1), hb));
            if (hb < 3) return;
            hUsed = hb;

            for (int i = 0; i < hb; i++) yBuf[i] = closeRing[(cHead - i + cap) % cap];
            eng.Fit(yBuf, mult, hb, hb, cfg.Kernel, cfg.Regression);
            if (!eng.Ok) return;

            alpha = eng.Alpha;
            beta = cfg.Regression == SwRegressionType.NadarayaWatson
                 ? (double.IsNaN(prevAlpha) ? 0.0 : alpha - prevAlpha) : eng.Beta;
            prevAlpha = alpha;
            fits++;
            try { Observe(); } catch { }                                                 // v1.2: наблюдение
        }

        /// <summary>v1.2. Выборочное отклонение доходностей окна (Э65) и z отношения дисперсий q = 4 (Э73).</summary>
        private void VarianceRatio()
        {
            int N = retRing.Length, q = 4;
            double s1 = 0, ss1 = 0;
            for (int k = 0; k < N; k++) { double r = Ret(k); s1 += r; ss1 += r * r; }
            double m1 = s1 / N, v1 = ss1 / N - m1 * m1;
            retSd = v1 > 0 ? Math.Sqrt(v1) : double.NaN;
            double sq = 0, ssq = 0; int nq = 0;
            for (int k = q - 1; k < N; k++)
            {
                double acc = 0; for (int i = 0; i < q; i++) acc += Ret(k - i);
                sq += acc; ssq += acc * acc; nq++;
            }
            if (nq < 8 || !(v1 > 0)) { vrZ = double.NaN; return; }
            double mq = sq / nq, vq = ssq / nq - mq * mq, vr = vq / (q * v1);
            vrZ = (vr - 1.0) / Math.Sqrt(2.0 * (2 * q - 1) * (q - 1) / (3.0 * q * N));
        }

        private void Observe()
        {
            if (!Host.VisualOn(SwVisualRole.HurstKernelLine)) return;
            DateTime t = Host.TimeOf(cfg.Slot, 0);
            if (t == DateTime.MinValue || double.IsNaN(alpha)) return;
            if (visPrevT != DateTime.MinValue && !double.IsNaN(visPrevA))
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.HurstKernelLine, visPrevT, visPrevA, t, alpha)
                    .WithVariant(double.IsNaN(beta) ? 0 : Math.Sign(beta)).InRing("H"));
            EmitVisual(SwVisualEvent.Label(SwVisualRole.HurstKernelLine, t, alpha, "H " + hurst.ToString("F2") + " окно " + hUsed).AsState("HLBL"));
            visPrevT = t; visPrevA = alpha;
        }

        /// <summary>
        /// Показатель Хёрста по нескольким масштабам с вычетом ожидаемого R/S блуждания.
        /// Окна одного масштаба не перекрываются; R/S масштаба — среднее по окнам.
        /// </summary>
        private double HurstRS()
        {
            int N = retRing.Length;
            double sx = 0, sy = 0, sxx = 0, sxy = 0; int m = 0;
            for (int j = 0; j < nScales; j++)
            {
                int n = scaleN[j], windows = N / n;
                double acc = 0; int ok = 0;
                for (int w = 0; w < windows; w++)
                {
                    double mean = 0;
                    for (int i = 0; i < n; i++) mean += Ret(w * n + i);
                    mean /= n;
                    double cum = 0, mn = 0, mx = 0, ss = 0;
                    for (int i = 0; i < n; i++)
                    {
                        double d = Ret(w * n + i) - mean;
                        cum += d; ss += d * d;
                        if (cum < mn) mn = cum; if (cum > mx) mx = cum;
                    }
                    double sd = Math.Sqrt(ss / n);
                    if (sd > 0) { acc += (mx - mn) / sd; ok++; }
                }
                if (ok == 0) continue;
                double x = logN[j], y = Math.Log(acc / ok) - logE[j];
                sx += x; sy += y; sxx += x * x; sxy += x * y; m++;
            }
            if (m < 2) return 0.5;
            double den = m * sxx - sx * sx;
            return den > 0 ? 0.5 + (m * sxy - sx * sy) / den : 0.5;
        }

        /// <summary>k-я доходность окна по порядку времени, от старой к новой.</summary>
        private double Ret(int k) { return retRing[(rHead - (retRing.Length - 1) + k + 2 * retRing.Length) % retRing.Length]; }

        public string StateLine
        {
            get
            {
                double tick = Host != null && Host.TickSize > 0 ? Host.TickSize : 1.0;
                return string.Format("H {0:F3} → окно {1} | {2} | наклон {3:F2} т/бар",
                    hurst, hUsed, cfg.Regression == SwRegressionType.LocalLinear ? "LL" : "NW",
                    double.IsNaN(beta) ? 0.0 : beta / tick);
            }
        }
    }

//
//  v1.1 — ПОРЦИЯ 3 ОТЛАДКИ: РАСЧЁТ НЕ ИЗМЕНЁН. Индексы настроек в комментариях и
//    текстах приведены к меню: прежние [Kxx] указывали на чужие настройки — группа K
//    в меню принадлежит адаптивному ядру, канал Калмана — группа C.
//
//  ЧТО ДЕЛАЕТ. На младшем, текущем и старшем ТФ ведёт фильтр Калмана с
//  моделью «уровень + наклон» и строит вокруг его середины канал. По наклону
//  середины и границ, их параллельности и симметрии классифицирует канал:
//  боковик, тренд вверх, тренд вниз, неопределённость. Дополнительно видит
//  сжатие и расширение канала и дисбаланс расстояний от середины до границ.
//
//  КАК ИСПОЛЬЗУЕТСЯ В РЕШЕНИИ.
//    роль Filter — вердикт по ТФ направления [C20]: в тренде — согласие с
//                  направлением, в боковике — по политике [C07];
//    роль Signal — пробой границы боковика ТФ входа; ТФ входа — младший,
//                  при его отсутствии — текущий;
//    в обеих ролях, по выбору — сопровождение стопом по границе канала ТФ
//                  сопровождения [C21] и выход при развороте середины и
//                  коридора против позиции.
//  Младший ТФ отвечает за вход, текущий или старший — за сопровождение.
//
//  МОДЕЛЬ. Состояние x = [уровень, наклон за бар], переход F = [[1,1],[0,1]],
//  измерение — цена закрытия. Шум процесса — модель дискретного белого
//  ускорения: Q = λ²·R·[[1/4,1/2],[1/2,1]]. Индекс слежения λ выводится из
//  эквивалентного периода N [C05] по формулам Калаты: α = 2/(N+1),
//  β = 2(2−α) − 4√(1−α), λ = β/√(1−α). Установившиеся коэффициенты фильтра
//  тогда равны α и β, как у альфа-бета фильтра с тем же периодом; численно
//  проверено на случайном блуждании: при N = 20 коэффициент уровня 0.099
//  против расчётных 0.095.
//
//  ШУМ ИЗМЕРЕНИЯ R — адаптивный, по Сейджу–Хьюзу с забыванием 0.97:
//  R ← (1−d)·R + d·(ν² − P⁻₀₀). Поскольку Q пропорционален R, коэффициенты
//  от масштаба цены не зависят: одни настройки годятся для золота и серебра.
//
//  ГРАНИЦЫ КАНАЛА — ОТДЕЛЬНЫЙ БЛОК [C08]:
//    GaussianSemiDeviation — полуотклонения остатков цены от середины с
//      гауссовыми весами по плавающему окну; верхняя граница по остаткам
//      выше середины, нижняя — по остаткам ниже. Канал асимметричен, и его
//      асимметрия — это и есть дисбаланс, мера импульса;
//    KalmanInnovation — середина ± k·√S, где S — дисперсия обновления
//      фильтра: интервал прогноза на один бар. Симметричен; дисбаланс в этом
//      режиме всё равно считается по полуотклонениям.
//
//  ДВА ПЛАВАЮЩИХ ОКНА. Окно входа [C09]-[C10] — гибкое, для ТФ входа.
//  Окно сопровождения [C11]-[C12] — жёсткое, для текущего и старшего ТФ.
//  Длина окна плавает между минимумом и максимумом по коэффициенту
//  эффективности Кауфмана: в направленном движении окно короче, в боковике
//  длиннее, и границы боковика оцениваются по большей выборке.
//
//  НА КАКОЙ СВЕЧЕ ОБЪЯВЛЯЕТСЯ БОКОВИК. Условия «горизонтально, параллельно,
//  симметрично» проверяются на каждом баре по сглаженным мерам; боковик
//  объявляется на [C15]-й подряд свече, где все они выполнены (0 — авто:
//  половина минимума окна входа, от 3 до 12; по умолчанию 5). Выход из
//  боковика — пробой границы предыдущего бара или появление тренда, а не
//  первый же «неплоский» бар: одиночный шум состояние не сбрасывает.
//  Число 5 взято из проверки на синтетических рядах и требует калибровки на
//  ваших данных: модуль печатает для каждого m долю случаев, когда цена
//  после m плоских свечей удержалась в канале [C30] баров.
//
//  ЭКСПЕРИМЕНТАЛЬНЫЙ БЛОК [C28]. Взаимодействие канала младшего ТФ с каналом
//  старшего: пересечения середин, выход границ и середины младшего за
//  границы старшего, смена согласия наклонов, боковик младшего внутри тренда
//  старшего. И взаимодействие цены с линиями ТФ входа. Для каждого события —
//  доходность через [C29] баров в ширинах канала и доля движений в сторону
//  события; рядом — база по всем барам. В торговлю блок не вмешивается.
//
//  ПРЕДВАРИТЕЛЬНОЕ СОСТОЯНИЕ СТАРШИХ ТФ. На закрытии бара младшего ТФ
//  модуль считает, где была бы середина старшего ТФ, если бы его бар закрылся
//  сейчас по текущей цене: шаг фильтра без сохранения. Только для окраски
//  линии (наблюдение [V01]) и для экспериментального блока; вердикт, вход, стоп и выход —
//  по закрытым барам.
//
//  ЧЕГО ЗДЕСЬ НЕТ. Взгляда в будущее: каждое значение опирается только на
//  закрытые бары своей серии и, для предварительного состояния, на последнюю
//  известную цену. Проверка «удержался ли боковик» и доходности событий
//  используют будущие бары только для статистики отчёта, не для решений.
// ============================================================================

namespace NinjaTrader.NinjaScript
{
    /// <summary>Роль модуля Кальмана в решении.</summary>
    public enum SwKalmanRole
    {
        Filter,     // вердикт по состоянию канала ТФ направления
        Signal      // пробой границы боковика ТФ входа
    }

    /// <summary>Как считаются границы канала.</summary>
    public enum SwKalmanBands
    {
        GaussianSemiDeviation,  // полуотклонения остатков, гауссовы веса; асимметричный канал
        KalmanInnovation        // середина ± k·√S; симметричный канал
    }

    /// <summary>Вердикт модуля в роли Filter.</summary>
    public enum SwKalmanFilterPolicy
    {
        TrendDirection,     // в тренде — только по направлению; в остальном не мешает
        TrendOnly,          // вход только в тренде и только по направлению
        NoRange             // запрет входа в боковике; направления не задаёт
    }

    /// <summary>Выбор таймфрейма для направления и сопровождения.</summary>
    public enum SwKalmanTfChoice
    {
        Current,
        Higher
    }

    /// <summary>Состояние канала одного ТФ.</summary>
    public enum SwKalmanState
    {
        Warmup,
        Undefined,
        Range,
        TrendUp,
        TrendDown
    }
}

    /// <summary>Кольцо последних значений. Get(0) — последнее.</summary>
    public class SwRing
    {
        private readonly double[] a;
        private int head, count;

        public SwRing(int capacity) { a = new double[Math.Max(2, capacity)]; }

        public int Count { get { return count; } }

        public void Push(double v)
        {
            head = (head + 1) % a.Length;
            a[head] = v;
            if (count < a.Length) count++;
        }

        public double Get(int ago)
        {
            if (ago < 0 || ago >= count) return double.NaN;
            int i = head - ago;
            if (i < 0) i += a.Length;
            return a[i];
        }
    }

    /// <summary>
    /// ПЛАВАЮЩЕЕ ОКНО. Длина — между Min и Max по коэффициенту эффективности
    /// Кауфмана за Max баров: ER = |чистый ход| / сумма |шагов|. ER близко к
    /// единице — направленное движение, окно короче; близко к нулю — пила,
    /// окно длиннее. Длина сглажена, чтобы окно не дёргалось с бара на бар.
    /// </summary>
    public class SwFloatingWindow
    {
        public readonly int Min, Max;
        private double lf;
        private int length;
        private double er;

        public SwFloatingWindow(int min, int max)
        {
            int lo = Math.Max(3, Math.Min(min, max));
            int hi = Math.Max(lo, Math.Max(min, max));
            Min = lo; Max = hi;
            lf = hi; length = hi;
        }

        public int Length { get { return length; } }
        public double Efficiency { get { return er; } }

        public void Update(SwRing closes)
        {
            if (closes.Count <= Max)
            {
                length = Math.Max(Min, Math.Min(Max, closes.Count - 1));
                return;
            }
            double net = Math.Abs(closes.Get(0) - closes.Get(Max));
            double path = 0;
            for (int i = 0; i < Max; i++) path += Math.Abs(closes.Get(i) - closes.Get(i + 1));
            er = path > 0 ? net / path : 0;

            double target = Min + (Max - Min) * (1.0 - er);
            lf += 0.2 * (target - lf);
            length = Math.Max(Min, Math.Min(Max, (int)Math.Round(lf)));
        }
    }

    /// <summary>
    /// ОДИН ТАЙМФРЕЙМ: фильтр Калмана, канал, геометрия, состояние. Чистое
    /// вычисление: данные получает аргументами, к Host не обращается, поэтому
    /// проверяется отдельно от платформы. Update вызывается ровно один раз на
    /// закрытии бара своей серии.
    /// </summary>
    public class SwKalmanTrack
    {
        public readonly int Tf;          // 0 младший, 1 текущий, 2 старший
        public readonly string Name;     // МЛ, ТК, СТ
        public readonly int Slot;
        public readonly SwFloatingWindow Window;

        // --- параметры ---
        private readonly double lam2, forget, tick, k, rMin;
        private readonly SwKalmanBands method;
        private readonly double flatTilt, symMax, trendZ, trendTilt, compRatio, expRatio;
        private readonly int warmupBars, confirmBars;

        // --- фильтр ---
        private int n, steps;
        private readonly double[] boot = new double[5];
        private double x0, x1, p00, p01, p11, r, s;
        public double NisAvg = double.NaN;     // Э21: скользящее среднее ν²/f
        private double nisA;

        // --- кольца ---
        private readonly SwRing closes, resid, uppers, lowers, widths;

        // --- сглаженные меры геометрии ---
        private bool smoothed;
        private double zS, tiltS, tuS, tlS, imbS;
        private const double SmoothA = 0.3;

        public SwKalmanTrack(int tf, string name, int slot, SwKalmanConfig c, SwFloatingWindow window,
                             double tickSize, int confirm)
        {
            Tf = tf; Name = name; Slot = slot; Window = window;

            int per = Math.Max(2, c.Period);
            double a = 2.0 / (per + 1.0);
            double b = 2.0 * (2.0 - a) - 4.0 * Math.Sqrt(1.0 - a);
            double lam = b / Math.Sqrt(1.0 - a);
            lam2 = lam * lam;
            nisA = 2.0 / (per + 1.0);
            forget = 0.97;
            tick = tickSize > 0 ? tickSize : 1e-6;
            rMin = (0.25 * tick) * (0.25 * tick);
            k = Math.Max(0.1, c.BandSigma);
            method = c.Bands;

            flatTilt = Math.Max(0.01, c.FlatTilt);
            symMax = Math.Max(0.01, c.SymmetryMax);
            trendZ = Math.Max(0.5, c.TrendZ);
            trendTilt = Math.Max(0.01, c.TrendTilt);
            compRatio = c.CompressionRatio;
            expRatio = c.ExpansionRatio;
            warmupBars = Math.Max(10, Math.Max(1, c.WarmupPeriods) * per);
            confirmBars = Math.Max(1, confirm);

            int cap = window.Max + 8;
            closes = new SwRing(cap); resid = new SwRing(cap);
            uppers = new SwRing(cap); lowers = new SwRing(cap); widths = new SwRing(cap);

            Level = Upper = Lower = PrevLevel = PrevUpper = PrevLower = double.NaN;
            State = SwKalmanState.Warmup;
        }

        // ---------------- результаты последнего бара ----------------
        public DateTime Time;
        public double Close, High, Low;
        public int BarsSeen;
        public bool Ready;

        public double Level, Upper, Lower, Width;
        public double PrevLevel, PrevUpper, PrevLower;
        public double Slope;                // наклон середины, цена за бар
        public double SlopeZ;               // значимость наклона: наклон / его стандартная ошибка
        public double Tilt;                 // смещение середины за окно, доля ширины канала
        public double TiltUpper, TiltLower; // то же для границ
        public double Dup, Ddn;             // полуотклонения выше и ниже середины
        public double Imbalance;            // (Dup − Ddn) / (Dup + Ddn): + импульс вверх, − вниз
        public double WidthRatio = 1.0;     // ширина к своей средней: сжатие и расширение

        public double SmoothZ { get { return zS; } }
        public double SmoothTilt { get { return tiltS; } }
        public double SmoothImbalance { get { return imbS; } }
        public double CorridorTilt { get { return (tuS + tlS) / 2.0; } }
        public int ConfirmBars { get { return confirmBars; } }

        public bool FlatNow, InRange, RangeDeclared, RangeEnded, BreakUp, BreakDown, RangeCompressed;
        public int FlatStreak, RangeBars, RangeSeq;
        public DateTime StreakStart, RangeStart;
        public double RangeUpperAtBreak, RangeLowerAtBreak;

        public bool Compression, Expansion, CompressionStarted, ExpansionStarted;
        public bool CrossMidUp, CrossMidDown, TouchUpper, TouchLower, CloseAbove, CloseBelow;

        public SwKalmanState State;
        public int MidVariant, UpperVariant, LowerVariant;   // окраска: +1 рост, −1 падение, 0 горизонталь

        public double ZFlat { get { return trendZ / 2.0; } }

        /// <summary>Один шаг: закрытие бара своей серии.</summary>
        public void Update(double z, double hi, double lo, DateTime time)
        {
            RangeDeclared = RangeEnded = BreakUp = BreakDown = false;
            CrossMidUp = CrossMidDown = TouchUpper = TouchLower = CloseAbove = CloseBelow = false;
            CompressionStarted = ExpansionStarted = false;

            if (double.IsNaN(z)) return;
            if (double.IsNaN(hi)) hi = z;
            if (double.IsNaN(lo)) lo = z;

            double prevClose = closes.Count > 0 ? closes.Get(0) : double.NaN;
            closes.Push(z);
            Window.Update(closes);
            Time = time; Close = z; High = hi; Low = lo;
            BarsSeen++;

            // --- затравка: пять баров на начальную оценку шума и наклона ---
            if (n < 5)
            {
                boot[n] = z;
                n++;
                if (n == 5)
                {
                    double acc = 0;
                    for (int i = 1; i < 5; i++) { double d = boot[i] - boot[i - 1]; acc += d * d; }
                    r = Math.Max(acc / 4.0, rMin);
                    x0 = z; x1 = (boot[4] - boot[0]) / 4.0;
                    p00 = 10 * r; p01 = 0; p11 = r;
                    s = p00 + r;
                }
                Ready = false;
                State = SwKalmanState.Warmup;
                return;
            }

            // --- прогноз ---
            double q = lam2 * r;
            double a00 = p00 + 2 * p01 + p11 + q / 4.0;
            double a01 = p01 + p11 + q / 2.0;
            double a11 = p11 + q;
            double xp0 = x0 + x1, xp1 = x1;

            // --- обновление ---
            double nu = z - xp0;
            double sv = a00 + r;
            double k0 = a00 / sv, k1 = a01 / sv;
            x0 = xp0 + k0 * nu;
            x1 = xp1 + k1 * nu;
            p00 = (1 - k0) * a00;
            p01 = (1 - k0) * a01;
            p11 = a11 - k1 * a01;
            if (p11 < 1e-18) p11 = 1e-18;
            if (p00 < 1e-18) p00 = 1e-18;
            s = sv;
            double nis = sv > 0 ? nu * nu / sv : double.NaN;   // Э21
            if (!double.IsNaN(nis)) NisAvg = double.IsNaN(NisAvg) ? nis : NisAvg + nisA * (nis - NisAvg);

            // --- шум измерения по Сейджу–Хьюзу ---
            steps++;
            double dsh = (1 - forget) / (1 - Math.Pow(forget, steps + 1));
            r = Math.Max((1 - dsh) * r + dsh * (nu * nu - a00), rMin);
            n++;

            // --- канал ---
            resid.Push(z - x0);
            int L = Math.Min(Window.Length, resid.Count);
            if (L < 3) { Ready = false; State = SwKalmanState.Warmup; return; }

            double sig = L / 2.0, W = 0, U2 = 0, D2 = 0;
            for (int i = 0; i < L; i++)
            {
                double rr = resid.Get(i);
                double u = i / sig;
                double w = Math.Exp(-0.5 * u * u);
                W += w;
                if (rr > 0) U2 += w * rr * rr; else D2 += w * rr * rr;
            }
            Dup = Math.Sqrt(U2 / W);
            Ddn = Math.Sqrt(D2 / W);
            Imbalance = Dup + Ddn > 0 ? (Dup - Ddn) / (Dup + Ddn) : 0;

            double halfUp, halfDn;
            if (method == SwKalmanBands.KalmanInnovation)
            {
                halfUp = halfDn = k * Math.Sqrt(s);
            }
            else
            {
                // Множитель √2 приводит полуотклонение к сигме: у симметричного
                // распределения средний квадрат положительной части — σ²/2.
                halfUp = k * Math.Sqrt(2.0) * Dup;
                halfDn = k * Math.Sqrt(2.0) * Ddn;
            }
            if (halfUp < tick) halfUp = tick;
            if (halfDn < tick) halfDn = tick;

            PrevLevel = Level; PrevUpper = Upper; PrevLower = Lower;
            Level = x0; Slope = x1;
            Upper = x0 + halfUp; Lower = x0 - halfDn;
            Width = Upper - Lower;
            uppers.Push(Upper); lowers.Push(Lower); widths.Push(Width);

            // --- ширина к своей средней по окну: сжатие и расширение ---
            int lw = Math.Min(widths.Count, Window.Max);
            double sw = Math.Max(1.0, lw / 2.0), ww = 0, swd = 0;
            for (int i = 0; i < lw; i++)
            {
                double u = i / sw;
                double w = Math.Exp(-0.5 * u * u);
                ww += w; swd += w * widths.Get(i);
            }
            bool wasComp = Compression, wasExp = Expansion;
            WidthRatio = swd > 0 ? Width / (swd / ww) : 1.0;
            Compression = WidthRatio < compRatio;
            Expansion = WidthRatio > expRatio;
            CompressionStarted = Compression && !wasComp;
            ExpansionStarted = Expansion && !wasExp;

            // --- геометрия ---
            int m = Math.Max(5, L / 2);
            double wd = Math.Max(Width, 2 * tick);
            Tilt = x1 * L / wd;
            TiltUpper = OlsSlope(uppers, m) * m / wd;
            TiltLower = OlsSlope(lowers, m) * m / wd;
            SlopeZ = x1 / Math.Sqrt(p11);

            if (!smoothed)
            {
                zS = SlopeZ; tiltS = Tilt; tuS = TiltUpper; tlS = TiltLower; imbS = Imbalance;
                smoothed = true;
            }
            else
            {
                zS += SmoothA * (SlopeZ - zS);
                tiltS += SmoothA * (Tilt - tiltS);
                tuS += SmoothA * (TiltUpper - tuS);
                tlS += SmoothA * (TiltLower - tlS);
                imbS += SmoothA * (Imbalance - imbS);
            }

            // --- боковик: горизонтально, параллельно, симметрично ---
            double zFlat = trendZ / 2.0, tauB = 1.25 * flatTilt;
            FlatNow = Math.Abs(zS) < zFlat && Math.Abs(tiltS) < flatTilt
                   && Math.Abs(tuS) < tauB && Math.Abs(tlS) < tauB
                   && Math.Abs(tuS - tlS) < tauB && Math.Abs(imbS) < symMax;

            if (FlatNow)
            {
                if (FlatStreak == 0) StreakStart = time;
                FlatStreak++;
            }
            else FlatStreak = 0;

            bool readyNow = n >= warmupBars && resid.Count >= Window.Min;
            bool trendNow = Math.Abs(zS) >= trendZ && Math.Abs(tiltS) >= trendTilt;

            if (InRange)
            {
                // Выход — пробой границы ПРЕДЫДУЩЕГО бара: собственный остаток
                // бара пробоя уже расширил текущую границу. Либо появился тренд.
                bool up = !double.IsNaN(PrevUpper) && z > PrevUpper;
                bool dn = !double.IsNaN(PrevLower) && z < PrevLower;
                if (up || dn || trendNow)
                {
                    InRange = false;
                    RangeEnded = true;
                    BreakUp = up;
                    BreakDown = dn;
                    RangeUpperAtBreak = PrevUpper;
                    RangeLowerAtBreak = PrevLower;
                    FlatStreak = 0;
                }
                else
                {
                    RangeBars++;
                    if (Compression) RangeCompressed = true;
                }
            }
            else if (readyNow && FlatStreak >= confirmBars)
            {
                InRange = true;
                RangeDeclared = true;
                RangeSeq++;
                RangeStart = StreakStart;
                RangeBars = FlatStreak;
                RangeCompressed = Compression;
            }

            // --- состояние и окраска ---
            if (!readyNow) State = SwKalmanState.Warmup;
            else if (InRange) State = SwKalmanState.Range;
            else if (trendNow) State = x1 > 0 ? SwKalmanState.TrendUp : SwKalmanState.TrendDown;
            else State = SwKalmanState.Undefined;
            Ready = readyNow;

            MidVariant = zS >= zFlat ? 1 : zS <= -zFlat ? -1 : 0;
            UpperVariant = tuS >= tauB ? 1 : tuS <= -tauB ? -1 : 0;
            LowerVariant = tlS >= tauB ? 1 : tlS <= -tauB ? -1 : 0;

            // --- цена и линии ---
            if (!double.IsNaN(PrevLevel) && !double.IsNaN(prevClose))
            {
                CrossMidUp = prevClose <= PrevLevel && z > Level;
                CrossMidDown = prevClose >= PrevLevel && z < Level;
            }
            if (!double.IsNaN(PrevUpper))
            {
                CloseAbove = z > PrevUpper;
                TouchUpper = hi >= PrevUpper && !CloseAbove;
            }
            if (!double.IsNaN(PrevLower))
            {
                CloseBelow = z < PrevLower;
                TouchLower = lo <= PrevLower && !CloseBelow;
            }
        }

        /// <summary>
        /// Предварительное состояние: где была бы середина, закройся бар сейчас
        /// по цене z. Шаг фильтра без сохранения; границы — нынешние
        /// полуширины от новой середины. Для окраски и исследования.
        /// </summary>
        public bool Peek(double z, out double level, out double upper, out double lower, out int variant)
        {
            level = upper = lower = double.NaN;
            variant = 0;
            if (n <= 5 || double.IsNaN(Level) || double.IsNaN(z)) return false;

            double q = lam2 * r;
            double a00 = p00 + 2 * p01 + p11 + q / 4.0;
            double a01 = p01 + p11 + q / 2.0;
            double a11 = p11 + q;
            double xp0 = x0 + x1, xp1 = x1;
            double nu = z - xp0;
            double sv = a00 + r;
            double k0 = a00 / sv, k1 = a01 / sv;

            level = xp0 + k0 * nu;
            double slope = xp1 + k1 * nu;
            double pp11 = Math.Max(a11 - k1 * a01, 1e-18);
            double zz = slope / Math.Sqrt(pp11);

            upper = level + (Upper - Level);
            lower = level - (Level - Lower);
            double zFlat = trendZ / 2.0;
            variant = zz >= zFlat ? 1 : zz <= -zFlat ? -1 : 0;
            return true;
        }

        /// <summary>Наклон по методу наименьших квадратов за m последних значений, за бар.</summary>
        private static double OlsSlope(SwRing ring, int m)
        {
            int mm = Math.Min(m, ring.Count);
            if (mm < 3) return 0;
            double xb = (mm - 1) / 2.0, yb = 0;
            for (int i = 0; i < mm; i++) yb += ring.Get(i);
            yb /= mm;
            double num = 0, den = 0;
            for (int i = 0; i < mm; i++)
            {
                double x = (mm - 1 - i) - xb;
                num += x * (ring.Get(i) - yb);
                den += x * x;
            }
            return den > 0 ? num / den : 0;
        }

        public static string StateText(SwKalmanState st)
        {
            switch (st)
            {
                case SwKalmanState.Range: return "боковик";
                case SwKalmanState.TrendUp: return "тренд ↑";
                case SwKalmanState.TrendDown: return "тренд ↓";
                case SwKalmanState.Undefined: return "неопр.";
                default: return "прогрев";
            }
        }
    }

    /// <summary>Накопитель статистики события: число, средняя, доля в сторону события.</summary>
    public class SwKalmanStat
    {
        public int N, Hits;
        public double Sum, Sum2;

        public void Add(double v)
        {
            if (double.IsNaN(v) || double.IsInfinity(v)) return;
            N++; Sum += v; Sum2 += v * v;
            if (v > 0) Hits++;
        }

        public double Mean { get { return N > 0 ? Sum / N : double.NaN; } }
        public double HitRate { get { return N > 0 ? (double)Hits / N : double.NaN; } }

        /// <summary>Стандартная ошибка средней.</summary>
        public double StdErr
        {
            get
            {
                if (N < 2) return double.NaN;
                double m = Sum / N;
                double v = (Sum2 - N * m * m) / (N - 1);
                return v > 0 ? Math.Sqrt(v / N) : 0;
            }
        }
    }

    public class SwKalmanChannelModule : SwModule
    {
        private readonly SwKalmanConfig cfg;
        private readonly SwKalmanTrack[] tr = new SwKalmanTrack[3];
        private readonly bool[] tfOn = new bool[3];
        private int entryTf = 1, escortTf = 1, dirTf = 1;
        private int confirmBars = 5;

        // --- сигнал пробоя, ожидающий такта исполнения ---
        private int pendDir, pendBarsLeft;
        private double pendConf, pendLevel;
        private string pendId = string.Empty, pendText = string.Empty, pendShort = string.Empty;

        // --- выход и сопровождение ---
        private int exitAgainst;
        private bool exitArmed;
        private string exitReason = string.Empty;
        private int lastPosSide;
        private double lastAvg = double.NaN;
        private int exitShownBar = -1;

        // --- счётчики для панели и итога ---
        private int breakoutsUp, breakoutsDown, suppressedHigher, cancelledInside, exitsRequested, stopRequests;
        private int rangesDeclared;

        private static readonly string[] TfNames = { "МЛ", "ТК", "СТ" };

        public SwKalmanChannelModule(string id, SwKalmanConfig config)
        {
            Id = id;
            cfg = config ?? new SwKalmanConfig();
            Role = cfg.Role == SwKalmanRole.Signal ? SwRole.Signal : SwRole.Filter;
        }

        public override void Attach(SwHostView host)
        {
            base.Attach(host);

            confirmBars = cfg.RangeConfirmBars > 0
                ? cfg.RangeConfirmBars
                : Math.Max(3, Math.Min(12, (int)Math.Round(Math.Min(cfg.EntryWindowMin, cfg.EntryWindowMax) / 2.0)));

            // Доступность таймфреймов. Невключённый слот пула читался бы с
            // графика и молча дублировал бы текущий ТФ — такой ТФ выключается.
            int[] slots = { cfg.LowerSlot, cfg.CurrentSlot, cfg.HigherSlot };
            tfOn[1] = true;
            tfOn[0] = Usable(slots[0]) && Host.Bip(slots[0]) != Host.Bip(slots[1]);
            tfOn[2] = Usable(slots[2]) && Host.Bip(slots[2]) != Host.Bip(slots[1])
                    && (!tfOn[0] || Host.Bip(slots[2]) != Host.Bip(slots[0]));

            entryTf = tfOn[0] ? 0 : 1;
            escortTf = cfg.EscortTf == SwKalmanTfChoice.Higher && tfOn[2] ? 2 : 1;
            dirTf = cfg.DirectionTf == SwKalmanTfChoice.Higher && tfOn[2] ? 2 : 1;

            // Окно входа — у ТФ входа; окно сопровождения — у остальных.
            for (int i = 0; i < 3; i++)
            {
                bool entryWin = i == entryTf;
                SwFloatingWindow w = entryWin
                    ? new SwFloatingWindow(cfg.EntryWindowMin, cfg.EntryWindowMax)
                    : new SwFloatingWindow(cfg.EscortWindowMin, cfg.EscortWindowMax);
                tr[i] = new SwKalmanTrack(i, TfNames[i], slots[i], cfg, w, Host.TickSize, confirmBars);
            }

            Host.Trace(Id, "KALMAN_ATTACHED",
                "роль " + cfg.Role + ", ТФ: МЛ слот " + slots[0] + (tfOn[0] ? "" : " выкл")
                + ", ТК слот " + slots[1] + ", СТ слот " + slots[2] + (tfOn[2] ? "" : " выкл")
                + "; вход " + TfNames[entryTf] + ", направление " + TfNames[dirTf] + ", сопровождение " + TfNames[escortTf]
                + "; период " + cfg.Period + ", границы " + cfg.Bands + " " + cfg.BandSigma.ToString("F1") + "σ"
                + ", боковик на " + confirmBars + "-й свече");

            if (!tfOn[0])
                Host.Trace(Id, "KALMAN_TF_OFF", "младший ТФ выключен: слот " + slots[0]
                    + " не загружен или совпадает с текущим; вход берётся с текущего ТФ");
            if (!tfOn[2])
                Host.Trace(Id, "KALMAN_TF_OFF", "старший ТФ выключен: слот " + slots[2]
                    + " не загружен или совпадает с другим ТФ; направление и сопровождение — по текущему");
        }

        private bool Usable(int slot)
        {
            return slot == 0 || Host.SlotLoaded(slot);
        }

        public override void Compute(int bip)
        {
            // --- 1. Шаг каждого ТФ на закрытии бара его серии ---
            for (int i = 0; i < 3; i++)
                if (tfOn[i] && Host.IsBarCloseOf(tr[i].Slot, bip)) StepTrack(i);

            // --- 2. На закрытии младшего: предварительные старшие ТФ и исследование ---
            if (tfOn[0] && Host.IsBarCloseOf(tr[0].Slot, bip))
            {
                try { OnLowerClose(); }
                catch { }
            }

            // --- 3. Решение — только в такте серии исполнения ---
            if (bip != Host.ExecBip) return;
            ExecTick();
        }

        #region Шаг таймфрейма

        private void StepTrack(int i)
        {
            SwKalmanTrack t = tr[i];
            double z = Host.CloseOf(t.Slot, 0);
            if (double.IsNaN(z)) return;

            SwKalmanState before = t.State;
            t.Update(z, Host.HighOf(t.Slot, 0), Host.LowOf(t.Slot, 0), Host.TimeOf(t.Slot, 0));

            // Смена состояния — только для ТФ, по которым принимаются решения:
            // младший ТФ меняет состояние слишком часто для трассировки.
            if (t.Ready && t.State != before && before != SwKalmanState.Warmup && (i == dirTf || i == escortTf))
                Host.Trace(Id, "KALMAN_STATE", t.Name + ": " + SwKalmanTrack.StateText(before) + " → "
                    + SwKalmanTrack.StateText(t.State) + ", z " + t.SmoothZ.ToString("F2")
                    + ", наклон за окно " + t.SmoothTilt.ToString("F2") + " ширины, дисбаланс "
                    + t.SmoothImbalance.ToString("+0.00;-0.00;0.00"));

            if (t.RangeDeclared)
            {
                if (i == entryTf) rangesDeclared++;
                Host.Trace(Id, "KALMAN_RANGE", t.Name + ": боковик на " + t.FlatStreak + "-й свече, ширина "
                    + t.Width.ToString("F2") + (t.Compression ? ", сжатие" : ""));
            }

            // --- вход: пробой боковика ТФ входа ---
            if (i == entryTf && (t.BreakUp || t.BreakDown)) OnBreakout(t);

            // --- выход: разворот середины и коридора ТФ сопровождения ---
            if (i == escortTf) StepExit(t);

            try { ObserveTrack(i); }
            catch { }
        }

        /// <summary>
        /// Пробой границы боковика. В роли Signal взводит сигнал до ближайшего
        /// такта исполнения; в роли Filter только учитывается и рисуется.
        /// </summary>
        private void OnBreakout(SwKalmanTrack t)
        {
            int dir = t.BreakUp ? 1 : -1;
            if (dir > 0) breakoutsUp++; else breakoutsDown++;

            string sid = "KLM_" + t.Time.ToString("yyyyMMdd_HHmmss", System.Globalization.CultureInfo.InvariantCulture)
                       + "_" + t.Name + (dir > 0 ? "_U" : "_D");
            string desc = "КЛМ " + t.Name + " пробой боковика " + (dir > 0 ? "вверх" : "вниз");

            // Старший ТФ против — пробой не отправляется, но попадает в журнал.
            SwKalmanTrack h = tfOn[2] && entryTf != 2 ? tr[2] : null;
            bool hAgainst = h != null && h.Ready
                && ((dir > 0 && h.State == SwKalmanState.TrendDown) || (dir < 0 && h.State == SwKalmanState.TrendUp));
            bool hWith = h != null && h.Ready
                && ((dir > 0 && h.State == SwKalmanState.TrendUp) || (dir < 0 && h.State == SwKalmanState.TrendDown));

            double conf = 0.6 + (t.RangeCompressed ? 0.15 : 0) + (hWith ? 0.15 : 0)
                        + (Math.Sign(t.SmoothImbalance) == dir ? 0.1 : 0);
            if (conf > 1.0) conf = 1.0;

            Host.Trace(Id, "KALMAN_BREAKOUT", desc + " @ " + t.Close.ToString("F2") + ", боковик " + t.RangeBars
                + " св." + (t.RangeCompressed ? ", было сжатие" : "")
                + (h == null ? "" : ", старший ТФ " + SwKalmanTrack.StateText(h.State))
                + ", уверенность " + conf.ToString("F2"));

            if (Role != SwRole.Signal) return;

            if (cfg.RequireHigherAgreement && hAgainst)
            {
                suppressedHigher++;
                try
                {
                    Host.RequestSignalJournalEntry(Id, t.Time, sid, dir, conf, desc,
                        "старший ТФ против: " + SwKalmanTrack.StateText(h.State) + " [C22]");
                }
                catch { }
                return;
            }

            pendDir = dir;
            pendConf = conf;
            pendBarsLeft = Math.Max(1, cfg.SignalBars);
            pendLevel = dir > 0 ? t.RangeUpperAtBreak : t.RangeLowerAtBreak;
            pendId = sid;
            pendShort = desc;
            pendText = desc + " @ " + t.Close.ToString("F2") + ", уверенность " + conf.ToString("F2");
        }

        /// <summary>
        /// Против позиции на закрытии бара ТФ сопровождения: середина значимо
        /// повернула против и коридор — среднее смещение границ — тоже.
        /// Подтверждение [C27] баров подряд; затем требование закрытия повторяется
        /// каждый такт, пока позиция открыта: исполнитель пропускает не больше
        /// одного выхода на бар и мог отклонить первое.
        /// </summary>
        private void StepExit(SwKalmanTrack t)
        {
            int pos = Host.PositionSide;
            if (!cfg.ExitOnSlopeFlip || pos == 0 || !t.Ready)
            {
                exitAgainst = 0;
                return;
            }

            bool against = pos > 0
                ? t.SmoothZ <= -t.ZFlat && t.CorridorTilt < 0
                : t.SmoothZ >= t.ZFlat && t.CorridorTilt > 0;

            if (against)
            {
                exitAgainst++;
                if (exitAgainst >= Math.Max(1, cfg.ExitConfirmBars) && !exitArmed)
                {
                    exitArmed = true;
                    exitReason = "Кальман " + t.Name + ": середина и коридор против позиции "
                               + exitAgainst + " бар., z " + t.SmoothZ.ToString("F2");
                    Host.Trace(Id, "KALMAN_EXIT_REQUESTED", exitReason);
                }
            }
            else
            {
                exitAgainst = 0;
                exitArmed = false;
            }
        }

        #endregion

        #region Такт исполнения

        private void ExecTick()
        {
            int bar = Host.CurrentBar;
            int pos = Host.PositionSide;

            // Новая сделка — счёт выхода заново: разворот, накопленный в прошлой
            // сделке, не должен закрыть новую.
            double avg = Host.PositionAvgPrice;
            bool newTrade = pos != lastPosSide
                || (pos != 0 && !double.IsNaN(lastAvg) && Math.Abs(avg - lastAvg) > 1e-9);
            if (newTrade)
            {
                exitAgainst = 0;
                exitArmed = false;
            }
            lastPosSide = pos;
            lastAvg = pos != 0 ? avg : double.NaN;

            if (pos != 0)
            {
                EmitEscort(bar, pos);
                if (exitArmed) EmitExit(bar);
            }

            if (Role == SwRole.Signal) EmitSignalRole(bar);
            else EmitFilterRole(bar);
        }

        private void EmitEscort(int bar, int pos)
        {
            if (!cfg.EscortStop) return;
            SwKalmanTrack t = tr[escortTf];
            if (!t.Ready) return;

            double off = Math.Max(0, cfg.StopOffsetTicks) * Host.TickSize;
            if (cfg.ExpEscortOffsetSigma > 0 && !double.IsNaN(t.Upper) && !double.IsNaN(t.Lower))        // Э72
                off = cfg.ExpEscortOffsetSigma * Math.Max(Host.TickSize, 0.5 * (t.Upper - t.Lower));
            double stop = pos > 0 ? t.Lower - off : t.Upper + off;
            double px = Host.Close(0);
            if (double.IsNaN(stop) || double.IsNaN(px)) return;

            // Только по защитную сторону: иначе перенос увеличил бы риск.
            if (pos > 0 ? stop >= px : stop <= px) return;

            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.MoveStop;
            a.Price = stop;
            a.Reason = "Кальман " + t.Name + ": стоп по " + (pos > 0 ? "нижней" : "верхней") + " границе канала "
                     + stop.ToString("F2");
            stopRequests++;
            Emit(a);
        }

        private void EmitExit(int bar)
        {
            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.FullExit;
            a.Reason = exitReason;
            Emit(a);
            exitsRequested++;

            try
            {
                if (exitShownBar != bar && Host.VisualOn(SwVisualRole.KalmanExit))
                {
                    exitShownBar = bar;
                    int pos = Host.PositionSide;
                    double y = pos > 0 ? Host.High(0) + 5 * Host.TickSize : Host.Low(0) - 5 * Host.TickSize;
                    EmitVisual(SwVisualEvent.Label(SwVisualRole.KalmanExit, Host.Time, y, "KLM EXIT")
                        .WithVariant(-pos));
                }
            }
            catch { }
        }

        private void EmitSignalRole(int bar)
        {
            SwKalmanTrack t = tr[entryTf];
            if (!t.Ready)
            {
                Health = SwHealth.Warmup;
                Emit(SwSignal.Empty(bar));
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;

            if (pendBarsLeft > 0 && cfg.ExpNisGate > 0 && !double.IsNaN(t.NisAvg) && t.NisAvg > cfg.ExpNisGate)
            {
                nisGated++;                       // Э21: сигнал при несогласованном фильтре снимается
                pendBarsLeft = 0;
            }

            if (pendBarsLeft > 0)
            {
                pendBarsLeft--;

                // Цена успела вернуться в диапазон — пробой не подтверждён.
                double px = t.Close;
                bool inside = !double.IsNaN(pendLevel) && (pendDir > 0 ? px <= pendLevel : px >= pendLevel);
                if (inside)
                {
                    cancelledInside++;
                    try { Host.RequestSignalJournalEntry(Id, t.Time, pendId, pendDir, pendConf, pendShort, "цена вернулась в диапазон"); }
                    catch { }
                    pendBarsLeft = 0;
                    s.Direction = 0;
                    s.Text = "пробой отменён: цена в диапазоне";
                }
                else
                {
                    s.Direction = pendDir;
                    s.Confidence = pendConf;
                    s.SignalId = pendId;
                    s.ShortDescription = pendShort;
                    s.Text = pendText;
                }
            }
            else
            {
                s.Direction = 0;
                s.Text = StatusShort();
            }

            Health = SwHealth.Ok;
            Emit(s);
        }

        private int nisGated;   // Э21
        private SwKalmanState expLastState; private int expStateBars;   // Э49

        private void EmitFilterRole(int bar)
        {
            SwKalmanTrack t = tr[dirTf];
            if (!t.Ready)
            {
                Health = SwHealth.Warmup;
                SwSignal w = SwSignal.Empty(bar);
                w.Text = "прогрев " + t.Name + ": " + t.BarsSeen + " баров";
                Emit(w);
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.Direction = 0;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;

            // Уверенность — качество тренда: чем круче наклон относительно
            // ширины канала, тем спокойнее тренд.
            s.Confidence = Math.Min(1.0, Math.Abs(t.SmoothTilt) / Math.Max(0.01, 2 * cfg.TrendTilt));

            bool trend = t.State == SwKalmanState.TrendUp || t.State == SwKalmanState.TrendDown;
            int dir = t.State == SwKalmanState.TrendUp ? 1 : t.State == SwKalmanState.TrendDown ? -1 : 0;

            switch (cfg.Policy)
            {
                case SwKalmanFilterPolicy.TrendOnly:
                    if (trend) { s.Verdict = SwVerdict.Agree; s.Direction = dir; }
                    else s.Verdict = SwVerdict.Veto;
                    break;
                case SwKalmanFilterPolicy.NoRange:
                    s.Verdict = t.State == SwKalmanState.Range ? SwVerdict.Veto : SwVerdict.Neutral;
                    break;
                default:   // TrendDirection
                    if (trend) { s.Verdict = SwVerdict.Agree; s.Direction = dir; }
                    else s.Verdict = SwVerdict.Neutral;
                    break;
            }

            if (t.State != expLastState) { expLastState = t.State; expStateBars = 0; } else expStateBars++;
            if (cfg.ExpMinStateBars > 0 && expStateBars < cfg.ExpMinStateBars)
            {
                // Э49: новое состояние ещё не выдержано — мнение не выдаётся.
                s.Verdict = SwVerdict.Neutral; s.Direction = 0;
                s.Text = t.Name + ": состояние " + SwKalmanTrack.StateText(t.State) + " держится " + expStateBars + " из " + cfg.ExpMinStateBars;
                Health = SwHealth.Ok;
                Emit(s);
                return;
            }

            if (cfg.ExpNisGate > 0 && !double.IsNaN(t.NisAvg) && t.NisAvg > cfg.ExpNisGate)
            {
                // Э21: модель фильтра не согласуется с данными — мнение канала снимается.
                nisGated++;
                s.Verdict = SwVerdict.Neutral; s.Direction = 0;
                s.Text = t.Name + ": фильтр не согласован с данными, ν²/f " + t.NisAvg.ToString("F1");
                Health = SwHealth.Ok;
                Emit(s);
                return;
            }

            s.Text = (s.Verdict == SwVerdict.Veto ? "вето: " : "") + t.Name + " " + SwKalmanTrack.StateText(t.State)
                   + ", z " + t.SmoothZ.ToString("F1") + ", наклон " + t.SmoothTilt.ToString("F2")
                   + ", дисб " + t.SmoothImbalance.ToString("+0.00;-0.00;0.00");

            Health = SwHealth.Ok;
            Emit(s);
        }

        private void AddEval(int type, int dir, double start, double norm, int nowBar)
        {
            if (dir == 0 || double.IsNaN(start) || !(norm > 0)) return;
            if (pend.Count >= 20000) return;   // предохранитель памяти
            PendingEval e = new PendingEval();
            e.Type = type; e.Dir = dir; e.Start = start; e.Norm = norm;
            e.Due = nowBar + Math.Max(1, cfg.ResearchHorizon);
            pend.Add(e);
        }

        /// <summary>
        /// Бар ТФ входа: закрываются созревшие оценки, проверяются боковики,
        /// регистрируются события цены и линий. Будущие бары используются только
        /// для статистики отчёта.
        /// </summary>
        private void ResearchEntryBar(SwKalmanTrack t)
        {
            int now = t.BarsSeen;
            double z = t.Close;

            for (int k = pend.Count - 1; k >= 0; k--)
            {
                PendingEval e = pend[k];
                if (e.Due > now) continue;
                stats[e.Type].Add(e.Dir * (z - e.Start) / e.Norm);
                stDirSum[e.Type] += e.Dir;
                pend.RemoveAt(k);
            }

            // --- удержание боковика: цена не закрылась за границами m-й свечи ---
            for (int k = checks.Count - 1; k >= 0; k--)
            {
                RangeCheck c = checks[k];
                if (z > c.Up || z < c.Lo) { checks.RemoveAt(k); continue; }
                if (now >= c.Due) { held[c.M]++; checks.RemoveAt(k); }
            }
            if (t.FlatStreak >= 1 && t.FlatStreak <= MaxM && checks.Count < 5000)
            {
                reached[t.FlatStreak]++;
                RangeCheck c = new RangeCheck();
                c.M = t.FlatStreak; c.Up = t.Upper; c.Lo = t.Lower;
                c.Due = now + Math.Max(1, cfg.RangeHoldBars);
                checks.Add(c);
            }

            // --- события цены и линий ---
            double norm = Math.Max(t.Width, 2 * Host.TickSize);
            AddEval(0, 1, z, norm, now);
            if (t.CrossMidUp) AddEval(1, 1, z, norm, now);
            if (t.CrossMidDown) AddEval(2, -1, z, norm, now);
            if (t.TouchUpper) AddEval(3, -1, z, norm, now);
            if (t.TouchLower) AddEval(4, 1, z, norm, now);
            if (t.CloseAbove && !t.BreakUp && t.State != SwKalmanState.Range) AddEval(5, 1, z, norm, now);
            if (t.CloseBelow && !t.BreakDown && t.State != SwKalmanState.Range) AddEval(6, -1, z, norm, now);
            if (t.BreakUp) AddEval(7, 1, z, norm, now);
            if (t.BreakDown) AddEval(8, -1, z, norm, now);
            if (t.CompressionStarted) AddEval(9, Math.Sign(t.SmoothImbalance), z, norm, now);
            if (t.ExpansionStarted) AddEval(10, Math.Sign(t.Slope), z, norm, now);
        }

        /// <summary>
        /// Закрытие бара младшего ТФ: предварительные состояния текущего и
        /// старшего ТФ для окраски и события пары «младший — старший».
        /// </summary>
        private void OnLowerClose()
        {
            SwKalmanTrack l = tr[0];
            double zL = l.Close;
            if (double.IsNaN(zL)) return;

            bool hValid = false;
            double hMid = double.NaN, hUp = double.NaN, hLo = double.NaN;
            int hVar = 0;

            for (int i = 1; i < 3; i++)
            {
                if (!tfOn[i] || !tr[i].Ready) continue;
                double lv, up, lo;
                int pv;
                if (!tr[i].Peek(zL, out lv, out up, out lo, out pv)) continue;
                if (i == 2) { hMid = lv; hUp = up; hLo = lo; hVar = pv; hValid = true; }
                if (cfg.ProvisionalDisplay)
                {
                    Host.SetChannelPlot(i, 0, lv, pv);
                    Host.SetChannelPlot(i, 1, up, tr[i].UpperVariant);
                    Host.SetChannelPlot(i, 2, lo, tr[i].LowerVariant);
                }
            }

            if (!cfg.Research || !l.Ready || !hValid || entryTf != 0)
            {
                prevPairValid = false;
                return;
            }

            int now = l.BarsSeen;
            double norm = Math.Max(l.Width, 2 * Host.TickSize);
            bool vis = Host.VisualOn(SwVisualRole.KalmanInteraction) && Host.VisualDebugOn;

            if (prevPairValid)
            {
                if (pLMid <= pHMid && l.Level > hMid) { AddEval(11, 1, zL, norm, now); Mark(vis, l, l.Level, 1, false); }
                if (pLMid >= pHMid && l.Level < hMid) { AddEval(12, -1, zL, norm, now); Mark(vis, l, l.Level, -1, false); }
                if (pLUp <= pHUp && l.Upper > hUp) { AddEval(13, 1, zL, norm, now); Mark(vis, l, l.Upper, 1, true); }
                if (pLLo >= pHLo && l.Lower < hLo) { AddEval(14, -1, zL, norm, now); Mark(vis, l, l.Lower, -1, true); }
                if (pLMid <= pHUp && l.Level > hUp) { AddEval(15, 1, zL, norm, now); Mark(vis, l, l.Level, 1, true); }
                if (pLMid >= pHLo && l.Level < hLo) { AddEval(16, -1, zL, norm, now); Mark(vis, l, l.Level, -1, true); }

                if (l.MidVariant != pLVar && l.MidVariant != 0 && hVar != 0)
                {
                    if (l.MidVariant == hVar) AddEval(17, hVar, zL, norm, now);
                    else AddEval(18, l.MidVariant, zL, norm, now);
                }

                SwKalmanState hs = tr[2].State;
                if (l.RangeDeclared && (hs == SwKalmanState.TrendUp || hs == SwKalmanState.TrendDown))
                    AddEval(19, hs == SwKalmanState.TrendUp ? 1 : -1, zL, norm, now);
            }

            pLMid = l.Level; pLUp = l.Upper; pLLo = l.Lower;
            pHMid = hMid; pHUp = hUp; pHLo = hLo;
            pLVar = l.MidVariant;
            prevPairValid = true;
        }

                    SwVisualRole.KalmanInteraction, l.Time, price)
                .WithVariant(dir).AsDebug().InRing("TFX"));
        }

        public string ResearchReport(string pad)
        {
            StringBuilder sb = new StringBuilder();
            sb.AppendLine(pad + "Кальман: боковиков на ТФ входа " + rangesDeclared + ", пробоев ↑" + breakoutsUp + " ↓" + breakoutsDown
                + ", не отправлено — старший ТФ против " + suppressedHigher + ", возврат в диапазон " + cancelledInside
                + "; требований выхода " + exitsRequested + ", просьб о стопе " + stopRequests);
            if (!cfg.Research) return sb.ToString();

            SwKalmanStat b = stats[0];
            double baseMean = b.Mean;
            sb.AppendLine(pad + "Кальман, исследование: ТФ входа " + tr[entryTf].Name + ", горизонт " + Math.Max(1, cfg.ResearchHorizon)
                + " баров, доходность в ширинах канала ТФ входа, знак — в сторону события. События пары МЛ–СТ — только при обоих ТФ.");
            sb.AppendLine(pad + "  событие; n; средняя; ст. ошибка; в сторону, %; средняя минус база");
            for (int i = 0; i < stats.Length; i++)
            {
                SwKalmanStat s = stats[i];
                if (s.N == 0) continue;
                double avgDir = s.N > 0 ? (double)stDirSum[i] / Math.Max(1, s.N) : 0;
                double excess = i == 0 || double.IsNaN(baseMean) ? double.NaN : s.Mean - avgDir * baseMean;
                sb.AppendLine(pad + "  " + EventNames[i] + "; " + s.N + "; " + F3(s.Mean) + "; " + F3(s.StdErr) + "; "
                    + (double.IsNaN(s.HitRate) ? "—" : (100 * s.HitRate).ToString("F1")) + "; " + F3(excess));
            }

            sb.AppendLine(pad + "Кальман, боковик ТФ входа: m-я плоская свеча подряд; достигнуто; удержалось в канале "
                + Math.Max(1, cfg.RangeHoldBars) + " баров; доля, %");
            for (int m = 1; m <= MaxM; m++)
            {
                if (reached[m] == 0) continue;
                sb.AppendLine(pad + "  m=" + m + "; " + reached[m] + "; " + held[m] + "; "
                    + (100.0 * held[m] / reached[m]).ToString("F1"));
            }
            sb.AppendLine(pad + "  " + SuggestM());
            return sb.ToString();
        }

        /// <summary>
        /// Наименьшее m, после которого доля удержания растёт меньше чем на два
        /// пункта за свечу два шага подряд: дальше ждать — терять время без
        /// выигрыша в надёжности. Считается только по m, где случаев не меньше 30.
        /// </summary>
        private string SuggestM()
        {
            double[] rate = new double[MaxM + 1];
            for (int m = 1; m <= MaxM; m++) rate[m] = reached[m] >= 30 ? (double)held[m] / reached[m] : double.NaN;
            for (int m = 1; m <= MaxM - 2; m++)
            {
                if (double.IsNaN(rate[m]) || double.IsNaN(rate[m + 1]) || double.IsNaN(rate[m + 2])) continue;
                if (rate[m + 1] - rate[m] < 0.02 && rate[m + 2] - rate[m + 1] < 0.02)
                    return "по этим данным боковик разумно объявлять на " + m + "-й свече [C15]; сейчас — на "
                         + confirmBars + "-й. Гипотеза: проверьте на другом участке истории";
            }
            return "данных для выбора m мало: нужно не меньше 30 случаев на каждое m";
        }

        private static string F3(double v)
        {
            return double.IsNaN(v) ? "—" : v.ToString("+0.000;-0.000;0.000");
        }

                    SwKalmanTrack t = tr[i];
                    sb.Append(' ').Append(t.Name).Append(' ').Append(SwKalmanTrack.StateText(t.State));
                    if (t.Ready && t.State != SwKalmanState.Range && t.FlatStreak > 0)
                        sb.Append(" (плоско ").Append(t.FlatStreak).Append('/').Append(t.ConfirmBars).Append(')');
                    if (t.Compression) sb.Append(" сжатие");
                    else if (t.Expansion) sb.Append(" расширение");
                    sb.Append(" ·");

                SwKalmanTrack d = tr[dirTf];

                    sb.Append(" дисб ").Append(d.SmoothImbalance.ToString("+0.00;-0.00;0.00"))
                      .Append(" · окна ").Append(tr[entryTf].Window.Length).Append('/').Append(tr[escortTf].Window.Length);
                return sb.ToString();
            }
        }

                 + (t.State != SwKalmanState.Range && t.FlatStreak > 0 ? " плоско " + t.FlatStreak + "/" + t.ConfirmBars : "");

//                          внутри тика; смещение от опоры в долях плеча — «m»;

//                          кластеров на их плечи: Σ избыток·плечо / Σ избыток·|плечо|;
//    баланс свечи        — [B09]: среднее m и T, только T или только m;
//    сглаживание         — взвешенная медиана балансов последних [B10] свечей,
//                          веса — масса свечей: тяжёлая свеча весит больше;
//    масштаб             — 1.4826·MAD баланса за [B11] свечей; ноль не
//                          сдвигается: равновесие рычага — естественный ноль;
//    тренд для контекста — наклон Тейла–Сена (медиана попарных наклонов) по
//                          закрытиям [B13] свечей;
//    нормировка доходностей в исследовании — медианный размах свечи.
//  Голос: z = сглаженный баланс / масштаб; z ≥ [B05] — вверх, ≤ −[B05] —
//  вниз, иначе — середина. На случайном блуждании при пороге 0.5 это примерно
//  четверть свечей вверх, четверть вниз, половина середина (прототип Python).
//
//  КОГДА ГОЛОС ГОТОВ. Профиль свечи должен быть полон. Подбар, закрывшийся в
//  ту же секунду, что и свеча анализа, приходит в очереди тактов ПОСЛЕ неё.
//  Поэтому [B18]:
//    Exact — дождаться поздних кластеров; голос — в следующем такте исполнения;
//    Fast  — голос по закрытию свечи; кластеры с меткой времени её конца
//            отбрасываются (для тиков это доли процента объёма);
//    Auto  — Fast для тикового источника (подбары без высоты), иначе Exact.
//
//  ЧЕГО НЕТ. Взгляда в будущее: голос опирается только на кластеры закрытой
//  свечи и прошлые балансы. Будущие свечи — только в статистике отчёта.
//  Гипотеза «масса выше опоры → продолжение вверх» проверяется блоком
//  исследования [B19] на ваших данных: таблица доходностей по корзинам
//  рычага и медианного центра должна показать монотонность, если связь есть.
// ============================================================================

    /// <summary>Роль модуля рычага в решении.</summary>
    public enum SwLeverRole
    {
        Filter,     // вердикт по текущему голосу
        Signal      // новый голос вверх или вниз
    }

    /// <summary>Чем взвешивается кластер.</summary>
    public enum SwLeverWeight
    {
        Volume,     // объём

    }

    public enum SwLeverFulcrum
    {
        RangeMid,   // середина между максимумом и минимумом
        BodyMid     // середина тела
    }

    /// <summary>Какой мерой голосует модуль.</summary>
    public enum SwLeverMeasure
    {
        Both,           // среднее рычага богатых кластеров и медианного центра
        Lever,          // только рычаг богатых кластеров
        MedianCenter    // только медианный центр массы
    }

    /// <summary>Вердикт роли Filter в середине.</summary>
    public enum SwLeverPolicy
    {
        DirectionOnly,  // середина не мешает
        BlockMiddle     // середина — вето: движение затухает
    }

    /// <summary>Когда профиль свечи считается полным.</summary>
    public enum SwLeverFinalize
    {
        Auto,
        Exact,
        Fast
    }

    /// <summary>Голос рычага.</summary>
    public enum SwLeverState
    {
        Warmup,
        Up,
        Down,
        Middle
    }

    /// <summary>Накопитель статистики события: число, средняя, доля в сторону события.</summary>
    public class SwLeverStat
    {
        public int N, Hits;
        public double Sum, Sum2;

        public void Add(double v)
        {
            if (double.IsNaN(v) || double.IsInfinity(v)) return;
            N++; Sum += v; Sum2 += v * v;
            if (v > 0) Hits++;
        }

        public double Mean { get { return N > 0 ? Sum / N : double.NaN; } }
        public double HitRate { get { return N > 0 ? (double)Hits / N : double.NaN; } }

        public double StdErr
        {
            get
            {
                if (N < 2) return double.NaN;
                double m = Sum / N;
                double v = (Sum2 - N * m * m) / (N - 1);
                return v > 0 ? Math.Sqrt(v / N) : 0;
            }
        }
    }

        public SwLeverModule(string id, SwLeverConfig config)

            Id = id;

        public override void Attach(SwHostView host)
        {
            base.Attach(host);
            tick = Host.TickSize > 0 ? Host.TickSize : 1e-6;
            cap = Math.Max(Math.Max(cfg.ScaleWindow, cfg.TrendWindow), Math.Max(cfg.SmoothCandles, 10)) + 4;

            // Кластеры должны идти из другой серии: невключённый слот пула читался
            // бы с графика, и свеча «взвешивала» бы сама себя.
            bool loaded = cfg.ClusterSlot == 0 || Host.SlotLoaded(cfg.ClusterSlot);
            srcOk = loaded && Host.Bip(cfg.ClusterSlot) != Host.Bip(cfg.AnalysisSlot);

            Host.Trace(Id, "LEVER_ATTACHED",
                "роль " + cfg.Role + ", свечи — слот " + cfg.AnalysisSlot + ", кластеры — слот " + cfg.ClusterSlot
                + (srcOk ? "" : " (НЕДОСТУПЕН)") + ", вес " + cfg.Weight + ", кластеров " + cfg.Clusters
                + ", опора " + cfg.Fulcrum + ", мера " + cfg.Measure + ", сглаживание " + cfg.SmoothCandles
                + ", масштаб " + cfg.ScaleWindow + ", порог z " + cfg.VoteZ.ToString("F2") + ", профиль " + cfg.Finalize);

            if (!srcOk)
                Host.Trace(Id, "LEVER_NO_SOURCE", "слот кластеров " + cfg.ClusterSlot
                    + " не загружен или совпадает с серией свечей: взвешивать нечем, модуль не готов");
        }

        public override void Compute(int bip)
        {
            if (srcOk)
            {
                if (Host.IsBarCloseOf(cfg.ClusterSlot, bip)) OnSubBar();
                if (Host.IsBarCloseOf(cfg.AnalysisSlot, bip)) OnAnalysisClose();
            }

            if (bip != Host.ExecBip) return;
            ExecTick();
        }

        #region Профиль свечи

        private bool FastMode()
        {
            if (cfg.Finalize == SwLeverFinalize.Fast) return true;
            if (cfg.Finalize == SwLeverFinalize.Exact) return false;
            // Auto: тиковый источник — подбары без высоты.
            bool fast = subSeen >= 200 && subPoint >= 0.95 * subSeen;
            if (!modeTraced && subSeen >= 200)
            {
                modeTraced = true;
                Host.Trace(Id, "LEVER_MODE", fast
                    ? "источник тиковый: профиль по закрытию свечи (Fast)"
                    : "источник с высотой подбаров: ожидание поздних кластеров (Exact), голос на такт позже");
            }
            return fast;
        }

        private bool TickLike()
        {
            return subSeen >= 50 && subPoint >= 0.95 * subSeen;
        }

        private void OnSubBar()
        {
            DateTime t = Host.TimeOf(cfg.ClusterSlot, 0);
            double h = Host.HighOf(cfg.ClusterSlot, 0);
            double l = Host.LowOf(cfg.ClusterSlot, 0);
            double v = Host.VolumeOf(cfg.ClusterSlot, 0);
            if (t == DateTime.MinValue || double.IsNaN(h) || double.IsNaN(l)) return;

            subSeen++;
            if (Math.Abs(h - l) < tick * 0.5) subPoint++;
            lastSubTime = t;

            if (pending)
            {
                if (t <= pendSnap.End)
                {
                    Accumulate(pendProf, h, l, v);
                    // У источника с высотой подбаров (минуты) в момент конца свечи
                    // кончается ровно один подбар: после него профиль полон. У тиков
                    // метку конца могут нести несколько сделок — ждём следующую.
                    if (t == pendSnap.End && !TickLike()) FinalizePending();
                    return;
                }
                // Первый подбар после конца свечи: её профиль полон.
                FinalizePending();
            }

            if (lastClosedEnd != DateTime.MinValue && t <= lastClosedEnd)
            {
                lateDropped++;
                return;
            }
            Accumulate(curProf, h, l, v);
        }

        /// <summary>
        /// Вес подбара — поровну на все цены, которые он прошёл. У тика высота
        /// нулевая: весь вес на одну цену.
        /// </summary>
        private void Accumulate(Dictionary<int, double> prof, double h, double l, double v)
        {
            int kl = (int)Math.Round(l / tick), kh = (int)Math.Round(h / tick);
            if (kh < kl) { int x = kl; kl = kh; kh = x; }
            int n = kh - kl + 1;
            if (n <= 0 || n > 5000) return;

            double w;
            if (cfg.Weight == SwLeverWeight.Volume)
            {
                if (!(v > 0)) return;
                w = v / n;
            }
            else w = 1.0;

            for (int k = kl; k <= kh; k++)
            {
                double cur;
                prof.TryGetValue(k, out cur);
                prof[k] = cur + w;
            }
        }

        private void OnAnalysisClose()
        {
            anaIndex++;

            // Запасной путь: поздний кластер так и не пришёл до следующей свечи.
            if (pending) FinalizePending();

            try { if (cfg.Research) ResolveResearch(); }
            catch { }

            SwLeverCandle c = new SwLeverCandle();
            c.Index = anaIndex;
            c.End = Host.TimeOf(cfg.AnalysisSlot, 0);
            c.Start = Host.TimeOf(cfg.AnalysisSlot, 1);
            c.Open = Host.OpenOf(cfg.AnalysisSlot, 0);
            c.High = Host.HighOf(cfg.AnalysisSlot, 0);
            c.Low = Host.LowOf(cfg.AnalysisSlot, 0);
            c.Close = Host.CloseOf(cfg.AnalysisSlot, 0);

            // Профиль закрытой свечи уходит в ожидающий, текущий начинается пустым.
            Dictionary<int, double> closed = curProf;
            curProf = pendProf;
            curProf.Clear();
            pendProf = closed;
            lastClosedEnd = c.End;

            if (firstCandle)
            {
                firstCandle = false;
                pendProf.Clear();
                return;
            }

            if (!FastMode() && lastSubTime < c.End)
            {
                pending = true;
                pendSnap = c;
                return;
            }

            FinalizeCandle(c, pendProf);
            pendProf.Clear();
        }

        private void FinalizePending()
        {
            pending = false;
            SwLeverCandle c = pendSnap;
            pendSnap = null;
            if (c != null) FinalizeCandle(c, pendProf);
            pendProf.Clear();
        }

        #endregion

        #region Взвешивание свечи

        private static double ArmOf(double p, double g, double armUp, double armDn)
        {
            double a = p >= g ? (p - g) / armUp : (p - g) / armDn;
            return a > 1 ? 1 : a < -1 ? -1 : a;
        }

        private void FinalizeCandle(SwLeverCandle c, Dictionary<int, double> prof)
        {
            if (double.IsNaN(c.High) || double.IsNaN(c.Low) || double.IsNaN(c.Close))
            {
                NoData(c, "нет цен свечи");
                return;
            }

            int kL = (int)Math.Round(c.Low / tick), kH = (int)Math.Round(c.High / tick);
            int nLev = kH - kL + 1;
            c.Levels = nLev;
            if (nLev < Math.Max(2, cfg.MinHeightTicks) || nLev > 5000)
            {
                NoData(c, "свеча ниже " + cfg.MinHeightTicks + " тиков [B17]");
                return;
            }

            double[] w = new double[nLev];
            double W = 0;
            foreach (KeyValuePair<int, double> kv in prof)
            {
                int j = kv.Key - kL;
                if (j < 0 || j >= nLev) { outsideWeight += kv.Value; continue; }
                w[j] += kv.Value;
                W += kv.Value;
            }
            if (!(W > 0))
            {
                NoData(c, "нет кластеров");
                return;
            }
            totalWeight += W;

            // --- опора и плечи ---
            double g = cfg.Fulcrum == SwLeverFulcrum.BodyMid ? (c.Open + c.Close) / 2.0 : (c.High + c.Low) / 2.0;
            double armUp = Math.Max(c.High - g, tick), armDn = Math.Max(g - c.Low, tick);

            // --- медиана массы с интерполяцией внутри тика ---
            double half = W / 2.0, cum = 0, p50 = c.High;
            for (int j = 0; j < nLev; j++)
            {
                if (w[j] > 0 && cum + w[j] >= half)
                {
                    p50 = (kL + j) * tick - tick / 2.0 + tick * (half - cum) / w[j];
                    break;
                }
                cum += w[j];
            }

            // --- кластеры по высоте ---
            int nc = Math.Max(1, Math.Min(Math.Max(2, cfg.Clusters), nLev));
            double[] dens = new double[nc], lo = new double[nc], hi = new double[nc], cen = new double[nc];
            int[] cnt = new int[nc];
            for (int b = 0; b < nc; b++)
            {
                int a0 = b * nLev / nc, a1 = (b + 1) * nLev / nc - 1;
                double mass = 0;
                for (int j = a0; j <= a1; j++) mass += w[j];
                cnt[b] = a1 - a0 + 1;
                dens[b] = mass / cnt[b];
                lo[b] = (kL + a0) * tick - tick / 2.0;
                hi[b] = (kL + a1) * tick + tick / 2.0;
                cen[b] = (kL + (a0 + a1) / 2.0) * tick;
            }
            double md = MedianOf(dens, nc);

            // --- рычаг богатых кластеров: момент избытка над медианной плотностью ---
            double num = 0, den = 0;
            bool[] rich = new bool[nc];
            for (int b = 0; b < nc; b++)
            {
                double e = Math.Max(0, dens[b] - md) * cnt[b];
                rich[b] = e > 0;
                double d = ArmOf(cen[b], g, armUp, armDn);
                num += e * d;
                den += e * Math.Abs(d);
            }

            c.HasData = true;
            c.Fulcrum = g;
            c.Center = p50;
            c.MedianOffset = ArmOf(p50, g, armUp, armDn);
            c.Lever = den > 0 ? num / den : 0;
            int jp = 0;
            for (int j = 1; j < nLev; j++) if (w[j] > w[jp]) jp = j;
            c.Poc = (kL + jp) * tick;
            c.PocOffset = ArmOf(c.Poc, g, armUp, armDn);
            c.Balance = cfg.ExpPocMeasure ? c.PocOffset     // Э22
                      : cfg.Measure == SwLeverMeasure.Lever ? c.Lever
                      : cfg.Measure == SwLeverMeasure.MedianCenter ? c.MedianOffset
                      : 0.5 * (c.Lever + c.MedianOffset);
            c.Weight = W;
            c.Clusters = nc;
            c.ClLo = lo; c.ClHi = hi; c.ClDensity = dens; c.ClRich = rich;
            c.MedianDensity = md;

            OnCandle(c);
        }

        private void NoData(SwLeverCandle c, string reason)
        {
            c.HasData = false;
            c.NoDataReason = reason;
            candlesSkipped++;
            last = c;
        }

        #region Временной ряд балансов и голос

        private static void PushCapped(List<double> l, double v, int cap)
        {
            l.Add(v);
            while (l.Count > cap) l.RemoveAt(0);
        }

        private static double MedianOf(double[] a, int n)
        {
            if (n <= 0) return double.NaN;
            double[] b = new double[n];
            Array.Copy(a, b, n);
            Array.Sort(b);
            return n % 2 == 1 ? b[n / 2] : 0.5 * (b[n / 2 - 1] + b[n / 2]);
        }

        /// <summary>Медиана последних n значений ряда.</summary>
        private static double TailMedian(List<double> l, int n)
        {
            int m = Math.Min(n, l.Count);
            if (m <= 0) return double.NaN;
            double[] a = new double[m];
            for (int i = 0; i < m; i++) a[i] = l[l.Count - m + i];
            return MedianOf(a, m);
        }

        /// <summary>Медианное абсолютное отклонение последних n значений.</summary>
        private static double TailMad(List<double> l, int n)
        {
            int m = Math.Min(n, l.Count);
            if (m <= 1) return double.NaN;
            double med = TailMedian(l, m);
            double[] a = new double[m];
            for (int i = 0; i < m; i++) a[i] = Math.Abs(l[l.Count - m + i] - med);
            return MedianOf(a, m);
        }

        /// <summary>Взвешенная медиана последних n значений: веса — масса свечей.</summary>
        private static double TailWeightedMedian(List<double> v, List<double> w, int n)
        {
            int m = Math.Min(n, Math.Min(v.Count, w.Count));
            if (m <= 0) return double.NaN;
            double[] vals = new double[m], wts = new double[m];
            double tot = 0;
            for (int i = 0; i < m; i++)
            {
                vals[i] = v[v.Count - m + i];
                wts[i] = Math.Max(0, w[w.Count - m + i]);
                tot += wts[i];
            }
            if (!(tot > 0)) return MedianOf(vals, m);
            Array.Sort(vals, wts);
            double half = tot / 2.0, cum = 0;
            for (int i = 0; i < m; i++)
            {
                cum += wts[i];
                if (cum >= half) return vals[i];
            }
            return vals[m - 1];
        }

        // Буферы Тейла–Сена: растут только при увеличении окна [B13], на свече
        // память не выделяется. Значения за пределами текущего окна не читаются.
        private double[] tsPairs = new double[0];
        private double[] tsDiffs = new double[0];

        /// <summary>
        /// Направление тренда по Тейлу–Сену: медиана попарных наклонов закрытий.
        /// Тренд засчитывается, если наклон за окно больше робастного шума
        /// одного шага, умноженного на √окна: как у блуждания за то же время.
        /// </summary>
        private int TheilSenDir(List<double> c, int n)
        {
            int m = Math.Min(Math.Max(3, n), c.Count);
            if (m < 3) return 0;
            int off = c.Count - m;
            int np = m * (m - 1) / 2, nd = m - 1;
            if (tsPairs.Length < np) tsPairs = new double[np];
            if (tsDiffs.Length < nd) tsDiffs = new double[nd];

            int k = 0;
            for (int i = 0; i < m; i++)
            {
                if (i > 0) tsDiffs[i - 1] = c[off + i] - c[off + i - 1];
                for (int j = i + 1; j < m; j++) tsPairs[k++] = (c[off + j] - c[off + i]) / (j - i);
            }

            double slope = SortedMedian(tsPairs, np);
            double med = SortedMedian(tsDiffs, nd);
            for (int i = 0; i < nd; i++) tsDiffs[i] = Math.Abs(tsDiffs[i] - med);
            double noise = 1.4826 * SortedMedian(tsDiffs, nd);

            if (double.IsNaN(slope) || double.IsNaN(noise)) return 0;
            if (Math.Abs(slope) * Math.Sqrt(m) < Math.Max(noise, 1e-12)) return 0;
            return Math.Sign(slope);
        }

        /// <summary>Медиана первых n значений массива; сортирует их на месте, без выделения памяти.</summary>
        private static double SortedMedian(double[] a, int n)
        {
            if (n <= 0) return double.NaN;
            Array.Sort(a, 0, n);
            return n % 2 == 1 ? a[n / 2] : 0.5 * (a[n / 2 - 1] + a[n / 2]);
        }

        private int WarmupCount { get { return Math.Max(10, Math.Min(cfg.ScaleWindow, 20)); } }

        private void OnCandle(SwLeverCandle c)
        {
            last = c;
            candlesWeighed++;
            PushCapped(hB, c.Balance, cap);
            PushCapped(hW, c.Weight, cap);
            PushCapped(hC, c.Close, cap);
            PushCapped(hR, c.High - c.Low, cap);

            SwLeverState prev = state;
            if (hB.Count < WarmupCount)
            {
                state = SwLeverState.Warmup;
                try { ObserveCandle(c, prev); }
                catch { }
                return;
            }

            smooth = TailWeightedMedian(hB, hW, Math.Max(1, cfg.SmoothCandles));
            double mad = TailMad(hB, Math.Max(5, cfg.ScaleWindow));
            scale = Math.Max(0.02, 1.4826 * (double.IsNaN(mad) ? 0 : mad));
            biasMedian = TailMedian(hB, Math.Max(5, cfg.ScaleWindow));
            z = smooth / scale;
            trendDir = TheilSenDir(hC, cfg.TrendWindow);

            double vz = Math.Max(0.01, cfg.VoteZ);
            state = z >= vz ? SwLeverState.Up : z <= -vz ? SwLeverState.Down : SwLeverState.Middle;

            if (state != prev)
                Host.Trace(Id, "LEVER_STATE", StateText(prev) + " → " + StateText(state) + ", баланс сглаж. "
                    + F2(smooth) + ", z " + F2(z) + ", рычаг " + F2(c.Lever) + ", медиана " + F2(c.MedianOffset)
                    + ", тренд Т–С " + (trendDir > 0 ? "↑" : trendDir < 0 ? "↓" : "нет"));

            bool newUp = state == SwLeverState.Up && prev != SwLeverState.Up;
            bool newDn = state == SwLeverState.Down && prev != SwLeverState.Down;
            bool fade = state == SwLeverState.Middle && (prev == SwLeverState.Up || prev == SwLeverState.Down);
            if (newUp) votesUp++;
            if (newDn) votesDown++;
            if (fade) fades++;

            if ((newUp || newDn) && Role == SwRole.Signal) ArmSignal(c, newUp ? 1 : -1);
            StepExit();

            try { if (cfg.Research) RegisterResearch(c, prev, newUp, newDn, fade); }
            catch { }
            try { ObserveCandle(c, prev); }
            catch { }
        }

        private void ArmSignal(SwLeverCandle c, int dir)
        {
            string sid = "LEV_" + c.End.ToString("yyyyMMdd_HHmmss", System.Globalization.CultureInfo.InvariantCulture)
                       + (dir > 0 ? "_U" : "_D");
            string desc = "РЫЧ масса " + (dir > 0 ? "выше" : "ниже") + " опоры";
            double vz = Math.Max(0.01, cfg.VoteZ);
            double conf = Math.Min(1.0, 0.5 + 0.25 * Math.Abs(z) / vz);

            if (cfg.SignalWithTrend && trendDir != dir)
            {
                suppressedTrend++;
                try
                {
                    Host.RequestSignalJournalEntry(Id, c.End, sid, dir, conf, desc,
                        "тренд Тейла–Сена " + (trendDir > 0 ? "вверх" : trendDir < 0 ? "вниз" : "не выражен") + " [B12]");
                }
                catch { }
                return;
            }

            pendDir = dir;
            pendConf = conf;
            pendBarsLeft = Math.Max(1, cfg.SignalBars);
            pendId = sid;
            pendShort = desc;
            pendText = desc + ", z " + F2(z) + ", рычаг " + F2(c.Lever) + ", медиана " + F2(c.MedianOffset);
            Host.Trace(Id, "LEVER_VOTE", pendText + ", уверенность " + conf.ToString("F2"));
        }

        /// <summary>
        /// Против позиции на закрытой свече: голос на противоположной стороне
        /// [B16] свечей подряд. Требование повторяется каждый такт, пока позиция
        /// открыта: исполнитель пропускает не больше одного выхода на бар.
        /// </summary>
        private void StepExit()
        {
            int pos = Host.PositionSide;
            if (!cfg.ExitAgainst || pos == 0)
            {
                exitAgainst = 0;
                return;
            }
            bool against = (pos > 0 && state == SwLeverState.Down) || (pos < 0 && state == SwLeverState.Up);
            if (against)
            {
                exitAgainst++;
                if (exitAgainst >= Math.Max(1, cfg.ExitConfirm) && !exitArmed)
                {
                    exitArmed = true;
                    exitReason = "Рычаг: масса " + (pos > 0 ? "ниже" : "выше") + " опоры " + exitAgainst + " св. подряд, z " + F2(z);
                    Host.Trace(Id, "LEVER_EXIT_REQUESTED", exitReason);
                }
            }
            else
            {
                exitAgainst = 0;
                exitArmed = false;
            }
        }

        private void ExecTick()
        {
            int bar = Host.CurrentBar;
            int pos = Host.PositionSide;

            double avg = Host.PositionAvgPrice;
            bool newTrade = pos != lastPosSide
                || (pos != 0 && !double.IsNaN(lastAvg) && Math.Abs(avg - lastAvg) > 1e-9);
            if (newTrade)
            {
                exitAgainst = 0;
                exitArmed = false;
            }
            lastPosSide = pos;
            lastAvg = pos != 0 ? avg : double.NaN;

            if (pos != 0 && exitArmed) EmitExit(bar, pos);

            if (Role == SwRole.Signal) EmitSignalRole(bar);
            else EmitFilterRole(bar);
        }

        private void EmitExit(int bar, int pos)
        {
            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.FullExit;
            a.Reason = exitReason;
            Emit(a);
            exitsRequested++;

            try
            {
                if (exitShownBar != bar && Host.VisualOn(SwVisualRole.LeverVote))
                {
                    exitShownBar = bar;
                    double y = pos > 0 ? Host.High(0) + 5 * Host.TickSize : Host.Low(0) - 5 * Host.TickSize;
                    EmitVisual(SwVisualEvent.Label(SwVisualRole.LeverVote, Host.Time, y, "LEV EXIT").WithVariant(-pos));
                }
            }
            catch { }
        }

        private bool Ready { get { return srcOk && state != SwLeverState.Warmup; } }

        private void EmitSignalRole(int bar)
        {
            if (!Ready)
            {
                Health = SwHealth.Warmup;
                SwSignal w = SwSignal.Empty(bar);
                w.Text = WarmupText();
                Emit(w);
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;

            if (pendBarsLeft > 0)
            {
                pendBarsLeft--;
                s.Direction = pendDir;
                s.Confidence = pendConf;
                s.SignalId = pendId;
                s.ShortDescription = pendShort;
                s.Text = pendText;
            }
            else
            {
                s.Direction = 0;
                s.Text = StatusShort();
            }

            Health = SwHealth.Ok;
            Emit(s);
        }

        private void EmitFilterRole(int bar)
        {
            if (!Ready)
            {
                Health = SwHealth.Warmup;
                SwSignal w = SwSignal.Empty(bar);
                w.Text = WarmupText();
                Emit(w);
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.Direction = 0;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;
            s.Confidence = Math.Min(1.0, Math.Abs(z) / Math.Max(0.02, 2 * cfg.VoteZ));

            switch (state)
            {
                case SwLeverState.Up:
                    s.Verdict = SwVerdict.Agree; s.Direction = 1;
                    break;
                case SwLeverState.Down:
                    s.Verdict = SwVerdict.Agree; s.Direction = -1;
                    break;
                default:
                    s.Verdict = cfg.Policy == SwLeverPolicy.BlockMiddle ? SwVerdict.Veto : SwVerdict.Neutral;
                    break;
            }

            string gate = string.Empty;
            if (s.Verdict == SwVerdict.Agree && last != null)                                   // Э69
            {
                if (cfg.ExpClrMin > 0)
                {
                    double rng = last.High - last.Low;
                    double clr = rng > 0 ? Math.Abs(last.Close - last.Open) / rng : 0.0;
                    if (clr < cfg.ExpClrMin) { s.Verdict = SwVerdict.Neutral; s.Direction = 0; gate = "CLR " + clr.ToString("F2") + " ниже порога; "; }
                }
                if (cfg.ExpVolZMin > 0 && s.Verdict == SwVerdict.Agree && hW.Count >= 5)
                {
                    int n = Math.Min(20, hW.Count); double m = 0, v = 0;
                    for (int i = hW.Count - n; i < hW.Count; i++) m += hW[i];
                    m /= n;
                    for (int i = hW.Count - n; i < hW.Count; i++) v += (hW[i] - m) * (hW[i] - m);
                    double sd = Math.Sqrt(v / Math.Max(1, n - 1));
                    double zv = sd > 0 ? (last.Weight - m) / sd : 0.0;
                    if (zv < cfg.ExpVolZMin) { s.Verdict = SwVerdict.Neutral; s.Direction = 0; gate = "z объёма " + zv.ToString("F2") + " ниже порога; "; }
                }
            }
            s.Text = (s.Verdict == SwVerdict.Veto ? "вето: середина, движение затухает; " : "") + gate + StatusShort();
            Health = SwHealth.Ok;
            Emit(s);
        }

        {

            return "прогрев: свечей со взвешиванием " + hB.Count + " из " + WarmupCount;
        }

        private static SwLeverStat[] CreateStats(int n)

            SwLeverStat[] a = new SwLeverStat[n];

        private static int Bin(double x)
        {
            return x < -0.6 ? 0 : x < -0.2 ? 1 : x <= 0.2 ? 2 : x <= 0.6 ? 3 : 4;
        }

        private void AddEval(int type, int dir, double start, double norm, long due)
        {
            if (double.IsNaN(start) || !(norm > 0) || pend.Count >= 20000) return;
            PendingEval e = new PendingEval();
            e.Type = type; e.Dir = dir; e.Start = start; e.Norm = norm; e.Due = due;
            pend.Add(e);
        }

        /// <summary>
        /// Свеча взвешена: регистрируются оценки. Доходность нормируется на
        /// медианный размах свечи — робастная мера типичной свечи участка.
        /// </summary>
        private void RegisterResearch(SwLeverCandle c, SwLeverState prev, bool newUp, bool newDn, bool fade)
        {
            double norm = Math.Max(TailMedian(hR, Math.Max(5, cfg.ScaleWindow)), 2 * tick);
            long due = c.Index + Math.Max(1, cfg.ResearchHorizon);
            double p = c.Close;

            AddEval(0, 1, p, norm, due);
            AddEval(100, 0, p, norm, due);   // база модуля хода

            if (state == SwLeverState.Up) AddEval(1, 1, p, norm, due);
            if (state == SwLeverState.Down) AddEval(2, -1, p, norm, due);
            if (newUp) AddEval(3, 1, p, norm, due);
            if (newDn) AddEval(4, -1, p, norm, due);
            if (fade) AddEval(prev == SwLeverState.Up ? 5 : 6, prev == SwLeverState.Up ? -1 : 1, p, norm, due);

            if (state == SwLeverState.Up && trendDir != 0) AddEval(trendDir > 0 ? 7 : 8, 1, p, norm, due);
            if (state == SwLeverState.Down && trendDir != 0) AddEval(trendDir < 0 ? 9 : 10, -1, p, norm, due);

            AddEval(11 + Bin(c.Lever), 1, p, norm, due);
            AddEval(16 + Bin(c.MedianOffset), 1, p, norm, due);

            if (state == SwLeverState.Middle) AddEval(101, 0, p, norm, due);
            if (state == SwLeverState.Up || state == SwLeverState.Down) AddEval(102, 0, p, norm, due);
        }

        /// <summary>Закрытие свечи анализа: созревшие оценки получают результат.</summary>
        private void ResolveResearch()
        {
            double p = Host.CloseOf(cfg.AnalysisSlot, 0);
            if (double.IsNaN(p)) return;
            for (int k = pend.Count - 1; k >= 0; k--)
            {
                PendingEval e = pend[k];
                if (e.Due > anaIndex) continue;
                double r = (p - e.Start) / e.Norm;
                if (e.Type >= 100)
                {
                    int mi = e.Type - 100;
                    if (mi >= 0 && mi < mags.Length) mags[mi].Add(Math.Abs(r));
                }
                else if (e.Type >= 0 && e.Type < stats.Length)
                {
                    stats[e.Type].Add(e.Dir * r);
                    stDirSum[e.Type] += e.Dir;
                }
                pend.RemoveAt(k);
            }
        }

        /// <summary>
        /// Центр массы — отрезок на уровне медианы массы через всю свечу, цвет —
        /// сторона перевеса этой свечи. Кластеры — только у последней взвешенной
        /// свечи: прямоугольники, непрозрачность по плотности. Опора — отладка.
        /// </summary>
        private void ObserveCandle(SwLeverCandle c, SwLeverState prev)
        {
            if (!c.HasData || c.Start == DateTime.MinValue) return;

            double sc = double.IsNaN(scale) ? 0.25 : scale;
            double vz = Math.Max(0.01, cfg.VoteZ);
            int side = c.Balance / sc >= vz ? 1 : c.Balance / sc <= -vz ? -1 : 0;

            if (Host.VisualOn(SwVisualRole.LeverCenter))
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.LeverCenter, c.Start, c.Center, c.End, c.Center)
                    .WithVariant(side).InRing("LEVC"));

            if (Host.VisualOn(SwVisualRole.LeverFulcrum) && Host.VisualDebugOn)
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.LeverFulcrum, c.Start, c.Fulcrum, c.End, c.Fulcrum)
                    .AsDebug().InRing("LEVG"));

            if (Host.VisualOn(SwVisualRole.LeverCluster))
            {
                double maxD = 0;
                for (int b = 0; b < c.Clusters; b++) if (c.ClDensity[b] > maxD) maxD = c.ClDensity[b];
                for (int b = 0; b < c.Clusters; b++)
                {
                    int op = maxD > 0 ? 8 + (int)Math.Round(52.0 * c.ClDensity[b] / maxD) : 8;
                    double mid = (c.ClLo[b] + c.ClHi[b]) / 2.0;
                    int v = c.ClRich[b] ? (mid >= c.Fulcrum ? 1 : -1) : 0;
                    EmitVisual(SwVisualEvent.Rect(SwVisualRole.LeverCluster, c.Start, c.ClLo[b], c.End, c.ClHi[b])
                        .AsState("CL" + b).WithVariant(v).WithOpacity(op));
                }
                for (int b = c.Clusters; b < lastClusters; b++) EmitVisual(SwVisualEvent.RemoveState("CL" + b));
                lastClusters = c.Clusters;
            }

            if (state == prev || prev == SwLeverState.Warmup) return;
            double pad = 3 * tick;
            if ((state == SwLeverState.Up || state == SwLeverState.Down) && Host.VisualOn(SwVisualRole.LeverVote))
            {
                bool up = state == SwLeverState.Up;
                EmitVisual(SwVisualEvent.Marker(up ? SwVisualKind.ArrowUp : SwVisualKind.ArrowDown,
                        SwVisualRole.LeverVote, c.End, up ? c.Low - pad : c.High + pad)
                    .WithVariant(up ? 1 : -1).InRing("LEVV"));
            }
            else if (state == SwLeverState.Middle && (prev == SwLeverState.Up || prev == SwLeverState.Down)
                     && Host.VisualOn(SwVisualRole.LeverFade))
            {
                EmitVisual(SwVisualEvent.Label(SwVisualRole.LeverFade, c.End, c.High + 2 * pad, "затухание")
                    .WithRank(1).InRing("LEVF"));
            }
        }

        public string ResearchReport(string pad)
        {
            StringBuilder sb = new StringBuilder();
            sb.AppendLine(pad + "Рычаг: свечей взвешено " + candlesWeighed + ", пропущено " + candlesSkipped
                + ", поздних кластеров отброшено " + lateDropped
                + (totalWeight > 0 ? ", масса вне диапазона свечи " + (100.0 * outsideWeight / (totalWeight + outsideWeight)).ToString("F2") + " %" : "")
                + "; голосов ↑" + votesUp + " ↓" + votesDown + ", затуханий " + fades
                + ", не отправлено против тренда " + suppressedTrend + ", требований выхода " + exitsRequested);
            if (!double.IsNaN(biasMedian))
                sb.AppendLine(pad + "Рычаг: медиана баланса за окно " + F2(biasMedian) + ", масштаб 1.4826·MAD " + F2(scale)
                    + " — заметный сдвиг медианы от нуля говорит о смещении метода на этом инструменте");
            if (!cfg.Research) return sb.ToString();

            SwLeverStat b = stats[0];
            double baseMean = b.Mean;
            sb.AppendLine(pad + "Рычаг, исследование: горизонт " + Math.Max(1, cfg.ResearchHorizon)
                + " свечей анализа, доходность в медианных размахах свечи, знак — в сторону события");
            sb.AppendLine(pad + "  событие; n; средняя; ст. ошибка; в сторону, %; средняя минус база");
            for (int i = 0; i < stats.Length; i++)
            {
                SwLeverStat s = stats[i];
                if (s.N == 0) continue;
                double avgDir = (double)stDirSum[i] / Math.Max(1, s.N);
                double excess = i == 0 || double.IsNaN(baseMean) ? double.NaN : s.Mean - avgDir * baseMean;
                sb.AppendLine(pad + "  " + EventNames[i] + "; " + s.N + "; " + F3(s.Mean) + "; " + F3(s.StdErr) + "; "
                    + (double.IsNaN(s.HitRate) ? "—" : (100 * s.HitRate).ToString("F1")) + "; " + F3(excess));
            }
            sb.AppendLine(pad + "  модуль хода: событие; n; средний |ход|; ст. ошибка");
            for (int i = 0; i < mags.Length; i++)
            {
                if (mags[i].N == 0) continue;
                sb.AppendLine(pad + "  " + MagNames[i] + "; " + mags[i].N + "; " + F3(mags[i].Mean) + "; " + F3(mags[i].StdErr));
            }
            sb.AppendLine(pad + "  как читать: связь «масса выше опоры → рост» есть, если средние по корзинам T или m растут"
                + " слева направо и крайние корзины отличаются от базы больше чем на две ст. ошибки; затухание подтверждается,"
                + " если |ход| после середины меньше базового");
            return sb.ToString();
        }

        private static string StateText(SwLeverState s)
        {
            switch (s)

        private static string F2(double v)

        private int KernelDir()

        #region Бар своей серии: признаки, самопроверка, нормализация, классификация

        private void OnOwnBar()
        {
            if (!Host.SlotReady(cfg.Slot, 22)) return;
            double h = Host.HighOf(cfg.Slot, 0), l = Host.LowOf(cfg.Slot, 0), c = Host.CloseOf(cfg.Slot, 0);
            double v = Host.VolumeOf(cfg.Slot, 0);
            if (double.IsNaN(c) || c <= 0) return;

            // --- признаки ---
            double tr = double.IsNaN(prevClose) ? h - l : Math.Max(h - l, Math.Max(Math.Abs(h - prevClose), Math.Abs(l - prevClose)));
            atr = double.IsNaN(prevClose) ? tr : atr + (tr - atr) / 14.0;
            double chg = double.IsNaN(prevClose) ? 0 : c - prevClose;
            rsiN++;
            rsiUp += ((chg > 0 ? chg : 0) - rsiUp) / Math.Min(14, rsiN);
            rsiDn += ((chg < 0 ? -chg : 0) - rsiDn) / Math.Min(14, rsiN);
            double rsi = rsiDn > 0 ? 100 - 100 / (1 + rsiUp / rsiDn) : 100;

            // Калман: уровень и скорость, модель постоянной скорости.
            if (double.IsNaN(kx)) { kx = c; kv = 0; }
            else
            {
                double xp = kx + kv, q = cfg.KalmanQ, r = cfg.KalmanR;
                double a00 = p00 + 2 * p01 + p11 + q, a01 = p01 + p11, a11 = p11 + q;
                double s = a00 + r, g0 = a00 / s, g1 = a01 / s, e = c - xp;
                kx = xp + g0 * e; kv = kv + g1 * e;
                p00 = (1 - g0) * a00; p01 = (1 - g0) * a01; p11 = a11 - g1 * a01;
            }

            volRing[volIdx] = v; tpvRing[volIdx] = v * (h + l + c) / 3.0;
            volIdx = (volIdx + 1) % volRing.Length; if (volCnt < volRing.Length) volCnt++;
            double vs = 0, tps = 0; for (int i = 0; i < volCnt; i++) { vs += volRing[i]; tps += tpvRing[i]; }
            double vdens = vs > 0 ? v / (vs / volCnt) : 1.0;
            double com = vs > 0 ? tps / vs : c;
            if (vs > 0) zeroVolRun = 0;
            else if (++zeroVolRun >= Math.Max(50, cfg.FitWindow) && !zeroVolTraced)
            {
                zeroVolTraced = true;
                Host.Trace(Id, "LOR_NO_VOLUME", "объёма нет " + zeroVolRun + " баров: признаки объёма и центра масс "
                    + "постоянны, классификатор работает на трёх признаках из пяти");
            }
            prevClose = c;

            // --- запись в кольцо ---
            head = (head + 1) % cap; if (count < cap) count++;
            double a = atr > 0 ? atr : 1e-9;
            raw[head, 0] = atr / c;
            raw[head, 1] = kv / a;
            raw[head, 2] = rsi;
            raw[head, 3] = vdens;
            raw[head, 4] = (c - com) / a;
            closeRing[head] = c;
            atrRing[head] = a;

            Evaluate();
            if (++sinceRefit >= cfg.RefitBars || !normReady) { if (Refit()) sinceRefit = 0; }
            if (normReady) { Classify(); try { Observe(); } catch { } }
        }

        private int Idx(int back) { return (head - back + cap) % cap; }

        /// <summary>L17. Исход сделки соседа в направлении dir тройным барьером: 1 — цель первой, 0 — стоп, оба одним баром
        /// или без касания против направления (худший путь). Бары соседа доступны у хоста по barsAgo = back − k.</summary>
        private int TripleOutcome(int back, int dir)
        {
            double entry = closeRing[Idx(back)], at = atrRing[Idx(back)];
            if (!(at > 0) || double.IsNaN(entry)) return 0;
            double tgt = entry + dir * cfg.ExpMetaAtr * at, stp = entry - dir * cfg.ExpMetaAtr * at;
            for (int k = 1; k <= cfg.Horizon; k++)
            {
                double h = Host.HighOf(cfg.Slot, back - k), l = Host.LowOf(cfg.Slot, back - k);
                if (double.IsNaN(h) || double.IsNaN(l)) return 0;
                bool hitT = dir > 0 ? h >= tgt : l <= tgt, hitS = dir > 0 ? l <= stp : h >= stp;
                if (hitS) return 0;
                if (hitT) return 1;
            }
            return (closeRing[Idx(back - cfg.Horizon)] - entry) * dir > 0 ? 1 : 0;
        }

        private void Observe()
        {
            if (dir == 0 || !Host.VisualOn(SwVisualRole.LorentzVote)) return;
            DateTime t = Host.TimeOf(cfg.Slot, 0);
            if (t == DateTime.MinValue) return;
            double px = Host.CloseOf(cfg.Slot, 0), tick = Host.TickSize > 0 ? Host.TickSize : 1.0;
            EmitVisual(SwVisualEvent.Marker(dir > 0 ? SwVisualKind.ArrowUp : SwVisualKind.ArrowDown, SwVisualRole.LorentzVote, t,
                dir > 0 ? px - 3 * tick : px + 3 * tick).WithVariant(dir).WithDetail("перевес " + strength.ToString("F2")).InRing("LV"));
        }

        /// <summary>
        /// Причинная самопроверка: прогноз, сделанный H баров назад, сверяется с тем,
        /// что произошло. Точность сравнивается с базовой частотой класса: при частоте
        /// роста 60% точность 65% означает перевес всего в 5 пунктов.
        /// </summary>
        private void Evaluate()
        {
            if (predCnt > cfg.Horizon && count > cfg.Horizon)
            {
                int pred = predRing[(predHead - cfg.Horizon + predRing.Length) % predRing.Length];
                bool up = closeRing[head] > closeRing[Idx(cfg.Horizon)];
                if (pred != 0)
                {
                    evalHit[evalHead] = (pred > 0) == up;
                    evalUp[evalHead] = up;          // фактический исход
                    evalLong[evalHead] = pred > 0;  // направление прогноза
                    evalHead = (evalHead + 1) % evalHit.Length; if (evalCnt < evalHit.Length) evalCnt++;
                    predsTotal++;
                }
            }
        }

        public double Precision()
        {
            if (evalCnt == 0) return double.NaN;
            int n = 0; for (int i = 0; i < evalCnt; i++) if (evalHit[i]) n++;
            return (double)n / evalCnt;
        }

        /// <summary>
        /// Эталон точности: «монетка» с рыночной частотой роста p и тем же соотношением
        /// прогнозов вверх и вниз. Её точность = доля_вверх·p + доля_вниз·(1−p). Точность
        /// классификатора имеет смысл только как превышение над этим числом.
        /// </summary>
        public double BaseRate()
        {
            if (evalCnt == 0) return double.NaN;
            int up = 0, lng = 0;
            for (int i = 0; i < evalCnt; i++) { if (evalUp[i]) up++; if (evalLong[i]) lng++; }
            double p = (double)up / evalCnt, fl = (double)lng / evalCnt;
            return fl * p + (1 - fl) * (1 - p);
        }

        #endregion

        #region Нормализация: стандартизация → Йео–Джонсон → z-оценка

        private static double YJ(double x, double l)
        {
            if (x >= 0) return Math.Abs(l) < 1e-9 ? Math.Log(x + 1) : (Math.Pow(x + 1, l) - 1) / l;
            return Math.Abs(l - 2) < 1e-9 ? -Math.Log(1 - x) : -(Math.Pow(1 - x, 2 - l) - 1) / (2 - l);
        }

        /// <summary>Логарифм правдоподобия Йео–Джонсона с точностью до константы.</summary>
        private double LogLik(int n, double l)
        {
            double m = 0; for (int i = 0; i < n; i++) m += YJ(scratch[i], l); m /= n;
            double v = 0, j = 0;
            for (int i = 0; i < n; i++) { double y = YJ(scratch[i], l) - m; v += y * y; j += Math.Sign(scratch[i]) * Math.Log(Math.Abs(scratch[i]) + 1); }
            v /= n;
            return v > 0 ? -0.5 * n * Math.Log(v) + (l - 1) * j : double.NegativeInfinity;
        }

        /// <summary>
        /// Пересчёт параметров по окну [L09] ПРОШЛЫХ баров. λ ищется золотым сечением
        /// на [−2; 2]. Используются только бары, уже записанные в кольцо: будущего нет.
        /// </summary>
        private bool Refit()
        {
            int n = Math.Min(count, cfg.FitWindow);
            if (n < 50) return false;
            for (int f = 0; f < F; f++)
            {
                double m = 0; for (int i = 0; i < n; i++) m += raw[Idx(i), f]; m /= n;
                double v = 0; for (int i = 0; i < n; i++) { double z = raw[Idx(i), f] - m; v += z * z; }
                preMu[f] = m; preSd[f] = Math.Sqrt(v / Math.Max(1, n - 1)); if (!(preSd[f] > 0)) preSd[f] = 1;
                for (int i = 0; i < n; i++) scratch[i] = (raw[Idx(i), f] - preMu[f]) / preSd[f];

                double lo = -2, hi = 2, gr = 0.6180339887;
                if (cfg.UseYeoJohnson)
                {
                    double x1 = hi - gr * (hi - lo), x2 = lo + gr * (hi - lo), f1 = LogLik(n, x1), f2 = LogLik(n, x2);
                    for (int it = 0; it < 30; it++)
                    {
                        if (f1 < f2) { lo = x1; x1 = x2; f1 = f2; x2 = lo + gr * (hi - lo); f2 = LogLik(n, x2); }
                        else { hi = x2; x2 = x1; f2 = f1; x1 = hi - gr * (hi - lo); f1 = LogLik(n, x1); }
                    }
                    lam[f] = 0.5 * (lo + hi);
                }
                else lam[f] = 1.0;

                double pm = 0; for (int i = 0; i < n; i++) pm += YJ(scratch[i], lam[f]); pm /= n;
                double pv = 0; for (int i = 0; i < n; i++) { double y = YJ(scratch[i], lam[f]) - pm; pv += y * y; }
                postMu[f] = pm; postSd[f] = Math.Sqrt(pv / Math.Max(1, n - 1)); if (!(postSd[f] > 0)) postSd[f] = 1;
            }
            normReady = true;
            return true;
        }

        private double Z(int back, int f)
        {
            return (YJ((raw[Idx(back), f] - preMu[f]) / preSd[f], lam[f]) - postMu[f]) / postSd[f];
        }

        #endregion

        #region K ближайших по расстоянию Лоренца

        private void Classify()
        {
            int K = kDist.Length; kN = 0; nwN = 0;
            double q0 = Z(0, 0), q1 = Z(0, 1), q2 = Z(0, 2), q3 = Z(0, 3), q4 = Z(0, 4);
            int maxBack = Math.Min(count - 1, cfg.Lookback);

            // Соседи — только с известным исходом: back ≥ H. Шаг снижает зависимость меток.
            for (int back = cfg.Horizon + Math.Max(0, cfg.ExpEmbargoBars); back <= maxBack; back += Math.Max(1, cfg.Stride))   // Э17
            {
                if (back + 0 > count - 1) break;
                double d = Math.Log(1 + Math.Abs(q0 - Z(back, 0))) + Math.Log(1 + Math.Abs(q1 - Z(back, 1)))
                         + Math.Log(1 + Math.Abs(q2 - Z(back, 2))) + Math.Log(1 + Math.Abs(q3 - Z(back, 3)))
                         + Math.Log(1 + Math.Abs(q4 - Z(back, 4)));
                int lab = cfg.ExpMetaLabels ? (TripleOutcome(back, 1) * 2 + TripleOutcome(back, -1))                 // L17
                        : (closeRing[Idx(back - cfg.Horizon)] > closeRing[Idx(back)] ? 1 : -1);

                // Для Надарая–Ватсона — доходность за H баров в ATR на момент ситуации:
                // регрессии нужна величина хода, а не только знак.
                if (nwN < nwD.Length)
                {
                    double at = atrRing[Idx(back)];
                    nwD[nwN] = d;
                    nwY[nwN] = at > 0 ? (closeRing[Idx(back - cfg.Horizon)] - closeRing[Idx(back)]) / at : 0.0;
                    nwW[nwN] = DecayWeight(back);   // Э18
                    nwN++;
                }

                if (kN < K) { Insert(d, lab, back, kN); kN++; }
                else if (d < kDist[K - 1]) Insert(d, lab, back, K - 1);
            }

            double sw = 0, vote = 0, h2 = 0, upW = 0, dnW = 0;
            if (cfg.ExpKernelVote && kN > 0) { double md = kDist[kN / 2]; h2 = md > 0 ? 2 * md * md : 1; }
            for (int i = 0; i < kN; i++)
            {
                double w = (cfg.ExpKernelVote ? Math.Exp(-kDist[i] * kDist[i] / h2) : 1.0) * DecayWeight(kBack[i]);   // Э18
                sw += w;
                if (cfg.ExpMetaLabels) { upW += w * (kLab[i] >> 1); dnW += w * (kLab[i] & 1); }                   // L17
                else vote += w * kLab[i];
            }
            double knnStrength;
            if (cfg.ExpMetaLabels)                                                                                 // L17
            {
                double pUp = sw > 0 ? upW / sw : 0, pDn = sw > 0 ? dnW / sw : 0;
                knnDir = (pUp >= cfg.VoteThreshold && pUp > pDn) ? 1 : (pDn >= cfg.VoteThreshold && pDn > pUp) ? -1 : 0;
                knnStrength = Math.Max(pUp, pDn);
            }
            else
            {
                knnStrength = sw > 0 ? Math.Abs(vote) / sw : 0;
                knnDir = knnStrength >= cfg.VoteThreshold ? Math.Sign(vote) : 0;
            }

            double nwStrength = 0; nwDir = 0; nwPred = double.NaN; nwEff = 0;
            if (cfg.Method != SwLorentzMethod.KnnVote && nwN >= 3) nwStrength = NadarayaWatson();

            switch (cfg.Method)
            {
                case SwLorentzMethod.NadarayaWatson: dir = nwDir; strength = nwStrength; break;
                case SwLorentzMethod.Consensus:
                    dir = (knnDir != 0 && knnDir == nwDir) ? knnDir : 0;
                    strength = dir != 0 ? Math.Min(knnStrength, nwStrength) : 0;
                    break;
                default: dir = knnDir; strength = knnStrength; break;
            }

            predRing[predHead] = dir; predHead = (predHead + 1) % predRing.Length; if (predCnt < predRing.Length + cfg.Horizon + 1) predCnt++;
        }

        /// <summary>
        /// Надарая–Ватсон в пространстве признаков: ŷ = Σ K(dᵢ/h)·yᵢ / Σ K(dᵢ/h) по ВСЕМ
        /// допустимым ситуациям. Ширина h = 3 · множитель · медиана расстояний: у гауссова ядра
        /// вычислителя MOD9 z = 3u, поэтому σ = множитель × медиана. Функция ядра — из MOD9,
        /// не переписана. Эффективный объём (Σw)²/Σw² защищает от решения по одной-двум точкам.
        /// </summary>
        private double NadarayaWatson()
        {
            for (int i = 0; i < nwN; i++) nwScratch[i] = nwD[i];
            for (int i = 1; i < nwN; i++)
            {
                double key = nwScratch[i]; int j = i - 1;
                while (j >= 0 && nwScratch[j] > key) { nwScratch[j + 1] = nwScratch[j]; j--; }
                nwScratch[j + 1] = key;
            }
            double med = nwScratch[nwN / 2];
            double h = 3.0 * Math.Max(1e-9, cfg.NwBandMult * med);

            double sw = 0, sw2 = 0, sy = 0;
            for (int i = 0; i < nwN; i++)
            {
                double w = SwAdaptiveKernelEngine.K(cfg.NwKernel, nwD[i] / h) * nwW[i];   // Э18
                if (w <= 0) continue;
                sw += w; sw2 += w * w; sy += w * nwY[i];
            }
            if (sw <= 0) return 0;
            nwPred = sy / sw;
            nwEff = sw * sw / Math.Max(1e-12, sw2);
            if (nwEff < cfg.NwMinEff || Math.Abs(nwPred) < cfg.NwThreshold) return 0;
            nwDir = Math.Sign(nwPred);
            return Math.Min(1.0, Math.Abs(nwPred) / (2.0 * Math.Max(1e-9, cfg.NwThreshold)));
        }

        /// <summary>Вставка в отсортированный массив длины K без выделения памяти.</summary>
        private void Insert(double d, int lab, int back, int pos)
        {
            int j = pos;
            while (j > 0 && kDist[j - 1] > d) { kDist[j] = kDist[j - 1]; kLab[j] = kLab[j - 1]; kBack[j] = kBack[j - 1]; j--; }
            kDist[j] = d; kLab[j] = lab; kBack[j] = back;
        }

        /// <summary>Э18: вес давности — 0,5^(давность / полураспад); без полураспада — 1.</summary>
        private double DecayWeight(int back)
        {
            return cfg.ExpDecayHalfLife > 0 ? Math.Pow(0.5, (double)back / cfg.ExpDecayHalfLife) : 1.0;
        }

        #endregion

        public string StateLine
        {
            get
            {
                return string.Format("{0} | K {1}, NW {2:F2} ATR (эфф. {3:F0}) → {4} | точность {5:P0} при базе {6:P0} ({7})",
                    cfg.Method, knnDir > 0 ? "↑" : knnDir < 0 ? "↓" : "—", double.IsNaN(nwPred) ? 0.0 : nwPred, nwEff,
                    dir > 0 ? "вверх" : dir < 0 ? "вниз" : "нет", Precision(), BaseRate(), evalCnt);
            }
        }
    }
}

        public bool ExpChochPlusFailedExtreme = false;             // Э76: правила PACV025
        public SwPacStrongWeak ExpStrongWeak = SwPacStrongWeak.Off; // Э77: сильные/слабые экстремумы
        public double ExpStrongBonus = 0.10;                        // Э77: ±к уверенности по силе пробитого уровня
    }

    /// <summary>
    /// Состояние структуры одного масштаба на одной серии.
    /// Хранит подтверждённые экстремумы и действующее направление.
    /// </summary>
    public class SwPacState
    {
        public int Strength;

        // Подтверждённые экстремумы. Номер бара подтверждения хранится рядом:
        // без него невозможно доказать, что уровень существовал на момент
        // использования.
        public double LastHigh = double.NaN;
        public int LastHighBar = -1;
        public double PrevHigh = double.NaN;

        public double LastLow = double.NaN;
        public int LastLowBar = -1;
        public double PrevLow = double.NaN;

        public int Trend;                 // +1 бычий, -1 медвежий, 0 не определён

        // Пробитые уровни помечаются, чтобы одно и то же событие не
        // порождалось на каждом последующем баре.
        public bool HighBroken;
        public bool LowBroken;

        public int BosCount, ChochCount, ChochPlusCount;

        // v1.5 — Э76/Э77: несостоявшиеся экстремумы, ожидание первого CHoCH, объём ног и сила уровней
        public bool HadFailedHigh, HadFailedLow, PriorEventConfirmed;
        public double VolSinceHigh, VolSinceLow, PrevLegVolHigh = double.NaN, PrevLegVolLow = double.NaN;
        public int LastHighStrong = -1, LastLowStrong = -1;   // 1 сильный, 0 слабый, −1 нет метки

        public void Reset(int strength)
        {
            Strength = Math.Max(1, strength);
            LastHigh = LastLow = PrevHigh = PrevLow = double.NaN;
            LastHighBar = LastLowBar = -1;
            Trend = 0;
            HighBroken = LowBroken = false;
            BosCount = ChochCount = ChochPlusCount = 0;
            HadFailedHigh = HadFailedLow = PriorEventConfirmed = false;
            VolSinceHigh = VolSinceLow = 0; PrevLegVolHigh = PrevLegVolLow = double.NaN; LastHighStrong = LastLowStrong = -1;
        }

        /// <summary>v1.5. Объём бара в ноги обоих типов (PACV025: накопление каждый бар, сброс на пивоте).</summary>
        public void AccumulateVolume(double v) { if (!double.IsNaN(v) && v > 0) { VolSinceHigh += v; VolSinceLow += v; } }

        /// <summary>v1.5. Сила подтверждённого пивота по PACV025; обновляет ноги объёма. −1 — метки нет.</summary>
        public int Classify(bool isHigh, SwPacStrongWeak mode)
        {
            int res = -1;
            if (mode == SwPacStrongWeak.VolumeRatio)
            {
                double leg = isHigh ? VolSinceHigh : VolSinceLow, prev = isHigh ? PrevLegVolHigh : PrevLegVolLow;
                if (!double.IsNaN(prev) && prev > 0 && leg > 0) res = Math.Min(prev, leg) / Math.Max(prev, leg) >= 0.5 ? 1 : 0;
                if (isHigh) { PrevLegVolHigh = leg; VolSinceHigh = 0; } else { PrevLegVolLow = leg; VolSinceLow = 0; }
            }
            else if (mode == SwPacStrongWeak.TrendSide)
            {
                if (Trend != 0) res = ((Trend < 0 && isHigh) || (Trend > 0 && !isHigh)) ? 1 : 0;
            }
            return res;
        }

        public void PushHigh(double px, int bar)
        {
            if (!double.IsNaN(LastHigh)) { if (px < LastHigh) HadFailedHigh = true; else if (px > LastHigh) HadFailedHigh = false; }   // Э76
            PrevHigh = LastHigh;
            LastHigh = px;
            LastHighBar = bar;
            HighBroken = false;
        }

        public void PushLow(double px, int bar)
        {
            if (!double.IsNaN(LastLow)) { if (px > LastLow) HadFailedLow = true; else if (px < LastLow) HadFailedLow = false; }       // Э76
            PrevLow = LastLow;
            LastLow = px;
            LastLowBar = bar;
            LowBroken = false;
        }
    }

    public class SwPacStructureModule : SwModule
    {
        private readonly SwPacConfig cfg;

        private readonly SwPacState lowInternal = new SwPacState();
        private readonly SwPacState lowSwing = new SwPacState();
        private readonly SwPacState highInternal = new SwPacState();
        private readonly SwPacState highSwing = new SwPacState();

        // Событие, отобранное на текущем такте решения.
        private SwPacEvent pendingEvent = SwPacEvent.None;
        private int pendingDir;
        private double pendingConfidence;
        private string pendingText = string.Empty;
        private int pendingRank;           // приоритет отбора

        // --- v1.1: наблюдение, на отбор не влияет ---
        private string pendingSignalId = string.Empty;
        private string pendingShort = string.Empty;
        private int labelsThisTick;

        private sealed class PacCandidate
        {
            public string Id;
            public string Desc;
            public int Dir;
            public double Conf;
        }

        // Допущенные события такта, уступившие отобранному: для журнала сигналов.
        private readonly List<PacCandidate> tickCandidates = new List<PacCandidate>();

        // --- v1.2: события такта для отрисовки после отбора ---
        private sealed class PacEventVis
        {
            public string Id;
            public bool Filtered;
            public int Flag;
            public double Conf;
            public int Dir;
            public string Text;
            public DateTime TBreak, TPivot;
            public double Level, Hi, Lo;
        }
        private readonly List<PacEventVis> tickEventVis = new List<PacEventVis>();

        // Показанные незавершённые опорные точки: ключ → номер бара кандидата.
        private readonly Dictionary<string, int> candShown = new Dictionary<string, int>();

        public SwPacStructureModule(string id, SwPacConfig config)
        {
            Id = id;
            Role = SwRole.Signal;
            cfg = config ?? new SwPacConfig();
        }

        public override void Attach(SwHostView host)
        {
            base.Attach(host);

            lowInternal.Reset(cfg.InternalStrength);
            lowSwing.Reset(cfg.SwingStrength);
            highInternal.Reset(cfg.InternalStrength);
            highSwing.Reset(cfg.SwingStrength);

            Host.Trace(Id, "PAC_ATTACHED",
                "младший слот " + cfg.LowerSlot + ", старший слот " + cfg.HigherSlot
                + ", сила " + cfg.InternalStrength + "/" + cfg.SwingStrength);
        }

        public override void Compute(int bip)
        {
            // --- 1. Обработка закрытия бара младшей серии ---
            if (Host.IsBarCloseOf(cfg.LowerSlot, bip))
                ProcessSeries(cfg.LowerSlot, lowInternal, lowSwing, false);

            // --- 2. Обработка закрытия бара старшей серии ---
            if (cfg.UseHigherTimeframe && Host.IsBarCloseOf(cfg.HigherSlot, bip))
                ProcessSeries(cfg.HigherSlot, highInternal, highSwing, true);

            // --- 3. Отправка только в такте серии исполнения ---
            if (bip != Host.ExecBip) return;

            int bar = Host.CurrentBar;

            if (!Ready())
            {
                Health = SwHealth.Warmup;
                Emit(SwSignal.Empty(bar));
                ClearPending();
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;

            if (pendingEvent == SwPacEvent.None)
            {
                s.Direction = 0;
                s.Confidence = 0;
                s.Text = "структура: событий нет";
            }
            else if (cfg.ExpMinBarsBetweenSignals > 0 && lastSignalBar >= 0 && bar - lastSignalBar < cfg.ExpMinBarsBetweenSignals)
            {
                // Э42: событие слишком близко к прошлому сигналу — не независимое.
                s.Direction = 0;
                s.Confidence = 0;
                s.Text = "структура: " + pendingShort + " пропущено — до прошлого сигнала " + (bar - lastSignalBar) + " баров";
                Host.Trace(Id, "PAC_UNIQUENESS_SKIP", s.Text);
            }
            else
            {
                s.Direction = pendingDir;
                s.Confidence = Math.Min(1.0, pendingConfidence);
                s.Text = pendingText;
                s.SignalId = pendingSignalId;
                s.ShortDescription = pendingShort;
                lastSignalBar = bar;
            }

            Health = SwHealth.Ok;
            Emit(s);

            // Наблюдение — после отправки: его сбой не может помешать сигналу.
            AfterEmit(s);

            // Событие живёт один такт решения. Без сброса один пробой
            // порождал бы вход на каждом последующем баре.
            ClearPending();
        }

        private bool Ready()
        {
            int needLow = cfg.SwingStrength * 2 + 2;
            if (!Host.SlotReady(cfg.LowerSlot, needLow)) return false;

            if (cfg.UseHigherTimeframe)
            {
                int needHigh = cfg.SwingStrength * 2 + 2;
                if (!Host.SlotReady(cfg.HigherSlot, needHigh)) return false;
            }
            return true;
        }

        private void ClearPending()
        {
            pendingEvent = SwPacEvent.None;
            pendingDir = 0;
            pendingConfidence = 0;
            pendingText = string.Empty;
            pendingRank = 0;
            pendingSignalId = string.Empty;
            pendingShort = string.Empty;
            labelsThisTick = 0;
            tickCandidates.Clear();
            tickEventVis.Clear();
        }

        #region Обработка серии

        private void ProcessSeries(int slot, SwPacState internalState, SwPacState swingState, bool isHigher)
        {
            DetectPivots(slot, internalState, isHigher, SwPacScope.Internal);
            DetectPivots(slot, swingState, isHigher, SwPacScope.Swing);

            DetectEvents(slot, internalState, isHigher, SwPacScope.Internal);
            DetectEvents(slot, swingState, isHigher, SwPacScope.Swing);

            // v1.2: наблюдение, на состояние не влияет.
            VisualCandidates(slot, internalState, isHigher, SwPacScope.Internal);
            VisualCandidates(slot, swingState, isHigher, SwPacScope.Swing);
        }

        /// <summary>
        /// Обнаружение опорной точки.
        ///
        /// Кандидат находится на баре, отстоящем на strength баров назад:
        /// именно столько баров должно закрыться СПРАВА от него, чтобы
        /// экстремум подтвердился. Раньше этого момента точка не существует.
        /// Отсюда неустранимое запаздывание в strength баров — плата за то,
        /// что уровень реален, а не подогнан задним числом.
        /// </summary>
        private void DetectPivots(int slot, SwPacState st, bool isHigher, SwPacScope scope)
        {
            int k = st.Strength;
            int need = k * 2 + 1;
            if (!Host.SlotReady(slot, need)) return;

            int centre = k;   // barsAgo центрального бара

            double hC = Host.HighOf(slot, centre);
            double lC = Host.LowOf(slot, centre);
            if (double.IsNaN(hC) || double.IsNaN(lC)) return;

            bool isHigh = true, isLow = true;

            for (int i = 1; i <= k; i++)
            {
                double hL = Host.HighOf(slot, centre + i);
                double hR = Host.HighOf(slot, centre - i);
                double lL = Host.LowOf(slot, centre + i);
                double lR = Host.LowOf(slot, centre - i);

                if (double.IsNaN(hL) || double.IsNaN(hR) || double.IsNaN(lL) || double.IsNaN(lR))
                    return;

                // Строгое сравнение слева и нестрогое справа: при равенстве
                // экстремум относится к более раннему бару. Иначе плато
                // порождало бы несколько опорных точек подряд.
                if (hC <= hL || hC < hR) isHigh = false;
                if (lC >= lL || lC > lR) isLow = false;
            }

            int barOfCentre = Host.BarOf(slot) - centre;
            if (cfg.ExpStrongWeak != SwPacStrongWeak.Off) st.AccumulateVolume(Host.VolumeOf(slot, 0));   // Э77

            if (isHigh && barOfCentre != st.LastHighBar)
            {
                st.PushHigh(hC, barOfCentre);
                if (cfg.ExpStrongWeak != SwPacStrongWeak.Off) st.LastHighStrong = st.Classify(true, cfg.ExpStrongWeak);   // Э77
                VisualPivot(slot, centre, hC, true, isHigher, scope);
            }
            if (isLow && barOfCentre != st.LastLowBar)
            {
                st.PushLow(lC, barOfCentre);
                if (cfg.ExpStrongWeak != SwPacStrongWeak.Off) st.LastLowStrong = st.Classify(false, cfg.ExpStrongWeak);   // Э77
                VisualPivot(slot, centre, lC, false, isHigher, scope);
            }
        }

        /// <summary>
        /// Обнаружение структурных событий по подтверждённым экстремумам.
        /// </summary>
        /// <summary>
        /// v1.3, Э37 и Э38: подтверждение пробоя глубиной в долях ATR и объёмом бара пробоя.
        /// Без экспериментов — всегда «да», как прежде.
        /// </summary>
        private int lastSignalBar = -1;   // Э42

        private bool BreakConfirmed(int slot, double depth)
        {
            int n = Math.Max(2, cfg.ExpBreakWindow);
            if (cfg.ExpBreakAtr > 0)
            {
                double sum = 0; int k = 0;
                for (int i = 0; i < n; i++)
                {
                    double h = Host.HighOf(slot, i), l = Host.LowOf(slot, i), pc = Host.CloseOf(slot, i + 1);
                    if (double.IsNaN(h) || double.IsNaN(l)) break;
                    double tr = h - l;
                    if (!double.IsNaN(pc)) tr = Math.Max(tr, Math.Max(Math.Abs(h - pc), Math.Abs(l - pc)));
                    sum += tr; k++;
                }
                if (k < n || !(depth >= cfg.ExpBreakAtr * sum / k)) return false;
            }
            if (cfg.ExpBreakVolMult > 0)
            {
                double v0 = Host.VolumeOf(slot, 0), sum = 0; int k = 0;
                for (int i = 1; i <= n; i++)
                {
                    double v = Host.VolumeOf(slot, i);
                    if (double.IsNaN(v)) break;
                    sum += v; k++;
                }
                if (k < n || !(sum > 0) || !(v0 >= cfg.ExpBreakVolMult * sum / k)) return false;
            }
            return true;
        }

        private void DetectEvents(int slot, SwPacState st, bool isHigher, SwPacScope scope)
        {
            double probeUp = cfg.BreakBy == SwPacBreakBy.BarExtreme
                ? Host.HighOf(slot, 0) : Host.CloseOf(slot, 0);
            double probeDn = cfg.BreakBy == SwPacBreakBy.BarExtreme
                ? Host.LowOf(slot, 0) : Host.CloseOf(slot, 0);

            if (double.IsNaN(probeUp) || double.IsNaN(probeDn)) return;

            bool upHit = !double.IsNaN(st.LastHigh) && !st.HighBroken && probeUp > st.LastHigh
                         && BreakConfirmed(slot, probeUp - st.LastHigh);
            bool dnHit = !double.IsNaN(st.LastLow) && !st.LowBroken && probeDn < st.LastLow
                         && BreakConfirmed(slot, st.LastLow - probeDn);

            // v1.3: оба экстремума одним баром — порядок внутри бара неизвестен (Книга, раздел 10).
            if (upHit && dnHit)
            {
                st.HighBroken = true; st.LowBroken = true;
                Host.Trace(Id, "PAC_AMBIGUOUS_BAR", (isHigher ? "СТ " : "МЛ ") + scope
                    + ": пробиты оба экстремума одним баром — порядок неизвестен, события нет");
                return;
            }

            // ----- пробой верхнего экстремума -----
            if (upHit)
            {
                st.HighBroken = true;

                // ПЕРВЫЙ ПРОБОЙ УСТАНАВЛИВАЕТ ТРЕНД, А НЕ МЕНЯЕТ ЕГО. Пока тренд
                // не определён, разворачивать нечего. Прежде первый пробой
                // уходил в ветку смены характера и порождал ложный сигнал
                // разворота с уверенностью 0.65 или 0.85 — по одному на каждое
                // из четырёх состояний структуры сразу после прогрева.
                if (st.Trend == 0)
                {
                    st.Trend = 1;
                    Host.Trace(Id, "PAC_TREND_INIT",
                        (isHigher ? "СТ " : "МЛ ") + scope + ": тренд установлен вверх, сигнала нет");
                    return;
                }

                SwPacEvent ev;
                if (st.Trend > 0)
                {
                    ev = SwPacEvent.Bos;            // продолжение бычьего
                    if (cfg.ExpChochPlusFailedExtreme && !st.PriorEventConfirmed) return;   // Э76: BOS лишь после первого CHoCH
                }
                else if (cfg.ExpChochPlusFailedExtreme)
                {
                    ev = st.HadFailedLow ? SwPacEvent.ChochPlus : SwPacEvent.Choch;   // Э76: несостоявшийся LL перед пробоем вверх
                    st.HadFailedLow = false; st.PriorEventConfirmed = true;
                }
                else
                {
                    // Смена характера вверх. Усиленной считается, когда
                    // пробит и предыдущий верхний экстремум: слом глубже.
                    ev = (!double.IsNaN(st.PrevHigh) && probeUp > st.PrevHigh)
                        ? SwPacEvent.ChochPlus : SwPacEvent.Choch;
                }

                st.Trend = 1;
                Register(ev, 1, isHigher, scope, st, st.LastHigh, slot, st.LastHighStrong);
            }

            // ----- пробой нижнего экстремума -----
            if (dnHit)
            {
                st.LowBroken = true;

                if (st.Trend == 0)
                {
                    st.Trend = -1;
                    Host.Trace(Id, "PAC_TREND_INIT",
                        (isHigher ? "СТ " : "МЛ ") + scope + ": тренд установлен вниз, сигнала нет");
                    return;
                }

                SwPacEvent ev;
                if (st.Trend < 0)
                {
                    ev = SwPacEvent.Bos;            // продолжение медвежьего
                    if (cfg.ExpChochPlusFailedExtreme && !st.PriorEventConfirmed) return;   // Э76
                }
                else if (cfg.ExpChochPlusFailedExtreme)
                {
                    ev = st.HadFailedHigh ? SwPacEvent.ChochPlus : SwPacEvent.Choch;   // Э76: несостоявшийся HH перед пробоем вниз
                    st.HadFailedHigh = false; st.PriorEventConfirmed = true;
                }
                else
                {
                    ev = (!double.IsNaN(st.PrevLow) && probeDn < st.PrevLow)
                        ? SwPacEvent.ChochPlus : SwPacEvent.Choch;
                }

                st.Trend = -1;
                Register(ev, -1, isHigher, scope, st, st.LastLow, slot, st.LastLowStrong);
            }
        }

        /// <summary>
        /// Регистрация события и отбор сильнейшего за такт.
        ///
        /// За один такт может сработать несколько событий: на двух сериях и
        /// в двух масштабах. Отправляется ОДНО — с наибольшим приоритетом.
        /// Порядок приоритета: тип события, затем старший таймфрейм,
        /// затем свинговый масштаб.
        /// </summary>
        private void Register(SwPacEvent ev, int dir, bool isHigher, SwPacScope scope,
                              SwPacState st, double level, int slot, int levelStrength)
        {
            if (ev == SwPacEvent.Bos) st.BosCount++;
            else if (ev == SwPacEvent.Choch) st.ChochCount++;
            else if (ev == SwPacEvent.ChochPlus) st.ChochPlusCount++;

            // Чистые вычисления: описание, уверенность, идентификатор. Прежде
            // уверенность считалась после проверки ранга — значение то же.
            string desc = Describe(ev, dir, isHigher, scope);
            double conf = ev == SwPacEvent.ChochPlus ? cfg.ConfidenceChochPlus
                        : ev == SwPacEvent.Choch ? cfg.ConfidenceChoch
                        : cfg.ConfidenceBos;
            if (isHigher) conf += cfg.HigherTimeframeBonus;
            if (cfg.ExpStrongWeak != SwPacStrongWeak.Off && levelStrength >= 0)                      // Э77
            {
                conf = Math.Max(0.05, Math.Min(1.0, conf + (levelStrength == 1 ? cfg.ExpStrongBonus : -cfg.ExpStrongBonus)));
                desc += levelStrength == 1 ? " (уровень сильный)" : " (уровень слабый)";
            }
            DateTime evTime = Host.TimeOf(slot, 0);
            string sid = MakeSignalId(evTime, ev, dir, isHigher, scope);

            if (!IsPermitted(ev, isHigher, scope))
            {
                Host.Trace(Id, "PAC_EVENT_SKIPPED", desc + " — тип выключен настройкой");
                int flag = FlagIndex(ev, isHigher, scope);
                PrepareEventVisual(slot, st, ev, dir, isHigher, scope, level, true, flag, sid, conf);
                JournalNotSent(evTime, sid, dir, conf, desc, "флаг [M" + flag + "] выключен");
                return;
            }

            PrepareEventVisual(slot, st, ev, dir, isHigher, scope, level, false, 0, sid, conf);

            int rank = (ev == SwPacEvent.ChochPlus ? 300 : ev == SwPacEvent.Choch ? 200 : 100)
                     + (isHigher ? 40 : 0)
                     + (scope == SwPacScope.Swing ? 20 : 0);

            if (rank <= pendingRank)
            {
                // Допущено, но уступает событию, уже отобранному за этот такт.
                AddCandidate(sid, dir, conf, desc);
                return;
            }

            // Прежде отобранное событие, если было, уступает новому.
            if (pendingEvent != SwPacEvent.None)
                AddCandidate(pendingSignalId, pendingDir, pendingConfidence, pendingShort);

            pendingRank = rank;
            pendingEvent = ev;
            pendingDir = dir;
            pendingConfidence = conf;
            pendingText = desc + " @ " + level.ToString("F2");
            pendingSignalId = sid;
            pendingShort = desc;

            Host.Trace(Id, "PAC_EVENT", pendingText + ", уверенность " + conf.ToString("F2"));
        }

        private bool IsPermitted(SwPacEvent ev, bool isHigher, SwPacScope scope)
        {
            if (!isHigher)
            {
                if (scope == SwPacScope.Internal)
                    return ev == SwPacEvent.Bos ? cfg.LowInternalBos
                         : ev == SwPacEvent.Choch ? cfg.LowInternalChoch
                         : cfg.LowInternalChochPlus;

                return ev == SwPacEvent.Bos ? cfg.LowSwingBos
                     : ev == SwPacEvent.Choch ? cfg.LowSwingChoch
                     : cfg.LowSwingChochPlus;
            }

            if (scope == SwPacScope.Internal)
                return ev == SwPacEvent.Bos ? cfg.HighInternalBos
                     : ev == SwPacEvent.Choch ? cfg.HighInternalChoch
                     : cfg.HighInternalChochPlus;

            return ev == SwPacEvent.Bos ? cfg.HighSwingBos
                 : ev == SwPacEvent.Choch ? cfg.HighSwingChoch
                 : cfg.HighSwingChochPlus;
        }

        private static string Describe(SwPacEvent ev, int dir, bool isHigher, SwPacScope scope)
        {
            return (isHigher ? "СТ" : "МЛ") + " "
                 + (scope == SwPacScope.Swing ? "swg" : "int") + " "
                 + (ev == SwPacEvent.ChochPlus ? "CHoCH+" : ev == SwPacEvent.Choch ? "CHoCH" : "BOS")
                 + " " + (dir > 0 ? "long" : "short");
        }

        #endregion

        #region Наблюдение: идентификаторы, визуализация, журнал (v1.1)

        /// <summary>
        /// Идентификатор события по времени бара пробоя на его серии:
        /// PAC_ГГГГММДД_ЧЧММСС_{L|H}{I|S}_{BOS|CH|CHP}_{U|D}.
        /// </summary>
        private static string MakeSignalId(DateTime t, SwPacEvent ev, int dir, bool isHigher, SwPacScope scope)
        {
            return "PAC_" + t.ToString("yyyyMMdd_HHmmss", System.Globalization.CultureInfo.InvariantCulture)
                 + "_" + (isHigher ? "H" : "L") + (scope == SwPacScope.Swing ? "S" : "I")
                 + "_" + (ev == SwPacEvent.Bos ? "BOS" : ev == SwPacEvent.Choch ? "CH" : "CHP")
                 + "_" + (dir > 0 ? "U" : "D");
        }

        /// <summary>Номер флага допуска в окне параметров ядра: [M87]-[M98].</summary>
        private static int FlagIndex(SwPacEvent ev, bool isHigher, SwPacScope scope)
        {
            return (isHigher ? 67 : 61) + (scope == SwPacScope.Swing ? 3 : 0)
                 + (ev == SwPacEvent.Bos ? 0 : ev == SwPacEvent.Choch ? 1 : 2);
        }

        private static string EventName(SwPacEvent ev)
        {
            return ev == SwPacEvent.ChochPlus ? "CHoCH+" : ev == SwPacEvent.Choch ? "CHoCH" : "BOS";
        }

        /// <summary>
        /// Подтверждённая опорная точка — на баре экстремума, с его ценой. На
        /// графике она появляется на баре подтверждения: так, как её видит модуль.
        /// </summary>
        private void VisualPivot(int slot, int centre, double price, bool isHighPivot, bool isHigher, SwPacScope scope)
        {
            try
            {
                SwVisualRole role = isHigher
                    ? (scope == SwPacScope.Swing ? SwVisualRole.PivotSwingHigh : SwVisualRole.PivotInternalHigh)
                    : (scope == SwPacScope.Swing ? SwVisualRole.PivotSwing : SwVisualRole.PivotInternal);
                if (!Host.VisualOn(role)) return;

                DateTime t = Host.TimeOf(slot, centre);
                if (t == DateTime.MinValue) return;

                double pad = Host.TickSize;
                SwVisualKind kind = (isHigher || scope == SwPacScope.Swing) ? SwVisualKind.Diamond : SwVisualKind.Dot;
                EmitVisual(SwVisualEvent.Marker(kind, role, t, isHighPivot ? price + pad : price - pad));
            }
            catch { }
        }

        /// <summary>
        /// Событие такта запоминается для отрисовки после отбора: до отправки
        /// сигнала неизвестно, какое из событий станет отобранным. Отсеянные
        /// флагом не запоминаются вовсе, если отладочные объекты не видны.
        /// </summary>
        private void PrepareEventVisual(int slot, SwPacState st, SwPacEvent ev, int dir, bool isHigher,
                                        SwPacScope scope, double level, bool filtered, int flag,
                                        string sid, double conf)
        {
            try
            {
                if (filtered && !Host.VisualDebugOn) return;
                bool wantLabel = Host.VisualOn(filtered ? SwVisualRole.StructureFiltered : SwVisualRole.StructureEvent);
                bool wantLine = Host.VisualOn(filtered ? SwVisualRole.BreakLevelFiltered : SwVisualRole.BreakLevel);
                if (!wantLabel && !wantLine) return;

                DateTime tBreak = Host.TimeOf(slot, 0);
                if (tBreak == DateTime.MinValue) return;

                PacEventVis v = new PacEventVis();
                v.Id = sid;
                v.Filtered = filtered;
                v.Flag = flag;
                v.Conf = conf;
                v.Dir = dir;
                v.Text = EventName(ev) + " " + (isHigher ? "СТ" : "МЛ") + "·" + (scope == SwPacScope.Swing ? "swg" : "int");
                v.TBreak = tBreak;
                v.Level = level;
                v.Hi = Host.HighOf(slot, 0);
                v.Lo = Host.LowOf(slot, 0);
                if (double.IsNaN(v.Hi) || double.IsNaN(v.Lo)) { v.Hi = level; v.Lo = level; }

                int pivotBar = dir > 0 ? st.LastHighBar : st.LastLowBar;
                int ago = Host.BarOf(slot) - pivotBar;
                v.TPivot = pivotBar >= 0 && ago >= 0 ? Host.TimeOf(slot, ago) : DateTime.MinValue;

                tickEventVis.Add(v);
            }
            catch { }
        }

        /// <summary>
        /// Метки и уровни событий такта. Отобранное событие — рабочий объект
        /// с главной подписью. Уступившие и отсеянные — отладочные: ранг
        /// подписи 1, если уверенность выше [D02], иначе 2.
        /// </summary>
        private void EmitEventVisuals(SwSignal s)
        {
            if (tickEventVis.Count == 0) return;

            double minConf = Host.MinConfidence;
            double step = 3 * Host.TickSize;

            foreach (PacEventVis v in tickEventVis)
            {
                bool selected = !v.Filtered && s.Direction != 0 && v.Id == pendingSignalId;
                SwVisualRole labelRole = v.Filtered ? SwVisualRole.StructureFiltered : SwVisualRole.StructureEvent;
                SwVisualRole lineRole = v.Filtered ? SwVisualRole.BreakLevelFiltered : SwVisualRole.BreakLevel;
                int rank = selected ? 0 : (!v.Filtered && v.Conf > minConf ? 1 : 2);

                if (Host.VisualOn(labelRole))
                {
                    double y = v.Dir > 0 ? v.Hi + step * (1 + labelsThisTick) : v.Lo - step * (1 + labelsThisTick);
                    labelsThisTick++;
                    SwVisualEvent e = SwVisualEvent.Label(labelRole, v.TBreak, y, v.Text).WithRank(rank);
                    if (!selected) e = e.AsDebug();
                    if (v.Filtered) e = e.WithDetail("[M" + v.Flag + " выкл]");
                    EmitVisual(e);
                }

                if (Host.VisualOn(lineRole) && v.TPivot != DateTime.MinValue)
                {
                    SwVisualEvent e = SwVisualEvent.Segment(lineRole, v.TPivot, v.Level, v.TBreak, v.Level);
                    if (!selected) e = e.AsDebug();
                    EmitVisual(e);
                }
            }
        }

        /// <summary>
        /// Незавершённая опорная точка ([V02]): бар среди последних k, который
        /// станет опорным, если следующие бары его не превысят. Правило то же,
        /// что при подтверждении: строго слева, нестрого справа. Кандидат
        /// может быть только один — бар с экстремумом среди последних k,
        /// самый ранний при равенстве; для него проверяется левая сторона.
        /// </summary>
        private int FindCandidate(int slot, int k, bool high)
        {
            int best = -1;
            double bv = 0;
            for (int j = 0; j < k; j++)
            {
                double v = high ? Host.HighOf(slot, j) : Host.LowOf(slot, j);
                if (double.IsNaN(v)) return -1;
                if (best < 0 || (high ? v >= bv : v <= bv)) { best = j; bv = v; }
            }
            if (best < 0 || !Host.SlotReady(slot, best + k + 1)) return -1;

            for (int i = k; i <= best + k; i++)
            {
                double v = high ? Host.HighOf(slot, i) : Host.LowOf(slot, i);
                if (double.IsNaN(v) || (high ? v >= bv : v <= bv)) return -1;
            }
            return best;
        }

        private void VisualCandidates(int slot, SwPacState st, bool isHigher, SwPacScope scope)
        {
            try
            {
                SwVisualRole role = scope == SwPacScope.Swing
                    ? SwVisualRole.PivotCandidateSwing : SwVisualRole.PivotCandidateInternal;
                SwVisualRole tfRole = isHigher
                    ? (scope == SwPacScope.Swing ? SwVisualRole.PivotSwingHigh : SwVisualRole.PivotInternalHigh)
                    : (scope == SwPacScope.Swing ? SwVisualRole.PivotSwing : SwVisualRole.PivotInternal);
                if (!Host.VisualOn(role) || !Host.VisualOn(tfRole)) return;

                string kb = "CAND_" + (isHigher ? "H" : "L") + (scope == SwPacScope.Swing ? "S" : "I");
                bool diamond = isHigher || scope == SwPacScope.Swing;
                ShowCandidate(slot, st.Strength, true, kb + "_HI", role, diamond);
                ShowCandidate(slot, st.Strength, false, kb + "_LO", role, diamond);
            }
            catch { }
        }

        /// <summary>Объект состояния по ключу: перерисовывается только при смене бара кандидата.</summary>
        private void ShowCandidate(int slot, int k, bool high, string key, SwVisualRole role, bool diamond)
        {
            int j = FindCandidate(slot, k, high);
            int shown;
            bool had = candShown.TryGetValue(key, out shown);

            if (j < 0)
            {
                if (had)
                {
                    EmitVisual(SwVisualEvent.RemoveState(key));
                    candShown.Remove(key);
                }
                return;
            }

            int barNo = Host.BarOf(slot) - j;
            if (had && shown == barNo) return;

            DateTime t = Host.TimeOf(slot, j);
            if (t == DateTime.MinValue) return;
            double pad = Host.TickSize;
            double px = high ? Host.HighOf(slot, j) + pad : Host.LowOf(slot, j) - pad;
            EmitVisual(SwVisualEvent.Marker(diamond ? SwVisualKind.Diamond : SwVisualKind.Dot, role, t, px).AsState(key));
            candShown[key] = barNo;
        }

        private void AddCandidate(string sid, int dir, double conf, string desc)
        {
            try
            {
                PacCandidate c = new PacCandidate();
                c.Id = sid;
                c.Dir = dir;
                c.Conf = conf;
                c.Desc = desc;
                tickCandidates.Add(c);
            }
            catch { }
        }

        private void JournalNotSent(DateTime t, string sid, int dir, double conf, string desc, string reason)
        {
            try { Host.RequestSignalJournalEntry(Id, t, sid, dir, conf, desc, reason); }
            catch { }
        }

        /// <summary>
        /// После отправки: стрелка отправленного сигнала, метки и уровни событий
        /// такта, записи журнала о событиях, уступивших отобранному.
        /// </summary>
        private void AfterEmit(SwSignal s)
        {
            try
            {
                if (s.Direction != 0)
                {
                    SwVisualRole role = s.Direction > 0 ? SwVisualRole.SignalLong : SwVisualRole.SignalShort;
                    if (Host.VisualOn(role))
                    {
                        double pad = 2 * Host.TickSize;
                        double y = s.Direction > 0 ? Host.Low(0) - pad : Host.High(0) + pad;
                        EmitVisual(SwVisualEvent.Marker(s.Direction > 0 ? SwVisualKind.ArrowUp : SwVisualKind.ArrowDown,
                            role, Host.Time, y));
                    }
                }

                // Отрисовка событий отдельно обёрнута: её сбой не лишает журнал записей.
                try { EmitEventVisuals(s); }
                catch { }

                foreach (PacCandidate c in tickCandidates)
                    if (c.Id != pendingSignalId)
                        JournalNotSent(Host.Time, c.Id, c.Dir, c.Conf, c.Desc,
                            "уступило событию выше рангом: " + (pendingShort.Length > 0 ? pendingShort : "—"));
            }
            catch { }
        }

        #endregion

        #region Диагностика

        /// <summary>Сводка состояния для панели и разбора.</summary>
        public string StateLine
        {
            get
            {
                return string.Format(
                    "МЛ int тренд {0} B{1} C{2} C+{3} | МЛ swg тренд {4} B{5} C{6} C+{7} | " +
                    "СТ swg тренд {8} B{9} C{10} C+{11}",
                    lowInternal.Trend, lowInternal.BosCount, lowInternal.ChochCount, lowInternal.ChochPlusCount,
                    lowSwing.Trend, lowSwing.BosCount, lowSwing.ChochCount, lowSwing.ChochPlusCount,
                    highSwing.Trend, highSwing.BosCount, highSwing.ChochCount, highSwing.ChochPlusCount);
            }
        }

        /// <summary>Действующее направление структуры старшего свинга.</summary>
        public int HigherSwingTrend { get { return highSwing.Trend; } }

        /// <summary>Действующее направление структуры младшего свинга.</summary>
        public int LowerSwingTrend { get { return lowSwing.Trend; } }

        #endregion
    }
}

## 1. Устройство

Одна стратегия — коммутатор `DZ_SW_Hub_S021` — и подключаемые модули `DZ_SW_Mod_*_S021`. Модули считают и голосуют, заявки отправляет только исполнитель внутри коммутатора. Модуль видит ядро через `SwHostView` — представление без ордерного API: вызвать `EnterLong` из модуля невозможно физически. Обмен — через ящики: модуль кладёт `SwSignal` или `SwAction`, коммутатор забирает и сверяет «отправлено — принято»: расхождение — потеря.

- Код NinjaScript сверяется с руководством NinjaTrader 8 со ссылкой на страницу.
- Статистика и проверки — по AFML, MLAM и банковской книге; уровни объёма — по Volume Profile.
