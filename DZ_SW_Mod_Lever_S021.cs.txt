//  v1.4 — РЕВИЗИЯ ВЕТКИ ИССЛЕДОВАНИЙ. Расчёт умолчаний не изменён. Эксперимент Э69, выключен: две
//    классические меры свечи как дополнительные условия согласия в роли Filter — CLR = |закрытие − открытие| /
//    диапазон не ниже ExpClrMin и z массы свечи относительно последних 20 свечей не ниже ExpVolZMin
//    (спецификация каскада от 2026-10-05; Volume Profile: тяжёлый объём = агрессивная инициация). Ниже порога —
//    вердикт Neutral вместо Agree.
//
//  v1.3 — ЭКСПЕРИМЕНТ Э22 [B21], выключен: мера POC. Volume Profile (Trader Dale,
//    с. 52–54): точка контроля — цена с наибольшим объёмом, «самое значимое место профиля»,
//    на неё цена реагирует сильнее всего. Баланс свечи — смещение POC от опоры в долях
//    плеча вместо медианы массы и рычага кластеров. Медиана устойчивее к хвостам, POC —
//    ближе к тому, что видит участник рынка; какая мера несёт информацию, покажет [B19].
//
//  v1.2 — ПОРЦИЯ 3 ОТЛАДКИ: РАСЧЁТ НЕ ИЗМЕНЁН. Индексы настроек в комментариях и
//    текстах приведены к меню: прежние [Lxx] совпадали с настройками классификатора
//    Лоренца — [L10] здесь был порогом голоса, а в меню [L10] — шум процесса Калмана.
//
//  v1.1 — РАСЧЁТ, ГОЛОС И СИГНАЛ НЕ ИЗМЕНЕНЫ. Наклон Тейла–Сена считается в
//  буферах модуля, выделенных один раз: прежде на каждой свече создавались
//  два списка и их копии для медиан — при окне [B13] = 200 это по 155 КБ в
//  куче больших объектов, дважды на свечу. Результат тот же: 18 000 сравнений
//  с v1.0 на прототипе Python без единого расхождения.
//
//  ИДЕЯ — ВЕСЫ-КАЧЕЛИ. Опора — геометрический центр закрытой свечи:
//  середина между экстремумами или середина тела [B08]. Свеча разбивается по
//  высоте на кластеры, в каждом — объём или число сделок [B07]. Где масса
//  богаче — туда и перевес:
//    богатые кластеры выше опоры — голос за продолжение вверх;
//    ниже опоры                 — голос за продолжение вниз;
//    масса у опоры              — середина: затухание движения, предпосылка
//                                 разворота или боковика.
//
//  ОТКУДА КЛАСТЕРЫ. Из более мелкой серии пула [B03]:
//    тиковая серия (1 тик)  — точный профиль: каждая сделка ложится на свою
//                             цену. Для воспроизведения рынка и реальной
//                             торговли — основной вариант;
//    минутная серия         — приближение: объём подбара делится поровну между
//                             ценами, которые он прошёл; в режиме «сделки» —
//                             число подбаров на цене, время у цены.
//  Алгоритм один: у тика высота нулевая, и его вес ложится на одну цену.
//
//  ВСЁ НА МЕДИАНАХ.
//    центр массы свечи   — взвешенная медиана цены по профилю, с интерполяцией
//                          внутри тика; смещение от опоры в долях плеча — «m»;
//    богатый кластер     — плотность выше медианной плотности кластеров этой
//                          свечи; рычаг «T» — момент избытка массы богатых
//                          кластеров на их плечи: Σ избыток·плечо / Σ избыток·|плечо|;
//    баланс свечи        — [B09]: среднее m и T, только T или только m;
//    сглаживание         — взвешенная медиана балансов последних [B10] свечей,
//                          веса — масса свечей: тяжёлая свеча весит больше;
//    масштаб             — 1.4826·MAD баланса за [B11] свечей; ноль не
//                          сдвигается: равновесие рычага — естественный ноль;
//    тренд для контекста — наклон Тейла–Сена (медиана попарных наклонов) по
//                          закрытиям [B13] свечей;
//    нормировка доходностей в исследовании — медианный размах свечи.
//  Голос: z = сглаженный баланс / масштаб; z ≥ [B05] — вверх, ≤ −[B05] —
//  вниз, иначе — середина. На случайном блуждании при пороге 0.5 это примерно
//  четверть свечей вверх, четверть вниз, половина середина (прототип Python).
//
//  КОГДА ГОЛОС ГОТОВ. Профиль свечи должен быть полон. Подбар, закрывшийся в
//  ту же секунду, что и свеча анализа, приходит в очереди тактов ПОСЛЕ неё.
//  Поэтому [B18]:
//    Exact — дождаться поздних кластеров; голос — в следующем такте исполнения;
//    Fast  — голос по закрытию свечи; кластеры с меткой времени её конца
//            отбрасываются (для тиков это доли процента объёма);
//    Auto  — Fast для тикового источника (подбары без высоты), иначе Exact.
//
//  ЧЕГО НЕТ. Взгляда в будущее: голос опирается только на кластеры закрытой
//  свечи и прошлые балансы. Будущие свечи — только в статистике отчёта.
//  Гипотеза «масса выше опоры → продолжение вверх» проверяется блоком
//  исследования [B19] на ваших данных: таблица доходностей по корзинам
//  рычага и медианного центра должна показать монотонность, если связь есть.
// ============================================================================

namespace NinjaTrader.NinjaScript
{
    /// <summary>Роль модуля рычага в решении.</summary>
    public enum SwLeverRole
    {
        Filter,     // вердикт по текущему голосу
        Signal      // новый голос вверх или вниз
    }

    /// <summary>Чем взвешивается кластер.</summary>
    public enum SwLeverWeight
    {
        Volume,     // объём
        Ticks       // число сделок; для минутного источника — время у цены
    }

    /// <summary>Опора рычага — геометрический центр свечи.</summary>
    public enum SwLeverFulcrum
    {
        RangeMid,   // середина между максимумом и минимумом
        BodyMid     // середина тела
    }

    /// <summary>Какой мерой голосует модуль.</summary>
    public enum SwLeverMeasure
    {
        Both,           // среднее рычага богатых кластеров и медианного центра
        Lever,          // только рычаг богатых кластеров
        MedianCenter    // только медианный центр массы
    }

    /// <summary>Вердикт роли Filter в середине.</summary>
    public enum SwLeverPolicy
    {
        DirectionOnly,  // середина не мешает
        BlockMiddle     // середина — вето: движение затухает
    }

    /// <summary>Когда профиль свечи считается полным.</summary>
    public enum SwLeverFinalize
    {
        Auto,
        Exact,
        Fast
    }

    /// <summary>Голос рычага.</summary>
    public enum SwLeverState
    {
        Warmup,
        Up,
        Down,
        Middle
    }
}

    /// <summary>Накопитель статистики события: число, средняя, доля в сторону события.</summary>
    public class SwLeverStat
    {
        public int N, Hits;
        public double Sum, Sum2;

        public void Add(double v)
        {
            if (double.IsNaN(v) || double.IsInfinity(v)) return;
            N++; Sum += v; Sum2 += v * v;
            if (v > 0) Hits++;
        }

        public double Mean { get { return N > 0 ? Sum / N : double.NaN; } }
        public double HitRate { get { return N > 0 ? (double)Hits / N : double.NaN; } }

        public double StdErr
        {
            get
            {
                if (N < 2) return double.NaN;
                double m = Sum / N;
                double v = (Sum2 - N * m * m) / (N - 1);
                return v > 0 ? Math.Sqrt(v / N) : 0;
            }
        }
    }

    public class SwLeverModule : SwModule
    {
        private readonly SwLeverConfig cfg;
        private bool srcOk;
        private double tick = 1e-6;

        // --- профиль ---
        private Dictionary<int, double> curProf = new Dictionary<int, double>();
        private Dictionary<int, double> pendProf = new Dictionary<int, double>();
        private bool pending;                       // закрытая свеча ждёт поздних кластеров
        private SwLeverCandle pendSnap;
        private bool firstCandle = true;            // первая свеча неполная: начало данных
        private DateTime lastSubTime = DateTime.MinValue;
        private DateTime lastClosedEnd = DateTime.MinValue;
        private long anaIndex;
        private int subSeen, subPoint;
        private bool modeTraced;
        private long lateDropped, candlesSkipped, candlesWeighed;
        private double outsideWeight, totalWeight;

        // --- ряды по свечам с данными ---
        private readonly List<double> hB = new List<double>();
        private readonly List<double> hW = new List<double>();
        private readonly List<double> hC = new List<double>();
        private readonly List<double> hR = new List<double>();
        private int cap = 64;

        // --- состояние ---
        private SwLeverState state = SwLeverState.Warmup;
        private double smooth = double.NaN, scale = double.NaN, z = double.NaN;
        private double biasMedian = double.NaN;
        private int trendDir;
        private SwLeverCandle last;
        private int lastClusters;

        // --- сигнал, выход ---
        private int pendDir, pendBarsLeft;
        private double pendConf;
        private string pendId = string.Empty, pendText = string.Empty, pendShort = string.Empty;
        private int exitAgainst;
        private bool exitArmed;
        private string exitReason = string.Empty;
        private int lastPosSide;
        private double lastAvg = double.NaN;
        private int exitShownBar = -1;

        // --- счётчики ---
        private int votesUp, votesDown, fades, suppressedTrend, exitsRequested;

        public SwLeverModule(string id, SwLeverConfig config)
        {
            Id = id;
            cfg = config ?? new SwLeverConfig();
            Role = cfg.Role == SwLeverRole.Signal ? SwRole.Signal : SwRole.Filter;
        }

        public override void Attach(SwHostView host)
        {
            base.Attach(host);
            tick = Host.TickSize > 0 ? Host.TickSize : 1e-6;
            cap = Math.Max(Math.Max(cfg.ScaleWindow, cfg.TrendWindow), Math.Max(cfg.SmoothCandles, 10)) + 4;

            // Кластеры должны идти из другой серии: невключённый слот пула читался
            // бы с графика, и свеча «взвешивала» бы сама себя.
            bool loaded = cfg.ClusterSlot == 0 || Host.SlotLoaded(cfg.ClusterSlot);
            srcOk = loaded && Host.Bip(cfg.ClusterSlot) != Host.Bip(cfg.AnalysisSlot);

            Host.Trace(Id, "LEVER_ATTACHED",
                "роль " + cfg.Role + ", свечи — слот " + cfg.AnalysisSlot + ", кластеры — слот " + cfg.ClusterSlot
                + (srcOk ? "" : " (НЕДОСТУПЕН)") + ", вес " + cfg.Weight + ", кластеров " + cfg.Clusters
                + ", опора " + cfg.Fulcrum + ", мера " + cfg.Measure + ", сглаживание " + cfg.SmoothCandles
                + ", масштаб " + cfg.ScaleWindow + ", порог z " + cfg.VoteZ.ToString("F2") + ", профиль " + cfg.Finalize);

            if (!srcOk)
                Host.Trace(Id, "LEVER_NO_SOURCE", "слот кластеров " + cfg.ClusterSlot
                    + " не загружен или совпадает с серией свечей: взвешивать нечем, модуль не готов");
        }

        public override void Compute(int bip)
        {
            if (srcOk)
            {
                if (Host.IsBarCloseOf(cfg.ClusterSlot, bip)) OnSubBar();
                if (Host.IsBarCloseOf(cfg.AnalysisSlot, bip)) OnAnalysisClose();
            }

            if (bip != Host.ExecBip) return;
            ExecTick();
        }

        #region Профиль свечи

        private bool FastMode()
        {
            if (cfg.Finalize == SwLeverFinalize.Fast) return true;
            if (cfg.Finalize == SwLeverFinalize.Exact) return false;
            // Auto: тиковый источник — подбары без высоты.
            bool fast = subSeen >= 200 && subPoint >= 0.95 * subSeen;
            if (!modeTraced && subSeen >= 200)
            {
                modeTraced = true;
                Host.Trace(Id, "LEVER_MODE", fast
                    ? "источник тиковый: профиль по закрытию свечи (Fast)"
                    : "источник с высотой подбаров: ожидание поздних кластеров (Exact), голос на такт позже");
            }
            return fast;
        }

        private bool TickLike()
        {
            return subSeen >= 50 && subPoint >= 0.95 * subSeen;
        }

        private void OnSubBar()
        {
            DateTime t = Host.TimeOf(cfg.ClusterSlot, 0);
            double h = Host.HighOf(cfg.ClusterSlot, 0);
            double l = Host.LowOf(cfg.ClusterSlot, 0);
            double v = Host.VolumeOf(cfg.ClusterSlot, 0);
            if (t == DateTime.MinValue || double.IsNaN(h) || double.IsNaN(l)) return;

            subSeen++;
            if (Math.Abs(h - l) < tick * 0.5) subPoint++;
            lastSubTime = t;

            if (pending)
            {
                if (t <= pendSnap.End)
                {
                    Accumulate(pendProf, h, l, v);
                    // У источника с высотой подбаров (минуты) в момент конца свечи
                    // кончается ровно один подбар: после него профиль полон. У тиков
                    // метку конца могут нести несколько сделок — ждём следующую.
                    if (t == pendSnap.End && !TickLike()) FinalizePending();
                    return;
                }
                // Первый подбар после конца свечи: её профиль полон.
                FinalizePending();
            }

            if (lastClosedEnd != DateTime.MinValue && t <= lastClosedEnd)
            {
                lateDropped++;
                return;
            }
            Accumulate(curProf, h, l, v);
        }

        /// <summary>
        /// Вес подбара — поровну на все цены, которые он прошёл. У тика высота
        /// нулевая: весь вес на одну цену.
        /// </summary>
        private void Accumulate(Dictionary<int, double> prof, double h, double l, double v)
        {
            int kl = (int)Math.Round(l / tick), kh = (int)Math.Round(h / tick);
            if (kh < kl) { int x = kl; kl = kh; kh = x; }
            int n = kh - kl + 1;
            if (n <= 0 || n > 5000) return;

            double w;
            if (cfg.Weight == SwLeverWeight.Volume)
            {
                if (!(v > 0)) return;
                w = v / n;
            }
            else w = 1.0;

            for (int k = kl; k <= kh; k++)
            {
                double cur;
                prof.TryGetValue(k, out cur);
                prof[k] = cur + w;
            }
        }

        private void OnAnalysisClose()
        {
            anaIndex++;

            // Запасной путь: поздний кластер так и не пришёл до следующей свечи.
            if (pending) FinalizePending();

            try { if (cfg.Research) ResolveResearch(); }
            catch { }

            SwLeverCandle c = new SwLeverCandle();
            c.Index = anaIndex;
            c.End = Host.TimeOf(cfg.AnalysisSlot, 0);
            c.Start = Host.TimeOf(cfg.AnalysisSlot, 1);
            c.Open = Host.OpenOf(cfg.AnalysisSlot, 0);
            c.High = Host.HighOf(cfg.AnalysisSlot, 0);
            c.Low = Host.LowOf(cfg.AnalysisSlot, 0);
            c.Close = Host.CloseOf(cfg.AnalysisSlot, 0);

            // Профиль закрытой свечи уходит в ожидающий, текущий начинается пустым.
            Dictionary<int, double> closed = curProf;
            curProf = pendProf;
            curProf.Clear();
            pendProf = closed;
            lastClosedEnd = c.End;

            if (firstCandle)
            {
                firstCandle = false;
                pendProf.Clear();
                return;
            }

            if (!FastMode() && lastSubTime < c.End)
            {
                pending = true;
                pendSnap = c;
                return;
            }

            FinalizeCandle(c, pendProf);
            pendProf.Clear();
        }

        private void FinalizePending()
        {
            pending = false;
            SwLeverCandle c = pendSnap;
            pendSnap = null;
            if (c != null) FinalizeCandle(c, pendProf);
            pendProf.Clear();
        }

        #endregion

        #region Взвешивание свечи

        private static double ArmOf(double p, double g, double armUp, double armDn)
        {
            double a = p >= g ? (p - g) / armUp : (p - g) / armDn;
            return a > 1 ? 1 : a < -1 ? -1 : a;
        }

        private void FinalizeCandle(SwLeverCandle c, Dictionary<int, double> prof)
        {
            if (double.IsNaN(c.High) || double.IsNaN(c.Low) || double.IsNaN(c.Close))
            {
                NoData(c, "нет цен свечи");
                return;
            }

            int kL = (int)Math.Round(c.Low / tick), kH = (int)Math.Round(c.High / tick);
            int nLev = kH - kL + 1;
            c.Levels = nLev;
            if (nLev < Math.Max(2, cfg.MinHeightTicks) || nLev > 5000)
            {
                NoData(c, "свеча ниже " + cfg.MinHeightTicks + " тиков [B17]");
                return;
            }

            double[] w = new double[nLev];
            double W = 0;
            foreach (KeyValuePair<int, double> kv in prof)
            {
                int j = kv.Key - kL;
                if (j < 0 || j >= nLev) { outsideWeight += kv.Value; continue; }
                w[j] += kv.Value;
                W += kv.Value;
            }
            if (!(W > 0))
            {
                NoData(c, "нет кластеров");
                return;
            }
            totalWeight += W;

            // --- опора и плечи ---
            double g = cfg.Fulcrum == SwLeverFulcrum.BodyMid ? (c.Open + c.Close) / 2.0 : (c.High + c.Low) / 2.0;
            double armUp = Math.Max(c.High - g, tick), armDn = Math.Max(g - c.Low, tick);

            // --- медиана массы с интерполяцией внутри тика ---
            double half = W / 2.0, cum = 0, p50 = c.High;
            for (int j = 0; j < nLev; j++)
            {
                if (w[j] > 0 && cum + w[j] >= half)
                {
                    p50 = (kL + j) * tick - tick / 2.0 + tick * (half - cum) / w[j];
                    break;
                }
                cum += w[j];
            }

            // --- кластеры по высоте ---
            int nc = Math.Max(1, Math.Min(Math.Max(2, cfg.Clusters), nLev));
            double[] dens = new double[nc], lo = new double[nc], hi = new double[nc], cen = new double[nc];
            int[] cnt = new int[nc];
            for (int b = 0; b < nc; b++)
            {
                int a0 = b * nLev / nc, a1 = (b + 1) * nLev / nc - 1;
                double mass = 0;
                for (int j = a0; j <= a1; j++) mass += w[j];
                cnt[b] = a1 - a0 + 1;
                dens[b] = mass / cnt[b];
                lo[b] = (kL + a0) * tick - tick / 2.0;
                hi[b] = (kL + a1) * tick + tick / 2.0;
                cen[b] = (kL + (a0 + a1) / 2.0) * tick;
            }
            double md = MedianOf(dens, nc);

            // --- рычаг богатых кластеров: момент избытка над медианной плотностью ---
            double num = 0, den = 0;
            bool[] rich = new bool[nc];
            for (int b = 0; b < nc; b++)
            {
                double e = Math.Max(0, dens[b] - md) * cnt[b];
                rich[b] = e > 0;
                double d = ArmOf(cen[b], g, armUp, armDn);
                num += e * d;
                den += e * Math.Abs(d);
            }

            c.HasData = true;
            c.Fulcrum = g;
            c.Center = p50;
            c.MedianOffset = ArmOf(p50, g, armUp, armDn);
            c.Lever = den > 0 ? num / den : 0;
            int jp = 0;
            for (int j = 1; j < nLev; j++) if (w[j] > w[jp]) jp = j;
            c.Poc = (kL + jp) * tick;
            c.PocOffset = ArmOf(c.Poc, g, armUp, armDn);
            c.Balance = cfg.ExpPocMeasure ? c.PocOffset     // Э22
                      : cfg.Measure == SwLeverMeasure.Lever ? c.Lever
                      : cfg.Measure == SwLeverMeasure.MedianCenter ? c.MedianOffset
                      : 0.5 * (c.Lever + c.MedianOffset);
            c.Weight = W;
            c.Clusters = nc;
            c.ClLo = lo; c.ClHi = hi; c.ClDensity = dens; c.ClRich = rich;
            c.MedianDensity = md;

            OnCandle(c);
        }

        private void NoData(SwLeverCandle c, string reason)
        {
            c.HasData = false;
            c.NoDataReason = reason;
            candlesSkipped++;
            last = c;
        }

        private static void PushCapped(List<double> l, double v, int cap)
        {
            l.Add(v);
            while (l.Count > cap) l.RemoveAt(0);
        }

        private static double MedianOf(double[] a, int n)
        {
            if (n <= 0) return double.NaN;
            double[] b = new double[n];
            Array.Copy(a, b, n);
            Array.Sort(b);
            return n % 2 == 1 ? b[n / 2] : 0.5 * (b[n / 2 - 1] + b[n / 2]);
        }

        /// <summary>Медиана последних n значений ряда.</summary>
        private static double TailMedian(List<double> l, int n)
        {
            int m = Math.Min(n, l.Count);
            if (m <= 0) return double.NaN;
            double[] a = new double[m];
            for (int i = 0; i < m; i++) a[i] = l[l.Count - m + i];
            return MedianOf(a, m);
        }

        /// <summary>Медианное абсолютное отклонение последних n значений.</summary>
        private static double TailMad(List<double> l, int n)
        {
            int m = Math.Min(n, l.Count);
            if (m <= 1) return double.NaN;
            double med = TailMedian(l, m);
            double[] a = new double[m];
            for (int i = 0; i < m; i++) a[i] = Math.Abs(l[l.Count - m + i] - med);
            return MedianOf(a, m);
        }

        /// <summary>Взвешенная медиана последних n значений: веса — масса свечей.</summary>
        private static double TailWeightedMedian(List<double> v, List<double> w, int n)
        {
            int m = Math.Min(n, Math.Min(v.Count, w.Count));
            if (m <= 0) return double.NaN;
            double[] vals = new double[m], wts = new double[m];
            double tot = 0;
            for (int i = 0; i < m; i++)
            {
                vals[i] = v[v.Count - m + i];
                wts[i] = Math.Max(0, w[w.Count - m + i]);
                tot += wts[i];
            }
            if (!(tot > 0)) return MedianOf(vals, m);
            Array.Sort(vals, wts);
            double half = tot / 2.0, cum = 0;
            for (int i = 0; i < m; i++)
            {
                cum += wts[i];
                if (cum >= half) return vals[i];
            }
            return vals[m - 1];
        }

        // Буферы Тейла–Сена: растут только при увеличении окна [B13], на свече
        // память не выделяется. Значения за пределами текущего окна не читаются.
        private double[] tsPairs = new double[0];
        private double[] tsDiffs = new double[0];

        /// <summary>
        /// Направление тренда по Тейлу–Сену: медиана попарных наклонов закрытий.
        /// Тренд засчитывается, если наклон за окно больше робастного шума
        /// одного шага, умноженного на √окна: как у блуждания за то же время.
        /// </summary>
        private int TheilSenDir(List<double> c, int n)
        {
            int m = Math.Min(Math.Max(3, n), c.Count);
            if (m < 3) return 0;
            int off = c.Count - m;
            int np = m * (m - 1) / 2, nd = m - 1;
            if (tsPairs.Length < np) tsPairs = new double[np];
            if (tsDiffs.Length < nd) tsDiffs = new double[nd];

            int k = 0;
            for (int i = 0; i < m; i++)
            {
                if (i > 0) tsDiffs[i - 1] = c[off + i] - c[off + i - 1];
                for (int j = i + 1; j < m; j++) tsPairs[k++] = (c[off + j] - c[off + i]) / (j - i);
            }

            double slope = SortedMedian(tsPairs, np);
            double med = SortedMedian(tsDiffs, nd);
            for (int i = 0; i < nd; i++) tsDiffs[i] = Math.Abs(tsDiffs[i] - med);
            double noise = 1.4826 * SortedMedian(tsDiffs, nd);

            if (double.IsNaN(slope) || double.IsNaN(noise)) return 0;
            if (Math.Abs(slope) * Math.Sqrt(m) < Math.Max(noise, 1e-12)) return 0;
            return Math.Sign(slope);
        }

        /// <summary>Медиана первых n значений массива; сортирует их на месте, без выделения памяти.</summary>
        private static double SortedMedian(double[] a, int n)
        {
            if (n <= 0) return double.NaN;
            Array.Sort(a, 0, n);
            return n % 2 == 1 ? a[n / 2] : 0.5 * (a[n / 2 - 1] + a[n / 2]);
        }

        private int WarmupCount { get { return Math.Max(10, Math.Min(cfg.ScaleWindow, 20)); } }

        private void OnCandle(SwLeverCandle c)
        {
            last = c;
            candlesWeighed++;
            PushCapped(hB, c.Balance, cap);
            PushCapped(hW, c.Weight, cap);
            PushCapped(hC, c.Close, cap);
            PushCapped(hR, c.High - c.Low, cap);

            SwLeverState prev = state;
            if (hB.Count < WarmupCount)
            {
                state = SwLeverState.Warmup;
                try { ObserveCandle(c, prev); }
                catch { }
                return;
            }

            smooth = TailWeightedMedian(hB, hW, Math.Max(1, cfg.SmoothCandles));
            double mad = TailMad(hB, Math.Max(5, cfg.ScaleWindow));
            scale = Math.Max(0.02, 1.4826 * (double.IsNaN(mad) ? 0 : mad));
            biasMedian = TailMedian(hB, Math.Max(5, cfg.ScaleWindow));
            z = smooth / scale;
            trendDir = TheilSenDir(hC, cfg.TrendWindow);

            double vz = Math.Max(0.01, cfg.VoteZ);
            state = z >= vz ? SwLeverState.Up : z <= -vz ? SwLeverState.Down : SwLeverState.Middle;

            if (state != prev)
                Host.Trace(Id, "LEVER_STATE", StateText(prev) + " → " + StateText(state) + ", баланс сглаж. "
                    + F2(smooth) + ", z " + F2(z) + ", рычаг " + F2(c.Lever) + ", медиана " + F2(c.MedianOffset)
                    + ", тренд Т–С " + (trendDir > 0 ? "↑" : trendDir < 0 ? "↓" : "нет"));

            bool newUp = state == SwLeverState.Up && prev != SwLeverState.Up;
            bool newDn = state == SwLeverState.Down && prev != SwLeverState.Down;
            bool fade = state == SwLeverState.Middle && (prev == SwLeverState.Up || prev == SwLeverState.Down);
            if (newUp) votesUp++;
            if (newDn) votesDown++;
            if (fade) fades++;

            if ((newUp || newDn) && Role == SwRole.Signal) ArmSignal(c, newUp ? 1 : -1);
            StepExit();

            try { if (cfg.Research) RegisterResearch(c, prev, newUp, newDn, fade); }
            catch { }
            try { ObserveCandle(c, prev); }
            catch { }
        }

        private void ArmSignal(SwLeverCandle c, int dir)
        {
            string sid = "LEV_" + c.End.ToString("yyyyMMdd_HHmmss", System.Globalization.CultureInfo.InvariantCulture)
                       + (dir > 0 ? "_U" : "_D");
            string desc = "РЫЧ масса " + (dir > 0 ? "выше" : "ниже") + " опоры";
            double vz = Math.Max(0.01, cfg.VoteZ);
            double conf = Math.Min(1.0, 0.5 + 0.25 * Math.Abs(z) / vz);

            if (cfg.SignalWithTrend && trendDir != dir)
            {
                suppressedTrend++;
                try
                {
                    Host.RequestSignalJournalEntry(Id, c.End, sid, dir, conf, desc,
                        "тренд Тейла–Сена " + (trendDir > 0 ? "вверх" : trendDir < 0 ? "вниз" : "не выражен") + " [B12]");
                }
                catch { }
                return;
            }

            pendDir = dir;
            pendConf = conf;
            pendBarsLeft = Math.Max(1, cfg.SignalBars);
            pendId = sid;
            pendShort = desc;
            pendText = desc + ", z " + F2(z) + ", рычаг " + F2(c.Lever) + ", медиана " + F2(c.MedianOffset);
            Host.Trace(Id, "LEVER_VOTE", pendText + ", уверенность " + conf.ToString("F2"));
        }

        /// <summary>
        /// Против позиции на закрытой свече: голос на противоположной стороне
        /// [B16] свечей подряд. Требование повторяется каждый такт, пока позиция
        /// открыта: исполнитель пропускает не больше одного выхода на бар.
        /// </summary>
        private void StepExit()
        {
            int pos = Host.PositionSide;
            if (!cfg.ExitAgainst || pos == 0)
            {
                exitAgainst = 0;
                return;
            }
            bool against = (pos > 0 && state == SwLeverState.Down) || (pos < 0 && state == SwLeverState.Up);
            if (against)
            {
                exitAgainst++;
                if (exitAgainst >= Math.Max(1, cfg.ExitConfirm) && !exitArmed)
                {
                    exitArmed = true;
                    exitReason = "Рычаг: масса " + (pos > 0 ? "ниже" : "выше") + " опоры " + exitAgainst + " св. подряд, z " + F2(z);
                    Host.Trace(Id, "LEVER_EXIT_REQUESTED", exitReason);
                }
            }
            else
            {
                exitAgainst = 0;
                exitArmed = false;
            }
        }

        private void ExecTick()
        {
            int bar = Host.CurrentBar;
            int pos = Host.PositionSide;

            double avg = Host.PositionAvgPrice;
            bool newTrade = pos != lastPosSide
                || (pos != 0 && !double.IsNaN(lastAvg) && Math.Abs(avg - lastAvg) > 1e-9);
            if (newTrade)
            {
                exitAgainst = 0;
                exitArmed = false;
            }
            lastPosSide = pos;
            lastAvg = pos != 0 ? avg : double.NaN;

            if (pos != 0 && exitArmed) EmitExit(bar, pos);

            if (Role == SwRole.Signal) EmitSignalRole(bar);
            else EmitFilterRole(bar);
        }

        private void EmitExit(int bar, int pos)
        {
            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.FullExit;
            a.Reason = exitReason;
            Emit(a);
            exitsRequested++;

            try
            {
                if (exitShownBar != bar && Host.VisualOn(SwVisualRole.LeverVote))
                {
                    exitShownBar = bar;
                    double y = pos > 0 ? Host.High(0) + 5 * Host.TickSize : Host.Low(0) - 5 * Host.TickSize;
                    EmitVisual(SwVisualEvent.Label(SwVisualRole.LeverVote, Host.Time, y, "LEV EXIT").WithVariant(-pos));
                }
            }
            catch { }
        }

        private bool Ready { get { return srcOk && state != SwLeverState.Warmup; } }

        private void EmitSignalRole(int bar)
        {
            if (!Ready)
            {
                Health = SwHealth.Warmup;
                SwSignal w = SwSignal.Empty(bar);
                w.Text = WarmupText();
                Emit(w);
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;

            if (pendBarsLeft > 0)
            {
                pendBarsLeft--;
                s.Direction = pendDir;
                s.Confidence = pendConf;
                s.SignalId = pendId;
                s.ShortDescription = pendShort;
                s.Text = pendText;
            }
            else
            {
                s.Direction = 0;
                s.Text = StatusShort();
            }

            Health = SwHealth.Ok;
            Emit(s);
        }

        private void EmitFilterRole(int bar)
        {
            if (!Ready)
            {
                Health = SwHealth.Warmup;
                SwSignal w = SwSignal.Empty(bar);
                w.Text = WarmupText();
                Emit(w);
                return;
            }

            SwSignal s = new SwSignal();
            s.IsReady = true;
            s.ProducedAtBar = bar;
            s.Direction = 0;
            s.EntryStopPrice = double.NaN;
            s.TrailTicksOverride = double.NaN;
            s.Confidence = Math.Min(1.0, Math.Abs(z) / Math.Max(0.02, 2 * cfg.VoteZ));

            switch (state)
            {
                case SwLeverState.Up:
                    s.Verdict = SwVerdict.Agree; s.Direction = 1;
                    break;
                case SwLeverState.Down:
                    s.Verdict = SwVerdict.Agree; s.Direction = -1;
                    break;
                default:
                    s.Verdict = cfg.Policy == SwLeverPolicy.BlockMiddle ? SwVerdict.Veto : SwVerdict.Neutral;
                    break;
            }

            string gate = string.Empty;
            if (s.Verdict == SwVerdict.Agree && last != null)                                   // Э69
            {
                if (cfg.ExpClrMin > 0)
                {
                    double rng = last.High - last.Low;
                    double clr = rng > 0 ? Math.Abs(last.Close - last.Open) / rng : 0.0;
                    if (clr < cfg.ExpClrMin) { s.Verdict = SwVerdict.Neutral; s.Direction = 0; gate = "CLR " + clr.ToString("F2") + " ниже порога; "; }
                }
                if (cfg.ExpVolZMin > 0 && s.Verdict == SwVerdict.Agree && hW.Count >= 5)
                {
                    int n = Math.Min(20, hW.Count); double m = 0, v = 0;
                    for (int i = hW.Count - n; i < hW.Count; i++) m += hW[i];
                    m /= n;
                    for (int i = hW.Count - n; i < hW.Count; i++) v += (hW[i] - m) * (hW[i] - m);
                    double sd = Math.Sqrt(v / Math.Max(1, n - 1));
                    double zv = sd > 0 ? (last.Weight - m) / sd : 0.0;
                    if (zv < cfg.ExpVolZMin) { s.Verdict = SwVerdict.Neutral; s.Direction = 0; gate = "z объёма " + zv.ToString("F2") + " ниже порога; "; }
                }
            }
            s.Text = (s.Verdict == SwVerdict.Veto ? "вето: середина, движение затухает; " : "") + gate + StatusShort();
            Health = SwHealth.Ok;
            Emit(s);
        }

            return "прогрев: свечей со взвешиванием " + hB.Count + " из " + WarmupCount;
        }

        private static int Bin(double x)
        {
            return x < -0.6 ? 0 : x < -0.2 ? 1 : x <= 0.2 ? 2 : x <= 0.6 ? 3 : 4;
        }

        private void AddEval(int type, int dir, double start, double norm, long due)
        {
            if (double.IsNaN(start) || !(norm > 0) || pend.Count >= 20000) return;
            PendingEval e = new PendingEval();
            e.Type = type; e.Dir = dir; e.Start = start; e.Norm = norm; e.Due = due;
            pend.Add(e);
        }

        /// <summary>
        /// Свеча взвешена: регистрируются оценки. Доходность нормируется на
        /// медианный размах свечи — робастная мера типичной свечи участка.
        /// </summary>
        private void RegisterResearch(SwLeverCandle c, SwLeverState prev, bool newUp, bool newDn, bool fade)
        {
            double norm = Math.Max(TailMedian(hR, Math.Max(5, cfg.ScaleWindow)), 2 * tick);
            long due = c.Index + Math.Max(1, cfg.ResearchHorizon);
            double p = c.Close;

            AddEval(0, 1, p, norm, due);
            AddEval(100, 0, p, norm, due);   // база модуля хода

            if (state == SwLeverState.Up) AddEval(1, 1, p, norm, due);
            if (state == SwLeverState.Down) AddEval(2, -1, p, norm, due);
            if (newUp) AddEval(3, 1, p, norm, due);
            if (newDn) AddEval(4, -1, p, norm, due);
            if (fade) AddEval(prev == SwLeverState.Up ? 5 : 6, prev == SwLeverState.Up ? -1 : 1, p, norm, due);

            if (state == SwLeverState.Up && trendDir != 0) AddEval(trendDir > 0 ? 7 : 8, 1, p, norm, due);
            if (state == SwLeverState.Down && trendDir != 0) AddEval(trendDir < 0 ? 9 : 10, -1, p, norm, due);

            AddEval(11 + Bin(c.Lever), 1, p, norm, due);
            AddEval(16 + Bin(c.MedianOffset), 1, p, norm, due);

            if (state == SwLeverState.Middle) AddEval(101, 0, p, norm, due);
            if (state == SwLeverState.Up || state == SwLeverState.Down) AddEval(102, 0, p, norm, due);
        }

        /// <summary>Закрытие свечи анализа: созревшие оценки получают результат.</summary>
        private void ResolveResearch()
        {
            double p = Host.CloseOf(cfg.AnalysisSlot, 0);
            if (double.IsNaN(p)) return;
            for (int k = pend.Count - 1; k >= 0; k--)
            {
                PendingEval e = pend[k];
                if (e.Due > anaIndex) continue;
                double r = (p - e.Start) / e.Norm;
                if (e.Type >= 100)
                {
                    int mi = e.Type - 100;
                    if (mi >= 0 && mi < mags.Length) mags[mi].Add(Math.Abs(r));
                }
                else if (e.Type >= 0 && e.Type < stats.Length)
                {
                    stats[e.Type].Add(e.Dir * r);
                    stDirSum[e.Type] += e.Dir;
                }
                pend.RemoveAt(k);
            }
        }

        /// <summary>
        /// Центр массы — отрезок на уровне медианы массы через всю свечу, цвет —
        /// сторона перевеса этой свечи. Кластеры — только у последней взвешенной
        /// свечи: прямоугольники, непрозрачность по плотности. Опора — отладка.
        /// </summary>
        private void ObserveCandle(SwLeverCandle c, SwLeverState prev)
        {
            if (!c.HasData || c.Start == DateTime.MinValue) return;

            double sc = double.IsNaN(scale) ? 0.25 : scale;
            double vz = Math.Max(0.01, cfg.VoteZ);
            int side = c.Balance / sc >= vz ? 1 : c.Balance / sc <= -vz ? -1 : 0;

            if (Host.VisualOn(SwVisualRole.LeverCenter))
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.LeverCenter, c.Start, c.Center, c.End, c.Center)
                    .WithVariant(side).InRing("LEVC"));

            if (Host.VisualOn(SwVisualRole.LeverFulcrum) && Host.VisualDebugOn)
                EmitVisual(SwVisualEvent.Segment(SwVisualRole.LeverFulcrum, c.Start, c.Fulcrum, c.End, c.Fulcrum)
                    .AsDebug().InRing("LEVG"));

            if (Host.VisualOn(SwVisualRole.LeverCluster))
            {
                double maxD = 0;
                for (int b = 0; b < c.Clusters; b++) if (c.ClDensity[b] > maxD) maxD = c.ClDensity[b];
                for (int b = 0; b < c.Clusters; b++)
                {
                    int op = maxD > 0 ? 8 + (int)Math.Round(52.0 * c.ClDensity[b] / maxD) : 8;
                    double mid = (c.ClLo[b] + c.ClHi[b]) / 2.0;
                    int v = c.ClRich[b] ? (mid >= c.Fulcrum ? 1 : -1) : 0;
                    EmitVisual(SwVisualEvent.Rect(SwVisualRole.LeverCluster, c.Start, c.ClLo[b], c.End, c.ClHi[b])
                        .AsState("CL" + b).WithVariant(v).WithOpacity(op));
                }
                for (int b = c.Clusters; b < lastClusters; b++) EmitVisual(SwVisualEvent.RemoveState("CL" + b));
                lastClusters = c.Clusters;
            }

            if (state == prev || prev == SwLeverState.Warmup) return;
            double pad = 3 * tick;
            if ((state == SwLeverState.Up || state == SwLeverState.Down) && Host.VisualOn(SwVisualRole.LeverVote))
            {
                bool up = state == SwLeverState.Up;
                EmitVisual(SwVisualEvent.Marker(up ? SwVisualKind.ArrowUp : SwVisualKind.ArrowDown,
                        SwVisualRole.LeverVote, c.End, up ? c.Low - pad : c.High + pad)
                    .WithVariant(up ? 1 : -1).InRing("LEVV"));
            }
            else if (state == SwLeverState.Middle && (prev == SwLeverState.Up || prev == SwLeverState.Down)
                     && Host.VisualOn(SwVisualRole.LeverFade))
            {
                EmitVisual(SwVisualEvent.Label(SwVisualRole.LeverFade, c.End, c.High + 2 * pad, "затухание")
                    .WithRank(1).InRing("LEVF"));
            }
        }

        #region Отчёт и панель

        public string ResearchReport(string pad)
        {
            StringBuilder sb = new StringBuilder();
            sb.AppendLine(pad + "Рычаг: свечей взвешено " + candlesWeighed + ", пропущено " + candlesSkipped
                + ", поздних кластеров отброшено " + lateDropped
                + (totalWeight > 0 ? ", масса вне диапазона свечи " + (100.0 * outsideWeight / (totalWeight + outsideWeight)).ToString("F2") + " %" : "")
                + "; голосов ↑" + votesUp + " ↓" + votesDown + ", затуханий " + fades
                + ", не отправлено против тренда " + suppressedTrend + ", требований выхода " + exitsRequested);
            if (!double.IsNaN(biasMedian))
                sb.AppendLine(pad + "Рычаг: медиана баланса за окно " + F2(biasMedian) + ", масштаб 1.4826·MAD " + F2(scale)
                    + " — заметный сдвиг медианы от нуля говорит о смещении метода на этом инструменте");
            if (!cfg.Research) return sb.ToString();

            SwLeverStat b = stats[0];
            double baseMean = b.Mean;
            sb.AppendLine(pad + "Рычаг, исследование: горизонт " + Math.Max(1, cfg.ResearchHorizon)
                + " свечей анализа, доходность в медианных размахах свечи, знак — в сторону события");
            sb.AppendLine(pad + "  событие; n; средняя; ст. ошибка; в сторону, %; средняя минус база");
            for (int i = 0; i < stats.Length; i++)
            {
                SwLeverStat s = stats[i];
                if (s.N == 0) continue;
                double avgDir = (double)stDirSum[i] / Math.Max(1, s.N);
                double excess = i == 0 || double.IsNaN(baseMean) ? double.NaN : s.Mean - avgDir * baseMean;
                sb.AppendLine(pad + "  " + EventNames[i] + "; " + s.N + "; " + F3(s.Mean) + "; " + F3(s.StdErr) + "; "
                    + (double.IsNaN(s.HitRate) ? "—" : (100 * s.HitRate).ToString("F1")) + "; " + F3(excess));
            }
            sb.AppendLine(pad + "  модуль хода: событие; n; средний |ход|; ст. ошибка");
            for (int i = 0; i < mags.Length; i++)
            {
                if (mags[i].N == 0) continue;
                sb.AppendLine(pad + "  " + MagNames[i] + "; " + mags[i].N + "; " + F3(mags[i].Mean) + "; " + F3(mags[i].StdErr));
            }
            sb.AppendLine(pad + "  как читать: связь «масса выше опоры → рост» есть, если средние по корзинам T или m растут"
                + " слева направо и крайние корзины отличаются от базы больше чем на две ст. ошибки; затухание подтверждается,"
                + " если |ход| после середины меньше базового");
            return sb.ToString();
        }

        public string StatusLine
        {
            get
            {
                StringBuilder sb = new StringBuilder("рычаг: ");
                if (!srcOk) { sb.Append("нет серии кластеров [B03]"); return sb.ToString(); }
                SwLeverCandle c = last;
                if (c == null) { sb.Append("ожидание первой свечи"); return sb.ToString(); }
                sb.Append(c.End.ToString("HH:mm")).Append(" · ");
                if (!c.HasData) sb.Append("свеча не взвешена: ").Append(c.NoDataReason).Append(" · ");
                else sb.Append("рычаг ").Append(F2(c.Lever)).Append(", медиана ").Append(F2(c.MedianOffset))
                       .Append(", кластеров ").Append(c.Clusters).Append(" · ");
                sb.Append(StatusShort());
                return sb.ToString();
            }
        }

        private string StatusShort()
        {
            if (state == SwLeverState.Warmup) return WarmupText();
            return "голос " + StateText(state) + ", сглаж. " + F2(smooth) + ", z " + F2(z)
                 + ", тренд Т–С " + (trendDir > 0 ? "↑" : trendDir < 0 ? "↓" : "нет");
        }

        private static string StateText(SwLeverState s)
        {
            switch (s)
            {
                case SwLeverState.Up: return "↑ масса выше";
                case SwLeverState.Down: return "↓ масса ниже";
                case SwLeverState.Middle: return "середина";
                default: return "прогрев";
            }
        }

        private static string F2(double v)
        {
            return double.IsNaN(v) ? "—" : v.ToString("+0.00;-0.00;0.00");
        }

        private static string F3(double v)
        {
            return double.IsNaN(v) ? "—" : v.ToString("+0.000;-0.000;0.000");
        }
