                        Order o = Orders[i];
                        if (o == null) continue;
                        if (o.OrderState == OrderState.Working || o.OrderState == OrderState.Accepted
                            || o.OrderState == OrderState.Submitted)

                            if (o.Name != null && o.Name.StartsWith("SW_EXIT")) return true;

    /// <summary>
    /// ЕДИНСТВЕННЫЙ КЛАСС, ВЫСТАВЛЯЮЩИЙ ЗАЯВКИ.
    ///
    /// Ни ядро, ни подключаемые модули ордерных вызовов не содержат. Все
    /// обращения к EnterLong, ExitLong и SetStopLoss проходят здесь и нигде
    /// более. Это проверяемо машинным поиском по файлу.
    ///
    /// Арбитраж встроен: один выход на один бар серии исполнения, кто бы его
    /// ни инициировал. Слот такта занимает первый успевший источник.
    /// </summary>
    /// <summary>
    /// v1.8. Общая математика ядра. Φ(z) — функция нормального распределения через erfc
    /// (Numerical Recipes); тот же алгоритм, что в модуле сигнала ПК v1.2, сверен со scipy
    /// до 4·10⁻⁸ на [−8, 8]. При следующей обработке модуля сигнала ПК он перейдёт на эту копию.
    /// </summary>
    public static class SwMath
    {
        public static double Phi(double z)
        {
            double x = -z / Math.Sqrt(2.0), a = Math.Abs(x), t = 1.0 / (1.0 + 0.5 * a);
            double e = t * Math.Exp(-a * a - 1.26551223 + t * (1.00002368 + t * (0.37409196 + t * (0.09678418
                     + t * (-0.18628806 + t * (0.27886807 + t * (-1.13520398 + t * (1.48851587
                     + t * (-0.82215223 + t * 0.17087277)))))))));
            return 0.5 * (x >= 0 ? e : 2.0 - e);
        }
    }

        public SwExecutionModule(DZ_SW_Hub_S021 h) { hub = h; }

        /// <summary>Разбор метки вида STEP3. Ноль означает «не ступень».</summary>
        private static int ParseStepTag(string tag)
        {
            if (string.IsNullOrEmpty(tag)) return 0;
            int i = tag.IndexOf("STEP", StringComparison.Ordinal);
            if (i < 0) return 0;
            int num;
            return int.TryParse(tag.Substring(i + 4), out num) ? num : 0;
        }

        /// <summary>Фактический стоп. Все прочие копии обязаны читать его отсюда.</summary>
        public double CurrentStop { get { return lastStop; } }

        public string SummaryText
        {
            get
            {
                return "входов " + EntriesSent + ", выходов " + ExitsSent
                     + ", переносов стопа " + StopMoves + ", отклонено " + Rejected
                     + ", блокировок арбитража " + ArbitrationBlocks;
            }
        }

        /// <summary>Запрос входа от ядра.</summary>
        public int StopEntriesArmed, StopEntryRefreshes;

        /// <summary>Вызывается ядром при исполнении виртуального стоп-входа.</summary>
        /// <summary>
        /// v1.4. Вход исполнен в бою: стоп выставлен платформой в тиках от средней цены
        /// входа, и фактический уровень известен только теперь. Единственный источник
        /// текущего стопа — исполнитель (Д-11), поэтому он обновляется здесь.
        /// </summary>
        private bool desyncExitTraced;

        /// <summary>
        /// v1.6. Аварийный выход с пульта: вся позиция рыночной заявкой сразу, без арбитража
        /// «минимум баров» и «один выход на бар» — но не поверх уже работающего выхода и не
        /// при рассинхроне, где выход стратегии перевернул бы счёт.
        /// </summary>
        public void EmergencyFlatten(string reason, int bar)
        {
            int pos = hub.HubPositionSide, qty = hub.HubPositionQty;
            if (pos == 0 || qty <= 0) { hub.HubTrace("EXEC", "REMOTE_FLAT_NOOP", "позиции нет"); return; }
            if (!hub.HubExitsSafe)
            {
                hub.HubTrace("EXEC", "REMOTE_FLAT_REFUSED", "позиция стратегии не совпадает со счётом: закрывайте вручную или аварийной кнопкой");
                return;
            }
            if (hub.HubExitPending) { hub.HubTrace("EXEC", "REMOTE_FLAT_PENDING", "выход уже работает"); return; }
            if (hub.HubDryRun)
            {
                ExitsSent++; lastExitBar = bar;
                hub.HubVirtualQueueExit(true, qty, 0, reason, bar);
                StatusText = "DRY " + reason + " — по открытию следующего бара";
                hub.HubTrace("EXEC", "DRY_REMOTE_EXIT", StatusText);
                return;
