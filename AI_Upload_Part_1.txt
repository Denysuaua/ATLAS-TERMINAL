"""Базовый класс баров, которые строятся по тикам (B04 гл.2, §2.3, с.25–32, PDF 52–59).

Тик — строка (date_time, price, volume). Бар закрывается по правилу подкласса;
незавершённые тики переносятся в следующий пакет, поэтому результат не зависит
от размера пакета (тест). Последний незавершённый бар в выход не попадает.

Столбцы выхода (как в каркасе): date_time — время последнего тика бара (для
баров времени — правая граница интервала), tick_num — сквозной номер
последнего тика, open, high, low, close, volume, cum_buy_volume — объём тиков
со знаком +1 по правилу тика, cum_ticks, cum_dollar_value — Σ price·volume.

Ссылки на источники — по каталогу библиотеки: B04 = AFML, страницы «печатная
(PDF)».
"""

from __future__ import annotations

import math
import numbers
import os
import sys
from collections import deque
from typing import Any, Iterable, Iterator

import numpy as np
import pandas as pd

from dz_afml.util.fast_ewma import ewma
from dz_afml.util.misc import crop_data_frame_in_batches

BAR_COLUMNS = ["date_time", "tick_num", "open", "high", "low", "close", "volume",
               "cum_buy_volume", "cum_ticks", "cum_dollar_value"]
TICK_COLUMNS = ["date_time", "price", "volume"]
_NO_WRITE = ("запись файлов библиотекой не выполняется (ТЗ, п.4): верните DataFrame "
             "и сохраните его сами")


def _crop_data_frame_in_batches(df: pd.DataFrame, chunksize: int) -> list[pd.DataFrame]:
    """Режет таблицу тиков на пакеты по chunksize строк.

    То же, что util.misc.crop_data_frame_in_batches: опись каркаса держит
    копию функции в base_bars. Статус: Недостаточно материалов (каркас не
    приложен; поведение однозначно из имени и проверено тестами).
    """
    return crop_data_frame_in_batches(df, chunksize)


def _positive_int(value: Any, name: str) -> int:
    if isinstance(value, bool) or not isinstance(value, numbers.Real) or value < 1:
        raise ValueError(f"{name} должно быть числом >= 1")
    return int(value)


class BaseBars:
    """Общая часть баров по тикам: чтение пакетами, правило тика, сборка строк.

    Входы конструктора:
        metric: что накапливается — 'cum_ticks', 'cum_volume',
            'cum_dollar_value' (для баров времени — None).
        batch_size: строк в пакете при чтении CSV и нарезке таблицы.

    Источник: B04 §2.3, с.25–32 (PDF 52–59): бары времени, тиков, объёма,
        денег; правило тика — §2.3.2.1, с.29 (PDF 56).
    Статус: Подтверждено по официальному учебнику.
    Причинность: бар собирается только из своих тиков; закрытие бара известно
        в момент его последнего тика (для баров времени — на правой границе).
    """

    METRICS = (None, "cum_ticks", "cum_volume", "cum_dollar_value",
               "tick_imbalance", "volume_imbalance", "dollar_imbalance",
               "tick_run", "volume_run", "dollar_run")

    def __init__(self, metric: str | None, batch_size: float = 2e7) -> None:
        if metric not in self.METRICS:
            raise ValueError(f"metric должен быть одним из {self.METRICS[1:]}")
        self.metric = metric
        self.batch_size = _positive_int(batch_size, "batch_size")
        self.prev_price: float | None = None
        self.prev_tick_rule = 0.0
        self.tick_num = 0
        self._pending: dict | None = None
        self._csv_header = False
        self._reset_cache()

    # --- чтение -----------------------------------------------------------
    def batch_run(self, file_path_or_df: Any, verbose: bool = True, to_csv: bool = False,
                  output_path: str | None = None) -> pd.DataFrame:
        """Строит бары по всему источнику пакетами и возвращает таблицу баров.

        Входы: file_path_or_df — DataFrame с тремя столбцами (время, цена,
            объём), список таких таблиц, путь к CSV или файловый объект;
            verbose — печатать номер пакета в stderr; to_csv, output_path —
            запись в файл не поддерживается (ValueError, ТЗ п.4).
        Выход: pd.DataFrame со столбцами BAR_COLUMNS.
        """
        if to_csv or output_path is not None:
            raise ValueError(_NO_WRITE)
        parts = []
        for number, batch in enumerate(self._batch_iterator(file_path_or_df), 1):
            if verbose:
                sys.stderr.write(f"пакет {number}: {len(batch)} строк\n")
            parts.append(self.run(batch))
        parts = [p for p in parts if len(p)]
        if not parts:
            return pd.DataFrame(columns=BAR_COLUMNS)
        return pd.concat(parts, ignore_index=True)

    def _batch_iterator(self, file_path_or_df: Any) -> Iterator[pd.DataFrame]:
        """Отдаёт пакеты тиков из таблицы, списка таблиц или CSV."""
        if isinstance(file_path_or_df, pd.DataFrame):
            if len(file_path_or_df):
                self._assert_csv(file_path_or_df.iloc[:1])
            yield from _crop_data_frame_in_batches(file_path_or_df, self.batch_size)
            return
        if isinstance(file_path_or_df, (list, tuple)):
            for frame in file_path_or_df:
                if len(frame):
                    self._assert_csv(frame.iloc[:1])
                yield from _crop_data_frame_in_batches(frame, self.batch_size)
            return
        first = self._read_first_row(file_path_or_df)
        self._assert_csv(first)
        source = file_path_or_df if hasattr(file_path_or_df, "read") else os.fspath(file_path_or_df)
        reader = pd.read_csv(source, header=0 if self._csv_header else None,
                             names=TICK_COLUMNS, chunksize=self.batch_size)
        yield from reader

    def _read_first_row(self, file_path: Any) -> pd.DataFrame:
        """Читает первую строку данных CSV; заголовок распознаётся по нечисловой цене."""
        source = file_path if hasattr(file_path, "read") else os.fspath(file_path)
        head = pd.read_csv(source, header=None, nrows=2)
        if hasattr(file_path, "seek"):
            file_path.seek(0)
        try:
            float(head.iloc[0, 1])
            self._csv_header = False
            first = head.iloc[:1]
        except (TypeError, ValueError):
            self._csv_header = True
            first = head.iloc[1:2]
        return first

    @staticmethod
    def _assert_csv(test_batch: pd.DataFrame) -> None:
        """Проверяет формат тиков: три столбца — время, цена, объём."""
        if test_batch.shape[1] != 3:
            raise ValueError("нужно ровно три столбца: date_time, price, volume")
        try:
            pd.to_datetime(test_batch.iloc[:, 0])
        except (TypeError, ValueError) as err:
            raise ValueError("первый столбец должен быть временем") from err
        for col, name in ((1, "price"), (2, "volume")):
            try:
                float(test_batch.iloc[0, col])
            except (TypeError, ValueError) as err:
                raise ValueError(f"столбец {name} должен быть числом") from err

    # --- расчёт -------------------------------------------------------------
    def run(self, data: pd.DataFrame) -> pd.DataFrame:
        """Строит бары по одному пакету тиков, продолжая состояние прошлых пакетов."""
        frame = data.copy()
        frame.columns = TICK_COLUMNS
        times = pd.DatetimeIndex(pd.to_datetime(frame["date_time"]))
        if len(times) > 1 and not times.is_monotonic_increasing:
            raise ValueError("время тиков не должно убывать")
        prices = frame["price"].to_numpy(dtype=np.float64)
        volumes = frame["volume"].to_numpy(dtype=np.float64)
        if not (np.all(np.isfinite(prices)) and np.all(np.isfinite(volumes))):
            raise ValueError("в ценах или объёмах есть пропуски")
        if (prices <= 0).any() or (volumes < 0).any():
            raise ValueError("цены должны быть > 0, объём >= 0")
        rows = self._extract_bars({"times": times, "prices": prices, "volumes": volumes})
        return pd.DataFrame(rows, columns=BAR_COLUMNS)

    def _extract_bars(self, data: dict) -> pd.DataFrame:
        raise NotImplementedError("определяется в подклассе")

    def _reset_cache(self) -> None:
        raise NotImplementedError("определяется в подклассе")

    def _update_high_low(self, price: Any) -> tuple[float, float]:
        """Обновляет максимум и минимум незавершённого бара ценой или массивом цен."""
        values = np.atleast_1d(np.asarray(price, dtype=np.float64))
        self.cache["high"] = max(self.cache.get("high", -np.inf), float(values.max()))
        self.cache["low"] = min(self.cache.get("low", np.inf), float(values.min()))
        return self.cache["high"], self.cache["low"]

    def _create_bars(self, date_time: Any, price: float, high_price: float, low_price: float,
                     bars_list: list) -> None:
        """Добавляет в bars_list строку бара из накопленного кэша и цены закрытия."""
        bars_list.append([date_time, self.tick_num, self.cache["open"], high_price, low_price, price,
                          self.cache["volume"], self.cache["buy_volume"], self.cache["ticks"],
                          self.cache["dollar"]])

    def _apply_tick_rule(self, price: Any) -> np.ndarray:
        """Правило тика: знак изменения цены, при нулевом изменении — прошлый знак.

        Источник: B04 §2.3.2.1, с.29 (PDF 56): b_t = b_{t−1} при Δp_t = 0,
            иначе |Δp_t|/Δp_t. Граничное значение переносится между пакетами;
            для самого первого тика ряда знак 0.
        Статус: Подтверждено по официальному учебнику.
        """
        prices = np.atleast_1d(np.asarray(price, dtype=np.float64))
        if prices.size == 0:
            return np.empty(0)
        start = prices[0] if self.prev_price is None else self.prev_price
        signs = np.sign(np.diff(np.r_[start, prices]))
        last = np.where(signs != 0, np.arange(signs.size), -1)
        np.maximum.accumulate(last, out=last)
        filled = np.where(last >= 0, signs[np.maximum(last, 0)], self.prev_tick_rule)
        self.prev_price = float(prices[-1])
        self.prev_tick_rule = float(filled[-1])
        return filled

    def _get_imbalance(self, price: Any, signed_tick: Any, volume: Any) -> np.ndarray:
        """Дисбаланс тика по метрике бара: b_t, b_t·v_t или b_t·p_t·v_t (B04, с.29–30)."""
        signed = np.asarray(signed_tick, dtype=np.float64)
        if self.metric is None or self.metric.startswith(("cum_ticks", "tick_")):
            return signed
        if self.metric.startswith(("cum_volume", "volume_")):
            return signed * np.asarray(volume, dtype=np.float64)
        return signed * np.asarray(volume, dtype=np.float64) * np.asarray(price, dtype=np.float64)

    # --- общие помощники подклассов ---------------------------------------------
    def _with_pending(self, data: dict) -> dict:
        """Добавляет к пакету тики незавершённого бара из прошлого пакета."""
        signs = self._apply_tick_rule(data["prices"])
        merged = {"times": data["times"], "prices": data["prices"], "volumes": data["volumes"],
                  "signs": signs, "first_num": self.tick_num + 1}
        self.tick_num += len(data["prices"])
        if self._pending is not None and len(self._pending["prices"]):
            p = self._pending
            merged = {"times": p["times"].append(data["times"]),
                      "prices": np.r_[p["prices"], data["prices"]],
                      "volumes": np.r_[p["volumes"], data["volumes"]],
                      "signs": np.r_[p["signs"], signs], "first_num": p["first_num"]}
        return merged

    def _rows_from_segments(self, merged: dict, ends: np.ndarray, labels: Any = None) -> list:
        """Строки баров по отрезкам тиков [start, end]; остаток уходит в _pending."""
        prices, volumes, signs = merged["prices"], merged["volumes"], merged["signs"]
        stop = int(ends[-1]) + 1 if len(ends) else 0
        rest = slice(stop, None)
        self._pending = {"times": merged["times"][rest], "prices": prices[rest],
                         "volumes": volumes[rest], "signs": signs[rest],
                         "first_num": merged["first_num"] + stop}
        if not len(ends):
            return []
        starts = np.r_[0, ends[:-1] + 1]
        p, v = prices[:stop], volumes[:stop]
        high = np.maximum.reduceat(p, starts)
        low = np.minimum.reduceat(p, starts)
        vol = np.add.reduceat(v, starts)
        buy = np.add.reduceat(np.where(signs[:stop] > 0, v, 0.0), starts)
        dollar = np.add.reduceat(p * v, starts)
        times = merged["times"][ends] if labels is None else labels
        nums = merged["first_num"] + ends
        return [[times[k], int(nums[k]), p[starts[k]], high[k], low[k], p[ends[k]], vol[k], buy[k],
                 int(ends[k] - starts[k] + 1), dollar[k]] for k in range(len(ends))]


# ---------------------------------------------------------------------------
# Бары по информации: дисбаланс и серии (B04 §2.3.2, с.29–32, PDF 56–59)
# ---------------------------------------------------------------------------
def _open_cache(cache: dict, price: float) -> None:
    if not cache:
        cache.update(open=price, high=price, low=price, volume=0.0, buy_volume=0.0, ticks=0, dollar=0.0)


def _add_tick(cache: dict, price: float, volume: float, sign: float) -> None:
    if price > cache["high"]:
        cache["high"] = price
    if price < cache["low"]:
        cache["low"] = price
    cache["volume"] += volume
    if sign > 0:
        cache["buy_volume"] += volume
    cache["ticks"] += 1
    cache["dollar"] += price * volume


class BaseImbalanceBars(BaseBars):
    """Общая часть баров дисбаланса: тиков (TIB), объёма (VIB), денег (DIB).

    Что делает: θ_T = Σ b_t·x_t, где x_t = 1, v_t или p_t·v_t, b_t — правило тика.
    Бар закрывается на первом T, при котором |θ_T| ≥ E0[T]·|E0[b_t·x_t]|.
    E0[b_t·x_t] — EWM-среднее b_t·x_t последних min(n, expected_imbalance_window)
    тиков; до прогрева (exp_num_ticks_init тиков) бар закрыться не может.
    E0[T] задаёт подкласс: EMA — EWM-среднее длин прошлых баров, Const — число.

    Входы: metric — 'tick_imbalance', 'volume_imbalance', 'dollar_imbalance';
        batch_size; expected_imbalance_window; exp_num_ticks_init — прогрев и
        начальное E0[T]; analyse_thresholds — сохранять пороги по каждому тику
        (bars_thresholds); min_expected_imbalance — ЭКСПЕРИМЕНТ E8: нижняя
        граница |E0[b_t·x_t]| (по умолчанию нет).
    Источник: B04 §2.3.2.1–2.3.2.2, с.29–31 (PDF 56–58).
    Статус: Подтверждено по официальному учебнику.
    На минутках GC неприменимо без тиков: знак сделки по правилу тика
        определяется только по последовательным сделкам.
    Причинность: пороги бара считаются только по тикам прошлых баров (тест).
    """

    _KINDS = ("tick_imbalance", "volume_imbalance", "dollar_imbalance")

    def __init__(self, metric: str, batch_size: float, expected_imbalance_window: int,
                 exp_num_ticks_init: int, analyse_thresholds: bool,
                 min_expected_imbalance: float | None = None) -> None:
        if metric not in self._KINDS:
            raise ValueError(f"metric — одно из {self._KINDS}")
        self.expected_imbalance_window = _positive_int(expected_imbalance_window, "expected_imbalance_window")
        self.exp_num_ticks_init = _positive_int(exp_num_ticks_init, "exp_num_ticks_init")
        if min_expected_imbalance is not None and not (isinstance(min_expected_imbalance, numbers.Real)
                                                       and min_expected_imbalance > 0):
            raise ValueError("min_expected_imbalance — число > 0 или None")
        self.min_expected_imbalance = min_expected_imbalance
        self.analyse_thresholds = bool(analyse_thresholds)
        self.num_ticks_bar: list = []
        self.bars_thresholds: list = []
        self._imbalances: deque = deque(maxlen=max(self.expected_imbalance_window, self.exp_num_ticks_init))
        self._seen = 0
        self.thresholds = {"cum_theta": 0.0, "expected_imbalance": math.nan,
                           "exp_num_ticks": float(self.exp_num_ticks_init)}
        super().__init__(metric, batch_size)

    def _reset_cache(self) -> None:
        """Сбрасывает накопления незавершённого бара (пороги остаются)."""
        self.cache = {}
        self._pending = None
        if hasattr(self, "thresholds"):
            self.thresholds["cum_theta"] = 0.0

    def _get_expected_imbalance(self, window: int) -> float:
        """E0[b_t·x_t]: EWM-среднее (span = k) последних k = min(n, window) значений; до прогрева NaN."""
        if self._seen < self.exp_num_ticks_init or not self._imbalances:
            return math.nan
        values = np.asarray(self._imbalances, dtype=np.float64)
        k = min(len(values), int(window))
        expected = float(ewma(values[-k:], k)[-1])
        floor = self.min_expected_imbalance
        if floor is not None and abs(expected) < floor:
            expected = math.copysign(floor, expected) if expected != 0 else floor
        return expected

    def _get_exp_num_ticks(self) -> float:
        raise NotImplementedError("определяется в подклассе")

    def _extract_bars(self, data: dict) -> list:
        """Проходит тики по одному: накапливает θ и закрывает бар по порогу книги."""
        times, prices, volumes = data["times"], data["prices"], data["volumes"]
        signs = self._apply_tick_rule(prices)
        imbalances = self._get_imbalance(prices, signs, volumes)
        rows: list = []
        th = self.thresholds
        for i in range(len(prices)):
            price, volume, sign, imb = float(prices[i]), float(volumes[i]), float(signs[i]), float(imbalances[i])
            self.tick_num += 1
            _open_cache(self.cache, price)
            _add_tick(self.cache, price, volume, sign)
            self._imbalances.append(imb)
            self._seen += 1
            th["cum_theta"] += imb
            if math.isnan(th["expected_imbalance"]) and self._seen >= self.exp_num_ticks_init:
                th["expected_imbalance"] = self._get_expected_imbalance(self.expected_imbalance_window)
            if self.analyse_thresholds:
                self.bars_thresholds.append({"timestamp": times[i], "cum_theta": th["cum_theta"],
                                             "expected_imbalance": th["expected_imbalance"],
                                             "exp_num_ticks": th["exp_num_ticks"]})
            expected = th["expected_imbalance"]
            if not math.isnan(expected) and abs(th["cum_theta"]) >= th["exp_num_ticks"] * abs(expected):
                self._create_bars(times[i], price, self.cache["high"], self.cache["low"], rows)
                self.num_ticks_bar.append(self.cache["ticks"])
                th["exp_num_ticks"] = self._get_exp_num_ticks()
                th["expected_imbalance"] = self._get_expected_imbalance(self.expected_imbalance_window)
                self._reset_cache()
        return rows


class BaseRunBars(BaseBars):
    """Общая часть баров серий: тиков (TRB), объёма (VRB), денег (DRB).

    Что делает: θ_T = max(Σ_{b=1} x_t, Σ_{b=−1} x_t) — объём (или число тиков)
    каждой стороны без взаимозачёта. Бар закрывается на первом T, при котором
    θ_T ≥ E0[T]·max(P·E0[x|b=1], (1 − P)·E0[x|b=−1]). P — EWM-среднее доли
    покупок в прошлых барах (num_prev_bars), E0[x|b] — EWM-средние x покупок и
    продаж по последним min(n, expected_imbalance_window) тикам каждой стороны.
    До прогрева (exp_num_ticks_init тиков и хотя бы по одной сделке каждой
    стороны) бар закрыться не может; на прогреве P — доля покупок в баре.

    Входы: metric — 'tick_run', 'volume_run', 'dollar_run'; batch_size;
        num_prev_bars; expected_imbalance_window; exp_num_ticks_init;
        analyse_thresholds.
    Источник: B04 §2.3.2.3–2.3.2.4, с.31–32 (PDF 58–59).
    Статус: Подтверждено по официальному учебнику.
    На минутках GC неприменимо без тиков.
    Причинность: пороги бара считаются только по тикам прошлых баров (тест).
    """

    _KINDS = ("tick_run", "volume_run", "dollar_run")

    def __init__(self, metric: str, batch_size: float, num_prev_bars: int, expected_imbalance_window: int,
                 exp_num_ticks_init: int, analyse_thresholds: bool) -> None:
        if metric not in self._KINDS:
            raise ValueError(f"metric — одно из {self._KINDS}")
        self.num_prev_bars = _positive_int(num_prev_bars, "num_prev_bars")
        self.expected_imbalance_window = _positive_int(expected_imbalance_window, "expected_imbalance_window")
        self.exp_num_ticks_init = _positive_int(exp_num_ticks_init, "exp_num_ticks_init")
        self.analyse_thresholds = bool(analyse_thresholds)
        size = max(self.expected_imbalance_window, self.exp_num_ticks_init)
        self._buys: deque = deque(maxlen=size)
        self._sells: deque = deque(maxlen=size)
        self._seen = 0
        self.num_ticks_bar: list = []
        self.buy_ticks_proportion: list = []
        self.bars_thresholds: list = []
        self.thresholds = {"cum_theta_buy": 0.0, "cum_theta_sell": 0.0, "exp_imbalance_buy": math.nan,
                           "exp_imbalance_sell": math.nan, "exp_num_ticks": float(self.exp_num_ticks_init),
                           "exp_buy_ticks_proportion": math.nan, "buy_ticks_num": 0}
        super().__init__(metric, batch_size)

    def _reset_cache(self) -> None:
        """Сбрасывает накопления незавершённого бара (ожидания остаются)."""
        self.cache = {}
        self._pending = None
        if hasattr(self, "thresholds"):
            self.thresholds.update(cum_theta_buy=0.0, cum_theta_sell=0.0, buy_ticks_num=0)

    def _get_expected_imbalance(self, array: Any, window: int, warm_up: bool = False) -> float:
        """EWM-среднее (span = k) последних k = min(n, window) значений стороны; NaN на прогреве."""
        if (warm_up and self._seen < self.exp_num_ticks_init) or len(array) == 0:
            return math.nan
        values = np.asarray(array, dtype=np.float64)
        k = min(len(values), int(window))
        return float(ewma(values[-k:], k)[-1])

    def _get_exp_num_ticks(self) -> float:
        raise NotImplementedError("определяется в подклассе")

    def _extract_bars(self, data: dict) -> list:
        """Проходит тики по одному: копит объём каждой стороны и закрывает бар по порогу книги."""
        times, prices, volumes = data["times"], data["prices"], data["volumes"]

        rows: list = []

        for i in range(len(prices)):

            self.tick_num += 1

            _add_tick(self.cache, price, volume, sign)
            if imb > 0:

                th["buy_ticks_num"] += 1
            elif imb < 0:
                self._sells.append(-imb)
                th["cum_theta_sell"] += -imb
            if math.isnan(th["exp_imbalance_buy"]) or math.isnan(th["exp_imbalance_sell"]):
                th["exp_imbalance_buy"] = self._get_expected_imbalance(self._buys, window, warm_up=True)
                th["exp_imbalance_sell"] = self._get_expected_imbalance(self._sells, window, warm_up=True)
                if not (math.isnan(th["exp_imbalance_buy"]) or math.isnan(th["exp_imbalance_sell"])):
                    th["exp_buy_ticks_proportion"] = th["buy_ticks_num"] / self.cache["ticks"]
            if self.analyse_thresholds:
                self.bars_thresholds.append({"timestamp": times[i], **{k: th[k] for k in th}})
            share = th["exp_buy_ticks_proportion"]
            expected = max(th["exp_imbalance_buy"] * share, th["exp_imbalance_sell"] * (1.0 - share))
            theta = max(th["cum_theta_buy"], th["cum_theta_sell"])
            if not math.isnan(expected) and theta >= th["exp_num_ticks"] * expected:
                self._create_bars(times[i], price, self.cache["high"], self.cache["low"], rows)
                self.num_ticks_bar.append(self.cache["ticks"])
                self.buy_ticks_proportion.append(th["buy_ticks_num"] / self.cache["ticks"])
                th["exp_num_ticks"] = self._get_exp_num_ticks()

def call(port, method, path, body=None):
    tok = open(os.path.join(HERE, "atlas_service_token.txt"), encoding="utf-8").read().strip()
    data = json.dumps(body).encode("utf-8") if body is not None else (b"{}" if method == "POST" else None)
    req = urllib.request.Request(f"http://127.0.0.1:{port}{path}", data=data, method=method,
                                 headers={"X-Atlas-Token": tok, "Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(req, timeout=3600) as r:
            return r.status, json.loads(r.read().decode("utf-8"))
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read().decode("utf-8") or "{}")
    except urllib.error.URLError as e:
        sys.exit(f"служба не отвечает на порту {port}: {e.reason} — запущена ли START_ATLAS_SERVICE_v2.bat?")

# -*- coding: utf-8 -*-
"""
==========================================================================================
DZ_ATLAS_Service.py — служба оптимизатора «Атласа»: работа без командной строки, v1.0 (2026-09-30)

  Пульт → POST /task → файл в task_queue → служба запускает оптимизатор (или проверку вперёд
  для плана WFA) → пульт читает GET /status и GET /summary?id=…

БЕЗОПАСНОСТЬ — как у сервера пульта: только 127.0.0.1; источник — только file://, null,
  app://atlas; ключ в заголовке X-Atlas-Token (файл atlas_service_token.txt рядом со службой).
ОЧЕРЕДЬ — task_queue/*.json → running/ → done/ либо failed/ (с хвостом ошибки). Состояние —
  results/_service_status.json; переживает перезапуск службы.

  Запуск: двойной щелчок по START_ATLAS_SERVICE.bat или python DZ_ATLAS_Service.py [--port 8090]
==========================================================================================
"""
import argparse, glob, json, os, secrets, shutil, subprocess, sys, threading, time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse, parse_qs

VERSION = "DZ_ATLAS_Service-v1.0 (2026-09-30)"
HERE = os.path.dirname(os.path.abspath(__file__))
ORIGINS = ("null", "file://", "app://atlas")
LOCK = threading.Lock()


class Svc:
    def __init__(self, root, workers):
        self.root, self.workers = root, workers
        self.q = os.path.join(root, "task_queue"); self.res = os.path.join(root, "results")
        for d in ("", "running", "done", "failed"): os.makedirs(os.path.join(self.q, d), exist_ok=True)
        os.makedirs(self.res, exist_ok=True)
        tf = os.path.join(root, "atlas_service_token.txt")
        if not os.path.exists(tf): open(tf, "w").write(secrets.token_hex(16))
        self.token = open(tf).read().strip()
        self.sf = os.path.join(self.res, "_service_status.json")
        self.status = json.load(open(self.sf, encoding="utf-8")) if os.path.exists(self.sf) else {}
        for f in glob.glob(os.path.join(self.q, "running", "*.json")):      # прерванные при остановке — снова в очередь
            shutil.move(f, os.path.join(self.q, os.path.basename(f)))

    def save(self):
        with LOCK:
            tmp = self.sf + ".tmp"; json.dump(self.status, open(tmp, "w", encoding="utf-8"), ensure_ascii=False, indent=1); os.replace(tmp, self.sf)

    def enqueue(self, body):
        tid = body.get("plan_id") or body.get("task_id") or time.strftime("T%Y%m%d_%H%M%S")
        if not ("jobs" in body or "optimize" in body or "target" in body or "hypothesis" in body or "windows" in body):
            raise ValueError("это не задача оптимизатора и не план проверки вперёд")
        path = os.path.join(self.q, tid + ".json"); tmp = path + ".tmp"
        json.dump(body, open(tmp, "w", encoding="utf-8"), ensure_ascii=False, indent=2); os.replace(tmp, path)
        self.status[tid] = {"state": "queued", "kind": "wfa" if "windows" in body else "task", "queued": time.strftime("%Y-%m-%d %H:%M:%S")}
        self.save(); return tid

    def worker(self):
        while True:
            files = sorted(glob.glob(os.path.join(self.q, "*.json")), key=os.path.getmtime)
            if not files: time.sleep(1.0); continue
            f = files[0]; tid = os.path.splitext(os.path.basename(f))[0]
            run = os.path.join(self.q, "running", os.path.basename(f)); shutil.move(f, run)
            wfa = "windows" in json.load(open(run, encoding="utf-8"))
            self.status.setdefault(tid, {}).update(state="running", started=time.strftime("%Y-%m-%d %H:%M:%S"), kind="wfa" if wfa else "task"); self.save()
            cmd = ([sys.executable, os.path.join(HERE, "DZ_OPT_WFA.py"), "--plan", run, "--results", self.res, "--workers", str(self.workers)] if wfa else
                   [sys.executable, os.path.join(HERE, "DZ_OPT_Engine_S021.py"), "run", "--task", run, "--results", self.res, "--workers", str(self.workers)])
            r = subprocess.run(cmd, capture_output=True, text=True, cwd=self.root)
            ok = r.returncode == 0
            shutil.move(run, os.path.join(self.q, "done" if ok else "failed", os.path.basename(run)))
            st = {"state": "done" if ok else "failed", "finished": time.strftime("%Y-%m-%d %H:%M:%S")}
            if ok and not wfa:
                s = json.load(open(os.path.join(self.res, tid, "summary.json"), encoding="utf-8"))
                st.update({k: s.get(k) for k in ("objective", "deflated_sharpe", "pbo", "verdict", "trials_this_task", "data_sha256")}, best=s.get("best_selected"))
            elif ok:
                w = os.path.join(self.res, f"wfa_{tid}_wfa.json") if not tid.startswith("wfa_") else os.path.join(self.res, f"{tid}_wfa.json")
                if os.path.exists(w): st["wfa"] = json.load(open(w, encoding="utf-8"))["summary"]
            else:
                st["error"] = (r.stderr or r.stdout)[-1500:]
            self.status[tid].update(st); self.save()


def handler(svc):
    class H(BaseHTTPRequestHandler):
        def log_message(self, *a): pass
        def _origin_ok(self):
            o = self.headers.get("Origin")
            return o is None or o == "null" or o.startswith(ORIGINS)
        def _send(self, code, obj):
            b = json.dumps(obj, ensure_ascii=False).encode("utf-8")
            self.send_response(code); self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Access-Control-Allow-Origin", self.headers.get("Origin") or "null")
            self.send_header("Content-Length", str(len(b))); self.end_headers(); self.wfile.write(b)
        def _guard(self):
            if not self._origin_ok(): self._send(403, {"error": "чужой источник"}); return False
            if self.headers.get("X-Atlas-Token") != svc.token: self._send(401, {"error": "неверный ключ"}); return False
            return True
        def do_OPTIONS(self):
            self.send_response(204 if self._origin_ok() else 403)
            self.send_header("Access-Control-Allow-Origin", self.headers.get("Origin") or "null")
            self.send_header("Access-Control-Allow-Headers", "X-Atlas-Token, Content-Type")
            self.send_header("Access-Control-Allow-Methods", "GET, POST"); self.end_headers()
        def do_GET(self):
            if not self._guard(): return
            u = urlparse(self.path)
            if u.path == "/status": self._send(200, {"service": VERSION, "jobs": svc.status})
            elif u.path == "/summary":
                tid = (parse_qs(u.query).get("id") or [""])[0]; p = os.path.join(svc.res, os.path.basename(tid), "summary.json")
                self._send(200, json.load(open(p, encoding="utf-8"))) if os.path.exists(p) else self._send(404, {"error": "нет итога"})
            else: self._send(404, {"error": "нет такого адреса"})
        def do_POST(self):
            if not self._guard(): return
            if urlparse(self.path).path != "/task": self._send(404, {"error": "нет такого адреса"}); return
            n = int(self.headers.get("Content-Length") or 0)
            if n > 1_000_000: self._send(413, {"error": "слишком большая задача"}); return
            try: self._send(200, {"queued": svc.enqueue(json.loads(self.rfile.read(n).decode("utf-8")))})
            except Exception as ex: self._send(400, {"error": str(ex)})
    return H


def main():
    ap = argparse.ArgumentParser(description=VERSION)

    ThreadingHTTPServer(("127.0.0.1", a.port), handler(svc)).serve_forever()


if __name__ == "__main__":

def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def read_json(path, default=None):
    try:
        with open(path, encoding="utf-8-sig") as f:
            return json.load(f)
    except (OSError, ValueError):
        return default


def write_json_atomic(path, obj, indent=1):
    tmp = path + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, indent=indent, default=str)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)


def load_module(path, name):

def merge(base, over):
    out = json.loads(json.dumps(base))
    for k, v in (over or {}).items():
        out[k] = merge(out[k], v) if isinstance(v, dict) and isinstance(out.get(k), dict) else v
    return out

def emax_sigma(n, engine=None):
    """
    Ожидаемый максимум n испытаний без преимущества в единицах σ(Шарпа). Формула — оптимизатора:
    SR₀ функции deflated_sharpe при Var[SR] = 1 (Bailey & López de Prado). Без оптимизатора —
    та же формула напрямую; источник указывается в ответе.
    """
    n = int(n)
    if n < 2:
        return 0.0
    if engine is not None:
        return float(engine.deflated_sharpe(0.0, 1.0, n, 10, 0.0, 3.0)[1])
    nd, g = NormalDist(), 0.5772156649015329
    return (1 - g) * nd.inv_cdf(1 - 1 / n) + g * nd.inv_cdf(1 - 1 / (n * math.e))


def trials_to_sigma(target, engine=None):
    """Наименьшее число испытаний, при котором порог «случайного лучшего» достигает target σ."""
    lo, hi = 2, 4
    while emax_sigma(hi, engine) < target:
        lo, hi = hi, hi * 2
        if hi > 10 ** 13:
            return None
    while lo < hi:
        mid = (lo + hi) // 2
        if emax_sigma(mid, engine) >= target:
            hi = mid
        else:
            lo = mid + 1
    return lo

class MiniWs:
    def __init__(self, port, origin="app://atlas", timeout=10.0):
        self.port, self.origin, self.timeout, self.sock = port, origin, timeout, None

    def connect(self):
        s = socket.create_connection(("127.0.0.1", self.port), timeout=self.timeout)
        key = base64.b64encode(os.urandom(16)).decode()
        s.sendall((f"GET / HTTP/1.1\r\nHost: 127.0.0.1:{self.port}\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"
                   f"Sec-WebSocket-Key: {key}\r\nSec-WebSocket-Version: 13\r\nOrigin: {self.origin}\r\n\r\n").encode())
        head = b""
        while b"\r\n\r\n" not in head:
            ch = s.recv(1)
            if not ch or len(head) > 8192:
                raise ConnectionError("рукопожатие оборвано")
            head += ch
        text = head.decode("latin-1")
        acc = base64.b64encode(hashlib.sha1((key + WS_GUID).encode()).digest()).decode()
        if " 101 " not in text.split("\r\n")[0] or acc not in text:
            raise ConnectionError("сервер не принял соединение: " + text.split("\r\n")[0])
        self.sock = s

    def _exact(self, n):
        b = b""
        while len(b) < n:
            c = self.sock.recv(n - len(b))
            if not c:
                raise ConnectionError("соединение закрыто")
            b += c
        return b

    def send_text(self, text):
        p = text.encode("utf-8")
        mask = os.urandom(4)
        if len(p) < 126:
            h = bytes([0x81, 0x80 | len(p)])
        elif len(p) < 65536:
            h = bytes([0x81, 0x80 | 126]) + struct.pack(">H", len(p))
        else:
            h = bytes([0x81, 0x80 | 127]) + struct.pack(">Q", len(p))
        self.sock.sendall(h + mask + bytes(b ^ mask[i % 4] for i, b in enumerate(p)))

    def recv_text(self):
        while True:
            h = self._exact(2)
            op, n = h[0] & 0x0F, h[1] & 0x7F
            if n == 126:
                n = struct.unpack(">H", self._exact(2))[0]
            elif n == 127:
                n = struct.unpack(">Q", self._exact(8))[0]
            mask = self._exact(4) if h[1] & 0x80 else None
            p = self._exact(n) if n else b""
            if mask:
                p = bytes(b ^ mask[i % 4] for i, b in enumerate(p))
            if op == 0x8:
                raise ConnectionError("сервер закрыл соединение")
            if op == 0x1:
                return p.decode("utf-8", "replace")

    def close(self):
        try:
            if self.sock:
                self.sock.close()
        except OSError:
            pass

class Refuse(Exception):
    def __init__(self, msg, code=400):
        super().__init__(msg)
        self.code = code


def need(x, msg, code=404):
    if not x:
        raise Refuse(msg, code)
    return x


def kill_tree(proc):
    """Остановить задачу вместе с рабочими процессами оптимизатора."""
    try:
        if os.name == "nt":
            subprocess.run(["taskkill", "/F", "/T", "/PID", str(proc.pid)], capture_output=True, timeout=60)
        else:
            import signal
            try:
                os.killpg(proc.pid, signal.SIGTERM)
                proc.wait(timeout=10)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
    except Exception:
        try:
            proc.kill()
        except Exception:
            pass


class Service:
    def __init__(self, root, cfg):
        self.root, self.cfg = root, cfg
        self.lock = threading.RLock()
        self.vlock = threading.Lock()
        self.stop_evt = threading.Event()
        self.q = os.path.join(root, "task_queue")
        self.res = os.path.join(root, "results")
        self.logs = os.path.join(self.res, "_logs")
        for d in ("",) + QUEUE_DIRS:
            os.makedirs(os.path.join(self.q, d), exist_ok=True)
        os.makedirs(self.logs, exist_ok=True)
        self.journal_path = os.path.join(self.res, "_service_journal.jsonl")
        self.status_path = os.path.join(self.res, "_service_status_v2.json")
        self.nt_root = os.path.expanduser(cfg["nt_root"])
        self.python = cfg.get("python") or sys.executable
        self.token = self._load_token()
        self.status = read_json(self.status_path, {}) or {}
        self.current = None
        self.alarms = []
        self.verify_results = read_json(os.path.join(self.res, "_service_verify.json"), {}) or {}
        self.desync = {"enabled": bool(cfg["desync_watch"].get("enabled")), "state": "выключен"}

    # ---------- основа ----------
    def _load_token(self):
        tf = os.path.join(self.root, "atlas_service_token.txt")
        if not os.path.exists(tf):
            with open(tf, "w", encoding="utf-8") as f:
                f.write(secrets.token_hex(32))
        with open(tf, encoding="utf-8") as f:
            tok = f.read().strip()
        if len(tok) < 32:
            raise SystemExit(f"ОТКАЗ: ключ в {tf} короче 32 знаков — сгенерируйте новый: DZ_HOUSE_Order_S021.py rotate-token")
        return tok

    def journal(self, event, **kw):
        rec = {"utc": utc_now(), "event": event, **kw}
        with self.lock:
            with open(self.journal_path, "a", encoding="utf-8") as f:
                f.write(json.dumps(rec, ensure_ascii=False, default=str) + "\n")
                f.flush()
                os.fsync(f.fileno())
        return rec

    def _set(self, tid, **kw):
        with self.lock:
            self.status.setdefault(tid, {}).update(kw)
            write_json_atomic(self.status_path, self.status)

    def snapshot(self):
        with self.lock:
            return json.loads(json.dumps(self.status, default=str))

    def mod(self, key):
        """Модули проекта загружаются лениво и один раз: оптимизатор, эталон, очиститель v2, граф."""
        with self.lock:
            if key not in self._mods:
                if key == "graph":
                    self._mods[key] = read_json(os.path.join(self.root, self.cfg["graph"]))
                    if not self._mods[key]:
                        raise Refuse(f"граф {self.cfg['graph']} не читается", 500)
                else:
                    f = {"engine": self.cfg["engine"], "reference": self.cfg["reference"], "san2": "DZ_DATA_Sanitizer_v2.py"}[key]
                    self._mods[key] = load_module(os.path.join(self.root, f), "dz_svc_" + key)
            return self._mods[key]

    def engine_or_none(self):
        try:
            return self.mod("engine")
        except Exception:
            return None

    def _check_deps(self):
        rc, out = run_script(self.root, self.python, ["-c", "import numpy, pandas; print(numpy.__version__, pandas.__version__)"], 120)
        ok = rc == 0
        return {"ok": ok, "python": self.python, "versions": out if ok else None,
                "error": None if ok else f"интерпретатор задач не видит numpy/pandas: {out[-300:]} — установите: "
                                           f"\"{self.python}\" -m pip install numpy pandas openpyxl, либо укажите \"python\" в конфигурации"}

    def _import_v1_status(self):
        v1 = read_json(os.path.join(self.res, "_service_status.json"), {}) or {}
        n = 0
        for tid, rec in v1.items():
            if tid in self.status:
                continue
            rec = dict(rec, source="служба v1")
            if rec.get("state") in ("queued", "running"):
                for d, st in (("done", "done"), ("failed", "failed")):
                    if os.path.exists(os.path.join(self.q, d, tid + ".json")):
                        rec.update(state=st, note=f"состояние сверено по папке {d}: в файле v1 стояло «{rec.get('state')}»")
                        break
            self.status[tid] = rec
            n += 1
        if n:
            write_json_atomic(self.status_path, self.status)
            self.journal("import_v1", records=n)

    def _recover(self):
        for f in glob.glob(os.path.join(self.q, "running", "*.json")):
            tid = os.path.splitext(os.path.basename(f))[0]
            os.replace(f, os.path.join(self.q, "interrupted", os.path.basename(f)))
            self._set(tid, state="interrupted", finished=local_now(),
                      note="служба остановилась во время расчёта; результат не принят. Повтор — POST /requeue (новый id)")
            self.journal("interrupted", id=tid)
        for f in sorted(glob.glob(os.path.join(self.q, "*.json")), key=os.path.getmtime):
            tid = os.path.splitext(os.path.basename(f))[0]
            if (self.status.get(tid) or {}).get("state") == "queued" and self.status[tid].get("version") == VERSION:
                continue
            try:                                     # задача из очереди v1: проверяется заново
                body = read_json(f)
                if body is None:
                    raise Refuse("файл не JSON")
                notes = []
                kind = "wfa" if "windows" in body else "task"
                self._validate(body, kind, notes)
                write_json_atomic(f, body, indent=2)
                self._set(tid, state="queued", kind=kind, queued=local_now(), queued_utc=utc_now(), notes=notes, version=VERSION)
            except Refuse as ex:
                os.replace(f, os.path.join(self.q, "failed", os.path.basename(f)))
                self._set(tid, state="failed", finished=local_now(), error=f"не прошла проверку службы v2: {ex}")

    # ---------- проверка задачи ----------
    def _id_taken(self, tid):
        if tid in self.status:
            return True
        if any(os.path.exists(os.path.join(self.q, d, tid + ".json")) for d in ("",) + QUEUE_DIRS):
            return True
        if os.path.exists(os.path.join(self.res, tid)) or os.path.exists(os.path.join(self.res, tid + "_wfa.json")):
            return True
        return bool(glob.glob(os.path.join(self.res, glob.escape(tid) + "_w[0-9][0-9]_*")))

    def _date(self, s, what):
        try:
            return datetime.strptime(str(s)[:10], "%Y-%m-%d")
        except ValueError:
            raise Refuse(f"{what}: дата ГГГГ-ММ-ДД, получено «{s}»")

    def _validate(self, body, kind, notes):
        if kind == "wfa":
            t = body.get("task")
            if not isinstance(t, dict):
                raise Refuse("план проверки вперёд: нет объекта «task»")
            w = body.get("windows")
            if not isinstance(w, list) or not w or len(w) > 60:
                raise Refuse("план проверки вперёд: «windows» — от 1 до 60 окон")
            for i, win in enumerate(w, 1):
                try:
                    a, b = (self._date(x, f"окно {i}") for x in win["train"])
                    c, d = (self._date(x, f"окно {i}") for x in win["test"])
                except (KeyError, TypeError, ValueError):
                    raise Refuse(f"окно {i}: нужны train [начало, конец] и test [начало, конец]")
                if not (a < b <= c < d):
                    raise Refuse(f"окно {i}: обучение раньше проверки, начало раньше конца")
        else:
            t = body
        g = self.mod("graph")
        P = g["params"]
        d = t.get("data")
        if not (isinstance(d, str) and SYM_RE.fullmatch(d)) and not self.cfg["allow_uncertified_data"]:
            raise Refuse("«data» — только символ сертифицированных данных (например, MGC): прогон на "
                         "несертифицированных данных ничего не доказывает (проверка №1)")
        for k in ("cache", "data_root"):
            if k in t and not self.cfg["allow_uncertified_data"]:
                raise Refuse(f"поле «{k}» запрещено: оптимизатор считал бы не сертифицированный файл, а {t[k]}")
        if isinstance(d, str) and SYM_RE.fullmatch(d):
            if not os.path.exists(os.path.join(self.root, "data_certified", d, f"{d}_M1.npz")):
                raise Refuse(f"нет сертифицированных данных {d}: DZ_CERT_Data.py certify --symbol {d}")
            prof = (g.get("instrument_profiles") or {})
            if not t.get("instrument") and d in prof:
                t["instrument"] = d
                notes.append(f"инструмент подставлен по символу данных: профиль {d} {prof[d].get('reference')}")
            ref = (prof.get(t.get("instrument") or "") or {}).get("reference") or {}
            for k, v in (t.get("base") or {}).items():
                if k in ref and float(v) != float(ref[k]):
                    notes.append(f"ВНИМАНИЕ, единицы: base.{k} = {v}, в профиле {t.get('instrument')} — {ref[k]} (проверка №4)")
        mode = t.get("mode", "dev")
        if mode == "dev":
            opt = t.get("optimize")
            if not isinstance(opt, dict) or not opt:
                raise Refuse("режим dev: нужен непустой «optimize»")
            for k, v in opt.items():
                if k not in P:
                    raise Refuse(f"«{k}» нет в графе {self.cfg['graph']}: оптимизатор его не исполнит")
                if not isinstance(v, list) or not v or len(v) > 500 or not all(isinstance(x, (int, float)) and not isinstance(x, bool) for x in v):
                    raise Refuse(f"«{k}»: список чисел, от 1 до 500 значений")
        elif mode == "param":
            tg = t.get("target")
            tg = [tg] if isinstance(tg, str) else tg
            if not isinstance(tg, list) or not tg or any(x not in P for x in tg):
                raise Refuse("режим param: «target» — параметры графа")
            if not isinstance(t.get("depth", 1), int) or not 0 <= t.get("depth", 1) <= 3:
                raise Refuse("режим param: «depth» от 0 до 3")
        elif mode == "system":
            if t.get("hypothesis") not in (g.get("groups") or {}):
                raise Refuse(f"режим system: гипотеза из графа: {list((g.get('groups') or {}).keys())}")
        else:
            raise Refuse("режим: dev, param или system")
        if t.get("objective", "SharpeDaily") not in OBJECTIVES:
            raise Refuse(f"цель — одна из {OBJECTIVES}: иначе оптимизатор упадёт после всех испытаний")
        mt = t.get("max_trials", 200)
        if not isinstance(mt, int) or isinstance(mt, bool) or not 1 <= mt <= 100000:
            raise Refuse("«max_trials» — целое от 1 до 100000")
        if t.get("profile", "chicago") not in ("chicago", "berlin"):
            raise Refuse("«profile» — chicago или berlin: профили не смешиваются")
        fields = {f.name for f in dataclasses.fields(self.mod("reference").Params)}
        unknown = [k for k in (t.get("base") or {}) if k not in fields]
        if unknown:
            raise Refuse(f"в «base» неизвестные эталону параметры: {unknown}")
        for c in t.get("constraints") or []:
            m = CONS_RE.fullmatch(str(c))
            if not m or any(not re.fullmatch(r"-?\d+(\.\d+)?", x) and x not in fields for x in (m.group(1), m.group(3))):
                raise Refuse(f"ограничение не разобрано или с неизвестным параметром: {c}")
        if kind == "task" and t.get("start") and t.get("end"):
            if self._date(t["start"], "start") >= self._date(t["end"], "end"):
                raise Refuse("начало периода не раньше конца")
        if t.get("nt_template") and not os.path.exists(os.path.join(self.root, t["nt_template"])):
            raise Refuse(f"шаблон {t['nt_template']} не найден")
        if "timeout_min" in body and not (isinstance(body["timeout_min"], (int, float)) and 1 <= body["timeout_min"] <= 10080):
            raise Refuse("«timeout_min» от 1 до 10080")
        self._holdout(body, kind, notes)
        self._budget(body, kind, notes)

    def _holdout(self, body, kind, notes):
        hf = self.cfg.get("holdout_from")
        if not hf:
            return
        hfd = self._date(hf, "holdout_from")
        if kind == "task":
            if body.get("unlock_holdout"):
                if len(str(body.get("unlock_reason") or "").strip()) < 10:
                    raise Refuse("вскрытие отложенной выборки — только с причиной «unlock_reason» (не короче 10 знаков)")
                notes.append(f"ВНИМАНИЕ: отложенная выборка с {hf} вскрывается — оптимизатор запишет это в holdout_log.csv")
                return
            end = body.get("end")
            if end is None:
                if self.cfg.get("holdout_policy") == "clip":
                    if body.get("start") and self._date(body["start"], "start") >= hfd:
                        raise Refuse(f"начало периода внутри отложенной выборки с {hf}")
                    body["end"], body["holdout_clipped"] = hf, True
                    notes.append(f"период обрезан замком отложенной выборки: конец {hf} (не включается)")
                else:
                    raise Refuse(f"задача без конца периода заходит в отложенную выборку с {hf}: укажите «end» не позже даты")
            elif self._date(end, "end") > hfd:
                raise Refuse(f"конец периода {end} заходит в отложенную выборку с {hf}: «end» не позже даты или "
                             f"«unlock_holdout»: true с причиной «unlock_reason»")
        else:
            last = max(self._date(w["test"][1], "окно") for w in body["windows"])
            if last > hfd:
                if not body.get("unlock_holdout") or len(str(body.get("unlock_reason") or "").strip()) < 10:
                    raise Refuse(f"окна плана заходят в отложенную выборку с {hf}: сдвиньте окна или "
                                 f"«unlock_holdout»: true с причиной «unlock_reason»")
                notes.append(f"ВНИМАНИЕ: план вскрывает отложенную выборку с {hf} — запись в holdout_log.csv")

    def trials_upper(self, body, kind):
        t = body["task"] if kind == "wfa" else body
        mt = int(t.get("max_trials", 200))
        if t.get("mode", "dev") == "dev":
            n = 1
            for v in t["optimize"].values():
                n *= len(set(v))
            mt = min(mt, n)
        return mt * len(body["windows"]) + len(body["windows"]) if kind == "wfa" else mt

    def registry_frame(self):
        import pandas as pd
        p = os.path.join(self.root, self.cfg["registry"])
        return pd.read_csv(p) if os.path.exists(p) else pd.DataFrame(columns=["TaskId", "Strategy", "Trials", "UTC"])

    def _budget(self, body, kind, notes):
        bl = self.cfg.get("budget_lock") or {}
        if not bl.get("enabled"):
            return
        strat = self.mod("graph")["strategy"]
        r = self.registry_frame()
        n_now = int(r[r.Strategy == strat].Trials.sum()) if len(r) else 0
        add = self.trials_upper(body, kind)
        e_after = emax_sigma(n_now + add, self.engine_or_none())
        sigma = float(bl.get("sigma", 4.0))
        budget = (bl.get("budgets") or {}).get(strat)
        problems = []
        if e_after > sigma:
            problems.append(f"порог «случайного лучшего» после задачи {e_after:.2f}σ > {sigma}σ (испытаний станет {n_now + add})")
        if budget and n_now + add > budget:
            problems.append(f"бюджет испытаний {strat} {budget}, станет {n_now + add}")
        if problems:
            reason = str(body.get("budget_override") or "").strip()
            if len(reason) < 20:
                raise Refuse("Э-Т1 бюджетный замок: " + "; ".join(problems) + ". Доказать уже нечего (критерии №1 и №7 "
                             "раздела «стоп»). Снять — «budget_override» с причиной не короче 20 знаков", 409)
            notes.append("Э-Т1: замок снят оператором: " + reason)
        notes.append(f"Э-Т1: испытаний {strat} станет {n_now + add}, порог {e_after:.2f}σ")

    # ---------- очередь ----------
    def enqueue(self, body):
        if not isinstance(body, dict):
            raise Refuse("задача — объект JSON")
        if not self.deps["ok"]:
            raise Refuse(self.deps["error"], 503)
        kind = "wfa" if "windows" in body else "task"
        if kind == "task" and not any(k in body for k in ("optimize", "target", "hypothesis")):
            raise Refuse("это не задача оптимизатора и не план проверки вперёд")
        key = "plan_id" if kind == "wfa" else "task_id"
        tid = str(body.get(key) or (("wfa_" if kind == "wfa" else "T") + time.strftime("%Y%m%d_%H%M%S")))
        if not ID_RE.fullmatch(tid):
            raise Refuse("id задачи: латиница, цифры, «_» и «-», до 80 знаков")
        body[key] = tid
        notes = []
        with self.lock:
            if self._id_taken(tid):
                raise Refuse(f"id {tid} уже использован: прежние результаты не перезаписываются — дайте новый id", 409)
            self._validate(body, kind, notes)
            write_json_atomic(os.path.join(self.q, tid + ".json"), body, indent=2)
            self._set(tid, state="queued", kind=kind, queued=local_now(), queued_utc=utc_now(), notes=notes, version=VERSION,
                      trials_upper=self.trials_upper(body, kind))
        self.journal("enqueue", id=tid, kind=kind, notes=notes)
        return tid, notes

    def cancel(self, tid):
        with self.lock:
            st = self.status.get(tid)
            if not st:
                raise Refuse("нет такой задачи", 404)
            if st.get("state") == "queued":
                src = os.path.join(self.q, tid + ".json")
                if os.path.exists(src):
                    os.replace(src, os.path.join(self.q, "cancelled", tid + ".json"))
                self._set(tid, state="cancelled", finished=local_now(), note="отменена до запуска")
            elif st.get("state") == "running" and self.current and self.current["id"] == tid:
                self.current["cancel"] = True
            else:
                raise Refuse(f"задача в состоянии «{st.get('state')}» не отменяется", 409)
        self.journal("cancel", id=tid)
        return st.get("state")

    def requeue(self, tid):
        st = self.status.get(tid) or {}
        if st.get("state") not in ("failed", "cancelled", "interrupted"):
            raise Refuse("повторяется только задача с ошибкой, отменённая или прерванная", 409)
        src = next((os.path.join(self.q, d, tid + ".json") for d in ("failed", "cancelled", "interrupted")
                    if os.path.exists(os.path.join(self.q, d, tid + ".json"))), None)
        if not src:
            raise Refuse("файл задачи не найден", 404)
        body = read_json(src)
        n = 2
        base = re.sub(r"_r\d+$", "", tid)
        while self._id_taken(f"{base}_r{n}"):
            n += 1
        body["plan_id" if "windows" in body else "task_id"] = f"{base}_r{n}"
        new, notes = self.enqueue(body)
        self.journal("requeue", id=tid, new=new)
        return new, notes

    def _next(self):
        with self.lock:
            q = sorted((v.get("queued_utc") or "", k) for k, v in self.status.items() if v.get("state") == "queued")
        for _, k in q:
            if os.path.exists(os.path.join(self.q, k + ".json")):
                return k
            self._set(k, state="failed", finished=local_now(), error="файл задачи пропал из очереди")
        return None

    def worker(self):
        while not self.stop_evt.is_set():
            try:
                self.worker_beat = time.time()
                tid = self._next() if self.deps["ok"] else None
                if tid is None:
                    self.stop_evt.wait(1.0)
                    continue
                self._run(tid)
            except Exception:
                self.journal("worker_error", error=traceback.format_exc()[-2000:])
                self.stop_evt.wait(2.0)

    def _command(self, kind, run, body):
        R = lambda k: os.path.join(self.root, self.cfg[k])
        hf = self.cfg.get("holdout_from")
        w = str(self.cfg.get("workers", 0))
        if kind == "task":
            cmd = [self.python, R("engine"), "run", "--task", run, "--results", self.res, "--workers", w,
                   "--registry", R("registry"), "--graph", R("graph"), "--ref", R("reference")]
            if hf:
                cmd += ["--holdout-from", hf]
            return cmd, child_env(drop=("DZ_HOLDOUT_FROM",))
        cmd = [self.python, R("wfa"), "--plan", run, "--results", self.res, "--workers", w, "--graph", R("graph")]
        if hf and body.get("unlock_holdout"):
            with open(os.path.join(self.root, "holdout_log.csv"), "a", encoding="utf-8") as f:
                f.write(f"{body['plan_id']},{hf},{utc_now()}\n")
            return cmd, child_env(drop=("DZ_HOLDOUT_FROM",))
        return cmd, child_env({"DZ_HOLDOUT_FROM": hf} if hf else None)

    def _tail(self, path, n=1500):
        try:
            with open(path, "rb") as f:
                f.seek(max(0, os.path.getsize(path) - n))
                return f.read().decode("utf-8", "replace")
        except OSError:
            return ""

    def _run(self, tid):
        src, run = os.path.join(self.q, tid + ".json"), os.path.join(self.q, "running", tid + ".json")
        os.replace(src, run)
        logp = os.path.join(self.logs, tid + ".log")
        try:
            body = read_json(run)
            if not isinstance(body, dict):
                raise ValueError("файл задачи не JSON")
            kind = "wfa" if "windows" in body else "task"
            cmd, env = self._command(kind, run, body)
        except Exception as ex:
            os.replace(run, os.path.join(self.q, "failed", tid + ".json"))
            self._set(tid, state="failed", finished=local_now(), error=f"задача не читается: {ex}")
            self.journal("failed", id=tid, error=str(ex))
            return
        self._set(tid, state="running", started=local_now(), started_utc=utc_now(), log=os.path.relpath(logp, self.root))
        self.journal("start", id=tid, cmd=cmd[1:])
        timeout = float(body.get("timeout_min") or self.cfg["task_timeout_min"]) * 60
        reason = None
        with open(logp, "ab") as lf:
            lf.write(f"# {VERSION} · {utc_now()} · {' '.join(cmd)}\n".encode("utf-8"))
            lf.flush()
            kw = {"cwd": self.root, "stdout": lf, "stderr": subprocess.STDOUT, "env": env}
            if os.name == "nt":
                kw["creationflags"] = subprocess.CREATE_NEW_PROCESS_GROUP
            else:
                kw["start_new_session"] = True
            t0 = time.time()
            try:
                proc = subprocess.Popen(cmd, **kw)
            except OSError as ex:
                proc = None
                lf.write(f"# запуск не удался: {ex}\n".encode("utf-8"))
        if proc is None:
            os.replace(run, os.path.join(self.q, "failed", tid + ".json"))
            self._set(tid, state="failed", finished=local_now(), error=f"оптимизатор не запустился: {self.python}")
            self.journal("failed", id=tid, error="popen")
            return
        with self.lock:
            self.current = {"id": tid, "proc": proc, "cancel": False, "t0": t0}
        while proc.poll() is None:
            self.worker_beat = time.time()
            if self.current["cancel"]:
                reason = "cancelled"
            elif time.time() - t0 > timeout:
                reason = "timeout"
            elif self.stop_evt.is_set():
                reason = "service_stop"
            if reason:
                kill_tree(proc)
                break
            time.sleep(0.5)
        try:
            rc = proc.wait(timeout=60)
        except subprocess.TimeoutExpired:
            rc = -9
        with self.lock:
            self.current = None
        fin = {"finished": local_now(), "finished_utc": utc_now(), "seconds": round(time.time() - t0, 1)}
        if reason == "cancelled":
            os.replace(run, os.path.join(self.q, "cancelled", tid + ".json"))
            self._set(tid, state="cancelled", note="отменена во время расчёта: итог не записан, испытания в реестр не внесены"
                      + ("; окна проверки вперёд, завершённые до отмены, — внесены" if kind == "wfa" else ""), **fin)
        elif reason == "service_stop":
            os.replace(run, os.path.join(self.q, "interrupted", tid + ".json"))
            self._set(tid, state="interrupted", note="служба остановлена во время расчёта", **fin)
        elif reason == "timeout":
            os.replace(run, os.path.join(self.q, "failed", tid + ".json"))
            self._set(tid, state="failed", error=f"превышено предельное время {timeout / 60:.0f} мин", **fin)
        elif rc == 0:
            st = self._result_fields(tid, kind)
            ok = st.pop("_ok")
            os.replace(run, os.path.join(self.q, "done" if ok else "failed", tid + ".json"))
            self._set(tid, state="done" if ok else "failed", **st, **fin)
        else:
            os.replace(run, os.path.join(self.q, "failed", tid + ".json"))
            self._set(tid, state="failed", error=self._tail(logp), returncode=rc, **fin)
        self.journal("finish", id=tid, state=self.status[tid]["state"], reason=reason)

    def trades(self, tid, limit=3000):
        import pandas as pd
        p = os.path.join(self.res, tid, "best_trades.csv")
        if not os.path.exists(p):
            raise Refuse("нет best_trades.csv", 404)
        df = pd.read_csv(p)
        if df.empty:
            return {"id": tid, "count": 0}
        t_exit = pd.to_datetime(df["ExitTimeUTC"], utc=True, errors="coerce")
        t_ent = pd.to_datetime(df["EntryTimeUTC"], utc=True, errors="coerce")
        df = df.assign(x_exit=t_exit, x_entry=t_ent).sort_values("x_exit")
        cum = df["NetUSD"].astype(float).cumsum()
        dd = cum - cum.cummax()
        idx = list(range(0, len(df), max(1, len(df) // 2000))) + [len(df) - 1]
        eq = [[int(df["x_exit"].iloc[i].timestamp()), round(float(cum.iloc[i]), 2), round(float(dd.iloc[i]), 2)] for i in sorted(set(idx))
              if pd.notna(df["x_exit"].iloc[i])]
        years = []
        for y, g in df.groupby(df["x_exit"].dt.year):
            w, l = g.NetUSD[g.NetUSD > 0].sum(), -g.NetUSD[g.NetUSD < 0].sum()
            years.append({"year": int(y), "trades": int(len(g)), "net": round(float(g.NetUSD.sum()), 2),
                          "win_pct": round(float((g.NetUSD > 0).mean() * 100), 1), "pf": round(float(w / l), 3) if l > 0 else None})
        cap = {}
        if {"MFE_Ticks", "GrossTicks"} <= set(df.columns):
            mfe, gross = df.MFE_Ticks.astype(float), df.GrossTicks.astype(float)
            pos = mfe > 0
            win = pos & (gross > 0)
            cap = {"capture_total": round(float(gross[pos].sum() / mfe[pos].sum()), 4) if mfe[pos].sum() > 0 else None,
                   "capture_winners_median": round(float((gross[win] / mfe[win]).median()), 4) if win.any() else None,
                   "definition": "доля забранного потенциала: GrossTicks / MFE_Ticks (по сделкам с MFE > 0)"}
        ent = df.head(limit)
        cols = [c for c in ("Direction", "EntryPriceWithSlip", "ExitPriceWithSlip", "ExitReason", "NetUSD", "Profile", "Session") if c in df.columns]
        tr = [{"entry": int(r.x_entry.timestamp()) if pd.notna(r.x_entry) else None, "exit": int(r.x_exit.timestamp()) if pd.notna(r.x_exit) else None,
               **{c: (None if pd.isna(getattr(r, c)) else getattr(r, c)) for c in cols}} for r in ent.itertuples()]
        return {"id": tid, "count": int(len(df)), "net": round(float(cum.iloc[-1]), 2), "max_drawdown": round(float(dd.min()), 2),
                "equity": eq, "years": years, "mfe_capture": cap,
                "exit_reasons": {str(k): int(v) for k, v in df["ExitReason"].value_counts().items()} if "ExitReason" in df else {},
                "ambiguous_trades": int((df["AmbiguousBars"] > 0).sum()) if "AmbiguousBars" in df else None,
                "trades": tr, "trades_truncated": len(df) > limit}

    def schema(self):

    @staticmethod
    def convert(p, v):
        t = p.get("type")
        if t == "bool":
            return (v, None) if isinstance(v, bool) else (None, "ожидалось true или false")
        if isinstance(v, bool) and t in ("int", "double"):
            return None, "ожидалось число"
        if t == "int":
            if not isinstance(v, (int, float)) or not float(v).is_integer():
                return None, f"ожидалось целое, получено {v}"
            v = int(v)
        elif t == "double":
            if not isinstance(v, (int, float)) or not math.isfinite(float(v)):
                return None, "ожидалось число"
            v = float(v)
        elif t == "enum":
            return (str(v), None) if str(v) in (p.get("enum") or []) else (None, f"нет значения {v} в перечислении")
        elif t == "string":
            return (v, None) if isinstance(v, str) else (None, "ожидалась строка")
        else:
            return None, f"тип {t} пресетом не задаётся"
        lo, hi = p.get("min"), p.get("max")
        if (lo is not None and v < lo) or (hi is not None and v > hi):
            return None, f"значение {v} вне предела {lo}…{hi}"
        return v, None

    def bars(self, q):
        import numpy as np
        import pandas as pd
        sym = (q.get("symbol") or ["MGC"])[0]
        try:
            tf = int((q.get("tf") or ["5"])[0])
            limit = max(1, min(int((q.get("limit") or ["5000"])[0]), 20000))
        except ValueError:
            raise Refuse("tf и limit — целые числа")
        if not SYM_RE.fullmatch(sym) or not 1 <= tf <= 1440:
            raise Refuse("символ и ТФ от 1 до 1440 минут")
        with self.lock:
            if sym not in self._m1:
                cert = os.path.join(self.root, "data_certified", sym, f"{sym}_DATA_CERTIFICATE.json")
                if not os.path.exists(cert):
                    raise Refuse(f"нет сертифицированных данных {sym}", 404)
                cd = load_module(os.path.join(self.root, "DZ_CERT_Data.py"), "dz_svc_certdata")
                ok, errs, _ = cd.verify(cert, quiet=True)
                if not ok:
                    raise Refuse(f"сертификат {sym} не прошёл проверку: {'; '.join(errs)}", 409)
                z = np.load(os.path.join(os.path.dirname(cert), f"{sym}_M1.npz"), allow_pickle=False)
                self._m1[sym] = {k: z[k] for k in ("t_open", "o", "h", "l", "c", "v") if k in z.files}
            m = self._m1[sym]
        t = m["t_open"]
        try:
            to = pd.Timestamp(q["to"][0], tz="UTC") if q.get("to") else pd.Timestamp(int(t[-1]), unit="ns", tz="UTC") + pd.Timedelta(minutes=1)
            fr = pd.Timestamp(q["from"][0], tz="UTC") if q.get("from") else to - pd.Timedelta(minutes=tf * limit * 3)
        except ValueError:
            raise Refuse("from и to — даты ГГГГ-ММ-ДД")
        i0 = int(np.searchsorted(t, (fr - pd.Timedelta(minutes=tf)).value, "left"))
        i1 = int(np.searchsorted(t, to.value, "left"))
        if i1 - i0 > 3_000_000:
            raise Refuse("слишком длинный период для одного запроса")
        df = pd.DataFrame({k: m[k][i0:i1] for k in ("o", "h", "l", "c", "v") if k in m}, index=pd.to_datetime(t[i0:i1], utc=True))
        if tf > 1 and len(df):
            b = self.mod("san2").resample(df, tf)
            b = b[(b.index >= fr) & (b.index < to)]
        else:
            b = df[(df.index >= fr) & (df.index < to)]
        b = b.tail(limit)
        return {"symbol": sym, "tf": tf, "count": int(len(b)), "t": [int(x.timestamp()) for x in b.index],
                **{k: [round(float(x), 6) for x in b[k].values] for k in ("o", "h", "l", "c", "v") if k in b},
                "source": "сертифицированные M1, старшие ТФ — очиститель v2 от 17:00 по Чикаго", "time": "UTC, открытие бара"}

    def desync_loop(self):
        c = self.cfg["desync_watch"]
        tokf = os.path.join(self.nt_root, "dz_export", "atlas_ws_token.txt")
        while not self.stop_evt.is_set():
            ws = None
            try:
                with open(tokf, encoding="utf-8") as f:
                    tok = f.read().strip()
                ws = MiniWs(int(c.get("ws_port", 8088)))
                ws.connect()
                ws.send_text(json.dumps({"cmd": "auth", "token": tok}))
                commands, since, paused, ep = False, None, False, 0
                self.desync.update(state="подключён")
                while not self.stop_evt.is_set():
                    m = json.loads(ws.recv_text())
                    if m.get("t") == "auth":
                        commands = bool(m.get("commands"))
                    if m.get("t") != "bar":
                        continue
                    side, qty = int(m.get("pos") or 0), abs(int(m.get("qty") or 0))
                    acc = [p for p in self.account()["positions"] if p["instrument"].startswith(c.get("instrument_prefix", ""))
                           and (not c.get("account") or p["account"] == c["account"])]
                    a_side = sum({"LONG": 1, "SHORT": -1}.get(p["side"], 0) for p in acc)
                    a_qty = sum(p["qty"] or 0 for p in acc if p["side"] in ("LONG", "SHORT"))
                    match = (side == a_side) and (side == 0 or qty == a_qty)
                    self.desync.update(strategy={"side": side, "qty": qty}, account={"side": a_side, "qty": a_qty, "files": len(acc)},
                                       match=match, utc=utc_now())
                    if match:
                        since, paused = None, False
                        continue
                    since = since or time.time()
                    if time.time() - since >= float(c.get("grace_sec", 20)) and ep != int(since):
                        ep = int(since)
                        al = {"utc": utc_now(), "kind": "desync", "strategy": {"side": side, "qty": qty},
                              "account": {"side": a_side, "qty": a_qty}, "seconds": round(time.time() - since, 1)}
                        if c.get("action") == "pause" and commands and not paused:
                            ws.send_text(json.dumps({"cmd": "pause", "token": tok}))
                            al["action"], paused = "пауза входов отправлена", True
                        with self.lock:
                            self.alarms = (self.alarms + [al])[-200:]
                        self.journal("alarm", **al)
            except Exception as ex:
                self.desync.update(state=f"нет связи: {str(ex)[:160]}")
            finally:
                if ws:
                    ws.close()
            self.stop_evt.wait(5.0)

def handler(svc):
    class H(BaseHTTPRequestHandler):
        server_version = "Atlas/2"

        def log_message(self, *a):
            pass

        def _origin(self):
            o = self.headers.get("Origin")
            return o is None or o in ALLOWED_ORIGINS, o

        def _send(self, code, obj):
            b = json.dumps(obj, ensure_ascii=False, default=str).encode("utf-8")
            ok, o = self._origin()
            self.send_response(code)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            if ok:
                self.send_header("Access-Control-Allow-Origin", o or "null")
            self.send_header("Cache-Control", "no-store")
            self.send_header("Content-Length", str(len(b)))
            self.end_headers()
            self.wfile.write(b)

        def _guard(self):
            ok, _ = self._origin()
            if not ok:
                self._send(403, {"error": "чужой источник"})
                return False
            if not hmac.compare_digest((self.headers.get("X-Atlas-Token") or "").encode("utf-8"), svc.token.encode("utf-8")):
                self._send(401, {"error": "неверный ключ"})
                return False
            return True

        def _body(self):
            n = int(self.headers.get("Content-Length") or 0)
            if n > MAX_BODY:
                raise Refuse("слишком большое тело запроса", 413)
            raw = self.rfile.read(n) if n else b"{}"
            try:
                return json.loads(raw.decode("utf-8") or "{}")
            except ValueError:
                raise Refuse("тело запроса — не JSON")

        def do_OPTIONS(self):
            ok, o = self._origin()
            self.send_response(204 if ok else 403)
            if ok:
                self.send_header("Access-Control-Allow-Origin", o or "null")
                self.send_header("Access-Control-Allow-Headers", "X-Atlas-Token, Content-Type")
                self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                if self.headers.get("Access-Control-Request-Private-Network") == "true":
                    self.send_header("Access-Control-Allow-Private-Network", "true")
            self.send_header("Content-Length", "0")
            self.end_headers()

        def _route(self, method):
            u = urlparse(self.path)
            q = parse_qs(u.query)
            p = u.path.rstrip("/") or "/"
            if method == "GET" and p == "/health":
                return self._send(200, {"service": VERSION, "ok": True, "worker_alive": time.time() - svc.worker_beat < 15,
                                        "deps_ok": svc.deps["ok"], "utc": utc_now()})
            if not self._guard():
                return
            try:
                if method == "GET":
                    R = {"/status": lambda: {"service": VERSION, "jobs": svc.snapshot(), "deps": svc.deps,
                                             "worker_alive": time.time() - svc.worker_beat < 15,
                                             "running": svc.current["id"] if svc.current else None,
                                             "holdout": {"from": svc.cfg.get("holdout_from"), "policy": svc.cfg.get("holdout_policy")},
                                             "experiments": {"budget_lock": bool(svc.cfg["budget_lock"].get("enabled")),
                                                             "desync_watch": svc.desync}, "alarms": svc.alarms[-20:]},
                         "/summary": lambda: need(read_json(os.path.join(svc.res, svc._rid(q), "summary.json")), "нет итога"),
                         "/log": lambda: self._log(q),
                         "/results": svc.results,
                         "/trades": lambda: svc.trades(svc._rid(q), min(int((q.get("limit") or ["3000"])[0]), 20000)),
                         "/trials": lambda: svc.trials(svc._rid(q)),
                         "/wfa": lambda: need(read_json(os.path.join(svc.res, svc._rid(q) + "_wfa.json")), "нет итога проверки вперёд"),
                         "/registry": svc.registry,
                         "/certificates": svc.certificates,
                         "/gates": svc.gates,
                         "/account": svc.account,
                         "/schema": lambda: svc.schema()[0],
                         "/graph": lambda: svc.mod("graph"),
                         "/bars": lambda: svc.bars(q),
                         "/alarms": lambda: {"alarms": svc.alarms, "desync": svc.desync},
                         "/journal": lambda: self._journal(q),
                         "/config": lambda: svc.cfg}
                    if p not in R:
                        return self._send(404, {"error": "нет такого адреса"})
                    return self._send(200, R[p]())
                body = self._body()
                if p == "/task":
                    tid, notes = svc.enqueue(body)
                    return self._send(200, {"queued": tid, "notes": notes})
                if p == "/cancel":
                    return self._send(200, {"cancelled": svc._rid(q), "was": svc.cancel(svc._rid(q))})
                if p == "/requeue":
                    new, notes = svc.requeue(svc._rid(q))
                    return self._send(200, {"queued": new, "notes": notes})
                if p == "/verify":
                    return self._send(200, svc.verify((q.get("what") or [""])[0], q))
                if p == "/preset":
                    return self._send(200, svc.preset(body))
                if p == "/decision":
                    return self._send(200, svc.decision(body))
                if p == "/emergency/flatten_all":
                    return self._send(200, svc.emergency("flatten_all", body))
                if p == "/emergency/close_position":
                    return self._send(200, svc.emergency("close_position", body))
                return self._send(404, {"error": "нет такого адреса"})
            except Refuse as ex:
                return self._send(ex.code, {"error": str(ex)})
            except Exception as ex:
                svc.journal("http_error", path=p, error=traceback.format_exc()[-1500:])
                return self._send(500, {"error": f"{type(ex).__name__}: {ex}"})

        def _log(self, q):
            tid = svc._rid(q)
            p = os.path.join(svc.logs, tid + ".log")
            if not os.path.exists(p):
                raise Refuse("журнала нет", 404)
            size = os.path.getsize(p)
            off = max(0, min(int((q.get("offset") or ["0"])[0]), size))
            with open(p, "rb") as f:
                f.seek(off)
                chunk = f.read(256 * 1024)
            return {"id": tid, "offset": off, "next": off + len(chunk), "size": size,
                    "text": chunk.decode("utf-8", "replace"), "state": (svc.status.get(tid) or {}).get("state")}

        def _journal(self, q):
            n = min(int((q.get("limit") or ["200"])[0]), 2000)
            if not os.path.exists(svc.journal_path):
                return {"records": []}
            with open(svc.journal_path, encoding="utf-8") as f:
                lines = f.readlines()[-n:]
            return {"records": [json.loads(x) for x in lines if x.strip()]}

        def do_GET(self):
            self._route("GET")

        def do_POST(self):
            self._route("POST")
    return H


def single_instance(path):
    """Вторая служба на той же очереди не стартует: файл-замок держится открытым до выхода процесса."""
    f = open(path, "a+")
    try:
        if os.name == "nt":
            import msvcrt
            f.seek(0)
            msvcrt.locking(f.fileno(), msvcrt.LK_NBLCK, 1)
        else:
            import fcntl
            fcntl.flock(f.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
    except OSError:
        f.close()
        return None
    return f

## 1. Ворота выпуска

| 3 | Сертификат стратегии | `DZ_CERT_Strategy.py certify --passport` | жёлтый; исходник совпадает с паспортом сборки |
| 4 | Паспорт инструмента и данные | `DZ_CERT_Instrument.py`, `DZ_CERT_Data.py` | выпущены, отпечатки целы |
| 5 | Шаблон | `DZ_CERT_Template.py` | жёлтый |

| 7 | Зелёный ранг | `DZ_CERT_Rank.py green` | присвоен |
| 8 | Воспроизведение рынка | NinjaTrader, режим без ордеров | `[HUB-ИТОГ]`: потерь 0, отказов 0 |
| 9 | Пульт | `DZ_XL_Cockpit.py` | баннер 🟢 при режиме «реальный счёт» |

## 4. Стратегия в NinjaTrader — `DZ_SW_Hub_S021`

**Коммутатор** — единственная стратегия NinjaTrader и **единственная точка ордеров**. Модули ордеров не подают: они выдают сигналы (`SwSignal`) и требования (`SwAction`), коммутатор решает и исполняет.

**Конвейер на каждом баре серии исполнения:**
1. модули вычисляют и кладут сигналы в свой ящик (`Emit`);
2. коммутатор разбирает ящики (`DrainSignals`, `DrainActions`);
3. команды пульта (`ProcessRemote`);
4. виртуальный стоп и сопровождение (финансовый модуль);
5. исполнение требований;
6. оценка входа (`EvaluateEntry`) — только без позиции;
7. снимок пульту, панель.

**Контракты:**
- `SwModule` — `Attach(SwHostView)`, `Compute(bip)`, `Emit`, `EmitVisual`, `Health`, `Role`, `TelemetryJson`;
- `SwHostView` — **единственный** доступ модуля к платформе. Модуль не трогает `BarsArray` и ордера;
- перечисления меню — только в пространстве `NinjaTrader.NinjaScript`, иначе ломаются обёртки NinjaTrader;
- модуль: файл `DZ_SW_Mod_<Имя>_S021.cs`, классы `Sw<Имя>Config` и `Sw<Имя>Module : SwModule`; буферы — в `Attach`, без выделения памяти на такте.

## 3. Как читать результат

| Показатель | Что значит | Решение |
|---|---|---|
| **Порог «случайного лучшего»** на пульте | Шарп, который даст лучший из N наборов без всякого преимущества | результат ниже порога — это шум |
| **Дефлированный Шарп (DSR)** | вероятность, что лучший Шарп выше ожидаемого максимума при нулевом преимуществе, с учётом **всех** испытаний проекта | ≥ 0,95 — преимущество доказано; ниже — нет |
| **PBO (CSCV)** | вероятность, что лучший на одной половине истории окажется хуже медианы на другой | ≤ 0,5 обязательно; ≤ 0,15 — для золота |
| **Плато против пика** | плато — лучший по среднему в окрестности ±1 шаг сетки; пик — одиночный максимум | выбирать плато; пик без плато — удача |
| **Эффективность проверки вперёд** | прибыль вне выборки в день / в выборке в день | около 1 — устойчиво; около 0 и ниже — подгонка |

**Каждое испытание повышает порог.** Широкая сетка и частые перезапуски делают доказательство преимущества дороже — это записывается в реестр и учитывается автоматически.

## 4. Если что-то не так

| Признак | Причина | Что сделать |
|---|---|---|
| «служба: …» под кнопкой | служба не запущена или неверен ключ | запустить `START_ATLAS_SERVICE.bat`, скопировать ключ из файла |
| Задача в состоянии «ошибка» с «нет сертифицированных данных» | инструмент не сертифицирован | раздел 1, шаг 2 |
| «сертификат данных … не прошёл проверку» | данные изменены после сертификации | сертифицировать заново; разобраться, кто изменил |
| Пульт не видит инструмент | в графе нет профиля | добавить профиль в `instrument_profiles` |

**Начальный риск** `[E03]` задаёт масштаб: все прочие расстояния в тиках осмысленны только относительно него. **Безубыток** переводит стоп за цену входа после триггера; отступ покрывает издержки круга и проскальзывание. **Трейлинг** подтягивает стоп за ценой; трейлинг короче триггера безубытка подтягивает стоп выше безубытка сразу. **Охрана** ограничивает день: после предела потерь или прибыли входы закрыты.

## Глава 7. Статистика доказательства

**Переподгонка** — выбор параметров, удачных на истории случайно. Её цена растёт с числом испытаний N: лучший из N наборов без всякого преимущества даёт Шарп около `E[max] ≈ (1−γ)Φ⁻¹(1−1/N) + γΦ⁻¹(1−1/(Ne))` в единицах σ (Бейли, Лопес де Прадо). Для N = 100 это ~2,5σ.

**DSR** — вероятность, что лучший Шарп выше этого порога с учётом длины выборки, асимметрии и эксцесса. **PBO (CSCV)** — доля разбиений истории, где лучший на одной половине хуже медианы на другой. **Плато** — устойчивость к соседним значениям: оптимум, окружённый хорошими соседями, переносится в будущее лучше одиночного пика. **Проверка вперёд** — единственная проверка на данных, которых выбор не видел; её эффективность — отношение доходности вне выборки к доходности в выборке.

"""DZ_CERT_Common.py — общие функции сертификации Атласа, v1.0 (2026-09-28)."""
import hashlib, json, os, stat, datetime as dt


def sha_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def sha_obj(obj):
    """Отпечаток содержимого JSON без поля content_sha256 — сертификат проверяет сам себя."""
    body = {k: v for k, v in obj.items() if k != "content_sha256"}
    return hashlib.sha256(json.dumps(body, ensure_ascii=False, sort_keys=True).encode("utf-8")).hexdigest()


def now_utc():
    return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def make_writable(path):
    if os.path.exists(path):
        os.chmod(path, stat.S_IREAD | stat.S_IWRITE)


def seal_readonly(path):
    os.chmod(path, stat.S_IREAD | stat.S_IRGRP | stat.S_IROTH)


def write_cert(path, obj):
    obj["content_sha256"] = sha_obj(obj)
    make_writable(path)
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, indent=2)
    seal_readonly(path)
    return obj


def read_cert(path):
    obj = json.load(open(path, encoding="utf-8"))
    return obj, obj.get("content_sha256") == sha_obj(obj)

  verify   все сертификаты папки: документ на месте, отпечаток совпадает, сертификат цел.
           ЭКСПЕРИМЕНТ: висячие ссылки — упомянутые в тексте файлы DZ_*, которых нет в хранилище.

  python DZ_CERT_Library.py certify --doc library_certified/X.md --author LEADER --reviewer OPUS_MAX

VERSION = "DZ_CERT_Library-v1.0 (2026-09-28)"
PRIMARY = "OPUS_MAX"


def certify(doc, author, reviewer):
    author, reviewer = author.upper(), (reviewer or "").upper()
    if author == PRIMARY: rank, why = "GOLD", "первоисточник"
    elif reviewer and reviewer != author: rank, why = "GREEN", f"перекрёстная проверка: {reviewer}"
    else: rank, why = "REJECTED", "нет перекрёстной проверки: рецензент не указан или совпадает с автором"
    cert = {"certificate": "DZ-LIBRARY-CERT-v1.0", "issued_utc": now_utc(), "document": os.path.basename(doc),
            "sha256": sha_file(doc), "author_declared": author, "reviewer_declared": reviewer or None,
            "rank": rank, "basis": why, "note": "автор и рецензент заявлены, а не доказаны"}
    write_cert(doc + ".doc.cert.json", cert)
    if rank != "REJECTED": seal_readonly(doc)
    print(f"[библиотека] {cert['document']}: {rank} — {why}")
    return rank != "REJECTED"


def verify(folder, root):
    names = {f for _, _, fs in os.walk(root) for f in fs}; bad = 0
    for c in sorted(glob.glob(os.path.join(folder, "*.doc.cert.json"))):
        cert, ok = read_cert(c); doc = c[: -len(".doc.cert.json")]
        state = ("ИЗМЕНЁН" if not ok else "НЕТ ДОКУМЕНТА" if not os.path.exists(doc)
                 else "ИЗМЕНЁН" if sha_file(doc) != cert["sha256"] else cert["rank"])
        refs = sorted(set(re.findall(r"\bDZ_[\w\-]+\.(?:py|cs|json|md|xlsx)\b", open(doc, encoding="utf-8", errors="replace").read()))) if os.path.exists(doc) else []
        dangling = [r for r in refs if r not in names]
        bad += state in ("ИЗМЕНЁН", "НЕТ ДОКУМЕНТА")
        print(f"  {state:14} {cert['document']}" + (f"\n    висячие ссылки: {', '.join(dangling)}" if dangling else ""))
    return bad == 0


def main():
    ap = argparse.ArgumentParser(description=VERSION)
    ap.add_argument("cmd", choices=["certify", "verify"]); ap.add_argument("--doc"); ap.add_argument("--author")
    ap.add_argument("--reviewer"); ap.add_argument("--dir", default="library_certified"); ap.add_argument("--root", default=".")
    a = ap.parse_args()

if __name__ == "__main__":

def fresh(path):
    """Сертификаты не перезаписываются: повторное присвоение получает номер версии."""
    if not os.path.exists(path):
        return path
    base = path[: -len(".cert.json")]; n = 2
    while os.path.exists(f"{base}_v{n:03d}.cert.json"):
        n += 1
    return f"{base}_v{n:03d}.cert.json"


def metrics(pnl):
    pnl = np.asarray(pnl, float); eq = np.cumsum(pnl)
    gp, gl = pnl[pnl > 0].sum(), -pnl[pnl < 0].sum()
    return {"trades": int(len(pnl)), "net": round(float(pnl.sum()), 2), "mean": round(float(pnl.mean()), 4),
            "std": round(float(pnl.std(ddof=1)), 4) if len(pnl) > 1 else 0.0, "pf": round(float(gp / gl), 4) if gl > 0 else None,
            "max_dd": round(float((np.maximum.accumulate(np.r_[0, eq]) - np.r_[0, eq]).max()), 2)}

# -*- coding: utf-8 -*-
"""
DZ_CERT_Strategy.py — аудит восьми элементов стратегии, v1.0 (2026-09-28)

  certify  по STRATEGY_ELEMENTS.json: наличие восьми элементов, их отпечатки, «только чтение»;

  verify   пересчёт отпечатков по сертификату.
Ранг: всё в порядке → ЖЁЛТЫЙ; иначе ОТКЛОНЁН. Зелёный и золотой — по сертификату допуска (DOC7).

  python DZ_CERT_Strategy.py certify --dir strategies_vault/DZ_S021 [--passport build_passport.json]
"""
import argparse, json, os, re, sys
from DZ_CERT_Common import sha_file, write_cert, read_cert, now_utc, seal_readonly

VERSION = "DZ_CERT_Strategy-v1.0 (2026-09-28)"
ELEMENTS = ["DOC1_TZ", "DOC2_MANUAL", "DOC3_DEV_GUIDE", "DOC4_SETTINGS_DICT", "DOC5_SETTINGS_LINKS",
            "DOC6_MODULES", "DOC7_TRADING_CERTIFICATE", "DOC8_ENGINE_CS"]


def cs_checks(path):
    """Версия в шапке и баланс скобок вне комментариев и строк."""
    src = open(path, encoding="utf-8", errors="replace").read()
    head = "\n".join(src.splitlines()[:40])
    ver = re.search(r"v\d+\.\d+(\.\d+)?", head)
    code = re.sub(r"//[^\n]*|/\*.*?\*/", "", src, flags=re.S)
    code = re.sub(r'@"(?:[^"]|"")*"|"(?:\\.|[^"\\])*"|\'(?:\\.|[^\'\\])\'', '""', code)
    bal = {p: code.count(p[0]) - code.count(p[1]) for p in ("{}", "()", "[]")}
    return (ver.group(0) if ver else None), {k: v for k, v in bal.items() if v}


def certify(d, passport):
    man = os.path.join(d, "STRATEGY_ELEMENTS.json")
    if not os.path.exists(man):
        sys.exit(f"ОТКАЗ: нет {man}")
    m = json.load(open(man, encoding="utf-8")); errors, warnings, found = [], [], {}
    for e in ELEMENTS:
        f = m.get(e)
        if not f or not os.path.exists(os.path.join(d, f)):
            errors.append(f"нет {e}"); continue
        found[e] = {"file": f, "sha256": sha_file(os.path.join(d, f))}
    eng = found.get("DOC8_ENGINE_CS")
    if eng:
        ver, bad = cs_checks(os.path.join(d, eng["file"])); eng["version"] = ver
        if not ver: errors.append("DOC8: в шапке исходника нет версии вида vN.N")
        if bad: errors.append(f"DOC8: скобки не сбалансированы {bad} — файл обрезан или повреждён")
        if passport:
            ps = json.load(open(passport, encoding="utf-8"))
            hit = [r for r, meta in ps.get("files", {}).items() if os.path.basename(r) == os.path.basename(eng["file"])]
            if not hit: warnings.append("DOC8 не найден в паспорте сборки")
            elif ps["files"][hit[0]]["sha256"] != eng["sha256"]:
                errors.append("DOC8 отличается от исходника, скомпилированного в NinjaTrader (паспорт сборки)")
            else:
                eng["nt_build_id"] = ps.get("build_id")
    for e in found.values():
        seal_readonly(os.path.join(d, e["file"]))
    cert = {"certificate": "DZ-STRATEGY-CERT-v1.0", "issued_utc": now_utc(), "strategy": os.path.basename(os.path.abspath(d)),
            "rank": "REJECTED" if errors else "YELLOW", "elements": found, "errors": errors, "warnings": warnings}
    write_cert(os.path.join(d, "STRATEGY_CERT.json"), cert)
    print(f"[стратегия] {cert['strategy']}: {cert['rank']}, элементов {len(found)} из 8"
          + (f", движок {eng.get('version')}" if eng else "") + (f", сборка NinjaTrader {eng.get('nt_build_id')}" if eng and eng.get("nt_build_id") else ""))
    for x in errors: print("  ОШИБКА:", x)
    for x in warnings: print("  ВНИМАНИЕ:", x)
    return not errors


def verify(d):
    c, ok = read_cert(os.path.join(d, "STRATEGY_CERT.json")); errs = [] if ok else ["сертификат изменён после выпуска"]
    for e, meta in c["elements"].items():
        p = os.path.join(d, meta["file"])

        elif sha_file(p) != meta["sha256"]: errs.append(f"{e}: отпечаток не совпадает")
    print(f"[проверка] {c['strategy']}: " + ("элементы соответствуют сертификату" if not errs else "НАРУШЕНИЕ: " + "; ".join(errs)))
    return not errs

def parse_hub(path):
    """Свойство → индекс, тип, пределы — из атрибутов исходника коммутатора."""
    src = open(path, encoding="utf-8").read()
    props, prev = {}, 0
    for m in re.finditer(r"public\s+([\w\.<>]+)\s+(\w+)\s*\{\s*get;\s*set;\s*\}", src):
        block = src[prev:m.start()]; prev = m.end()
        d = re.search(r'Display\(Name = "\[(\w+)\]', block)
        if not d:
            continue
        r = re.search(r"\[Range\(\s*([-\d\.]+)\s*,\s*([-\d\.]+)\s*\)\]", block)
        props[m.group(2)] = {"index": d.group(1), "type": m.group(1),
                             "range": (float(r.group(1)), float(r.group(2))) if r else None}
    return props


def load_values(path):
    if path.endswith(".json"):
        return {k: str(v) for k, v in json.load(open(path, encoding="utf-8")).items()}
    vals = {}
    for el in ET.parse(path).iter():
        if len(el) == 0 and el.text is not None:
            vals[el.tag.split("}")[-1]] = el.text.strip()
    return vals


def main():

    ap.add_argument("--template", required=True); ap.add_argument("--hub", required=True)
    ap.add_argument("--instrument-cert"); ap.add_argument("--out")

    props = parse_hub(a.hub); raw = load_values(a.template)
    by_idx = {p["index"]: n for n, p in props.items()}

    def val(name):
        if name not in raw: return None
        t = props[name]["type"]
        if t == "bool": return raw[name].lower() == "true"
        if t in ("int", "double", "float"):
            try: return float(raw[name])
            except ValueError: return raw[name]
        return raw[name]

    def v(idx):
        n = by_idx.get(idx)
        return val(n) if n else None

    errors, warnings, checked = [], [], 0
    for name, p in props.items():                                       # 1. пределы
        x = val(name)
        if x is None or p["range"] is None or not isinstance(x, float):
            continue
        checked += 1
        lo, hi = p["range"]
        if not (lo <= x <= hi):
            errors.append(f"[{p['index']}] {name} = {x:g} вне предела {lo:g}…{hi:g}")

    if a.instrument_cert:                                                 # 4. риск и издержки
        pas, ok = read_cert(a.instrument_cert)
        if not ok:
            errors.append("паспорт инструмента изменён после выпуска")

            usd = stop * tv
            if not (lo <= usd <= hi):
                errors.append(f"[E03] стоп {stop:g} т = {usd:g} за контракт вне паспорта {lo:g}…{hi:g}")
        off, be = v("F32"), v("F31")
        need = pas["costs_per_contract"]["min_breakeven_offset_ticks_costs_and_slippage"]
        if isinstance(off, float) and str(be) != "Off" and off < need:
            warnings.append(f"[F32] отступ безубытка {off:g} т меньше издержек круга и проскальзывания ({need} т)")
    if v("E01") is True:
        warnings.append("[E01] режим без ордеров включён — для реального счёта его нужно выключить")

    cert = {"certificate": "DZ-TEMPLATE-CERT-v1.0", "issued_utc": now_utc(),
            "rank": "REJECTED" if errors else "YELLOW",
            "rank_meaning": "отклонён" if errors else "жёлтый: допуск к бэктесту и воспроизведению рынка",
            "template_file": os.path.basename(a.template), "template_sha256": sha_file(a.template),
            "hub_source_sha256": sha_file(a.hub), "instrument": pas.get("symbol") if pas else None,
            "instrument_cert_sha256": pas.get("content_sha256") if pas else None,

    print(f"[шаблон] {cert['rank']}: ошибок {len(errors)}, предупреждений {len(warnings)}, пределов проверено {checked} → {out}")
    for e in errors: print("  ОШИБКА:", e)
    for w in warnings: print("  ВНИМАНИЕ:", w)
    sys.exit(1 if errors else 0)


if __name__ == "__main__":
    main()

Код возврата 0 — проверка пройдена, 1 — нет. Только чтение входных файлов.

import argparse, importlib.util, os, sys
import numpy as np
import pandas as pd

VERSION = "DZ_DATA_Check_S021-v1.0 (2026-10-04)"
HERE = os.path.dirname(os.path.abspath(__file__))


def _mod(fname, name):
    s = importlib.util.spec_from_file_location(name, os.path.join(HERE, fname))
    m = importlib.util.module_from_spec(s); s.loader.exec_module(m); return m


def compare(a):
    san = _mod("DZ_DATA_Sanitizer_v2.py", "san")
    old, new = san.load_clean(a.old), san.load_clean(a.new)
    lo, hi = max(old.index[0], new.index[0]), min(old.index[-1], new.index[-1])
    o, n = old.loc[lo:hi], new.loc[lo:hi]
    both = o.index.intersection(n.index)
    eq = (o.loc[both, ["o", "h", "l", "c"]].values == n.loc[both, ["o", "h", "l", "c"]].values).all(axis=1)
    diff = np.abs(o.loc[both, ["o", "h", "l", "c"]].values - n.loc[both, ["o", "h", "l", "c"]].values).max(axis=1)
    print(f"{VERSION} · compare\n  общий период {lo} … {hi}")
    print(f"  минуток: старая {len(o):,}, новая {len(n):,}, в обеих {len(both):,}; "
          f"только в старой {len(o.index.difference(n.index)):,}, только в новой {len(n.index.difference(o.index)):,}")
    print(f"  одинаковых OHLC {eq.mean():.4%}; наибольшее расхождение цены {diff.max() if len(diff) else 0:.4f}")
    yr = pd.Series(eq, index=both.year).groupby(level=0).mean()
    for y, v in yr.items():
        if v < 1.0:
            print(f"    {y}: одинаковых {v:.4%}")
    ok = len(both) > 0 and eq.mean() >= a.min_equal
    print("  ИТОГ: " + ("версии совпадают на общем периоде" if ok else f"совпадение ниже порога {a.min_equal:.2%} — разбор обязателен"))
    return ok

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
==========================================================================================
DZ_DATA_Sanitizer.py — очистка минутных данных и канонический ресемплинг, v1.0 (2026-09-26)

  clean     сырой CSV → проверенный кэш M1 в UTC (.npz, либо .parquet при наличии pyarrow)
  diff      сравнение двух манифестов: какие дни изменились между выгрузками
  resample  старший таймфрейм из кэша M1 в CSV — для сверки с графиком NinjaTrader

ПРИНЦИПЫ
  Единственный источник разбора. Формат даты и разметка бара определяются функциями
  эталона DZ_BT_DualBreakout_S021.py: второй разборщик разошёлся бы с первым.
  Выбросы не удаляются, а помечаются: удаление по порогу срезает именно те прострелы,
  что выбили бы стоп, и незаметно улучшает результат.
  Время хранится целыми НАНОСЕКУНДАМИ UTC, единица записана в метаданных (урок Д-23).
  Схема кэша совпадает с кэшем DZ_OPT_Engine_S021: один формат, а не два.
==========================================================================================
"""
import argparse, hashlib, importlib.util, json, os, sys
import numpy as np
import pandas as pd

VERSION = "DZ_DATA_Sanitizer-v1.0 (2026-09-26)"
TZ_EX = "America/Chicago"
NS_MIN = 60_000_000_000


def load_ref(path):
    spec = importlib.util.spec_from_file_location("dz_reference", path)
    m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m


def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def to_ns(idx):
    idx = pd.DatetimeIndex(idx)
    return idx.as_unit("ns").asi8 if hasattr(idx, "as_unit") else idx.asi8

def clean(raw, out, ref_path, spike_k, min_coverage, src_tz=TZ_EX):
    ref = load_ref(ref_path)
    rep = {"script": VERSION, "raw_file": os.path.basename(raw), "raw_sha256": sha256_file(raw)}

    stamp, df = ref.read_raw(raw)
    fmt, fmt_rep = ref.detect_date_format(stamp, "auto")
    ts = pd.to_datetime(stamp, format=fmt, errors="coerce")
    num = lambda c: pd.to_numeric(df[c], errors="coerce") if c in df.columns else pd.Series(np.nan, index=df.index)
    d = pd.DataFrame({"ts": ts, "o": num("Open"), "h": num("High"), "l": num("Low"),
                      "c": num("Close"), "v": num("Volume")})
    rep["rows_read"] = int(len(d))
    rep["volume_present"] = bool(d.v.notna().any())
    d["v"] = d.v.fillna(0.0)

    bad = d[["ts", "o", "h", "l", "c"]].isna().any(axis=1)
    rep["dropped_unparsed"] = int(bad.sum()); d = d[~bad]

    # 1. Геометрия бара — каждый инвариант считается отдельно.
    inv = {"high_lt_low": d.h < d.l, "high_lt_open": d.h < d.o, "high_lt_close": d.h < d.c,
           "low_gt_open": d.l > d.o, "low_gt_close": d.l > d.c, "price_le_zero": d.l <= 0, "volume_lt_zero": d.v < 0}
    rep["ohlc_violations"] = {k: int(v.sum()) for k, v in inv.items()}
    anyinv = np.logical_or.reduce([v.values for v in inv.values()])
    rep["dropped_bad_ohlc"] = int(anyinv.sum()); d = d[~anyinv]

    # 2. Пояс источника — по умолчанию Чикаго; выгрузка DZ_IND_BarExporter пишет UTC.
    loc = pd.DatetimeIndex(d.ts).tz_localize(src_tz, ambiguous="NaT", nonexistent="NaT")
    rep["dropped_dst_nat"] = int(loc.isna().sum())
    d = d[~loc.isna()]; loc = loc[~loc.isna()]

    label, lab_ev = ref.detect_bar_label(loc.tz_convert(TZ_EX).tz_localize(None), "auto")
    utc = loc.tz_convert("UTC")
    if label == "close":
        utc = utc - pd.Timedelta(minutes=1)
    d = d.set_index(pd.DatetimeIndex(utc, name="t")).drop(columns="ts")
    if hasattr(d.index, "as_unit"):
        d.index = d.index.as_unit("ns")
    d = d.sort_index()

    # 3. Дубликаты: одинаковые и конфликтующие — разные величины.
    dup = d.index.duplicated(keep=False)
    grp = d[dup].groupby(level=0).nunique()
    rep["duplicates_identical"] = int((grp[["o", "h", "l", "c"]].max(axis=1) <= 1).sum()) if len(grp) else 0
    rep["duplicates_conflicting"] = int((grp[["o", "h", "l", "c"]].max(axis=1) > 1).sum()) if len(grp) else 0
    d = d[~d.index.duplicated(keep="first")]

    # 4. Выходные — однозначно вне Globex: суббота и воскресенье до 17:00 по Чикаго.
    ct = d.index.tz_convert(TZ_EX)
    wk = (ct.dayofweek == 5) | ((ct.dayofweek == 6) & (ct.hour < 17))
    rep["dropped_weekend"] = int(wk.sum()); d = d[~wk]; ct = ct[~wk]

    # Ежедневный перерыв НЕ удаляется правилом: его границы менялись по годам, и
    # удаление по сегодняшнему расписанию выбросило бы настоящие бары прошлых лет.
    hm = ct.hour * 100 + ct.minute
    rep["bars_in_halt_1600_1700_ct_by_year"] = {int(k): int(v) for k, v in
        pd.Series(((hm >= 1600) & (hm < 1700)), index=ct.year).groupby(level=0).sum().items() if v}

    # 5. Выбросы: истинный диапазон больше k медиан за 60 баров при объёме не выше медианы.
    pc = d.c.shift(1)
    tr = pd.concat([d.h - d.l, (d.h - pc).abs(), (d.l - pc).abs()], axis=1).max(axis=1)
    med = tr.rolling(60, min_periods=20).median()
    vmed = d.v.rolling(60, min_periods=20).median()
    spike = (tr > spike_k * med) & (med > 0) & ((d.v <= vmed) | (~rep["volume_present"]))
    sp = d[spike].assign(TR=tr[spike], TRMedian60=med[spike])
    rep["spikes_flagged_not_removed"] = int(spike.sum())

    # ЭКСПЕРИМЕНТ 2. Покрытие сессии: доля минут Globex, присутствующих в торговом дне.
    tday = (ct - pd.Timedelta(hours=17)).normalize() + pd.Timedelta(days=1)
    cov = pd.Series(1, index=tday).groupby(level=0).sum()
    cov = pd.DataFrame({"TradingDay": cov.index.date, "Minutes": cov.values, "Coverage": (cov.values / 1380.0).round(4)})
    low = cov[cov.Coverage < min_coverage]
    rep["days_total"] = int(len(cov)); rep["days_low_coverage"] = int(len(low))

    rep.update({"date_format": fmt, "date_format_report": fmt_rep, "bar_label": label, "bar_label_evidence": lab_ev,
                "rows_clean": int(len(d)), "first_bar_utc": str(d.index[0]), "last_bar_utc": str(d.index[-1]),
                "time_unit": "ns", "time_zone": "UTC", "time_label": "bar_open"})

    # ЭКСПЕРИМЕНТ 1. Отпечаток по дням: какие дни изменятся при следующей выгрузке.
    t = to_ns(d.index)
    arr = np.stack([t.astype(np.float64), d.o.values, d.h.values, d.l.values, d.c.values, d.v.values], 1)
    rep["clean_sha256"] = hashlib.sha256(np.ascontiguousarray(arr).tobytes()).hexdigest()
    dd = pd.Series(range(len(d)), index=d.index.tz_convert(TZ_EX).date)
    day_hash = {}
    for day, pos in dd.groupby(level=0):
        day_hash[str(day)] = hashlib.sha256(np.ascontiguousarray(arr[pos.values]).tobytes()).hexdigest()[:16]

    base = os.path.splitext(out)[0]
    if out.endswith(".parquet"):
        try:
            import pyarrow as pa, pyarrow.parquet as pq
            tbl = pa.table({"t": pa.array(t, pa.timestamp("ns", tz="UTC")), "o": d.o.values, "h": d.h.values,
                            "l": d.l.values, "c": d.c.values, "v": d.v.values})
            tbl = tbl.replace_schema_metadata({"dz_meta": json.dumps(rep, ensure_ascii=False, default=str)})
            pq.write_table(tbl, out)
        except ImportError:
            out = base + ".npz"
            print("pyarrow не установлен — запись в .npz", file=sys.stderr)
    if out.endswith(".npz"):
        np.savez(out, t=t, o=d.o.values, h=d.h.values, l=d.l.values, c=d.c.values, v=d.v.values,
                 validation=np.array(json.dumps(rep, ensure_ascii=False, default=str)))

    with open(base + "_manifest.json", "w", encoding="utf-8") as f:
        json.dump({**rep, "day_hash": day_hash}, f, ensure_ascii=False, indent=2, default=str)
    cov.to_csv(base + "_coverage.csv", index=False)
    sp.reset_index().to_csv(base + "_spikes.csv", index=False)
    print(f"[очистка] {rep['rows_read']:,} → {rep['rows_clean']:,} баров | формат {fmt}, разметка {label} | "
          f"OHLC {rep['dropped_bad_ohlc']}, выходные {rep['dropped_weekend']}, дубли {rep['duplicates_identical']}+"
          f"{rep['duplicates_conflicting']} конфл. | выбросов помечено {rep['spikes_flagged_not_removed']} | "
          f"дней с покрытием ниже {min_coverage:.0%}: {rep['days_low_coverage']} из {rep['days_total']}")
    return out, rep


def load_clean(path):
    """Кэш в DataFrame с индексом UTC в наносекундах; столбцы o, h, l, c, v."""
    if path.endswith(".parquet"):
        import pyarrow.parquet as pq
        tb = pq.read_table(path).to_pandas()
        idx = pd.DatetimeIndex(tb.pop("t"))
        return tb.set_index(idx.tz_convert("UTC") if idx.tz else idx.tz_localize("UTC"))
    z = np.load(path, allow_pickle=False)
    idx = pd.to_datetime(z["t"], utc=True)
    v = z["v"] if "v" in z.files else np.zeros(len(idx))
    return pd.DataFrame({"o": z["o"], "h": z["h"], "l": z["l"], "c": z["c"], "v": v}, index=idx)

def resample(m1, minutes, anchor_hours=17):
    """
    Бар старшего таймфрейма по биржевому времени Чикаго с якорем начала сессии 17:00.
    Возвращает бары с open_utc и close_utc. close_utc — ПЛАНОВОЕ закрытие: значение бара
    пригодно для решения только с этого момента, и не раньше.
    """
    ct = m1.index.tz_convert(TZ_EX).tz_localize(None)
    sh = pd.Timedelta(hours=anchor_hours)
    start = (ct - sh).floor(f"{minutes}min") + sh
    g = m1.groupby(start)
    b = pd.DataFrame({"o": g.o.first(), "h": g.h.max(), "l": g.l.min(), "c": g.c.last(), "v": g.v.sum()})
    op = pd.DatetimeIndex(b.index).tz_localize(TZ_EX, ambiguous="NaT", nonexistent="NaT")
    b["open_utc"] = op.tz_convert("UTC")
    b["close_utc"] = (pd.DatetimeIndex(b.index) + pd.Timedelta(minutes=minutes)).tz_localize(
        TZ_EX, ambiguous="NaT", nonexistent="NaT").tz_convert("UTC")
    return b.dropna(subset=["open_utc", "close_utc"]).reset_index(drop=True)

ИЗВЕСТНОЕ ОГРАНИЧЕНИЕ. Последний бар перед перерывом 16:00–17:00 получает плановое
закрытие 17:00, а NinjaTrader закрывает его по концу сессии в 16:00. Для снимков «строго
до окна» плановое закрытие консервативно: раньше срока бар не используется.

ЭКСПЕРИМЕНТ — долларовые бары (AFML, гл. 2, с. 27–28): бар закрывается, когда
накопленный оборот цена × объём × стоимость пункта достигает порога. Из минуток это
приближение: бар не может быть короче минуты.

import numpy as np
import pandas as pd

VERSION = "DZ_DATA_Sanitizer-v2.0 (2026-10-04)"
TZ_EXCHANGE = "America/Chicago"


def load_clean(path):
    """Кэш движка (t) или сертифицированные массивы (t_open); объём — если есть, иначе нули."""
    z = np.load(path, allow_pickle=False)

        "v": z["v"] if "v" in z.files else np.zeros(len(t)),
    }, index=t)


def resample(df, minutes, anchor_hours=17):
    """
    Старшие бары от начала сессии в биржевом времени Чикаго. Объём — сумма минуток.
    Индекс и столбец open_utc — время открытия в UTC; close_utc — плановое закрытие.
    """
    ct = df.index.tz_convert(TZ_EXCHANGE).tz_localize(None)
    shift = pd.Timedelta(hours=anchor_hours)
    start = (ct - shift).floor(f"{minutes}min") + shift
    g = pd.DataFrame({"o": df.o.values, "h": df.h.values, "l": df.l.values, "c": df.c.values,
                      "v": df.v.values if "v" in df else np.zeros(len(df))}, index=start).groupby(level=0)
    b = pd.DataFrame({"o": g.o.first(), "h": g.h.max(), "l": g.l.min(), "c": g.c.last(), "v": g.v.sum()})
    open_loc = pd.DatetimeIndex(b.index).tz_localize(TZ_EXCHANGE, ambiguous="NaT", nonexistent="NaT")
    close_loc = pd.DatetimeIndex(b.index + pd.Timedelta(minutes=minutes)).tz_localize(
        TZ_EXCHANGE, ambiguous="NaT", nonexistent="NaT")
    b["open_utc"] = open_loc.tz_convert("UTC")
    b["close_utc"] = close_loc.tz_convert("UTC")
    b = b.dropna(subset=["open_utc", "close_utc"])
    b.index = pd.DatetimeIndex(b["open_utc"])
    return b


def dollar_bars(df, usd_per_bar, point_value):
    """Э8: долларовые бары из минуток. Минутка с нулевым объёмом оборота не добавляет."""
    usd = (df.c.values * np.maximum(df.v.values, 0.0) * point_value).astype(float)
    cum = np.cumsum(usd)
    bar = np.floor((cum - 1e-9) / float(usd_per_bar)).astype(np.int64)
    g = pd.DataFrame({"o": df.o.values, "h": df.h.values, "l": df.l.values, "c": df.c.values,
                      "v": df.v.values, "t": df.index.values}, index=bar).groupby(level=0)
    out = pd.DataFrame({"o": g.o.first(), "h": g.h.max(), "l": g.l.min(), "c": g.c.last(),
                        "v": g.v.sum(), "open_utc": g.t.first(), "last_minute_open_utc": g.t.last()})
    return out.reset_index(drop=True)

VALID_ESC = set("'\"\\0abfnrtv")


def lex(src):
    """Код без комментариев и содержимого строк + ошибки лексики. Позиции строк сохраняются."""
    out, errs, i, n, line = [], [], 0, len(src), 1
    while i < n:
        c, d = src[i], src[i:i + 2]
        if c == "\n":
            line += 1; out.append(c); i += 1; continue
        if d == "//":
            j = src.find("\n", i); j = n if j < 0 else j
            out.append(" " * (j - i)); i = j; continue
        if d == "/*":
            j = src.find("*/", i + 2); j = n if j < 0 else j + 2
            seg = src[i:j]; out.append("".join("\n" if ch == "\n" else " " for ch in seg))
            line += seg.count("\n"); i = j; continue
        if d == '@"' or (d == '$"') or src[i:i + 3] in ('$@"', '@$"'):
            verb = "@" in src[i:i + 3].split('"')[0]
            interp = "$" in src[i:i + 3].split('"')[0]
            k = src.index('"', i) + 1
            start_line = line
            while k < n:
                if verb and src[k:k + 2] == '""': k += 2; continue
                if not verb and src[k] == "\\": k += 2; continue
                if src[k] == "\n":
                    if not verb: errs.append((start_line, "незакрытая строка")); break
                    line += 1
                if src[k] == '"': k += 1; break
                k += 1
            out.append(("$" if interp else "") + '""' + "\n" * (src[i:k].count("\n"))); i = k; continue
        if c == '"':
            k, start_line = i + 1, line
            while k < n and src[k] != '"':
                if src[k] == "\\":
                    e = src[k + 1] if k + 1 < n else ""
                    if e in VALID_ESC: k += 2; continue
                    if e in "uUx": k += 2; continue
                    errs.append((line, f"недопустимая escape-последовательность \\{e}"))
                    k += 2; continue
                if src[k] == "\n":
                    errs.append((start_line, "незакрытая строка")); break
                k += 1
            out.append('""'); i = k + 1; continue
        if c == "'":
            k = i + 1
            if k < n and src[k] == "\\":
                e = src[k + 1] if k + 1 < n else ""
                if e not in VALID_ESC and e not in "uUx":
                    errs.append((line, f"недопустимый символ '\\{e}'"))
                k += 2
            else:
                k += 1
            while k < n and src[k] != "'" and src[k] != "\n":
                k += 1
            if k >= n or src[k] != "'":
                errs.append((line, "незакрытый символьный литерал"))
            out.append("'x'"); i = k + 1; continue
        out.append(c); i += 1
    return "".join(out), errs


def brackets(code):
    st, errs, line = [], [], 1
    pairs = {")": "(", "]": "[", "}": "{"}
    for ch in code:
        if ch == "\n": line += 1
        elif ch in "([{": st.append((ch, line))
        elif ch in ")]}":
            if not st or st[-1][0] != pairs[ch]:
                errs.append((line, f"непарная «{ch}»")); return errs
            st.pop()
    if st: errs.append((st[-1][1], f"незакрытая «{st[-1][0]}»"))
    return errs


FEATURES = [
    (r"\w\s*\?\.\s*\w", "?. — C# 6"), (r"\w\s*\?\[", "?[ — C# 6"), (r'\$""', '$"…" — C# 6'),
    (r"\bnameof\s*\(", "nameof — C# 6"),
    (r"^\s*(public|private|protected|internal)[^=;{(]*\([^)]*\)\s*=>", "метод через => — C# 6"),
    (r"^\s*(public|private|protected|internal)[^=;{(]*\w\s*=>\s*[^;]*;", "свойство через => — C# 6"),
    (r"\{\s*get\s*;\s*(private\s+)?(set\s*;\s*)?\}\s*=", "инициализатор автосвойства — C# 6"),
    (r"\bout\s+var\b", "out var — C# 7"), (r"\bis\s+[A-Z]\w*\s+[a-z]\w*\s*[)&|;]", "шаблон is — C# 7"),
    (r"\bswitch\s*\{", "switch-выражение — C# 8"),
]


def types_of(code, path):
    ns_stack, res, depth = [], [], 0
    for m in re.finditer(r"\bnamespace\s+([\w.]+)|\b(class|struct|enum|interface)\s+(\w+)|[{}]", code):
        tok = m.group(0)
        if tok == "{": depth += 1; continue
        if tok == "}":
            depth -= 1
            while ns_stack and ns_stack[-1][1] > depth: ns_stack.pop()
            continue
        if m.group(1):
            ns_stack.append((m.group(1), depth + 1)); continue
        # только типы верхнего уровня пространства имён: вложенные классы не конфликтуют
        ns_depth = ns_stack[-1][1] if ns_stack else 0
        if depth == ns_depth:
            res.append((".".join(n for n, _ in ns_stack) or "<глобальное>", m.group(3), path))
    return res


def menu(src):
    """Свойства меню: блок атрибутов — от последней пустой строки до объявления (строки описаний
    содержат «]» и «;», поэтому атрибуты не разбираются регулярным выражением по скобкам)."""
    props = []
    for m in re.finditer(r"public\s+([\w.<>]+)\s+(\w+)\s*\{\s*get;\s*set;\s*\}", src):
        typ, name = m.group(1), m.group(2)
        head = src[max(0, m.start() - 3000):m.start()]
        cut = max(head.rfind("\n\n"), head.rfind("\n\r\n"), head.rfind("{ get; set; }"), head.rfind("#region"))
        attrs = head[cut + 1:] if cut >= 0 else head
        if "NinjaScriptProperty" not in attrs:
            continue
        d = re.search(r'Name\s*=\s*"\[([A-Z]+\d+)\][^"]*",\s*Order\s*=\s*(-?\d+),\s*GroupName\s*=\s*"([^"]+)"', attrs)
        r = re.search(r"Range\(\s*([-\d.eE]+)\s*,\s*([-\d.eE]+|double\.MaxValue|int\.MaxValue)\s*\)", attrs)
        props.append({"name": name, "type": typ, "index": d.group(1) if d else None,
                      "order": int(d.group(2)) if d else None, "group": d.group(3) if d else None,
                      "range": (r.group(1), r.group(2)) if r else None, "hidden": "Browsable(false)" in attrs,
                      "line": src[:m.start()].count("\n") + 1})
    return props


def defaults(src):
    i = src.find("State == State.SetDefaults")
    j = src.find("State == State.Configure", i)
    body = src[i:j] if i >= 0 else ""
    res = {}
    for m in re.finditer(r"\b([A-Z]\w*)\s*=\s*([^;=][^;]*);", body):
        res.setdefault(m.group(1), m.group(2).strip())
    return res


def num(v):
    try:
        return float(v.replace("double.MaxValue", "1e308").replace("int.MaxValue", "2147483647"))
    except Exception:
        return None


def main():
    ap = argparse.ArgumentParser(description=VERSION)
    ap.add_argument("--src", required=True); ap.add_argument("--extra", nargs="*", default=[])
    a = ap.parse_args()
    files = sorted(glob.glob(os.path.join(a.src, "*.cs"))) + list(a.extra)
    errors, warns, alltypes = [], [], []
    hub_src = None
    for f in files:
        src = open(f, encoding="utf-8").read()
        base = os.path.basename(f)
        code, lerr = lex(src)
        errors += [f"{base}:{l}: {t}" for l, t in lerr]
        errors += [f"{base}:{l}: {t}" for l, t in brackets(code)]
        for li, text in enumerate(code.split("\n"), 1):
            for pat, what in FEATURES:
                if re.search(pat, text):
                    errors.append(f"{base}:{li}: {what} — компилятор NinjaTrader может не принять: {text.strip()[:90]}")
        alltypes += types_of(code, base)
        if base == "DZ_SW_Hub_S021.cs":
            hub_src = src
    seen = {}
    for ns, t, p in alltypes:
        if (ns, t) in seen and seen[(ns, t)] != p:
            errors.append(f"тип {ns}.{t} объявлен в {seen[(ns, t)]} и в {p} (CS0101)")
        seen.setdefault((ns, t), p)

    info = {}
    if hub_src:
        props = menu(hub_src)
        dfl = defaults(hub_src)
        by_idx, by_name, by_grp = {}, {}, {}
        for p in props:
            if not p["index"]:
                warns.append(f"{p['name']}: свойство схемы без индекса [Xnn] в названии")
            if p["index"]:
                if p["index"] in by_idx:
                    errors.append(f"индекс [{p['index']}] у {by_idx[p['index']]} и у {p['name']}")
                by_idx.setdefault(p["index"], p["name"])
            if p["name"] in by_name:
                errors.append(f"свойство {p['name']} объявлено дважды (CS0102)")
            by_name[p["name"]] = p
            key = (p["group"], p["order"])
            if p["group"] and key in by_grp:
                warns.append(f"Order {p['order']} в группе «{p['group']}» у {by_grp[key]} и у {p['name']}")
            by_grp.setdefault(key, p["name"])
            v = dfl.get(p["name"])
            if v is None:
                warns.append(f"[{p['index']}] {p['name']}: нет умолчания в SetDefaults — будет значение типа по умолчанию")
            elif p["range"] and num(v) is not None:
                lo, hi = num(p["range"][0]), num(p["range"][1])
                if not (lo - 1e-12 <= num(v) <= hi + 1e-12):
                    errors.append(f"[{p['index']}] {p['name']} = {v} вне Range({p['range'][0]}, {p['range'][1]})")
        # все подписи меню, включая свойства вне схемы (цвета панели и т. п.)
        all_labels = {}
        for m in re.finditer(r'Display\(Name\s*=\s*"\[([A-Z]+\d+)\]([^"]*)"', hub_src):
            all_labels.setdefault(m.group(1), []).append(m.group(2).strip()[:40])
        for k, v in all_labels.items():
            if len(v) > 1:
                errors.append(f"индекс [{k}] у нескольких подписей: {' | '.join(v)}")
        known = set(all_labels) | {f"MOD{i}" for i in range(1, 16)}
        refs = {}
        for f in files:
            if f in a.extra:
                continue                                   # чужие стратегии со своими индексами
            for m in re.finditer(r"\[([A-Z]{1,3}\d{2,3})\]", open(f, encoding="utf-8").read()):
                if m.group(1) not in known and not re.match(r"MOD\d+$", m.group(1)):
                    refs.setdefault(m.group(1), set()).add(os.path.basename(f))
        for k, fs in sorted(refs.items()):
            warns.append(f"Э33: ссылка на [{k}] — такого индекса нет в меню ({', '.join(sorted(fs))})")
        info = {"настроек в схеме": len(props), "с индексом": len(by_idx), "умолчаний найдено": len(dfl)}

    if hub_src:
        assigned = set(re.findall(r"\.(\w+)\s*=(?!=)", hub_src))
        assigned |= set(re.findall(r"\.(\w+)\.(?:Add|AddRange|Clear)\s*\(", hub_src))   # заполнение списка
        assigned |= set(re.findall(r"\.(\w+)\.\w+\s*=(?!=)", hub_src))               # поля вложенного объекта
        for f in files:
            if f in a.extra:
                continue
            src = open(f, encoding="utf-8").read()
            for cm in re.finditer(r"public\s+class\s+(Sw\w+Config)\b", src):
                body = src[cm.end():]
                body = body[body.find("{") + 1:]
                depth, k = 1, 0
                while k < len(body) and depth:
                    depth += {"{": 1, "}": -1}.get(body[k], 0); k += 1
                for fm in re.finditer(r"public\s+(?!class|static|const|readonly)[\w.<>\[\]]+\s+(\w+)\s*(?:=|;)", body[:k]):
                    if fm.group(1) not in assigned:
                        warns.append(f"Э34: {cm.group(1)}.{fm.group(1)} ({os.path.basename(f)}) — ядро не передаёт, всегда умолчание")
    print(f"{VERSION}: файлов {len(files)}, типов верхнего уровня {len(alltypes)}" + (f", {info}" if info else ""))
    for e in errors: print("  ОШИБКА  " + e)
    for w in warns: print("  внимание " + w)
    print(f"ИТОГ: ошибок {len(errors)}, предупреждений {len(warns)}")
    sys.exit(1 if errors else 0)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
==========================================================================================
DZ_DISPATCHER_DropZone.py — горячая папка терминала «Атлас», v1.0 (2026-09-28)

  init   создать хранилище:  DROP_ZONE_INBOX, DROP_ZONE_QUARANTINE, library_certified,
         data_certified, certified_vault, strategies_vault, status
  run    каждые 1,5 с разбирать DROP_ZONE_INBOX по правилам имён

ПРАВИЛА (первое совпавшее)
  LIB_* | *CANON* | *OPUS*                     → library_certified/
  <СИМВОЛ>.instrument.cert.json                → certified_vault/<СИМВОЛ>/
  <СТРАТЕГИЯ>__<СИМВОЛ>__*.xml | *.cert.json    → certified_vault/<СИМВОЛ>/<СТРАТЕГИЯ>/templates/
  DZ_*_S<NNN>*.dll | *.zip                     → strategies_vault/DZ_S<NNN>/releases/
  DZ_*_S<NNN>*.cs | .md | .json                → strategies_vault/DZ_S<NNN>/
  всё остальное                                → DROP_ZONE_QUARANTINE/ с причиной в журнале

ГАРАНТИИ
  • Файл берётся, только когда его размер не менялся два прохода подряд: недописанный не трогается.
  • Существующий файл НЕ перезаписывается: пришедший получает номер версии _vNNN.
  • Копия сверяется по SHA-256, получает «только чтение», и лишь затем исчезает из INBOX.
  • В папки NinjaTrader диспетчер НЕ пишет никогда: bin\\Custom — только руками руководителя.
==========================================================================================
"""
import argparse, datetime as dt, json, os, re, shutil, stat, sys, time

VERSION = "DZ_DISPATCHER_DropZone-v1.0 (2026-09-28)"
DIRS = ["DROP_ZONE_INBOX", "DROP_ZONE_QUARANTINE", "library_certified", "data_certified",
        "certified_vault", "strategies_vault", "status"]
HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
from DZ_CERT_Common import sha_file, seal_readonly  # noqa: E402


def symbols():
    p = os.path.join(HERE, "DZ_NTSIM_Instruments.json")
    return {k for k in json.load(open(p, encoding="utf-8")) if not k.startswith("_")} if os.path.exists(p) else set()


def route(name, syms):
    """Возвращает (относительная папка назначения, правило) либо (None, причина)."""
    up = name.upper()
    if up.startswith("LIB_") or "CANON" in up or "OPUS" in up:
        return "library_certified", "библиотека"
    m = re.match(r"^([A-Z0-9]+)\.instrument\.cert\.json$", name)
    if m and m.group(1) in syms:
        return os.path.join("certified_vault", m.group(1)), "паспорт инструмента"
    m = re.match(r"^(.+?)__([A-Z0-9]+)__.+\.(xml|cert\.json)$", name)
    if m and m.group(2) in syms:
        return os.path.join("certified_vault", m.group(2), m.group(1), "templates"), "шаблон"
    m = re.match(r"^DZ_.*?_(S\d{3}).*\.(dll|zip)$", name, re.I)
    if m:
        return os.path.join("strategies_vault", "DZ_" + m.group(1).upper(), "releases"), "релиз для импорта"
    m = re.match(r"^DZ_.*?_(S\d{3}).*\.(cs|md|json)$", name, re.I)
    if m:
        return os.path.join("strategies_vault", "DZ_" + m.group(1).upper()), "элемент стратегии"
    return None, "имя не соответствует ни одному правилу"


def free_name(folder, name):
    if not os.path.exists(os.path.join(folder, name)):
        return name
    stem, ext = os.path.splitext(name)
    if name.endswith(".cert.json"):
        stem, ext = name[:-10], ".cert.json"
    n = 2
    while os.path.exists(os.path.join(folder, f"{stem}_v{n:03d}{ext}")):
        n += 1
    return f"{stem}_v{n:03d}{ext}"


def log(root, text):
    with open(os.path.join(root, "DROP_ZONE_DISPATCH.log"), "a", encoding="utf-8") as f:
        f.write(f"[{dt.datetime.now():%Y-%m-%d %H:%M:%S}] {text}\n")


def dispatch(root, sizes, syms):
    inbox = os.path.join(root, "DROP_ZONE_INBOX")
    for name in sorted(os.listdir(inbox)):
        src = os.path.join(inbox, name)
        if not os.path.isfile(src) or name.startswith("~"):
            continue
        size = os.path.getsize(src)
        if sizes.get(name) != size:                 # размер ещё меняется либо файл новый
            sizes[name] = size; continue
        rel, rule = route(name, syms)
        dest_dir = os.path.join(root, rel if rel else "DROP_ZONE_QUARANTINE")
        os.makedirs(dest_dir, exist_ok=True)
        dn = free_name(dest_dir, name); dst = os.path.join(dest_dir, dn)
        h = sha_file(src)
        shutil.copy2(src, dst)
        if sha_file(dst) != h:
            os.remove(dst); log(root, f"ОШИБКА КОПИИ: {name} — отпечаток копии не совпал, файл оставлен в INBOX"); continue
        seal_readonly(dst); os.chmod(src, stat.S_IWRITE | stat.S_IREAD); os.remove(src); sizes.pop(name, None)
        if rel:
            log(root, f"ПРИНЯТО: {name} → {os.path.relpath(dst, root)} | {rule} | sha256 {h}")
        else:
            log(root, f"КАРАНТИН: {name} → {os.path.relpath(dst, root)} | {rule} | sha256 {h}")


def main():
    ap = argparse.ArgumentParser(description=VERSION)
    ap.add_argument("cmd", choices=["init", "run"]); ap.add_argument("--root", default=".")
    ap.add_argument("--interval", type=float, default=1.5); ap.add_argument("--cycles", type=int, default=0)
    a = ap.parse_args()
    for d in DIRS:
        os.makedirs(os.path.join(a.root, d), exist_ok=True)
    if a.cmd == "init":
        print(f"[хранилище] создано в {os.path.abspath(a.root)}: {', '.join(DIRS)}"); return
    syms, sizes, n = symbols(), {}, 0
    print(f"{VERSION}: наблюдение за {os.path.join(os.path.abspath(a.root), 'DROP_ZONE_INBOX')}")
    while True:
        try:
            dispatch(a.root, sizes, syms)
        except Exception as ex:
            log(a.root, f"ОШИБКА ПРОХОДА: {ex}")
        n += 1
        if a.cycles and n >= a.cycles:
            break
        time.sleep(a.interval)


if __name__ == "__main__":
    main()
