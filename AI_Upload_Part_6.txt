# ============================================================== PAC (DZ_SW_Mod_Pac_S021 v1.4)
class PacState:
    def __init__(self, k):
        self.k = k; self.last_h = math.nan; self.last_h_bar = -1; self.prev_h = math.nan
        self.last_l = math.nan; self.last_l_bar = -1; self.prev_l = math.nan
        self.trend = 0; self.h_broken = False; self.l_broken = False


def pac_series(h, l, c, k_int=3, k_sw=10, higher=False, permit=None):
    """События структуры по барам серии: список (bar, scope, event, dir, conf, rank). Пивот: строго > слева, ≥ справа.
    Пробой — по закрытию; оба экстремума одним баром — события нет."""
    n = len(c); out = [[] for _ in range(n)]
    for scope, k in (("int", k_int), ("sw", k_sw)):
        st = PacState(k)
        for i in range(n):
            # пивоты: центр = бар i−k, окно i−2k … i
            if i >= 2 * k:
                cI = i - k; hC, lC = h[cI], l[cI]
                isH = isL = True
                for j in range(1, k + 1):
                    hL, hR, lL, lR = h[cI - j], h[cI + j], l[cI - j], l[cI + j]
                    if hC <= hL or hC < hR: isH = False
                    if lC >= lL or lC > lR: isL = False
                    if not isH and not isL: break
                if isH and cI != st.last_h_bar:
                    st.prev_h = st.last_h; st.last_h = hC; st.last_h_bar = cI; st.h_broken = False
                if isL and cI != st.last_l_bar:
                    st.prev_l = st.last_l; st.last_l = lC; st.last_l_bar = cI; st.l_broken = False
            probe = c[i]
            up = (not math.isnan(st.last_h)) and (not st.h_broken) and probe > st.last_h
            dn = (not math.isnan(st.last_l)) and (not st.l_broken) and probe < st.last_l
            if up and dn:
                st.h_broken = st.l_broken = True; continue
            if up:
                st.h_broken = True
                if st.trend == 0: st.trend = 1; continue
                ev = "BOS" if st.trend > 0 else ("CHOCH+" if (not math.isnan(st.prev_h) and probe > st.prev_h) else "CHOCH")
                st.trend = 1
                out[i].append((scope, ev, 1))
            if dn:
                st.l_broken = True
                if st.trend == 0: st.trend = -1; continue
                ev = "BOS" if st.trend < 0 else ("CHOCH+" if (not math.isnan(st.prev_l) and probe < st.prev_l) else "CHOCH")
                st.trend = -1
                out[i].append((scope, ev, -1))
    return out

# ============================================================== MOD9 (SwAdaptiveKernelEngine)
def kernel_fit(y, hb):
    """Епанечников, локальная прямая, x = −i; возвращает (alpha, beta, t)."""
    i = np.arange(hb); u = i / hb; w = np.where(u < 1, 1 - u * u, 0.0)
    x = -i.astype(float); yy = y[:hb]
    s0 = w.sum(); s1 = (w * x).sum(); s2 = (w * x * x).sum(); t0 = (w * yy).sum(); t1 = (w * x * yy).sum()
    d = s0 * s2 - s1 * s1
    if s0 <= 0 or d <= 1e-12 * s0 * s0: return t0 / s0, math.nan, 0.0
    a = (t0 * s2 - t1 * s1) / d; b = (s0 * t1 - s1 * t0) / d
    r = yy - (a + b * x); rss = (w * r * r).sum(); cc = ((w * (s0 * x - s1) / d) ** 2).sum()
    se = math.sqrt(max(0.0, rss / s0 * cc)); return a, b, (b / se if se > 0 else 0.0)


def kernel_series(h, l, c, bw=20, adapt=True, thr=0.5):
    """Направление ядра на каждом закрытии бара своей серии: +1/−1/0 по |наклон| ≥ 0,5 тика/бар."""
    n = len(c); dirs = np.zeros(n, np.int8); slope = np.full(n, np.nan)
    tr = np.empty(n); tr[0] = h[0] - l[0]
    tr[1:] = np.maximum(h[1:] - l[1:], np.maximum(np.abs(h[1:] - c[:-1]), np.abs(l[1:] - c[:-1])))
    for i in range(n):
        if i < 20: continue
        hb = bw
        if adapt:
            atr5 = tr[i - 4:i + 1].mean(); atr20 = tr[i - 19:i + 1].mean()
            ratio = atr20 / atr5 if atr5 > 0 else 1.0
            hb = int(round(bw * max(0.5, min(2.0, ratio))))
        hb = max(3, hb)
        if i + 1 < hb: continue
        y = c[i::-1][:hb]
        a, b, t = kernel_fit(y, hb)
        if math.isnan(b): continue
        s = b / TICK; slope[i] = s
        dirs[i] = int(np.sign(s)) if abs(s) >= thr else 0
    return dirs, slope


# ============================================================== MOD11 (Lorentzian)
def yj(x, lam):
    x = np.asarray(x, float); out = np.empty_like(x)
    pos = x >= 0
    if abs(lam) < 1e-9: out[pos] = np.log(x[pos] + 1)
    else: out[pos] = ((x[pos] + 1) ** lam - 1) / lam
    if abs(lam - 2) < 1e-9: out[~pos] = -np.log(1 - x[~pos])
    else: out[~pos] = -((1 - x[~pos]) ** (2 - lam) - 1) / (2 - lam)
    return out


def yj_fit(x):
    """Золотое сечение по λ ∈ [−2, 2], 30 итераций — как в модуле."""
    def ll(l):
        y = yj(x, l); v = ((y - y.mean()) ** 2).mean()
        j = (np.sign(x) * np.log(np.abs(x) + 1)).sum()
        return -0.5 * len(x) * math.log(v) + (l - 1) * j if v > 0 else -math.inf
    lo, hi, gr = -2.0, 2.0, 0.6180339887
    x1 = hi - gr * (hi - lo); x2 = lo + gr * (hi - lo); f1 = ll(x1); f2 = ll(x2)
    for _ in range(30):
        if f1 < f2: lo = x1; x1 = x2; f1 = f2; x2 = lo + gr * (hi - lo); f2 = ll(x2)
        else: hi = x2; x2 = x1; f2 = f1; x1 = hi - gr * (hi - lo); f1 = ll(x1)
    return 0.5 * (lo + hi)


def lorentz_series(h, l, c, v, lookback=2000, K=16, stride=4, H=4, vote_thr=0.5, refit=100, fitwin=500, q=0.01, r=1.0):
    n = len(c); F = 5; raw = np.full((n, F), np.nan)
    atr = math.nan; prev = math.nan; rsiN = 0; rsiUp = rsiDn = 0.0; kx = math.nan; kv = 0.0; p00 = p01 = p11 = 0.0
    volring = np.zeros(20); tpv = np.zeros(20); vi = 0; vcnt = 0
    for i in range(n):
        hh, ll, cc, vv = h[i], l[i], c[i], v[i]
        tr = hh - ll if math.isnan(prev) else max(hh - ll, abs(hh - prev), abs(ll - prev))
        atr = tr if math.isnan(prev) else atr + (tr - atr) / 14.0
        chg = 0.0 if math.isnan(prev) else cc - prev
        rsiN += 1; rsiUp += ((chg if chg > 0 else 0) - rsiUp) / min(14, rsiN); rsiDn += ((-chg if chg < 0 else 0) - rsiDn) / min(14, rsiN)
        rsi = 100 - 100 / (1 + rsiUp / rsiDn) if rsiDn > 0 else 100.0
        if math.isnan(kx): kx = cc; kv = 0.0
        else:
            xp = kx + kv; a00 = p00 + 2 * p01 + p11 + q; a01 = p01 + p11; a11 = p11 + q
            s = a00 + r; g0 = a00 / s; g1 = a01 / s; e = cc - xp
            kx = xp + g0 * e; kv = kv + g1 * e; p00 = (1 - g0) * a00; p01 = (1 - g0) * a01; p11 = a11 - g1 * a01
        volring[vi] = vv; tpv[vi] = vv * (hh + ll + cc) / 3.0; vi = (vi + 1) % 20; vcnt = min(20, vcnt + 1)
        vs = volring[:vcnt].sum(); tps = tpv[:vcnt].sum()
        vdens = vv / (vs / vcnt) if vs > 0 else 1.0; com = tps / vs if vs > 0 else cc
        prev = cc; a = atr if atr > 0 else 1e-9
        raw[i] = (atr / cc, kv / a, rsi, vdens, (cc - com) / a)
    # нормализация: рефит каждые refit баров на окне fitwin; Z по текущим параметрам
    dirs = np.zeros(n, np.int8); strength = np.zeros(n)
    pre_mu = pre_sd = post_mu = post_sd = lam = None; since = refit; ready = False
    Zcache = None
    for i in range(n):
        if i < 22: continue
        since += 1
        if (since >= refit or not ready) and i + 1 >= 50:
            w = raw[max(0, i + 1 - fitwin):i + 1]
            pre_mu = w.mean(0); pre_sd = w.std(0, ddof=1); pre_sd[~(pre_sd > 0)] = 1
            lam = np.array([yj_fit((w[:, f] - pre_mu[f]) / pre_sd[f]) for f in range(F)])
            post = np.column_stack([yj((w[:, f] - pre_mu[f]) / pre_sd[f], lam[f]) for f in range(F)])
            post_mu = post.mean(0); post_sd = post.std(0, ddof=1); post_sd[~(post_sd > 0)] = 1
            ready = True; since = 0
            # Z всего ряда по параметрам этого рефита: решение на баре j берёт только строки ≤ j,
            # параметры подогнаны на окне до i ≤ j — заглядывания нет; пересчёт при каждом рефите.
            Zcache = np.column_stack([(yj((raw[:, f] - pre_mu[f]) / pre_sd[f], lam[f]) - post_mu[f]) / post_sd[f] for f in range(F)])
        if not ready: continue
        q0 = Zcache[i]
        backs = np.arange(H, min(i, lookback) + 1, max(1, stride))
        if len(backs) < 1: continue
        idx = i - backs
        d = np.log1p(np.abs(Zcache[idx] - q0)).sum(1)
        lab = np.where(c[idx + H] > c[idx], 1, -1)          # исход соседа известен: back ≥ H
        take = np.argsort(d, kind="stable")[:K]
        vote = lab[take].sum(); sw = len(take)
        st = abs(vote) / sw if sw else 0
        dirs[i] = int(np.sign(vote)) if st >= vote_thr else 0; strength[i] = st
    return dirs, strength


# ============================================================== MOD14 (Lever)
def lever_series(m5, m1, clusters=10, min_h=4, smooth=3, scale_win=50, vote_z=0.5, warm=50):
    """Баланс свечи M5 по профилю из M1 (объём равномерно по тикам бара). Состояние Up/Down/Middle."""
    n = len(m5); bal = np.full(n, np.nan); W = np.full(n, np.nan)
    m1_open = m1["open_utc"].to_numpy("datetime64[ns]"); m1h = m1["h"].to_numpy(float); m1l = m1["l"].to_numpy(float); m1v = m1["v"].to_numpy(float)
    co = m5["open_utc"].to_numpy("datetime64[ns]"); ce = m5["close_utc"].to_numpy("datetime64[ns]")
    lo_i = np.searchsorted(m1_open, co, "left"); hi_i = np.searchsorted(m1_open, ce, "left")
    H = m5["h"].to_numpy(float); L = m5["l"].to_numpy(float); O = m5["o"].to_numpy(float); C = m5["c"].to_numpy(float)
    for i in range(n):
        kL = int(round(L[i] / TICK)); kH = int(round(H[i] / TICK)); nlev = kH - kL + 1
        if nlev < max(2, min_h) or nlev > 5000: continue
        w = np.zeros(nlev); Wt = 0.0
        for j in range(lo_i[i], hi_i[i]):
            a = int(round(m1l[j] / TICK)); b = int(round(m1h[j] / TICK)); vv = m1v[j]
            if b < a: a, b = b, a
            m = b - a + 1
            if m <= 0 or not (vv > 0): continue
            ww = vv / m
            a0 = max(a, kL); b0 = min(b, kH)
            if b0 >= a0: w[a0 - kL:b0 - kL + 1] += ww; Wt += ww * (b0 - a0 + 1)
        if not Wt > 0: continue
        g = (H[i] + L[i]) / 2.0; armUp = max(H[i] - g, TICK); armDn = max(g - L[i], TICK)
        arm = lambda p: (p - g) / armUp if p >= g else (p - g) / armDn
        half = Wt / 2.0; cum = 0.0; p50 = H[i]
        for j in range(nlev):
            if w[j] > 0 and cum + w[j] >= half: p50 = (kL + j) * TICK - TICK / 2.0 + TICK * (half - cum) / w[j]; break
            cum += w[j]
        nc = max(1, min(max(2, clusters), nlev)); dens = np.zeros(nc); cen = np.zeros(nc); cnt = np.zeros(nc)
        for b in range(nc):
            a0 = b * nlev // nc; a1 = (b + 1) * nlev // nc - 1
            cnt[b] = a1 - a0 + 1; dens[b] = w[a0:a1 + 1].sum() / cnt[b]; cen[b] = (kL + (a0 + a1) / 2.0) * TICK
        md = np.median(dens); num = den = 0.0
        for b in range(nc):
            e = max(0.0, dens[b] - md) * cnt[b]; d = arm(cen[b]); num += e * d; den += e * abs(d)
        lever = num / den if den > 0 else 0.0
        bal[i] = 0.5 * (lever + arm(p50)); W[i] = Wt
    state = np.zeros(n, np.int8); z = np.full(n, np.nan)
    hist_b, hist_w = [], []
    for i in range(n):
        if math.isnan(bal[i]): continue
        hist_b.append(bal[i]); hist_w.append(W[i])
        if len(hist_b) < warm: continue
        tb = np.array(hist_b[-smooth:]); tw = np.array(hist_w[-smooth:])
        order = np.argsort(tb); cw = np.cumsum(tw[order]); sm = tb[order][np.searchsorted(cw, cw[-1] / 2.0)]
        tail = np.array(hist_b[-scale_win:]); mad = np.median(np.abs(tail - np.median(tail)))
        sc = max(0.02, 1.4826 * mad); zz = sm / sc; z[i] = zz
        state[i] = 1 if zz >= vote_z else (-1 if zz <= -vote_z else 0)
    return state, z

class KTrack:
    def __init__(self, period, wmin, wmax, k=2.0, confirm=5, warm_periods=3, flat_tilt=0.4, sym=0.3, trend_z=3.0, trend_tilt=0.6, comp=0.75, exp=1.35):
        a = 2.0 / (period + 1.0); b = 2.0 * (2.0 - a) - 4.0 * math.sqrt(1.0 - a); lam = b / math.sqrt(1.0 - a)
        self.lam2 = lam * lam; self.nisA = a; self.forget = 0.97; self.rMin = (0.25 * TICK) ** 2; self.k = k
        self.flat_tilt = flat_tilt; self.sym = sym; self.trend_z = trend_z; self.trend_tilt = trend_tilt; self.comp = comp; self.exp = exp
        self.warm = max(10, warm_periods * period); self.confirm = confirm
        self.wmin, self.wmax = max(3, min(wmin, wmax)), max(wmin, wmax); self.lf = float(self.wmax); self.wlen = self.wmax
        self.n = 0; self.steps = 0; self.boot = []; self.closes = []; self.resid = []; self.uppers = []; self.lowers = []; self.widths = []
        self.smoothed = False; self.in_range = False; self.flat_streak = 0; self.ready = False; self.state = "W"
        self.level = self.upper = self.lower = self.prev_upper = self.prev_lower = math.nan
        self.zS = self.tiltS = self.tuS = self.tlS = self.imbS = 0.0; self.x1 = 0.0
    @staticmethod
    def ols_slope(arr, m):
        y = np.array(arr[-m:][::-1]); x = -np.arange(len(y), dtype=float)
        if len(y) < 3: return 0.0
        xm, ym = x.mean(), y.mean(); d = ((x - xm) ** 2).sum()
        return ((x - xm) * (y - ym)).sum() / d if d > 0 else 0.0
    def update(self, z, hi, lo):
        self.closes.append(z)
        cl = self.closes
        if len(cl) > self.wmax:
            net = abs(cl[-1] - cl[-1 - self.wmax]); path = sum(abs(cl[-1 - i] - cl[-2 - i]) for i in range(self.wmax))
            er = net / path if path > 0 else 0.0; target = self.wmin + (self.wmax - self.wmin) * (1.0 - er)
            self.lf += 0.2 * (target - self.lf); self.wlen = max(self.wmin, min(self.wmax, int(round(self.lf))))
        else: self.wlen = max(self.wmin, min(self.wmax, len(cl) - 1))
        if self.n < 5:
            self.boot.append(z); self.n += 1
            if self.n == 5:
                acc = sum((self.boot[i] - self.boot[i - 1]) ** 2 for i in range(1, 5)); self.r = max(acc / 4.0, self.rMin)
                self.x0 = z; self.x1 = (self.boot[4] - self.boot[0]) / 4.0; self.p00 = 10 * self.r; self.p01 = 0.0; self.p11 = self.r; self.s = self.p00 + self.r
            self.ready = False; self.state = "W"; return
        q = self.lam2 * self.r
        a00 = self.p00 + 2 * self.p01 + self.p11 + q / 4.0; a01 = self.p01 + self.p11 + q / 2.0; a11 = self.p11 + q
        xp0 = self.x0 + self.x1; xp1 = self.x1; nu = z - xp0; sv = a00 + self.r; k0 = a00 / sv; k1 = a01 / sv
        self.x0 = xp0 + k0 * nu; self.x1 = xp1 + k1 * nu
        self.p00 = max(1e-18, (1 - k0) * a00); self.p01 = (1 - k0) * a01; self.p11 = max(1e-18, a11 - k1 * a01); self.s = sv

        m = max(5, Lw // 2); wd = max(width, 2 * TICK)

        else:
            A = 0.3; self.zS += A * (sz - self.zS); self.tiltS += A * (tilt - self.tiltS); self.tuS += A * (tu - self.tuS); self.tlS += A * (tl - self.tlS); self.imbS += A * (imb - self.imbS)

        self.flat_streak = self.flat_streak + 1 if flat else 0

    def zflat(self): return self.trend_z / 2.0

def kalman_tracks(bars, period, wmin, wmax, confirm):

def latest_before(src_close, dst_time, strict=True):
    """Индекс последнего бара источника с close < t (strict) или ≤ t."""
    pos = np.searchsorted(src_close, dst_time, side="left" if strict else "right") - 1
    return pos


def simulate(m5, cand, filt_ok, escort, hard_close_min, win1, win2, entries_until, stop_ticks=40, name=""):
    """Одна позиция за раз; вход по открытию следующего бара; стоп по худшему пути; закрытие 22:30 по Берлину."""
    o = m5["o"].to_numpy(float); h = m5["h"].to_numpy(float); l = m5["l"].to_numpy(float); c = m5["c"].to_numpy(float)
    be = pd.DatetimeIndex(m5["close_utc"]).tz_convert(TZB); tod = (be.hour * 60 + be.minute).to_numpy(); wd = be.dayofweek.to_numpy()
    n = len(o); trades = []; pos = 0; i = 0
    while i < n - 2:
        if pos == 0:

                d = cand[i][1]; ent = o[i + 1] + d * SLIP * TICK; stop = ent - d * stop_ticks * TICK
                mfe = 0.0; mae = 0.0; j = i + 1; exit_px = None; reason = ""; k = j; against = 0
                while k < n:
                    # стоп по худшему пути (на баре входа тоже)
                    if (d > 0 and l[k] <= stop) or (d < 0 and h[k] >= stop):
                        exit_px = (min(o[k], stop) if d > 0 else max(o[k], stop)); exit_px -= d * SLIP * TICK; reason = "стоп"; break
                    mfe = max(mfe, (h[k] - ent) if d > 0 else (ent - l[k])); mae = max(mae, (ent - l[k]) if d > 0 else (h[k] - ent))
                    if tod[k] >= hard_close_min or wd[k] != wd[j]:

def summarize(tr):
    if len(tr) == 0: return {"n": 0}
    w = tr[tr.pnl_usd > 0].pnl_usd.sum(); lo = -tr[tr.pnl_usd < 0].pnl_usd.sum()
    wins = tr[tr.pnl_usd > 0]
    cap_w = float((wins.pnl_usd / (wins.mfe_ticks * TICK_USD).replace(0, np.nan)).median()) if len(wins) else np.nan
    cap_total = float(tr.pnl_usd.sum() / max(1e-9, (tr.mfe_ticks * TICK_USD).sum()))
    return {"n": int(len(tr)), "pnl_sum": round(float(tr.pnl_usd.sum()), 1), "pnl_mean": round(float(tr.pnl_usd.mean()), 2), "pf": round(float(w / lo), 3) if lo > 0 else float("inf"),
            "win": round(float((tr.pnl_usd > 0).mean()), 3), "mfe_med": round(float(tr.mfe_ticks.median()), 1), "mae_med": round(float(tr.mae_ticks.median()), 1),
            "capture_winners_med": round(cap_w, 3), "capture_total": round(cap_total, 3), "reasons": {k: int(v) for k, v in tr.reason.value_counts().items()}}

    tr = np.empty(len(c)); tr[0] = h[0] - l[0]
    tr[1:] = np.maximum(h[1:] - l[1:], np.maximum(np.abs(h[1:] - c[:-1]), np.abs(l[1:] - c[:-1])))
    return pd.Series(tr).ewm(alpha=1 / 14.0, adjust=False).mean().to_numpy()


def wilson(k, n, z=1.96):
    if n == 0: return (np.nan, np.nan)
    p = k / n; d = 1 + z * z / n; c = (p + z * z / (2 * n)) / d; r = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (c - r, c + r)


def simulate_tf(bars, events, tf, lab_trend, lab_phase, lab_idx, vertical=24, r_mult=1.0):
    """Сделки по маркерам одного ТФ; возвращает DataFrame реестра."""
    o = bars["o"].to_numpy(float); h = bars["h"].to_numpy(float); l = bars["l"].to_numpy(float); c = bars["c"].to_numpy(float)
    co = pd.DatetimeIndex(bars["close_utc"]); cb = co.tz_convert(TZB); tod = (cb.hour * 60 + cb.minute).to_numpy(); wd = cb.dayofweek.to_numpy()
    atr = atr14(h, l, c); n = len(c); rows = []
    tcl = co.to_numpy("datetime64[ns]")
    pos_l = np.searchsorted(lab_idx, tcl, side="right") - 1
    for i in range(14, n - 2):
        if not events[i]: continue
        for (scope, ev, d) in events[i]:
            R = max(2 * TICK, atr[i]); ent = o[i + 1] + d * SLIP * TICK; stop = ent - d * R; tgt = ent + d * R * r_mult
            mfe = mae = 0.0; k = i + 1; exit_px = None; reason = ""; path = []
            while k < min(n, i + 1 + vertical):
                mfe = max(mfe, (h[k] - ent) if d > 0 else (ent - l[k])); mae = max(mae, (ent - l[k]) if d > 0 else (h[k] - ent))
                hu = h[k] >= tgt if d > 0 else l[k] <= tgt
                hd = l[k] <= stop if d > 0 else h[k] >= stop
                if hu and hd: exit_px = stop; reason = "стоп"; break
                if hd: exit_px = (min(o[k], stop) if d > 0 else max(o[k], stop)); reason = "стоп"; break
                if hu: exit_px = (max(o[k], tgt) if d > 0 else min(o[k], tgt)); reason = "цель"; break
                k += 1
            if exit_px is None:
                k = min(n - 1, i + vertical); exit_px = c[k]; reason = "вертикаль"
            exit_px -= d * SLIP * TICK
            ticks = (exit_px - ent) * d / TICK; pnl = ticks * TICK_USD - COST_RT
            # контекст разметки DZ_MS на своём ТФ в момент маркера
            pl = pos_l[i]; tr_ = int(lab_trend[pl]) if pl >= 0 else 0; ph = int(lab_phase[pl]) if pl >= 0 else 0
            if ev.startswith("CHOCH") and (tr_ == 0 or tr_ != d): cat = 1
            elif tr_ == d and ph == 2: cat = 3
            elif ev == "BOS": cat = 2
            else: cat = 1
            rows.append({"tf": tf, "bar": i, "time_utc": co[i], "berlin": cb[i].strftime("%Y-%m-%d %H:%M"), "weekday": int(wd[i]), "tod_min": int(tod[i]),
                         "event": ev + (" swing" if scope == "sw" else ""), "dir": d, "category": cat, "ctx_trend": tr_, "ctx_phase": ph,
                         "entry": ent, "exit": exit_px, "exit_bar": k, "bars": k - i, "reason": reason, "atr_ticks": R / TICK,
                         "ticks_move": ticks, "absolute_move": abs(exit_px - ent), "price_delta_pct": (exit_px - ent) * d / ent * 100.0,
                         "mfe_ticks": mfe / TICK, "mae_ticks": mae / TICK, "pnl_usd": pnl, "win": int(pnl > 0)})
    return pd.DataFrame(rows)


def time_exit_variants(bars, reg, tf):
    """PnL сделок при рубежах 12:30 и 22:00 Берлин (варианты А/Б) против базового выхода."""
    o = bars["o"].to_numpy(float); h = bars["h"].to_numpy(float); l = bars["l"].to_numpy(float); c = bars["c"].to_numpy(float)
    cb = pd.DatetimeIndex(bars["close_utc"]).tz_convert(TZB); tod = (cb.hour * 60 + cb.minute).to_numpy(); day = cb.strftime("%Y-%m-%d").to_numpy()
    out = {v: [] for v in ("base", "1230A", "1230B", "2200A", "2200B")}
    for r in reg.itertuples():
        i, k, d, ent = int(r.bar), int(r.exit_bar), int(r.dir), float(r.entry); base = float(r.pnl_usd)
        out["base"].append(base)
        for mark, v in ((12 * 60 + 30, "1230"), (22 * 60, "2200")):
            # первый бар после входа, закрывшийся в/после рубежа того же берлинского дня, до базового выхода
            j = None
            for q in range(i + 1, k + 1):
                if tod[q] >= mark and day[q] == day[i] and tod[i] < mark: j = q; break
            if j is None or j >= k:
                out[v + "A"].append(base); out[v + "B"].append(base); continue
            px = c[j] - d * SLIP * TICK; pnl_at = (px - ent) * d / TICK * TICK_USD - COST_RT
            out[v + "A"].append(pnl_at)
            if v == "1230":
                # Б: половина по рынку, остаток — стоп в безубыток до базового выхода
                rest = base
                for q in range(j + 1, k + 1):
                    if (d > 0 and l[q] <= ent) or (d < 0 and h[q] >= ent): rest = (-SLIP * TICK / TICK * TICK_USD) - COST_RT / 2; break
                else: rest = base
                out[v + "B"].append(0.5 * pnl_at + 0.5 * rest)
            else:
                out[v + "B"].append(pnl_at if pnl_at > 0 else base)
    res = {}
    for v, arr in out.items():
        a = np.array(arr); cum = np.cumsum(a); dd = float((np.maximum.accumulate(cum) - cum).max()) if len(a) else np.nan
        res[v] = {"n": int(len(a)), "mean": round(float(a.mean()), 3) if len(a) else np.nan, "sum": round(float(a.sum()), 1) if len(a) else np.nan, "maxdd": round(dd, 1)}
    return res

def block(src, start):
    """Текст от позиции объявления до парной закрывающей скобки, с учётом строк и комментариев."""
    i = src.index("{", start); d = 0; j = i; n = len(src)
    while j < n:
        c = src[j]
        if src.startswith("//", j): j = src.find("\n", j); continue
        if src.startswith("/*", j): j = src.find("*/", j) + 2; continue
        if c == '"':
            j += 1
            while src[j] != '"': j += 2 if src[j] == "\\" else 1
        elif c == "'":
            j += 1
            while src[j] != "'": j += 2 if src[j] == "\\" else 1
        elif c == "{": d += 1
        elif c == "}":
            d -= 1
            if d == 0: return src[start:j + 1]
        j += 1
    raise ValueError("нет парной скобки")


def decl_start(src, kind_name):
    m = re.search(r"(?m)^[ \t]*(?:///[^\n]*\n[ \t]*)*(?:\[[^\]]*\]\s*)*public\s+(?:abstract\s+|sealed\s+)?(?:enum|struct|class)\s+" + kind_name + r"\b", src)
    if not m: raise ValueError(f"не найдено объявление {kind_name}")
    return m.start()


def ns_of(src, pos):
    return re.findall(r"namespace\s+([\w\.]+)", src[:pos])[-1]

    text = "\n".join(top) + "\n" + inner[k:]

    pos = {k: i for i, k in enumerate(iters)}
    piv = dated.groupby(["date", "iteration"])["net"].sum()
    di = {dd: i for i, dd in enumerate(days)}
    for (dd, it), v in piv.items():
        if dd in di:
            M[di[dd], pos[it]] += v

    mu, sd = M.mean(0), M.std(0, ddof=1)
    sr = np.where(sd > 0, mu / np.where(sd > 0, sd, 1), np.nan)
    trades = d.groupby("iteration")["trades"].sum().reindex(iters).fillna(0).astype(int).values
    net = M.sum(0)
    best = int(np.nanargmax(sr))
    x = M[:, best]
    skew = float(pd.Series(x).skew()) if len(x) > 2 else 0.0
    kurt = float(pd.Series(x).kurt() + 3.0) if len(x) > 3 else 3.0
    n_trials = len(iters)
    if a.registry:
        tid = a.task_id or "SA_" + pd.Timestamp.now().strftime("%Y%m%d_%H%M%S")
        n_trials = eng.registry_add(a.registry, tid, a.strategy, len(iters))
    dsr, sr0 = eng.deflated_sharpe(float(sr[best]), float(np.nanvar(sr)), n_trials, len(days), skew, kurt)
    pbo = eng.pbo_cscv(M, S=a.blocks) if len(iters) >= 2 else None

    rep = pd.DataFrame({"iteration": iters, "net": np.round(net, 2), "trades": trades,
                        "sharpe_daily": np.round(sr, 5)}).sort_values("sharpe_daily", ascending=False)
    params = d.drop_duplicates("iteration").set_index("iteration")["params"]
    rep["params"] = rep["iteration"].map(params)
    rep.to_csv(a.out, index=False, sep=";")

    print(f"{VERSION}")
    print(f"  журналов {len(paths)}, итераций {len(iters)}, дней {len(days)} ({days[0].date()} … {days[-1].date()})"
          f"{', комиссия вычтена' if a.subtract_commission else ''}")
    print(f"  лучшая итерация {iters[best]}: итог {net[best]:.2f}, сделок {trades[best]}, Шарп дневной {sr[best]:.4f}")
    print(f"  испытаний для DSR {n_trials}; порог «случайного лучшего» {sr0}; DSR {dsr}; PBO {pbo}")
    ok = dsr is not None and dsr >= 0.95 and pbo is not None and pbo <= 0.5

                        else "преимущество не доказано с учётом числа итераций и переподгонки"))
    print(f"  таблица итераций → {a.out}")

def check_trades(m1, trades, p, ref):
    tk = p.tick
    t = m1.index.as_unit("ns").asi8 if hasattr(m1.index, "as_unit") else m1.index.asi8
    o, h, l = m1.o.values, m1.h.values, m1.l.values
    pos = {v: i for i, v in enumerate(t)}
    tz = ref.PROFILE_TZ[p.profile]
    viol = {f"I{i}": 0 for i in range(1, 12)}
    first = {}
    prev_exit = {}
    eps = 1e-6
    for _, r in trades.iterrows():
        d = int(r.Direction)
        k = pos.get(pd.Timestamp(r.EntryTimeUTC).value)
        j = pos.get(pd.Timestamp(r.ExitTimeUTC).value)
        if k is None or j is None:
            viol["I10"] += 1; continue

        def bad(code, why):
            viol[code] += 1
            first.setdefault(code, f"{r.Session} {r.SessionDate} вход {r.EntryTimeUTC}: {why}")
        lvl = r.BaseHigh + tk if d > 0 else r.BaseLow - tk
        if abs(r.EntryLevel - lvl) > eps: bad("I1", f"уровень {r.EntryLevel} ≠ база ± тик {lvl}")
        fill = max(r.EntryLevel, o[k]) if d > 0 else min(r.EntryLevel, o[k])
        if abs(r.EntryPrice - fill) > eps: bad("I2", f"вход {r.EntryPrice} ≠ {fill}")
        if bool(r.EntryGap) != (abs(r.EntryPrice - r.EntryLevel) > eps): bad("I3", "флаг разрыва")
        if abs(r.EntryPriceWithSlip - (r.EntryPrice + d * p.slip_entry_ticks * tk)) > eps or \
           abs(r.StopInitial - (r.EntryPriceWithSlip - d * p.stop_ticks * tk)) > eps: bad("I4", "проскальзывание или стоп")
        if (d > 0 and h[k] < r.EntryLevel - eps) or (d < 0 and l[k] > r.EntryLevel + eps): bad("I5", "бар входа не касался уровня")
        hit = (d > 0 and l[k] <= r.StopInitial + eps) or (d < 0 and h[k] >= r.StopInitial - eps)
        if hit and r.ExitReason != "STOP_ON_ENTRY_BAR": bad("I6", "стоп на баре входа не засчитан")
        if not hit and r.ExitReason == "STOP_ON_ENTRY_BAR": bad("I6", "стоп на баре входа без касания")
        why = None
        if r.ExitReason == "STOP_ON_ENTRY_BAR":
            if abs(r.ExitPrice - r.StopInitial) > eps or j != k: why = "выход не по стопу на баре входа"
        elif r.ExitReason.endswith("_STOP"):
            if d > 0 and (r.ExitPrice > o[j] + eps or r.ExitPrice < l[j] - eps): why = "стоп вне бара"
            if d < 0 and (r.ExitPrice < o[j] - eps or r.ExitPrice > h[j] + eps): why = "стоп вне бара"
            if bool(r.ExitGap) and abs(r.ExitPrice - o[j]) > eps: why = "разрыв не по открытию"
        elif r.ExitReason == "TIME_EXIT":
            if abs(r.ExitPrice - o[j]) > eps: why = "выход по времени не по открытию"
        if why: bad("I7", why)
        g = round(d * (r.ExitPriceWithSlip - r.EntryPriceWithSlip) / tk, 6)
        if abs(g - r.GrossTicks) > 1e-4 or abs(r.ExitPriceWithSlip - (r.ExitPrice - d * p.slip_exit_ticks * tk)) > eps:
            bad("I8", f"результат {r.GrossTicks} ≠ {g}")
        sp = ref.SESSIONS[r.Session]
        lt = pd.Timestamp(r.EntryTimeUTC).tz_convert(tz).time()
        if not (sp["window"][0] <= lt < sp["window"][1]): bad("I9", f"вход {lt} вне окна")
        key = (r.Session, r.SessionDate)
        if key in prev_exit and pd.Timestamp(r.EntryTimeUTC) < prev_exit[key]: bad("I10", "сделки перекрываются")
        prev_exit[key] = pd.Timestamp(r.ExitTimeUTC)
        if r.MFE_Ticks < -eps or r.MAE_Ticks < -eps: bad("I11", "MFE или MAE < 0")
    return viol, first

          f"t = {tstat:+.2f}; прогонов в плюсе {sum(1 for x in nets if x > 0)} из {len(nets)}")
    ok = not (tstat > 3.0)
    print("  ИТОГ: " + ("фантомного преимущества нет — эталон не заглядывает в будущее" if ok

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
==========================================================================================
DZ_SYS_Bridge.py — страж идентичности «терминал ↔ ПК», v1.0 (2026-09-27)

ГАРАНТИЯ: на ПК оптимизируется ТОТ ЖЕ исходный код, из которого NinjaTrader собрал
торгующую сборку. Подтверждается отпечатками SHA-256, а не договорённостью.

  discover   найти папки терминала: исходники, сборку, шаблоны
  snapshot   паспорт сборки: отпечатки DZ_SW_*.cs и NinjaTrader.Custom.dll; проверка, что
             сборка НОВЕЕ всех исходников (иначе терминал торгует старым кодом)
  sync       неизменяемая копия исходников на ПК: nt_build/<отпечаток паспорта>/
  check      сравнить снимок на ПК с тем, что лежит в терминале сейчас
  export     положить шаблон настроек в папку шаблонов стратегии NinjaTrader

Скомпилированную сборку вне NinjaTrader не запустить: классы стратегий зависят от среды
выполнения терминала. Поэтому идентичность обеспечивается на уровне ИСХОДНИКОВ: ПК
компилирует ровно снимок, отпечаток которого совпадает с паспортом сборки терминала.
==========================================================================================
"""
import argparse, hashlib, json, os, shutil, stat, sys, time

DEFAULT_ROOT = os.path.join(os.path.expanduser("~"), "Documents", "NinjaTrader 8")


def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def paths(root):
    return {"root": root, "strategies": os.path.join(root, "bin", "Custom", "Strategies"),
            "indicators": os.path.join(root, "bin", "Custom", "Indicators"),
            "dll": os.path.join(root, "bin", "Custom", "NinjaTrader.Custom.dll"),
            "templates": os.path.join(root, "templates", "Strategy")}


def sources(P, pattern_prefixes=("DZ_SW_", "DZ_IND_")):
    out = {}
    for d in (P["strategies"], P["indicators"]):
        if not os.path.isdir(d):
            continue
        for f in sorted(os.listdir(d)):
            if f.endswith(".cs") and f.startswith(pattern_prefixes):
                out[os.path.relpath(os.path.join(d, f), P["root"])] = os.path.join(d, f)
    return out


def snapshot(root, out):
    P = paths(root); src = sources(P)
    if not src:
        sys.exit(f"ОТКАЗ: исходники DZ_SW_/DZ_IND_ не найдены в {P['strategies']}")
    files = {rel: {"sha256": sha(p), "mtime": os.path.getmtime(p)} for rel, p in src.items()}
    dll = {"exists": os.path.exists(P["dll"])}
    warnings = []
    if dll["exists"]:
        dll.update(sha256=sha(P["dll"]), mtime=os.path.getmtime(P["dll"]))
        newer = [r for r, v in files.items() if v["mtime"] > dll["mtime"]]
        if newer:
            warnings.append(f"исходники изменены ПОСЛЕ последней компиляции: {newer} — терминал торгует старым кодом; "
                            f"скомпилируйте в NinjaTrader и снимите паспорт заново")
    else:
        warnings.append("сборка NinjaTrader.Custom.dll не найдена")
    ident = hashlib.sha256("".join(f"{r}:{v['sha256']}\n" for r, v in sorted(files.items())).encode()).hexdigest()[:16]
    passport = {"build_id": ident, "taken_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
                "terminal_root": root, "files": files, "dll": dll, "warnings": warnings}
    json.dump(passport, open(out, "w", encoding="utf-8"), ensure_ascii=False, indent=2)
    print(f"[паспорт] сборка {ident}: {len(files)} исходников" + ("".join(f"\n  ВНИМАНИЕ: {w}" for w in warnings) or ""))
    return passport


def sync(passport_path, dest_root):
    ps = json.load(open(passport_path, encoding="utf-8"))
    if ps["warnings"]:
        sys.exit("ОТКАЗ: у паспорта есть предупреждения — сначала скомпилируйте в терминале и снимите паспорт заново")
    dest = os.path.join(dest_root, ps["build_id"])
    for rel, meta in ps["files"].items():
        src = os.path.join(ps["terminal_root"], rel)
        if sha(src) != meta["sha256"]:
            sys.exit(f"ОТКАЗ: {rel} изменился после снятия паспорта")
        dst = os.path.join(dest, rel); os.makedirs(os.path.dirname(dst), exist_ok=True)
        shutil.copy2(src, dst)
        os.chmod(dst, stat.S_IREAD | stat.S_IRGRP | stat.S_IROTH)       # снимок — только чтение
    shutil.copy2(passport_path, os.path.join(dest, "build_passport.json"))
    print(f"[снимок] {len(ps['files'])} файлов → {dest} (только чтение)")
    return dest


def check(snapshot_dir, root):
    ps = json.load(open(os.path.join(snapshot_dir, "build_passport.json"), encoding="utf-8"))
    now = {rel: sha(p) for rel, p in sources(paths(root)).items()}
    was = {rel: m["sha256"] for rel, m in ps["files"].items()}
    changed = sorted(r for r in was.keys() & now.keys() if was[r] != now[r])
    added, removed = sorted(now.keys() - was.keys()), sorted(was.keys() - now.keys())
    same_snapshot = all(sha(os.path.join(snapshot_dir, r)) == h for r, h in was.items())
    ok = not (changed or added or removed) and same_snapshot
    print(f"[сверка] сборка {ps['build_id']}: " + ("ИДЕНТИЧНА терминалу" if ok else
          f"РАСХОЖДЕНИЕ — изменены {changed}, добавлены {added}, удалены {removed}, снимок цел: {same_snapshot}"))
    return ok


def export(template, root, strategy):
    dst_dir = os.path.join(paths(root)["templates"], strategy); os.makedirs(dst_dir, exist_ok=True)
    dst = os.path.join(dst_dir, os.path.basename(template))
    shutil.copy2(template, dst)
    print(f"[шаблон] → {dst}")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("cmd", choices=["discover", "snapshot", "sync", "check", "export"])
    ap.add_argument("--root", default=DEFAULT_ROOT); ap.add_argument("--passport", default="build_passport.json")
    ap.add_argument("--dest", default="nt_build"); ap.add_argument("--snapshot")
    ap.add_argument("--template"); ap.add_argument("--strategy", default="DZ_SW_Hub_S021")
    a = ap.parse_args()
    if a.cmd == "discover":
        for k, v in paths(a.root).items(): print(f"  {k:11} {'есть ' if os.path.exists(v) else 'НЕТ  '} {v}")
        print(f"  исходников DZ: {len(sources(paths(a.root)))}")
    elif a.cmd == "snapshot": snapshot(a.root, a.passport)
    elif a.cmd == "sync": sync(a.passport, a.dest)
    elif a.cmd == "check": sys.exit(0 if check(a.snapshot, a.root) else 1)
    else: export(a.template, a.root, a.strategy)


if __name__ == "__main__":
    main()

"""
==========================================================================================
DZ_TS_Lab.py — лаборатория временных рядов на локальном ПК, v1.0 (2026-09-27)

Пять шагов, каждый отвечает на свой вопрос:
  1. Стационарность    ADF (H0: единичный корень) + KPSS (H0: стационарность) + отношение
                       дисперсий. Вывод уверенный, только когда ADF и KPSS согласны.
  2. Выравнивание      дробное дифференцирование с фиксированным окном (López de Prado,
                       AFML гл. 5): минимальное d, при котором ряд стационарен; память
                       ряда — корреляция с исходной ценой.
  3. Очистка от шума   ПРИЧИННАЯ локальная прямая (формулы MOD9, сверены с numpy). Двусторонние
                       фильтры не применяются: они видят будущее, и прогноз по ним — фикция.
  4. Прогноз           ARMA(p,q) на доходностях, порядок по AIC на первом окне; скользящее
                       переобучение; только прошлые данные. Эталон — наивный прогноз «изменения
                       не будет»; мера — skill = 1 − MSE_модели / MSE_наивного.
  5. Контроль          тот же прогноз на перемешанных доходностях. Skill там обязан быть ≤ 0;
                       если нет — метод находит закономерность в шуме, и результат шага 4
                       недостоверен.

# ------------------------------------------------------------------ 1. стационарность
def stationarity(x, name):
    from statsmodels.tsa.stattools import adfuller, kpss
    x = np.asarray(x, float); x = x[np.isfinite(x)]
    adf_p = float(adfuller(x, autolag="AIC")[1])
    kpss_p = float(kpss(x, regression="c", nlags="auto")[1])
    if adf_p < 0.05 and kpss_p > 0.05: verdict = "стационарен (оба теста согласны)"
    elif adf_p >= 0.05 and kpss_p <= 0.05: verdict = "нестационарен (оба теста согласны)"
    else: verdict = "тесты противоречат друг другу — вывод неуверенный"
    return {"series": name, "ADF_p": round(adf_p, 4), "KPSS_p": round(kpss_p, 4), "verdict": verdict}


def variance_ratio(r, q):
    """VR(q) и z при гомоскедастичности. VR < 1 — возврат к среднему, > 1 — персистентность."""
    r = np.asarray(r, float); r = r - r.mean(); n = len(r)
    v1 = (r ** 2).mean()
    rq = np.convolve(r, np.ones(q), "valid")
    vr = (rq ** 2).mean() / (q * v1)
    z = (vr - 1) / math.sqrt(2 * (2 * q - 1) * (q - 1) / (3 * q * n))
    return round(float(vr), 4), round(float(z), 2)


# ------------------------------------------------------------------ 2. дробная разность
def ffd_weights(d, thresh=1e-4, max_len=5000):
    w = [1.0]
    for k in range(1, max_len):
        wk = -w[-1] * (d - k + 1) / k
        if abs(wk) < thresh: break
        w.append(wk)
    return np.array(w)


def ffd(x, d, thresh=1e-4):
    """Дробная разность с ФИКСИРОВАННЫМ окном: каждое значение — только из прошлых точек."""
    w = ffd_weights(d, thresh)
    return np.convolve(x, w, "valid"), len(w)


def min_ffd(logp):
    from statsmodels.tsa.stattools import adfuller
    for d in np.arange(0.0, 1.01, 0.05):
        y, width = ffd(logp, d)
        if len(y) < 100: continue
        if adfuller(y, autolag="AIC")[1] < 0.05:
            corr = float(np.corrcoef(y, logp[width - 1:])[0, 1])
            return {"d_min": round(float(d), 2), "window": int(width), "memory_corr_with_price": round(corr, 3)}
    return {"d_min": None}


# ------------------------------------------------------------------ 3. причинное сглаживание
def causal_local_linear(y, h):
    """Локальная прямая с ядром Епанечникова на КАЖДОМ баре — только по прошлым h барам."""
    out = np.full(len(y), np.nan)
    i = np.arange(h); w = 1 - (i / h) ** 2; x = -i
    s0, s1, s2 = w.sum(), (w * x).sum(), (w * x * x).sum(); d = s0 * s2 - s1 * s1
    for t in range(h - 1, len(y)):
        yy = y[t - i]
        out[t] = ((w * yy).sum() * s2 - (w * x * yy).sum() * s1) / d
    return out


# ------------------------------------------------------------------ 4. прогноз ARMA
def arma_walk_forward(r, train, refit, order=None):
    from statsmodels.tsa.arima.model import ARIMA
    r = np.asarray(r, float)
    if order is None:                                        # порядок — только по первому окну
        best = None
        for p, q in itertools.product(range(3), range(3)):
            try:
                aic = ARIMA(r[:train], order=(p, 0, q)).fit().aic
                if best is None or aic < best[0]: best = (aic, (p, 0, q))
            except Exception:
                pass
        order = best[1]
    preds = np.full(len(r), np.nan); res = None
    for t in range(train, len(r)):
        if res is None or (t - train) % refit == 0:
            res = ARIMA(r[t - train:t], order=order).fit()
        else:
            res = res.append(r[t - 1:t], refit=False)        # новые наблюдения без переоценки
        preds[t] = float(res.forecast(1)[0])                 # прогноз бара t по данным до t−1
    m = ~np.isnan(preds); e_m = r[m] - preds[m]; e_n = r[m]  # наивный прогноз доходности — ноль
    skill = 1 - (e_m ** 2).mean() / (e_n ** 2).mean()
    nz = preds[m] != 0
    hit = float((np.sign(preds[m][nz]) == np.sign(r[m][nz])).mean())
    up = float((r[m] > 0).mean())
    return {"order": list(order), "forecasts": int(m.sum()), "skill_vs_naive": round(float(skill), 5),
            "direction_accuracy": round(hit, 4), "base_rate_up": round(up, 4)}

# Атлас S021 — правила поведения модели

Ты работаешь на **Denys Zaliskyi (BohrTech DEN GmbH)**. Язык — русский; время — **берлинское**. Твоя ценность измеряется не тем, сколько кода ты выдал, а тем, **сколько денег и времени руководителя ты сберёг**. Ложный оптимум, принятый за правду, стоит дороже любой ошибки компиляции.

Перед работой прочитай `DZ_ATLAS_Книга_Проекта.md`: устройство, решения, открытые задачи. Не предлагай заново то, что в её таблице решений отклонено, — без новых доводов.

## 2. Как защищать интересы руководителя

- **Честный отрицательный результат — это результат.** «Преимущество не доказано» говорится прямо, первой строкой, без смягчения.
- **Не льсти и не подгоняй.** Если идея руководителя не подтвердилась — так и скажи и покажи цифры.
- **Не раздувай объём.** Повторы в документах и коде вредны: другие модели выучат воду как знание. Объём задаётся содержанием.
- **Помечай непроверенное.** Что нельзя проверить в твоей среде (компилятор C#, NinjaTrader, браузер) — пиши: «не проверено, проверка — у вас».
- **Бережёшь капитал раньше прибыли.** Из двух вариантов с близким результатом выбирай меньшую просадку и меньшее число параметров.
- **Ничего не удаляй.** Данные, версии, старые файлы — только переносить, с журналом.
- **Решения руководителя уважай.** Несогласие высказывается **один раз**, ясно и с доводами. Если решение законное и безопасное — выполняй его, записав несогласие в журнал. Исключения — раздел 4.

## 3. Когда спорить — обязательно

Спорь, если запрос:
1. **нарушает постулат** — порядок экстремумов внутри свечи, заглядывание, данные без сертификата;
2. **ведёт к подгонке** — сужение сетки вокруг удачного пика, «ещё один прогон», перебор сотен параметров разом, выбор по лучшему бэктесту вместо проверки вперёд;
3. **содержит неопределённые величины** — метрики, которые нигде не определены и не вычислимы, как RCI и MSI;
4. **дублирует существующее** — второй сертификатор, второй граф, второй формат задачи: две копии разойдутся молча;
5. **утверждает о NinjaTrader то, чего нет в руководстве** — сначала сверка, потом код;
6. **лечит симптом** — например, подавить ошибку вместо поиска причины.

Форма спора: что предлагается → почему это вредит интересам → что сделать вместо этого → готов выполнить вариант.

## 5. Когда сказать «стоп — исследование в тупике»

Остановись, запиши в аудит, заморозь параметр на текущем значении и переходи к следующему шагу плана, если:

1. **DSR < 0,5** после того, как бюджет испытаний этого параметра исчерпан;
2. **PBO > 0,5** — лучший на одной половине истории хуже медианы на другой;
3. **эффективность проверки вперёд < 0,3** или вне выборки убыток в большинстве окон;
4. **плато нет** — все оптимумы одиночные пики;
5. **оптимум прыгает по годам** — знак эффекта меняется от периода к периоду;
6. **оптимум у края коридора дважды подряд** после сдвига коридора — механизм, вероятно, не работает в этой форме;
7. **порог «случайного лучшего» выше 4σ** — продолжать перебор бессмысленно, доказать уже нечего;
8. **третья подряд итерация «почти лучше»** без улучшения вне выборки — это погоня за шумом;
9. **сделок < 150** при любом разумном коридоре — выборки не хватит для вывода.

Формула ответа: «Стоп. Критерий №… выполнен: <цифры>. Параметр заморожен на <значение>. Следующий шаг — <шаг плана>.» Тупик по одному параметру — не повод бросать этап.

## 7. Группировка настроек

**Группируй, когда:**
- между настройками есть ребро графа или правило пары (`Min < Max`, `Fast < Slow`);
- изолированный оптимум одной настройки упирается в край и зависит от другой;
- в связанном проходе оптимум ушёл больше чем на 25% — связь обнаружена данными;
- настройки компенсируют друг друга: отношение важнее значений (стоп / безубыток, быстрый / медленный ATR);
- экспериментальный флаг и его числовые параметры.

**Не группируй:** настройки разных этапов; больше трёх-четырёх измерений сетки одновременно — число испытаний растёт как произведение.

**Рабочие группы:**

| Группа | Состав | Цель |
|---|---|---|
| Геометрия базы | `[W31]`, `[W32]`, `[W33]`, затем `[E03]` | какие базы торгуются; широкая база — дальний стоп |
| Режим волатильности | `[W35]`, `[W36]`, `[W34]` | «плоская» база перед движением |
| Направление дня | `[W37]` и правило вечера | вечер только по тренду |
| Риск-якорь | `[E03]` и объём финансового модуля | масштаб потерь |
| Защита прибыли | безубыток (триггер, отступ), `[W16]`, `[W26]` | доля сделок около нуля против среднего выигрыша |
| Повторные входы | `[W39]` и триггер безубытка | второй, истинный пробой после ложного |
| Ядро | ширина окна мин/макс, порог наклона | скорость против запаздывания |
| Хёрст | окно, отображение H → ширина | шум оценки против эффекта режима |
| Лоренц | число соседей, горизонт, признаки, метод | память против переподгонки |
| Калман | период, ширина полос, политика | ложные пробои против запаздывания |
| Масса | окно, веса эшелонов, процентиль, порог | дрейф против шума |
| Охрана дня | пределы потерь и прибыли | хвостовой риск дня |

**Улучшать имеющееся:** только одну группу за раз; остальные — на подтверждённых значениях; сетка ±1 шаг вокруг подтверждённого; решение — проверкой вперёд.

//  ТРИ УРОКА ПРЕДЫДУЩЕЙ СИСТЕМЫ, ЗАЛОЖЕННЫЕ В ОСНОВАНИЕ.

//  1. СТУПЕНИ СЧИТАЮТСЯ ПО ПОДТВЕРЖДЁННЫМ ИСПОЛНЕНИЯМ, а не по факту отправки

//  только на границе через TickValue = TickSize × PointValue. Это устраняет
//  класс ошибок, где доллар движения цены путается с долларом результата.

    /// <summary>
    /// ФИНАНСОВЫЙ МОДУЛЬ. Считает объём, стартовый стоп, ступени, безубыток
    /// и трейлинг. Ордеров не создаёт.
    /// </summary>
    public class SwFinancialManager
    {
        private readonly DZ_SW_Hub_S021 hub;
        private readonly SwFinancialConfig cfg;

        // --- состояние сделки ---
        private int side;                     // +1 long, -1 short, 0 вне рынка
        private double entryPrice;
        private int entryBar;
        private int initialQty;
        private double initialStopPrice;
        private double riskTicks;             // R в тиках, база всех R-кратных величин

        private int stepsFilled;              // ПОДТВЕРЖДЁННЫХ ступеней
        private int stepsRequested;
        private int stepRequestBar = -1;      // бар запроса неподтверждённой ступени
        public int StepTimeouts;
        private readonly Dictionary<int, double> stepFillPrice = new Dictionary<int, double>();
        private readonly Dictionary<int, int> stepFillQty = new Dictionary<int, int>();

        private double currentStop = double.NaN;

        // Трейлинг этой сделки в тиках. Задаётся сигналом при входе; без
        // этого утренняя и вечерняя сделки шли бы с одним расстоянием, а
        // исследование требует 15 и 10 тиков соответственно.
        private double tradeTrailTicks = double.NaN;
        private int lastStopMoveBar = -1;
        private double bestPrice = double.NaN;
        private bool breakevenDone;

        // --- диагностика ---
        public int SizingBlocks, StepsRequestedTotal, StepsConfirmedTotal;
        public int StopMovesRequested, StopMovesSuppressed;
        public string LastNote = "—";

        public SwFinancialManager(DZ_SW_Hub_S021 h, SwFinancialConfig c)
        {
            hub = h; cfg = c;
        }

        #region Деньги и объём

        /// <summary>Стоимость одного тика одного контракта.</summary>
        private double TickValue
        {
            get { return hub.HubTickSize * hub.HubPointValue; }
        }

        /// <summary>
        /// Расстояние стопа в тиках по выбранному режиму.
        /// Режим PriceDollars означает доллары ДВИЖЕНИЯ ЦЕНЫ: для микрозолота
        /// с шагом 0.10 значение 25.0 даёт 250 тиков. Путаница этой величины
        /// с денежным результатом — источник ошибок на порядок.
        /// </summary>
        public double StopDistanceTicks()
        {
            switch (cfg.StopMode)
            {
                case SwStopMode.PriceDollars:
                    return hub.HubTickSize > 0 ? cfg.StopPriceDollars / hub.HubTickSize : 0;

                case SwStopMode.AtrMultiple:
                {
                    double atr = hub.HubAtrTicks;
                    return atr > 0 ? atr * cfg.StopAtrMultiple : cfg.StopFixedTicks;
                }

                default:
                    return cfg.StopFixedTicks;
            }
        }

        /// <summary>
        /// Полный риск одного контракта в деньгах: ход стопа плюс издержки
        /// круга плюс резерв на проскальзывание. Без последних двух слагаемых
        /// расчётный объём систематически завышается.
        /// </summary>
        public double TotalRiskPerContract()
        {
            double tv = TickValue;
            if (tv <= 0) return 0;

            double stopTicks = StopDistanceTicks();
            if (stopTicks <= 0) return 0;

            return stopTicks * tv
                 + cfg.RoundTurnCostPerContract
                 + cfg.SlippageReserveTicks * tv;
        }

        /// <summary>
        /// Объём позиции. Возвращает ноль, когда вход невозможен, и пишет
        /// причину в LastNote: молчаливый отказ недопустим.
        /// </summary>
        public int ResolveQuantity(out string note)
        {
            note = string.Empty;

            if (cfg.SizingMode == SwSizingMode.FixedQuantity)
                return Clamp(cfg.FixedQuantity);

            double risk = TotalRiskPerContract();
            if (risk <= 0)
            {
                note = "риск на контракт не определён: проверьте стоп и шаг цены";
                SizingBlocks++;
                return 0;
            }

            double budget;
            if (cfg.SizingMode == SwSizingMode.RiskMoneyPerTrade)
            {
                budget = cfg.RiskMoneyPerTrade;
                if (budget <= 0)
                {
                    note = "бюджет риска равен нулю";
                    SizingBlocks++;
                    return 0;
                }
            }
            else
            {
                double eq = cfg.AccountEquityReference;
                if (eq <= 0)
                {
                    note = "расчётный капитал равен нулю";
                    SizingBlocks++;
                    return 0;
                }
                budget = eq * cfg.RiskPercentOfEquity / 100.0;
            }

            int q = (int)Math.Floor(budget / risk);
            if (q < cfg.MinQuantity)
            {
                note = "расчётный объём " + q + " меньше минимума " + cfg.MinQuantity
                     + " (риск на контракт " + risk.ToString("F2") + ")";
                SizingBlocks++;
                return 0;
            }

            note = "бюджет " + budget.ToString("F2") + " / риск " + risk.ToString("F2")
                 + " = " + q + " конт.";
            return Clamp(q);
        }

        private int Clamp(int q)
        {
            if (q < cfg.MinQuantity) q = cfg.MinQuantity;
            if (q > cfg.MaxQuantity) q = cfg.MaxQuantity;
            return Math.Max(1, q);
        }

        /// <summary>
        /// Минимальный отступ безубытка, покрывающий издержки круга.
        /// При меньшем значении срабатывание безубыточного стопа даёт убыток.
        /// </summary>
        public int MinimumBreakevenOffsetTicks()
        {
            double tv = TickValue;
            if (tv <= 0 || cfg.RoundTurnCostPerContract <= 0) return 0;
            return (int)Math.Ceiling(cfg.RoundTurnCostPerContract / tv);
        }

        #endregion

        #region Жизненный цикл сделки

        /// <summary>Вызывается коммутатором при подтверждённом входе.</summary>
        public void OnTradeOpened(int dir, double avgPrice, int qty, int bar)
        {
            side = dir;
            entryPrice = avgPrice;
            entryBar = bar;
            initialQty = qty;

            riskTicks = StopDistanceTicks();

            double ov = hub.HubPendingTrailOverride;
            tradeTrailTicks = ov > 0 ? ov : hub.HubTrailDistance;
            hub.HubPendingTrailOverride = double.NaN;
            initialStopPrice = dir > 0
                ? entryPrice - riskTicks * hub.HubTickSize
                : entryPrice + riskTicks * hub.HubTickSize;

            currentStop = initialStopPrice;
            bestPrice = avgPrice;
            stepsFilled = 0; stepsRequested = 0; stepRequestBar = -1;
            breakevenDone = false;
            lastStopMoveBar = -1;
            stepFillPrice.Clear(); stepFillQty.Clear();

            LastNote = "вход " + (dir > 0 ? "LONG" : "SHORT") + " x" + qty
                     + ", R = " + riskTicks.ToString("F0") + " тиков";
            hub.HubTrace("FIN", "TRADE_OPENED", LastNote);
        }

        public void OnTradeClosed()
        {
            side = 0;
            stepsFilled = 0; stepsRequested = 0;
            currentStop = double.NaN;
            bestPrice = double.NaN;
            breakevenDone = false;
            stepFillPrice.Clear(); stepFillQty.Clear();
        }

        /// <summary>
        /// ПОДТВЕРЖДЁННОЕ ИСПОЛНЕНИЕ СТУПЕНИ. Вызывается коммутатором из
        /// OnExecutionUpdate. Это единственный путь, по которому ступень
        /// считается состоявшейся. Отправка заявки таковым не является.
        /// </summary>
        public void OnStepExecution(int stepNumber, double price, int qty)
        {
            if (stepNumber <= 0) return;

            int had; stepFillQty.TryGetValue(stepNumber, out had);
            double avg; stepFillPrice.TryGetValue(stepNumber, out avg);

            int now = had + qty;
            if (now > 0) avg = (avg * had + price * qty) / now;

            stepFillQty[stepNumber] = now;
            stepFillPrice[stepNumber] = avg;

            if (stepNumber > stepsFilled)
            {
                stepsFilled = stepNumber;
                stepRequestBar = -1;
                StepsConfirmedTotal++;
                hub.HubTrace("FIN", "STEP_CONFIRMED",
                    "ступень " + stepNumber + ": " + now + " конт. по " + avg.ToString("F2"));
            }
        }

        #endregion

        #region Оценка на баре

        /// <summary>
        /// Главный метод. Складывает требования в общий ящик коммутатора.
        /// Ордеров не создаёт.
        /// </summary>
        public void Evaluate(int bar, List<SwAction> outbox)
        {
            int pos = hub.HubPositionSide;

            if (pos == 0) { if (side != 0) OnTradeClosed(); return; }
            if (side == 0) return;                 // вход ещё не подтверждён
            if (riskTicks <= 0) return;

            // На баре входа безубыток и трейлинг не включаются, и лучшая цена не
            // обновляется: неизвестно, был ли максимум бара после входа. Так же
            // поступает контрольный бэктест.
            if (bar == entryBar) return;

            double px = hub.HubClose(0);

            bestPrice = side > 0
                ? Math.Max(double.IsNaN(bestPrice) ? px : bestPrice, hub.HubHigh(0))
                : Math.Min(double.IsNaN(bestPrice) ? px : bestPrice, hub.HubLow(0));

            double profitTicks = (side > 0 ? px - entryPrice : entryPrice - px) / hub.HubTickSize;
            double rNow = profitTicks / riskTicks;

            // ---------- 1. Ступени фиксации ----------
            if (cfg.UseScaleOut) EvaluateScaleOut(bar, profitTicks, outbox);

            // ---------- 2. Перенос стопа ----------
            // Два кандидата вычисляются независимо, затем берётся тот,
            // что СИЛЬНЕЕ сокращает риск. Спора между безубытком и
            // трейлингом не возникает по построению.
            // ЕДИНЫЙ ИСТОЧНИК ФАКТИЧЕСКОГО СТОПА — исполнитель. Прежде модуль
            // запоминал новый стоп до того, как исполнитель его принял, и не
            // знал о переносах, сделанных другими модулями сопровождения.
            // Две копии «текущего стопа» расходились.
            double actual = hub.HubCurrentStop;
            if (!double.IsNaN(actual)) currentStop = actual;

            double basisPx = cfg.BreakevenOnBestPrice ? bestPrice : px;
            double rBasis = ((side > 0 ? basisPx - entryPrice : entryPrice - basisPx) / hub.HubTickSize) / riskTicks;
            double candBe = ComputeBreakeven(rBasis);
            double candTr = ComputeTrail(rBasis);

            double target = BetterStop(candBe, candTr);
            if (double.IsNaN(target)) return;

            // ЗАПРЕТ ПОСЛЕ ПЕРВОЙ ФИКСАЦИИ — подавляет ОБЕ ветви сразу.
            if (cfg.DisableStopMoveAfterStep1 && stepsFilled >= 1)
            {
                StopMovesSuppressed++;
                LastNote = "перенос подавлен: после первой фиксации стоп не двигается";
                hub.HubTrace("FIN", "STOP_SUPPRESSED", LastNote);
                return;
            }

            if (cfg.MinBarsBetweenStopMoves > 0 && lastStopMoveBar >= 0
                && bar - lastStopMoveBar < cfg.MinBarsBetweenStopMoves)
                return;

            // Движение только в сторону сокращения риска.
            if (!double.IsNaN(currentStop))
            {
                bool better = side > 0 ? target > currentStop : target < currentStop;
                if (!better) return;
            }

            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.MoveStop;
            a.Price = target;
            a.Reason = DescribeStopSource(candBe, candTr, target, rNow);
            outbox.Add(a);

            // currentStop здесь НЕ присваивается: принятие решает исполнитель,
            // и фактическое значение будет прочитано у него на следующем баре.
            lastStopMoveBar = bar;
            StopMovesRequested++;
            LastNote = a.Reason;
        }

        private void EvaluateScaleOut(int bar, double profitTicks, List<SwAction> outbox)
        {
            // ТАЙМАУТ ПОДТВЕРЖДЕНИЯ. Ожидание подтверждения верно, но без
            // предела оно превращается в тупик: если исполнитель отклонил
            // ступень арбитражем — например, выход по линии пришёлся на тот
            // же бар, — подтверждение не придёт никогда. Прежде это означало,
            // что до конца сделки не будет ни одной ступени и не наступит
            // безубыток AfterStep1.
            if (stepsRequested > stepsFilled && stepRequestBar >= 0
                && bar - stepRequestBar > Math.Max(1, cfg.StepConfirmTimeoutBars))
            {
                StepTimeouts++;
                hub.HubTrace("FIN", "STEP_TIMEOUT",
                    "ступень " + stepsRequested + " не подтверждена за "
                    + (bar - stepRequestBar) + " баров, запрос снят");
                stepsRequested = stepsFilled;
                stepRequestBar = -1;
            }

            if (stepsRequested >= cfg.ScaleSteps) return;

            // Следующая ступень считается от ПОДТВЕРЖДЁННЫХ, а не отправленных:
            // отклонённая заявка не должна сдвигать очередь.
            int next = stepsFilled + 1;
            if (next > cfg.ScaleSteps) return;
            if (stepsRequested >= next) return;     // эта уже отправлена, ждём подтверждения

            double needTicks = StepDistanceTicks(next);
            if (needTicks <= 0 || profitTicks < needTicks) return;

            int qty = StepQuantity(next);
            if (qty <= 0) return;

            SwAction a = SwAction.None(bar);
            a.Kind = SwActionKind.PartialExit;
            a.Quantity = qty;
            a.Tag = "STEP" + next;
            a.Reason = "ступень " + next + " при +" + profitTicks.ToString("F0")
                     + " тиков (нужно " + needTicks.ToString("F0") + ")";
            outbox.Add(a);

            stepsRequested = next;
            stepRequestBar = bar;
            StepsRequestedTotal++;
            hub.HubTrace("FIN", "STEP_REQUESTED", a.Reason + ", " + qty + " конт.");
        }

        private double StepDistanceTicks(int stepNumber)
        {
            switch (cfg.StepDistanceMode)
            {
                case SwStepDistanceMode.PriceDollars:
                    return hub.HubTickSize > 0
                        ? stepNumber * cfg.StepPriceDollars / hub.HubTickSize : 0;

                case SwStepDistanceMode.RMultiple:
                    return stepNumber * cfg.StepRMultiple * riskTicks;

                default:
                    return stepNumber * cfg.StepFixedTicks;
            }
        }

        private int StepQuantity(int stepNumber)
        {
            int open = hub.HubPositionQty;
            int reserve = cfg.KeepRunner ? 1 : 0;

            int want;
            if (stepNumber == 1 && cfg.ContractsOnStep1 > 0)
                want = cfg.ContractsOnStep1;
            else if (cfg.ContractsPerStep > 0)
                want = cfg.ContractsPerStep;
            else
            {
                int divisor = Math.Max(1, cfg.ScaleSteps + reserve);
                want = Math.Max(1, initialQty / divisor);
            }

            return Math.Max(0, Math.Min(want, open - reserve));
        }

        private double ComputeBreakeven(double rNow)
        {
            if (breakevenDone) return double.NaN;

            bool due;
            switch (cfg.BreakevenMode)
            {
                case SwBreakevenMode.AfterStep1: due = stepsFilled >= 1; break;
                case SwBreakevenMode.AfterStep2: due = stepsFilled >= 2; break;
                case SwBreakevenMode.AtRMultiple: due = rNow >= cfg.BreakevenAtR; break;
                default: return double.NaN;
            }
            if (!due) return double.NaN;

            // Отступ не меньше того, что покрывает издержки круга.
            int off = Math.Max(cfg.BreakevenOffsetTicks, MinimumBreakevenOffsetTicks());

            breakevenDone = true;
            return side > 0
                ? entryPrice + off * hub.HubTickSize
                : entryPrice - off * hub.HubTickSize;
        }

        private double ComputeTrail(double rNow)
        {
            if (cfg.TrailMode == SwTrailMode.Off) return double.NaN;
            if (rNow < cfg.TrailActivateAtR) return double.NaN;

            switch (cfg.TrailMode)
            {
                case SwTrailMode.FixedDistance:
                {
                    double tt = tradeTrailTicks > 0 ? tradeTrailTicks : cfg.TrailFixedTicks;
                    return side > 0
                        ? bestPrice - tt * hub.HubTickSize
                        : bestPrice + tt * hub.HubTickSize;
                }

                case SwTrailMode.AtrMultiple:
                {
                    double atr = hub.HubAtrTicks;
                    if (atr <= 0) return double.NaN;
                    double d = atr * cfg.TrailAtrMultiple * hub.HubTickSize;
                    return side > 0 ? bestPrice - d : bestPrice + d;
                }

                case SwTrailMode.StepLock:
                {
                    // После ступени k стоп встаёт на уровень ступени k-1.
                    // При k = 1 это цена входа: прибыль первой ступени
                    // фиксируется, а остаток ведётся без риска.
                    if (stepsFilled < 1) return double.NaN;
                    int prev = stepsFilled - 1;
                    double lockTicks = prev == 0 ? 0 : StepDistanceTicks(prev);
                    return side > 0
                        ? entryPrice + lockTicks * hub.HubTickSize
                        : entryPrice - lockTicks * hub.HubTickSize;
                }

                default: return double.NaN;
            }
        }

        /// <summary>Из двух кандидатов берётся сильнее сокращающий риск.</summary>
        private double BetterStop(double a, double b)
        {
            if (double.IsNaN(a)) return b;
            if (double.IsNaN(b)) return a;
            return side > 0 ? Math.Max(a, b) : Math.Min(a, b);
        }

        private string DescribeStopSource(double be, double tr, double chosen, double rNow)
        {
            string who;
            if (!double.IsNaN(be) && Math.Abs(chosen - be) < 1e-9) who = "безубыток";
            else if (!double.IsNaN(tr) && Math.Abs(chosen - tr) < 1e-9) who = "трейлинг " + cfg.TrailMode;
            else who = "перенос";

            return who + " → " + chosen.ToString("F2")
                 + " (R = " + rNow.ToString("F2") + ", ступеней " + stepsFilled + ")";
        }

        #endregion

        #region Диагностика

        public double CurrentR
        {
            get
            {
                if (side == 0 || riskTicks <= 0) return 0;
                double px = hub.HubClose(0);
                double p = (side > 0 ? px - entryPrice : entryPrice - px) / hub.HubTickSize;
                return p / riskTicks;
            }
        }

        /// <summary>Открытый риск в деньгах: расстояние до текущего стопа.</summary>
        public double OpenRiskMoney
        {
            get
            {
                if (side == 0 || double.IsNaN(currentStop)) return 0;
                double px = hub.HubClose(0);
                double ticks = (side > 0 ? px - currentStop : currentStop - px) / hub.HubTickSize;
                return -ticks * TickValue * hub.HubPositionQty;   // отрицательно = риск
            }
        }

        public string StatusLine
        {
            get
            {
                if (side == 0)
                    return "вне рынка | R-база " + StopDistanceTicks().ToString("F0")
                         + " тиков, риск/контракт " + TotalRiskPerContract().ToString("F2");

                return string.Format("{0} x{1} @ {2:F2} | R {3:F2} | стоп {4} | ступеней {5}/{6} | открытый риск {7:F2}",
                    side > 0 ? "LONG" : "SHORT", hub.HubPositionQty, entryPrice,
                    CurrentR,
                    double.IsNaN(currentStop) ? "—" : currentStop.ToString("F2"),
                    stepsFilled, cfg.ScaleSteps, OpenRiskMoney);
            }
        }

        public string SummaryLine
        {
            get
            {
                return "ступеней запрошено " + StepsRequestedTotal
                     + ", подтверждено " + StepsConfirmedTotal
                     + ", переносов стопа " + StopMovesRequested
                     + ", подавлено " + StopMovesSuppressed
                     + ", таймаутов ступеней " + StepTimeouts
                     + ", отказов расчёта объёма " + SizingBlocks;
            }
        }

        #endregion
    }
